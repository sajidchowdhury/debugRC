    46→    46→    46→    /**
    47→    47→    47→     * Phase 1: Create a sales return (no stock, no GL).
    48→    48→    48→     * Looks up ORIGINAL avg_cost from the challan's stock_transaction.
    49→    49→    49→     *
    50→    50→    50→     * @param array $data {
    51→    51→    51→     *     sales_invoice_id: int,
    52→    52→    52→     *     return_date: string (Y-m-d),
    53→    53→    53→     *     reason: string|null,
    54→    54→    54→     *     created_by: int,
    55→    55→    55→     *     items: array each { product_id, warehouse_id, qty, rate, sales_invoice_item_id }
    56→    56→    56→     * }
    57→    57→    57→     * @return SalesReturn
    58→    58→    58→     */
    59→    59→    59→    public function createReturn(array $data): SalesReturn
    60→    60→    60→    {
    61→    61→    61→        $invoiceId = (int) ($data['sales_invoice_id'] ?? 0);
    62→    62→    62→        if ($invoiceId <= 0) {
    63→    63→    63→            throw new \InvalidArgumentException('sales_invoice_id is required.');
    64→    64→    64→        }
    65→    65→    65→
    66→    66→    66→        $invoice = DB::table('sales_invoices')->where('id', $invoiceId)->first();
    67→    67→    67→        if (!$invoice) {
    68→    68→    68→            throw new \InvalidArgumentException("Invoice {$invoiceId} not found.");
    69→    69→    69→        }
    70→    70→    70→        if ($invoice->is_reversed) {
    71→    71→    71→            throw new \RuntimeException("Cannot return against a reversed invoice.");
    72→    72→    72→        }
    73→    73→    73→
    74→    74→    74→        // P0-8: Defense-in-depth branch isolation check.
    75→    75→    75→        $this->salesAccess->assertBranchAccessible((int) $invoice->branch_id);
    76→    76→    76→
    77→    77→    77→        // Find the challan for this invoice.
    78→    78→    78→        $challan = DB::table('sales_challans')
    79→    79→    79→            ->where('sales_invoice_id', $invoiceId)
    80→    80→    80→            ->where('is_reversed', false)
    81→    81→    81→            ->first();
    82→    82→    82→        if (!$challan) {
    83→    83→    83→            throw new \RuntimeException("No active challan found for invoice {$invoiceId}. Returns require a completed challan.");
    84→    84→    84→        }
    85→    85→    85→
    86→    86→    86→        $items = $this->validateItems($data['items'], $invoiceId, $challan->id);
    87→    87→    87→        $totalRevenue = collect($items)->sum(fn($i) => $i['qty'] * $i['rate']);
    88→    88→    88→        $totalCogs = collect($items)->sum(fn($i) => $i['qty'] * $i['original_cost']);
    89→    89→    89→
    90→    90→    90→        $returnCode = $this->generateReturnCode();
    91→    91→    91→        $customerId = (int) $invoice->customer_id;
    92→    92→    92→        $branchId = (int) $invoice->branch_id;
    93→    93→    93→
    94→    94→    94→        return DB::transaction(function () use (
    95→    95→    95→            $data, $items, $totalRevenue, $totalCogs, $returnCode, $invoiceId, $customerId, $branchId
    96→    96→    96→        ) {
    97→    97→    97→            $returnId = DB::table('sales_returns')->insertGetId([
    98→    98→    98→                'return_code' => $returnCode,
    99→    99→    99→                'return_date' => $data['return_date'] ?? now()->format('Y-m-d'),
   100→   100→   100→                'sales_invoice_id' => $invoiceId,
   101→   101→   101→                'customer_id' => $customerId,
   102→   102→   102→                'branch_id' => $branchId,
   103→   103→   103→                'total_amount' => round($totalRevenue, 2),
   104→   104→   104→                'cogs_amount' => round($totalCogs, 2),
   105→   105→   105→                'status' => 'created',
   106→   106→   106→                'is_reversed' => false,
   107→   107→   107→                'reason' => $data['reason'] ?? null,
   108→   108→   108→                'created_by' => $data['created_by'] ?? null,
   109→   109→   109→                'created_at' => now(),
   110→   110→   110→                'updated_at' => now(),
   111→   111→   111→            ]);
   112→   112→   112→
   113→   113→   113→            $itemRows = [];
   114→   114→   114→            foreach ($items as $item) {
   115→   115→   115→                $itemRows[] = [
   116→   116→   116→                    'sales_return_id' => $returnId,
   117→   117→   117→                    'sales_invoice_item_id' => $item['sales_invoice_item_id'],
   118→   118→   118→                    'product_id' => $item['product_id'],
   119→   119→   119→                    'warehouse_id' => $item['warehouse_id'],
   120→   120→   120→                    'qty' => $item['qty'],
   121→   121→   121→                    'rate' => $item['rate'],
   122→   122→   122→                    'original_cost' => $item['original_cost'],
   123→   123→   123→                ];
   124→   124→   124→            }
   125→   125→   125→            DB::table('sales_return_items')->insert($itemRows);
   126→   126→   126→
   127→   127→   127→            // P1-3: Audit log — return_created.
   128→   128→   128→            $this->auditLogger->returnCreated(
   129→   129→   129→                $data['created_by'] ?? auth()->id() ?? 0,
   130→   130→   130→                $returnId, $returnCode, $invoiceId, $customerId, $branchId,
   131→   131→   131→                $totalRevenue, $totalCogs
   132→   132→   132→            );
   133→   133→   133→
   134→   134→   134→            // P1-7: Notify admins that a return was created.
   135→   135→   135→            $this->notifications->dispatch(
   136→   136→   136→                'return_created',
   137→   137→   137→                "Return {$returnCode} created for Tk " . number_format($totalRevenue, 2)
   138→   138→   138→                . " against invoice #{$invoiceId}.",
   139→   139→   139→                'sales_return',
   140→   140→   140→                $returnId
   141→   141→   141→            );
   142→   142→   142→
   143→   143→   143→            return SalesReturn::with(['items.product', 'salesInvoice', 'customer', 'branch'])->find($returnId);
   144→   144→   144→        });
   145→   145→   145→    }
   146→   146→   146→
   147→   147→   147→    /**
   148→   148→   148→     * Phase 2: Confirm a sales return — stock IN + GL + customer_ledger.
   149→   149→   149→     *
   150→   150→   150→     * @param int $returnId
   151→   151→   151→     * @param int $confirmedBy
   152→   152→   152→     * @return SalesReturn
   153→   153→   153→     * @throws \RuntimeException If not 'created' status, or stock/GL fails.
   154→   154→   154→     */
   155→   155→   155→    public function confirmReturn(int $returnId, int $confirmedBy): SalesReturn
   156→   156→   156→    {
   157→   157→   157→        return DB::transaction(function () use ($returnId, $confirmedBy) {
   158→   158→   158→            $return = SalesReturn::with('items')->lockForUpdate()->find($returnId);
   159→   159→   159→
   160→   160→   160→            if (!$return) {
   161→   161→   161→                throw new \RuntimeException("Return {$returnId} not found.");
   162→   162→   162→            }
   163→   163→   163→            if (!$return->isCreated()) {
   164→   164→   164→                throw new \RuntimeException("Only 'created' returns can be confirmed (current: {$return->status}).");
   165→   165→   165→            }
   166→   166→   166→
   167→   167→   167→            $returnDate = $return->return_date->format('Y-m-d');
   168→   168→   168→
   169→   169→   169→            // 1. Stock IN at ORIGINAL avg_cost (CRITICAL: not current avg_cost).
   170→   170→   170→            foreach ($return->items as $item) {
   171→   171→   171→                $this->stockService->applyTransaction([
   172→   172→   172→                    'warehouse_id' => $item->warehouse_id,
   173→   173→   173→                    'product_id' => $item->product_id,
   174→   174→   174→                    'qty' => (float) $item->qty, // positive = IN
   175→   175→   175→                    'rate' => (float) $item->original_cost, // ORIGINAL avg_cost from challan
   176→   176→   176→                    'reference_type' => 'sales_return',
   177→   177→   177→                    'reference_id' => $return->id,
   178→   178→   178→                    'notes' => 'Sales Return ' . $return->return_code . ' — stock restored at original cost',
   179→   179→   179→                    'transaction_date' => $returnDate,
   180→   180→   180→                    'created_by' => $confirmedBy,
   181→   181→   181→                ]);
   182→   182→   182→            }
   183→   183→   183→
   184→   184→   184→            // 1b. P1-5: Linked damage write-off for 'Damage' condition items.
   185→   185→   185→            // Legacy: SalesReturnModel::createLinkedDamageWriteOff — for each
   186→   186→   186→            // Damage-condition item, create a damage_invoices row + stock OUT
   187→   187→   187→            // (immediately back OUT the damaged goods) + GL Dr damage_loss / Cr inventory.
   188→   188→   188→            $linkedDamageIds = $this->createLinkedDamageWriteOffs($return, $confirmedBy);
   189→   189→   189→
   190→   190→   190→            // 2. GL Revenue Reversal: Dr Sales Return / Cr AR.
   191→   191→   191→            $journalEntryId = $this->postRevenueReversalGL($return, $confirmedBy);
   192→   192→   192→
   193→   193→   193→            // 3. GL COGS Reversal: Dr Inventory / Cr COGS.
   194→   194→   194→            $cogsJournalEntryId = $this->postCogsReversalGL($return, $confirmedBy);
   195→   195→   195→
   196→   196→   196→            // 4. Customer ledger: credit entry (customer owes less).
   197→   197→   197→            $this->postCustomerLedgerCredit($return, $journalEntryId, $confirmedBy);
   198→   198→   198→
   199→   199→   199→            // 5. Update return status.
   200→   200→   200→            DB::table('sales_returns')
   201→   201→   201→                ->where('id', $returnId)
   202→   202→   202→                ->update([
   203→   203→   203→                    'status' => 'confirmed',
   204→   204→   204→                    'journal_entry_id' => $journalEntryId,
   205→   205→   205→                    'cogs_journal_entry_id' => $cogsJournalEntryId,
   206→   206→   206→                    'updated_at' => now(),
   207→   207→   207→                ]);
   208→   208→   208→
   209→   209→   209→            // P1-3: Audit log — return_confirmed.
   210→   210→   210→            $this->auditLogger->returnConfirmed(
   211→   211→   211→                $confirmedBy,
   212→   212→   212→                $returnId, $return->return_code, (int) $return->branch_id,
   213→   213→   213→                (float) $return->total_amount, (float) $return->cogs_amount,
   214→   214→   214→                $journalEntryId
   215→   215→   215→            );
   216→   216→   216→
   217→   217→   217→            // P1-7: Notify warehouse managers + accountants that a return was confirmed.
   218→   218→   218→            $this->notifications->dispatch(
   219→   219→   219→                'return_confirmed',
   220→   220→   220→                "Return {$return->return_code} confirmed — stock restored, GL posted. "
   221→   221→   221→                . "Total: Tk " . number_format((float) $return->total_amount, 2)
   222→   222→   222→                . ", COGS reversed: Tk " . number_format((float) $return->cogs_amount, 2) . ".",
   223→   223→   223→                'sales_return',
   224→   224→   224→                $returnId
   225→   225→   225→            );
   226→   226→   226→
   227→   227→   227→            return SalesReturn::with([
   228→   228→   228→                'items.product', 'salesInvoice', 'customer', 'branch',
   229→   229→   229→                'journalEntry.lines.ledger', 'cogsJournalEntry.lines.ledger',
   230→   230→   230→            ])->find($returnId);
   231→   231→   231→        });
   232→   232→   232→    }
   233→   233→   233→
   234→   234→   234→    /**
   235→   235→   235→     * Phase 3: Reverse a confirmed return.
   236→   236→   236→     * Reverses stock + both GL journals + customer_ledger.
   237→   237→   237→     *
   238→   238→   238→     * @param int $returnId
   239→   239→   239→     * @param int $reversedBy
   240→   240→   240→     * @param string $reason
   241→   241→   241→     * @return SalesReturn
   242→   242→   242→     */
   243→   243→   243→    public function reverseReturn(int $returnId, int $reversedBy, string $reason = ''): SalesReturn
   244→   244→   244→    {
   245→   245→   245→        return DB::transaction(function () use ($returnId, $reversedBy, $reason) {
   246→   246→   246→            $return = SalesReturn::with('items')->lockForUpdate()->find($returnId);
   247→   247→   247→
   248→   248→   248→            if (!$return) {
   249→   249→   249→                throw new \RuntimeException("Return {$returnId} not found.");
   250→   250→   250→            }
   251→   251→   251→            if (!$return->isConfirmed()) {
   252→   252→   252→                throw new \RuntimeException("Only confirmed returns can be reversed (current: {$return->status}).");
   253→   253→   253→            }
   254→   254→   254→
   255→   255→   255→            // Reverse both GL journals.
   256→   256→   256→            if ($return->journal_entry_id) {
   257→   257→   257→                $this->journalPosting->reverseJournalEntry(
   258→   258→   258→                    $return->journal_entry_id, $reversedBy,
   259→   259→   259→                    "Return reversed: {$reason}"
   260→   260→   260→                );
   261→   261→   261→            }
   262→   262→   262→            if ($return->cogs_journal_entry_id) {
   263→   263→   263→                $this->journalPosting->reverseJournalEntry(
   264→   264→   264→                    $return->cogs_journal_entry_id, $reversedBy,
   265→   265→   265→                    "Return reversed: {$reason}"
   266→   266→   266→                );
   267→   267→   267→            }
   268→   268→   268→
   269→   269→   269→            // Reverse customer_ledger.
   270→   270→   270→            $this->reverseCustomerLedgerCredit($return, $reversedBy, $reason);
   271→   271→   271→
   272→   272→   272→            // P1-5: Reverse linked damage write-offs FIRST (before return stock reversal).
   273→   273→   273→            // Each linked damage_invoice has its own stock OUT + GL that must be reversed.
   274→   274→   274→            $this->reverseLinkedDamageForReturn($returnId, $reversedBy, $reason);
   275→   275→   275→
   276→   276→   276→            // Reverse each stock movement (stock OUT at original cost — append-only reversal).
   277→   277→   277→            $stockTxs = DB::table('stock_transactions')
   278→   278→   278→                ->where('reference_type', 'sales_return')
   279→   279→   279→                ->where('reference_id', $returnId)
   280→   280→   280→                ->where('is_reversed', false)
   281→   281→   281→                ->get();
   282→   282→   282→
   283→   283→   283→            foreach ($stockTxs as $tx) {
   284→   284→   284→                $this->stockService->reverseTransaction(
   285→   285→   285→                    $tx->id, $reversedBy,
   286→   286→   286→                    "Return reversed: {$reason}"
   287→   287→   287→                );
   288→   288→   288→            }
   289→   289→   289→
   290→   290→   290→            // Mark return as reversed.
   291→   291→   291→            DB::table('sales_returns')
   292→   292→   292→                ->where('id', $returnId)
   293→   293→   293→                ->update([
   294→   294→   294→                    'status' => 'reversed',
   295→   295→   295→                    'is_reversed' => true,
   296→   296→   296→                    'reversed_at' => now(),
   297→   297→   297→                    'reversed_by' => $reversedBy,
   298→   298→   298→                    'reverse_reason' => $reason,
   299→   299→   299→                    'updated_at' => now(),
   300→   300→   300→                ]);
   301→   301→   301→
   302→   302→   302→            // P1-3: Audit log — return_reversed.
   303→   303→   303→            $this->auditLogger->returnReversed(
   304→   304→   304→                $reversedBy,
   305→   305→   305→                $returnId, $return->return_code, (int) $return->branch_id,
   306→   306→   306→                (float) $return->total_amount, $reason
   307→   307→   307→            );
   308→   308→   308→
   309→   309→   309→            // P1-7: Notify accountants + admins that a return was reversed.
   310→   310→   310→            $this->notifications->dispatch(
   311→   311→   311→                'return_reversed',
   312→   312→   312→                "Return {$return->return_code} reversed — stock + GL + ledger rolled back. "
   313→   313→   313→                . "Reason: {$reason}",
   314→   314→   314→                'sales_return',
   315→   315→   315→                $returnId
   316→   316→   316→            );
   317→   317→   317→
   318→   318→   318→            return SalesReturn::find($returnId);
   319→   319→   319→        });
   320→   320→   320→    }
   321→   321→   321→
   322→   322→   322→    /**
   323→   323→   323→     * Post GL Revenue Reversal: Dr Sales Return / Cr AR.
   324→   324→   324→     *
   325→   325→   325→     * @return int journal_entry_id
   326→   326→   326→     */
   327→   327→   327→    private function postRevenueReversalGL(SalesReturn $return, int $createdBy): int
   328→   328→   328→    {
   329→   329→   329→        $amount = (float) $return->total_amount;
   330→   330→   330→        if ($amount < 0.01) return 0;
   331→   331→   331→
   332→   332→   332→        $returnLedgerId = $this->journalPosting->lookupLedgerByNature('sales_return');
   333→   333→   333→        $arLedgerId = $this->journalPosting->lookupLedgerByNature('ar');
   334→   334→   334→
   335→   335→   335→        if (!$returnLedgerId) {
   336→   336→   336→            throw new \RuntimeException('Sales Return ledger not found (nature: sales_return).');
   337→   337→   337→        }

   339→   339→   339→            throw new \RuntimeException('Accounts Receivable ledger not found (nature: ar).');
   340→   340→   340→        }
   341→   341→   341→
   342→   342→   342→        return $this->journalPosting->createJournalEntry([
   343→   343→   343→            'entry_date' => $return->return_date->format('Y-m-d'),

   346→   346→   346→            'branch_id' => $return->branch_id,

   348→   348→   348→            'source' => 'sales_return',
   349→   349→   349→            'created_by' => $createdBy,
   350→   350→   350→        ], [
   351→   351→   351→            [
   352→   352→   352→                'ledger_id' => $returnLedgerId,
   353→   353→   353→                'debit' => $amount, 'credit' => 0,
   354→   354→   354→                'entity_type' => 'sales_return', 'entity_id' => $return->id,

   356→   356→   356→            ],
   357→   357→   357→            [
   358→   358→   358→                'ledger_id' => $arLedgerId,
   359→   359→   359→                'debit' => 0, 'credit' => $amount,
   360→   360→   360→                'entity_type' => 'customer', 'entity_id' => $return->customer_id,
   361→   361→   361→                'memo' => 'Return ' . $return->return_code . ' — AR credit',
   362→   362→   362→            ],
   363→   363→   363→        ]);
   364→   364→   364→    }
   365→   365→   365→
   366→   366→   366→    /**
   367→   367→   367→     * Post GL COGS Reversal: Dr Inventory / Cr COGS (at ORIGINAL avg_cost).
   368→   368→   368→     *
   369→   369→   369→     * @return int journal_entry_id

   371→   371→   371→    private function postCogsReversalGL(SalesReturn $return, int $createdBy): int

   376→   376→   376→        $inventoryLedgerId = $this->journalPosting->lookupLedgerByNature('inventory');
   377→   377→   377→        $cogsLedgerId = $this->journalPosting->lookupLedgerByNature('cogs');

   384→   384→   384→        }
   385→   385→   385→
   386→   386→   386→        return $this->journalPosting->createJournalEntry([
   387→   387→   387→            'entry_date' => $return->return_date->format('Y-m-d'),
   388→   388→   388→            'reference_type' => 'sales_return',
   389→   389→   389→            'reference_id' => $return->id,
   390→   390→   390→            'branch_id' => $return->branch_id,

   392→   392→   392→            'source' => 'sales_return',

   394→   394→   394→        ], [

   398→   398→   398→                'entity_type' => 'sales_return', 'entity_id' => $return->id,
   399→   399→   399→                'memo' => 'Return ' . $return->return_code . ' — inventory restored',
   400→   400→   400→            ],
   401→   401→   401→            [
   402→   402→   402→                'ledger_id' => $cogsLedgerId,
   403→   403→   403→                'debit' => 0, 'credit' => $cogsAmount,
   404→   404→   404→                'entity_type' => 'sales_return', 'entity_id' => $return->id,
   405→   405→   405→                'memo' => 'Return ' . $return->return_code . ' — COGS reversal',
   406→   406→   406→            ],
   407→   407→   407→        ]);
   408→   408→   408→    }

   410→   410→   410→    /**
   411→   411→   411→     * Post customer_ledger credit (customer owes less).
   412→   412→   412→     */
   413→   413→   413→    private function postCustomerLedgerCredit(SalesReturn $return, ?int $journalEntryId, int $createdBy): void
   414→   414→   414→    {
   415→   415→   415→        $amount = (float) $return->total_amount;
   416→   416→   416→        if ($amount < 0.01) return;
   417→   417→   417→
   418→   418→   418→        $currentBalance = (float) DB::table('customer_ledger')
   419→   419→   419→            ->where('customer_id', $return->customer_id)
