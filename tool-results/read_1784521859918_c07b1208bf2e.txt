    13→/**
    14→ * Sales Return Service — Phase 8.5.
    15→ *
    16→ * Two-phase: create → confirm → reverse.
    17→ *
    18→ * CRITICAL CORRECTNESS (per avg_cost_rule.md §3):
    19→ *   Stock comes back IN at the ORIGINAL avg_cost from the challan's stock_transaction
    20→ *   (NOT current avg_cost). This ensures:
    21→ *   1. The COGS reversal matches the original COGS exactly.
    22→ *   2. The avg_cost is restored to its pre-sale value.
    23→ *
    24→ * The original_cost is looked up from the stock_transaction for the challan
    25→ * (reference_type='sales_challan', reference_id=challan_id, product_id).
    26→ * This is snapshotted into sales_return_items.original_cost.
    27→ *
    28→ * On confirm (4 operations, all atomic):
    29→ *   1. Stock IN via StockService at ORIGINAL avg_cost (ref='sales_return')
    30→ *      → avg_cost recalculated (IN rule: weighted average with original cost)
    31→ *   2. GL Revenue Reversal: Dr Sales Return / Cr AR (at sales rate)
    32→ *   3. GL COGS Reversal: Dr Inventory / Cr COGS (at original avg_cost)
    33→ *   4. Customer ledger: credit entry (customer owes less)
    34→ */
    35→class SalesReturnService
    36→{
    37→    public function __construct(
    38→        private StockService $stockService,
    39→        private JournalPostingService $journalPosting,
    40→        private SalesAccess $salesAccess,
    41→        private SalesAuditLogger $auditLogger,
    42→        private DamageService $damageService,
    43→        private NotificationService $notifications
    44→    ) {}
    45→
    46→    /**
    47→     * Phase 1: Create a sales return (no stock, no GL).
    48→     * Looks up ORIGINAL avg_cost from the challan's stock_transaction.
    49→     *
    50→     * @param array $data {
    51→     *     sales_invoice_id: int,
    52→     *     return_date: string (Y-m-d),
    53→     *     reason: string|null,
    54→     *     created_by: int,
    55→     *     items: array each { product_id, warehouse_id, qty, rate, sales_invoice_item_id }
    56→     * }
    57→     * @return SalesReturn
    58→     */
    59→    public function createReturn(array $data): SalesReturn
    60→    {
    61→        $invoiceId = (int) ($data['sales_invoice_id'] ?? 0);
    62→        if ($invoiceId <= 0) {
    63→            throw new \InvalidArgumentException('sales_invoice_id is required.');
    64→        }
    65→
    66→        $invoice = DB::table('sales_invoices')->where('id', $invoiceId)->first();
    67→        if (!$invoice) {
    68→            throw new \InvalidArgumentException("Invoice {$invoiceId} not found.");
    69→        }
    70→        if ($invoice->is_reversed) {
    71→            throw new \RuntimeException("Cannot return against a reversed invoice.");
    72→        }
    73→
    74→        // P0-8: Defense-in-depth branch isolation check.
    75→        $this->salesAccess->assertBranchAccessible((int) $invoice->branch_id);
    76→
    77→        // Find the challan for this invoice.
    78→        $challan = DB::table('sales_challans')
    79→            ->where('sales_invoice_id', $invoiceId)
    80→            ->where('is_reversed', false)
    81→            ->first();
    82→        if (!$challan) {
    83→            throw new \RuntimeException("No active challan found for invoice {$invoiceId}. Returns require a completed challan.");
    84→        }
    85→
    86→        $items = $this->validateItems($data['items'], $invoiceId, $challan->id);
    87→        $totalRevenue = collect($items)->sum(fn($i) => $i['qty'] * $i['rate']);
    88→        $totalCogs = collect($items)->sum(fn($i) => $i['qty'] * $i['original_cost']);
    89→
    90→        $returnCode = $this->generateReturnCode();
    91→        $customerId = (int) $invoice->customer_id;
    92→        $branchId = (int) $invoice->branch_id;
    93→
    94→        return DB::transaction(function () use (
    95→            $data, $items, $totalRevenue, $totalCogs, $returnCode, $invoiceId, $customerId, $branchId
    96→        ) {
    97→            $returnId = DB::table('sales_returns')->insertGetId([
    98→                'return_code' => $returnCode,
    99→                'return_date' => $data['return_date'] ?? now()->format('Y-m-d'),
   100→                'sales_invoice_id' => $invoiceId,
   101→                'customer_id' => $customerId,
   102→                'branch_id' => $branchId,
   103→                'total_amount' => round($totalRevenue, 2),
   104→                'cogs_amount' => round($totalCogs, 2),
   105→                'status' => 'created',
   106→                'is_reversed' => false,
   107→                'reason' => $data['reason'] ?? null,
   108→                'created_by' => $data['created_by'] ?? null,
   109→                'created_at' => now(),
   110→                'updated_at' => now(),
   111→            ]);
   112→
   113→            $itemRows = [];
   114→            foreach ($items as $item) {
   115→                $itemRows[] = [
   116→                    'sales_return_id' => $returnId,
   117→                    'sales_invoice_item_id' => $item['sales_invoice_item_id'],
   118→                    'product_id' => $item['product_id'],
   119→                    'warehouse_id' => $item['warehouse_id'],
   120→                    'qty' => $item['qty'],
   121→                    'rate' => $item['rate'],
   122→                    'original_cost' => $item['original_cost'],
   123→                ];
   124→            }
   125→            DB::table('sales_return_items')->insert($itemRows);
   126→
   127→            // P1-3: Audit log — return_created.
   128→            $this->auditLogger->returnCreated(
   129→                $data['created_by'] ?? auth()->id() ?? 0,
   130→                $returnId, $returnCode, $invoiceId, $customerId, $branchId,
   131→                $totalRevenue, $totalCogs
   132→            );
   133→
   134→            // P1-7: Notify admins that a return was created.
   135→            $this->notifications->dispatch(
   136→                'return_created',
   137→                "Return {$returnCode} created for Tk " . number_format($totalRevenue, 2)
   138→                . " against invoice #{$invoiceId}.",
   139→                'sales_return',
   140→                $returnId
   141→            );
   142→
   143→            return SalesReturn::with(['items.product', 'salesInvoice', 'customer', 'branch'])->find($returnId);
   144→        });
   145→    }

   147→    /**
   148→     * Phase 2: Confirm a sales return — stock IN + GL + customer_ledger.
   149→     *
   150→     * @param int $returnId
   151→     * @param int $confirmedBy
   152→     * @return SalesReturn
   153→     * @throws \RuntimeException If not 'created' status, or stock/GL fails.
   154→     */
   155→    public function confirmReturn(int $returnId, int $confirmedBy): SalesReturn
   156→    {
   157→        return DB::transaction(function () use ($returnId, $confirmedBy) {
   158→            $return = SalesReturn::with('items')->lockForUpdate()->find($returnId);
   159→
   160→            if (!$return) {
   161→                throw new \RuntimeException("Return {$returnId} not found.");
   162→            }
   163→            if (!$return->isCreated()) {
   164→                throw new \RuntimeException("Only 'created' returns can be confirmed (current: {$return->status}).");
   165→            }
   166→
   167→            $returnDate = $return->return_date->format('Y-m-d');
   168→
   169→            // 1. Stock IN at ORIGINAL avg_cost (CRITICAL: not current avg_cost).
   170→            foreach ($return->items as $item) {
   171→                $this->stockService->applyTransaction([
   172→                    'warehouse_id' => $item->warehouse_id,
   173→                    'product_id' => $item->product_id,
   174→                    'qty' => (float) $item->qty, // positive = IN
   175→                    'rate' => (float) $item->original_cost, // ORIGINAL avg_cost from challan
   176→                    'reference_type' => 'sales_return',
   177→                    'reference_id' => $return->id,
   178→                    'notes' => 'Sales Return ' . $return->return_code . ' — stock restored at original cost',
   179→                    'transaction_date' => $returnDate,
   180→                    'created_by' => $confirmedBy,
   181→                ]);
   182→            }
   183→
   184→            // 1b. P1-5: Linked damage write-off for 'Damage' condition items.
   185→            // Legacy: SalesReturnModel::createLinkedDamageWriteOff — for each
   186→            // Damage-condition item, create a damage_invoices row + stock OUT
   187→            // (immediately back OUT the damaged goods) + GL Dr damage_loss / Cr inventory.
   188→            $linkedDamageIds = $this->createLinkedDamageWriteOffs($return, $confirmedBy);
   189→
   190→            // 2. GL Revenue Reversal: Dr Sales Return / Cr AR.
   191→            $journalEntryId = $this->postRevenueReversalGL($return, $confirmedBy);
   192→
   193→            // 3. GL COGS Reversal: Dr Inventory / Cr COGS.
   194→            $cogsJournalEntryId = $this->postCogsReversalGL($return, $confirmedBy);
   195→
   196→            // 4. Customer ledger: credit entry (customer owes less).
   197→            $this->postCustomerLedgerCredit($return, $journalEntryId, $confirmedBy);
   198→
   199→            // 5. Update return status.
   200→            DB::table('sales_returns')
   201→                ->where('id', $returnId)
   202→                ->update([
   203→                    'status' => 'confirmed',
   204→                    'journal_entry_id' => $journalEntryId,
   205→                    'cogs_journal_entry_id' => $cogsJournalEntryId,
   206→                    'updated_at' => now(),
   207→                ]);
   208→
   209→            // P1-3: Audit log — return_confirmed.
   210→            $this->auditLogger->returnConfirmed(
   211→                $confirmedBy,
   212→                $returnId, $return->return_code, (int) $return->branch_id,
   213→                (float) $return->total_amount, (float) $return->cogs_amount,
   214→                $journalEntryId
   215→            );
   216→
   217→            // P1-7: Notify warehouse managers + accountants that a return was confirmed.
   218→            $this->notifications->dispatch(
   219→                'return_confirmed',
   220→                "Return {$return->return_code} confirmed — stock restored, GL posted. "
   221→                . "Total: Tk " . number_format((float) $return->total_amount, 2)
   222→                . ", COGS reversed: Tk " . number_format((float) $return->cogs_amount, 2) . ".",
   223→                'sales_return',
   224→                $returnId
   225→            );
   226→
   227→            return SalesReturn::with([
   228→                'items.product', 'salesInvoice', 'customer', 'branch',
   229→                'journalEntry.lines.ledger', 'cogsJournalEntry.lines.ledger',
   230→            ])->find($returnId);
   231→        });
   232→    }

   234→    /**
   235→     * Phase 3: Reverse a confirmed return.
   236→     * Reverses stock + both GL journals + customer_ledger.
   237→     *
   238→     * @param int $returnId
   239→     * @param int $reversedBy
   240→     * @param string $reason
   241→     * @return SalesReturn
   242→     */
   243→    public function reverseReturn(int $returnId, int $reversedBy, string $reason = ''): SalesReturn
   244→    {
   245→        return DB::transaction(function () use ($returnId, $reversedBy, $reason) {
   246→            $return = SalesReturn::with('items')->lockForUpdate()->find($returnId);
   247→
   248→            if (!$return) {
   249→                throw new \RuntimeException("Return {$returnId} not found.");
   250→            }
   251→            if (!$return->isConfirmed()) {
   252→                throw new \RuntimeException("Only confirmed returns can be reversed (current: {$return->status}).");
   253→            }
   254→
   255→            // Reverse both GL journals.
   256→            if ($return->journal_entry_id) {
   257→                $this->journalPosting->reverseJournalEntry(
   258→                    $return->journal_entry_id, $reversedBy,
   259→                    "Return reversed: {$reason}"
   260→                );
   261→            }
   262→            if ($return->cogs_journal_entry_id) {
   263→                $this->journalPosting->reverseJournalEntry(
   264→                    $return->cogs_journal_entry_id, $reversedBy,
   265→                    "Return reversed: {$reason}"
   266→                );
   267→            }
   268→
   269→            // Reverse customer_ledger.
   270→            $this->reverseCustomerLedgerCredit($return, $reversedBy, $reason);
   271→
   272→            // P1-5: Reverse linked damage write-offs FIRST (before return stock reversal).
   273→            // Each linked damage_invoice has its own stock OUT + GL that must be reversed.
   274→            $this->reverseLinkedDamageForReturn($returnId, $reversedBy, $reason);
   275→
   276→            // Reverse each stock movement (stock OUT at original cost — append-only reversal).
   277→            $stockTxs = DB::table('stock_transactions')
   278→                ->where('reference_type', 'sales_return')
   279→                ->where('reference_id', $returnId)
   280→                ->where('is_reversed', false)
   281→                ->get();
   282→
   283→            foreach ($stockTxs as $tx) {
   284→                $this->stockService->reverseTransaction(
   285→                    $tx->id, $reversedBy,
   286→                    "Return reversed: {$reason}"
   287→                );
   288→            }
   289→
   290→            // Mark return as reversed.
   291→            DB::table('sales_returns')
   292→                ->where('id', $returnId)
   293→                ->update([
   294→                    'status' => 'reversed',
   295→                    'is_reversed' => true,
   296→                    'reversed_at' => now(),
   297→                    'reversed_by' => $reversedBy,
   298→                    'reverse_reason' => $reason,
   299→                    'updated_at' => now(),
   300→                ]);
   301→
   302→            // P1-3: Audit log — return_reversed.
   303→            $this->auditLogger->returnReversed(
   304→                $reversedBy,
   305→                $returnId, $return->return_code, (int) $return->branch_id,
   306→                (float) $return->total_amount, $reason
   307→            );
   308→
   309→            // P1-7: Notify accountants + admins that a return was reversed.
   310→            $this->notifications->dispatch(
   311→                'return_reversed',
   312→                "Return {$return->return_code} reversed — stock + GL + ledger rolled back. "
   313→                . "Reason: {$reason}",
   314→                'sales_return',
   315→                $returnId
   316→            );
   317→
   318→            return SalesReturn::find($returnId);
   319→        });
   320→    }

   322→    /**
   323→     * Post GL Revenue Reversal: Dr Sales Return / Cr AR.
   324→     *
   325→     * @return int journal_entry_id
   326→     */
   327→    private function postRevenueReversalGL(SalesReturn $return, int $createdBy): int
   328→    {
   329→        $amount = (float) $return->total_amount;
   330→        if ($amount < 0.01) return 0;
   331→
   332→        $returnLedgerId = $this->journalPosting->lookupLedgerByNature('sales_return');
   333→        $arLedgerId = $this->journalPosting->lookupLedgerByNature('ar');
   334→
   335→        if (!$returnLedgerId) {
   336→            throw new \RuntimeException('Sales Return ledger not found (nature: sales_return).');
   337→        }
   338→        if (!$arLedgerId) {
   339→            throw new \RuntimeException('Accounts Receivable ledger not found (nature: ar).');
   340→        }
   341→
   342→        return $this->journalPosting->createJournalEntry([
   343→            'entry_date' => $return->return_date->format('Y-m-d'),
   344→            'reference_type' => 'sales_return',
   345→            'reference_id' => $return->id,
   346→            'branch_id' => $return->branch_id,
   347→            'description' => 'Sales Return Revenue Reversal ' . $return->return_code,
   348→            'source' => 'sales_return',
   349→            'created_by' => $createdBy,
   350→        ], [
   351→            [
   352→                'ledger_id' => $returnLedgerId,
   353→                'debit' => $amount, 'credit' => 0,
   354→                'entity_type' => 'sales_return', 'entity_id' => $return->id,
   355→                'memo' => 'Return ' . $return->return_code . ' — revenue reversal',
   356→            ],
   357→            [
   358→                'ledger_id' => $arLedgerId,
   359→                'debit' => 0, 'credit' => $amount,
   360→                'entity_type' => 'customer', 'entity_id' => $return->customer_id,
   361→                'memo' => 'Return ' . $return->return_code . ' — AR credit',
   362→            ],
   363→        ]);
   364→    }

   366→    /**
   367→     * Post GL COGS Reversal: Dr Inventory / Cr COGS (at ORIGINAL avg_cost).

   389→            'reference_id' => $return->id,
   390→            'branch_id' => $return->branch_id,
   391→            'description' => 'Sales Return COGS Reversal ' . $return->return_code,
   392→            'source' => 'sales_return',
   393→            'created_by' => $createdBy,
   394→        ], [
   395→            [
   396→                'ledger_id' => $inventoryLedgerId,

   399→                'memo' => 'Return ' . $return->return_code . ' — inventory restored',

   402→                'ledger_id' => $cogsLedgerId,

   410→    /**
   411→     * Post customer_ledger credit (customer owes less).
   412→     */
   413→    private function postCustomerLedgerCredit(SalesReturn $return, ?int $journalEntryId, int $createdBy): void
   414→    {
   415→        $amount = (float) $return->total_amount;
   416→        if ($amount < 0.01) return;
   417→
   418→        $currentBalance = (float) DB::table('customer_ledger')
   419→            ->where('customer_id', $return->customer_id)
   420→            ->orderByDesc('id')
   421→            ->value('balance');
   422→
   423→        $newBalance = $currentBalance - $amount;
   424→
   425→        DB::table('customer_ledger')->insert([
   426→            'customer_id' => $return->customer_id,
   427→            'branch_id' => $return->branch_id,
   428→            'transaction_date' => $return->return_date->format('Y-m-d'),
   429→            'transaction_type' => 'sales_return',
   430→            'reference_type' => 'sales_return',
   431→            'reference_id' => $return->id,
   432→            'debit' => 0,
   433→            'credit' => $amount,
   434→            'balance' => $newBalance,
   435→            'description' => 'Sales Return ' . $return->return_code . ($return->reason ? ' — ' . $return->reason : ''),
   436→            'journal_entry_id' => $journalEntryId,
   437→            'created_by' => $createdBy,
   438→            'created_at' => now(),
   439→        ]);
   440→    }

   442→    /**

   445→    private function reverseCustomerLedgerCredit(SalesReturn $return, int $reversedBy, string $reason): void

   473→    /**
   474→     * Generate atomic return code: SR-YYYYMMDD-NNNN.
   475→     */
   476→    private function generateReturnCode(): string
   477→    {
   478→        $datePart = now()->format('Ymd');
   479→        $periodKey = now()->format('Y-m');
   480→        $docType = 'sales_return';
   481→
   482→        return DB::transaction(function () use ($docType, $periodKey, $datePart) {
   483→            $seqRow = DB::table('document_sequences')
   484→                ->where('doc_type', $docType)
   485→                ->where('branch_id', 0)
   486→                ->where('period_key', $periodKey)
   487→                ->lockForUpdate()
   488→                ->first();
   489→
   490→            $nextNumber = $seqRow ? ((int) $seqRow->last_number + 1) : 1;
   491→
   492→            if ($seqRow) {
   493→                DB::table('document_sequences')->where('id', $seqRow->id)
   494→                    ->update(['last_number' => $nextNumber, 'updated_at' => now()]);
   495→            } else {
   496→                DB::table('document_sequences')->insert([
   497→                    'doc_type' => $docType, 'branch_id' => 0,
   498→                    'period_key' => $periodKey, 'last_number' => $nextNumber,
   499→                    'updated_at' => now(),
   500→                ]);
   501→            }
   502→
   503→            return "SR-{$datePart}-" . str_pad((string) $nextNumber, 4, '0', STR_PAD_LEFT);
   504→        });
   505→    }

   507→    /**
   508→     * Validate items — look up ORIGINAL avg_cost from the challan's stock_transaction.
   509→     *
   510→     * @param array $items
   511→     * @param int $invoiceId
   512→     * @param int $challanId
   513→     * @return array
   514→     */
   515→    private function validateItems(array $items, int $invoiceId, int $challanId): array
   516→    {
   517→        $validated = [];
   518→
   519→        foreach ($items as $item) {
   520→            $productId = (int) ($item['product_id'] ?? 0);
   521→            $qty = (float) ($item['qty'] ?? 0);
   522→            $warehouseId = (int) ($item['warehouse_id'] ?? 0);
   523→            $invoiceItemId = (int) ($item['sales_invoice_item_id'] ?? 0);
   524→
   525→            if ($productId <= 0 || $qty <= 0 || $warehouseId <= 0) continue;
   526→
   527→            // Check returnable_qty (invoice item qty - already returned).
   528→            if ($invoiceItemId > 0) {
   529→                $invoiceItem = DB::table('sales_invoice_items')->where('id', $invoiceItemId)->first();
   530→                if ($invoiceItem) {
   531→                    $alreadyReturned = (float) DB::table('sales_return_items as sri')
   532→                        ->join('sales_returns as sr', 'sr.id', '=', 'sri.sales_return_id')
   533→                        ->where('sri.sales_invoice_item_id', $invoiceItemId)
   534→                        ->whereIn('sr.status', ['created', 'confirmed'])
   535→                        ->where('sr.is_reversed', false)
   536→                        ->sum('sri.qty');
   537→
   538→                    $returnable = (float) $invoiceItem->qty - $alreadyReturned;
   539→                    if ($qty > $returnable + 0.0001) {
   540→                        throw new \RuntimeException(
   541→                            "Return qty {$qty} exceeds returnable qty {$returnable} for product {$productId}."
   542→                        );
   543→                    }
   544→                }
   545→            }
   546→
   547→            // CRITICAL: Look up ORIGINAL avg_cost from the challan's stock_transaction.
   548→            $challanStockTx = DB::table('stock_transactions')
   549→                ->where('reference_type', 'sales_challan')
   550→                ->where('reference_id', $challanId)
   551→                ->where('product_id', $productId)
   552→                ->where('warehouse_id', $warehouseId)
   553→                ->where('is_reversed', false)
   554→                ->first();
   555→
   556→            $originalCost = $challanStockTx ? (float) $challanStockTx->rate : 0;
   557→            if ($originalCost <= 0) {
   558→                throw new \RuntimeException(
   559→                    "Cannot determine original avg_cost for product {$productId} from challan. "
   560→                    . "The challan stock transaction may be missing or reversed."
   561→                );
   562→            }
   563→
   564→            // Rate = sales rate from the invoice item (for revenue reversal).
   565→            $rate = (float) ($item['rate'] ?? 0);
   566→            if ($rate <= 0 && $invoiceItemId > 0) {
   567→                $invoiceItem = DB::table('sales_invoice_items')->where('id', $invoiceItemId)->first();
   568→                $rate = $invoiceItem ? (float) $invoiceItem->rate : 0;
   569→            }
   570→
   571→            $validated[] = [
   572→                'product_id' => $productId,
   573→                'warehouse_id' => $warehouseId,
   574→                'qty' => $qty,
   575→                'rate' => $rate, // sales rate (for revenue reversal)
   576→                'original_cost' => $originalCost, // ORIGINAL avg_cost (for COGS + stock IN)
   577→                'sales_invoice_item_id' => $invoiceItemId > 0 ? $invoiceItemId : null,
   578→            ];
   579→        }
   580→
   581→        if (empty($validated)) {
   582→            throw new \InvalidArgumentException('At least one valid item is required.');
   583→        }
   584→
   585→        return $validated;
   586→    }

   588→    /**
   589→     * P1-5: Create linked damage write-offs for 'Damage' condition return items.
   590→     *
   591→     * Mirrors legacy SalesReturnModel::createLinkedDamageWriteOff:1599.
   592→     * For each Damage-condition item:
   593→     *   1. Create a damage_invoices row (linked via sales_return_id)
   594→     *   2. Create damage_invoice_items
   595→     *   3. Stock OUT via DamageService (immediately back OUT the damaged goods)
   596→     *   4. GL: Dr damage_loss / Cr inventory (via DamageService::confirmDamage)
   597→     *   5. Link sales_return_items.damage_invoice_id → damage_invoices.id
   598→     *
   599→     * Groups damage items by warehouse (one damage_invoice per warehouse).
   600→     *
   601→     * @param SalesReturn $return
   602→     * @param int $confirmedBy
   603→     * @return array List of created damage_invoice IDs
   604→     */
   605→    private function createLinkedDamageWriteOffs(SalesReturn $return, int $confirmedBy): array
   606→    {
   607→        // Group Damage-condition items by warehouse.
   608→        $damageByWarehouse = [];
   609→        foreach ($return->items as $item) {
   610→            if (($item->condition_state ?? 'Good') !== 'Damage') {
   611→                continue;
   612→            }
   613→            $wid = (int) $item->warehouse_id;
   614→            if ($wid <= 0) {
   615→                continue;
   616→            }
   617→            $damageByWarehouse[$wid][] = $item;
   618→        }
   619→
   620→        if (empty($damageByWarehouse)) {
   621→            return []; // No damage items — nothing to write off.
   622→        }
   623→
   624→        $linkedDamageIds = [];
   625→        $returnDate = $return->return_date->format('Y-m-d');
   626→
   627→        foreach ($damageByWarehouse as $warehouseId => $damageItems) {
   628→            // Build items array for DamageService::createDamage.
   629→            $items = [];
   630→            foreach ($damageItems as $item) {
   631→                $items[] = [
   632→                    'product_id' => (int) $item->product_id,
   633→                    'qty' => (float) $item->qty,
   634→                    'rate' => (float) $item->original_cost, // use original_cost (same as return stock IN)
   635→                ];
   636→            }
   637→
   638→            // Create the damage invoice (draft) via DamageService.
   639→            $damage = $this->damageService->createDamage([
   640→                'warehouse_id' => $warehouseId,
   641→                'damage_date' => $returnDate,
   642→                'reason' => 'Auto write-off for damaged sales return #' . $return->return_code,
   643→                'created_by' => $confirmedBy,
   644→                'items' => $items,
   645→            ]);
   646→
   647→            // Link damage_invoices.sales_return_id → this return.
   648→            DB::table('damage_invoices')
   649→                ->where('id', $damage->id)
   650→                ->update(['sales_return_id' => $return->id]);
   651→
   652→            // Confirm the damage (stock OUT + GL Dr damage_loss / Cr inventory).
   653→            $damage = $this->damageService->confirmDamage($damage->id, $confirmedBy);
   654→
   655→            // Link each return item to the damage invoice.
   656→            foreach ($damageItems as $item) {
   657→                DB::table('sales_return_items')
   658→                    ->where('id', $item->id)
   659→                    ->update(['damage_invoice_id' => $damage->id]);
   660→            }
   661→
   662→            $linkedDamageIds[] = $damage->id;
   663→        }
   664→
   665→        return $linkedDamageIds;
   666→    }

   668→    /**
   669→     * P1-5: Reverse linked damage write-offs when a return is reversed.
   670→     *
   671→     * Mirrors legacy SalesReturnModel::reverseLinkedDamageForReturn.
   672→     * Finds all damage_invoices linked to this return (via sales_return_id)
   673→     * and cancels each one via DamageService::cancelDamage (reverses stock + GL).
   674→     *
   675→     * Must be called BEFORE reversing the return's own stock movements
   676→     * (so the damage stock OUT is reversed first, then the return stock IN).
   677→     *
   678→     * @param int $returnId
   679→     * @param int $reversedBy
   680→     * @param string $reason
   681→     */
   682→    private function reverseLinkedDamageForReturn(int $returnId, int $reversedBy, string $reason): void
   683→    {
   684→        $linkedDamages = DB::table('damage_invoices')
   685→            ->where('sales_return_id', $returnId)
   686→            ->where('is_reversed', false)
   687→            ->get();
   688→
   689→        foreach ($linkedDamages as $damage) {
   690→            // Cancel the damage (reverses stock OUT + GL Dr damage_loss / Cr inventory).
   691→            $this->damageService->cancelDamage($damage->id, $reversedBy, "Return reversed: {$reason}");
   692→
   693→            // Clear the link on sales_return_items (set damage_invoice_id = null).
   694→            DB::table('sales_return_items')
   695→                ->where('damage_invoice_id', $damage->id)
   696→                ->update(['damage_invoice_id' => null]);
   697→        }
   698→    }
   699→}
