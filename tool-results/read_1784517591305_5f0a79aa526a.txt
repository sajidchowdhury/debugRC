   109→   109→     * Load invoice for return with clear errors (branch + challan rules).
   110→   110→     *
   111→   111→     * @return array{status:string,message?:string,invoice?:array}
   112→   112→     */
   113→   113→    public function resolveInvoiceForReturn(string $invoiceCode): array
   114→   114→    {
   115→   115→        $invoiceCode = trim($invoiceCode);
   116→   116→        if ($invoiceCode === '') {
   117→   117→            return ['status' => 'error', 'message' => 'Invoice code is required.'];
   118→   118→        }
   119→   119→
   120→   120→        $branchId = self::sessionBranchId();
   121→   121→        $this->db->query("
   122→   122→            SELECT si.*, c.shop_name, c.mobile, c.address
   123→   123→            FROM sales_invoices si
   124→   124→            JOIN customers c ON si.customer_id = c.id
   125→   125→            WHERE si.invoice_code = :code
   126→   126→              AND si.is_reversed = 0
   127→   127→        ");
   128→   128→        $this->db->bind(':code', $invoiceCode);
   129→   129→        $row = $this->db->single();
   130→   130→
   131→   131→        if (!$row) {
   132→   132→            return ['status' => 'error', 'message' => 'Invoice not found.'];
   133→   133→        }
   134→   134→
   135→   135→        if ((int)$row['branch_id'] !== $branchId) {
   136→   136→            return ['status' => 'error', 'message' => 'This invoice belongs to another branch. You can only return invoices from your branch.'];
   137→   137→        }
   138→   138→
   139→   139→        $status = (string)($row['status'] ?? '');
   140→   140→        if ($status === 'cancelled' || $status === 'reversed') {
   141→   141→            return ['status' => 'error', 'message' => 'This invoice is cancelled or reversed.'];
   142→   142→        }
   143→   143→
   144→   144→        if ($status !== self::RETURNABLE_INVOICE_STATUS) {
   145→   145→            $label = str_replace('_', ' ', $status);
   146→   146→            return [
   147→   147→                'status'  => 'error',
   148→   148→                'message' => 'Return is allowed only after challan is completed. This invoice status: ' . $label . '.',
   149→   149→            ];
   150→   150→        }
   151→   151→
   152→   152→        $invoice = $row;
   153→   153→        $this->db->query("
   154→   154→            SELECT sii.*, p.product_name, p.unit,
   155→   155→                   COALESCE(sii.qty - (SELECT COALESCE(SUM(sri.return_qty),0)
   156→   156→                                      FROM sales_return_items sri
   157→   157→                                      JOIN sales_returns sr ON sr.id = sri.sales_return_id
   158→   158→                                      WHERE sri.sales_invoice_item_id = sii.id
   159→   159→                                        AND sr.status != 'reversed'
   160→   160→                                        AND COALESCE(sr.is_reversed, 0) = 0), 0) as returnable_qty
   161→   161→            FROM sales_invoice_items sii
   162→   162→            JOIN products p ON sii.product_id = p.id
   163→   163→            WHERE sii.sales_invoice_id = :invoice_id
   164→   164→        ");
   165→   165→        $this->db->bind(':invoice_id', $invoice['id']);
   166→   166→        $invoice['items'] = $this->db->resultSet();
   167→   167→
   168→   168→        $hasReturnable = false;
   169→   169→        foreach ($invoice['items'] as $line) {
   170→   170→            if ((float)($line['returnable_qty'] ?? 0) > 0.0001) {
   171→   171→                $hasReturnable = true;
   172→   172→                break;
   173→   173→            }
   174→   174→        }
   175→   175→        if (!$hasReturnable) {
   176→   176→            return ['status' => 'error', 'message' => 'No returnable quantity left on this invoice.'];
   177→   177→        }
   178→   178→
   179→   179→        return ['status' => 'success', 'invoice' => $invoice];
   180→   180→    }

   195→   195→     * Max returnable qty for an invoice line (excludes other non-reversed returns).
   196→   196→     */
   197→   197→    public function getMaxReturnableQty(int $salesInvoiceItemId): float
   198→   198→    {

   200→   200→            return 0.0;
   201→   201→        }
   202→   202→

   204→   204→            SELECT GREATEST(0,
   205→   205→                COALESCE(sii.qty, 0) - COALESCE((
   206→   206→                    SELECT SUM(sri.return_qty)
   207→   207→                    FROM sales_return_items sri
   208→   208→                    JOIN sales_returns sr ON sr.id = sri.sales_return_id
   209→   209→                    WHERE sri.sales_invoice_item_id = sii.id
   210→   210→                      AND sr.status != 'reversed'
   211→   211→                      AND COALESCE(sr.is_reversed, 0) = 0
   212→   212→                ), 0)
   213→   213→            ) AS returnable_qty
   214→   214→            FROM sales_invoice_items sii
   215→   215→            WHERE sii.id = :id
   216→   216→        ");
   217→   217→        $this->db->bind(':id', $salesInvoiceItemId);
   218→   218→        $row = $this->db->single();
   219→   219→
   220→   220→        return (float)($row['returnable_qty'] ?? 0);
   221→   221→    }

   223→   223→    // PHASE 1: Create Return (Sales Manager) — ledger credit posts on warehouse confirm (Phase 2)
   224→   224→public function createReturn($data, $items) {
   225→   225→    $this->lastError = '';
   226→   226→    $this->db->beginTransaction();
   227→   227→    try {
   228→   228→        $invoiceId = (int)($data['sales_invoice_id'] ?? 0);
   229→   229→        $this->db->query("SELECT branch_id, status FROM sales_invoices WHERE id = :id AND is_reversed = 0");
   230→   230→        $this->db->bind(':id', $invoiceId);
   231→   231→        $inv = $this->db->single();
   232→   232→        if (!$inv) {
   233→   233→            throw new Exception('Invoice not found.');
   234→   234→        }
   235→   235→        $this->assertInvoiceAccessible((int)$inv['branch_id']);
   236→   236→        if (($inv['status'] ?? '') !== self::RETURNABLE_INVOICE_STATUS) {
   237→   237→            throw new Exception('Return is allowed only after challan is completed for this invoice.');
   238→   238→        }
   239→   239→
   240→   240→        $return_code = $this->generateSalesReturnCode();
   241→   241→
   242→   242→        $computedTotal = 0.0;
   243→   243→        $validatedItems = [];
   244→   244→
   245→   245→        foreach ($items as $item) {
   246→   246→            $return_qty = (float)($item['return_qty'] ?? 0);
   247→   247→            if ($return_qty <= 0) {
   248→   248→                continue;
   249→   249→            }
   250→   250→
   251→   251→            $siiId = (int)($item['sales_invoice_item_id'] ?? 0);
   252→   252→            $maxReturnable = $this->getMaxReturnableQty($siiId);
   253→   253→
   254→   254→            if ($return_qty > $maxReturnable + 0.0001) {
   255→   255→                throw new Exception(
   256→   256→                    "Return quantity exceeds returnable amount for invoice line #{$siiId}. "
   257→   257→                    . "Max: " . number_format($maxReturnable, 2)
   258→   258→                );
   259→   259→            }
   260→   260→
   261→   261→            $rate = (float)($item['rate'] ?? $item['sales_rate'] ?? 0);
   262→   262→            $amount = $return_qty * $rate;
   263→   263→            $computedTotal += $amount;
   264→   264→
   265→   265→            $validatedItems[] = [
   266→   266→                'sales_invoice_item_id' => $siiId,
   267→   267→                'product_id'            => (int)($item['product_id'] ?? 0),
   268→   268→                'return_qty'            => $return_qty,
   269→   269→                'rate'                  => $rate,
   270→   270→                'amount'                => $amount,
   271→   271→                'condition'             => $item['condition'] ?? 'Good',
   272→   272→            ];
   273→   273→        }
   274→   274→
   275→   275→        if (empty($validatedItems)) {
   276→   276→            throw new Exception('No valid return quantities entered.');
   277→   277→        }
   278→   278→
   279→   279→        // Insert main return
   280→   280→        $this->db->query("
   281→   281→            INSERT INTO sales_returns 
   282→   282→            (return_code, sales_invoice_id, customer_id, return_date, 
   283→   283→             total_amount, reason, status, created_by)
   284→   284→            VALUES (:code, :invoice_id, :customer_id, :return_date, 
   285→   285→                    :total_amount, :reason, 'pending', :created_by)
   286→   286→        ");
   287→   287→
   288→   288→        $this->db->bind(':code', $return_code);
   289→   289→        $this->db->bind(':invoice_id', $data['sales_invoice_id']);
   290→   290→        $this->db->bind(':customer_id', $data['customer_id']);
   291→   291→        $this->db->bind(':return_date', $data['return_date'] ?? date('Y-m-d'));
   292→   292→        $this->db->bind(':total_amount', $computedTotal);
   293→   293→        $this->db->bind(':reason', $data['reason'] ?? '');
   294→   294→        $this->db->bind(':created_by', $_SESSION['user_id'] ?? 1);
   295→   295→        $this->db->execute();
   296→   296→
   297→   297→        $return_id = $this->db->lastInsertId();
   298→   298→
   299→   299→        foreach ($validatedItems as $item) {
   300→   300→            $this->db->query("
   301→   301→                INSERT INTO sales_return_items 
   302→   302→                (sales_return_id, sales_invoice_item_id, product_id, 
   303→   303→                 return_qty, rate, amount, condition)
   304→   304→                VALUES (:rid, :sii, :pid, :rqty, :rate, :amt, :cond)
   305→   305→            ");
   306→   306→
   307→   307→            $this->db->bind(':rid', $return_id);
   308→   308→            $this->db->bind(':sii', $item['sales_invoice_item_id']);
   309→   309→            $this->db->bind(':pid', $item['product_id']);
   310→   310→            $this->db->bind(':rqty', $item['return_qty']);
   311→   311→            $this->db->bind(':rate', $item['rate']);
   312→   312→            $this->db->bind(':amt', $item['amount']);
   313→   313→            $this->db->bind(':cond', $item['condition']);
   314→   314→            $this->db->execute();
   315→   315→        }
   316→   316→
   317→   317→        $this->db->commit();
   318→   318→        return $return_id;
   319→   319→
   320→   320→    } catch (Exception $e) {
   321→   321→        $this->db->rollback();
   322→   322→        $this->lastError = $e->getMessage();
   323→   323→        error_log("Create Sales Return Error: " . $e->getMessage());
   324→   324→        return false;
   325→   325→    }
   326→   326→}

   428→   428→public function confirmReturn($return_id, $items) {
   429→   429→    $this->lastError = '';
   430→   430→    $return_id = (int)$return_id;
   431→   431→
   432→   432→    if ($return_id <= 0) {
   433→   433→        $this->lastError = 'Invalid Return ID';
   434→   434→        return ['success' => false, 'message' => $this->lastError];
   435→   435→    }
   436→   436→
   437→   437→    $this->db->beginTransaction();
   438→   438→    try {
   439→   439→        $this->db->query("
   440→   440→            SELECT * FROM sales_returns
   441→   441→            WHERE id = :id AND status = 'pending' AND COALESCE(is_reversed, 0) = 0
   442→   442→            FOR UPDATE
   443→   443→        ");
   444→   444→        $this->db->bind(':id', $return_id);
   445→   445→        $returnHeader = $this->db->single();
   446→   446→
   447→   447→        if (!$returnHeader) {
   448→   448→            throw new Exception('Return not found or already processed.');
   449→   449→        }
   450→   450→
   451→   451→        $this->db->query("SELECT branch_id FROM sales_invoices WHERE id = :id");
   452→   452→        $this->db->bind(':id', (int)$returnHeader['sales_invoice_id']);
   453→   453→        $invBranch = $this->db->single();
   454→   454→        $branchId = (int)($invBranch['branch_id'] ?? self::sessionBranchId());
   455→   455→        $this->assertInvoiceAccessible($branchId);
   456→   456→

   865→   865→    protected function estimateReturnCogsFromStockLog(int $returnId): float
   866→   866→    {

   870→   870→            if ($qty <= 0.0001) {
   871→   871→                continue;
   872→   872→            }
   873→   873→            $rate = (float)($movement['rate'] ?? 0);
   874→   874→            if ($rate <= 0) {
   875→   875→                $rate = $this->stock->getWarehouseAvgCost(
   876→   876→                    (int)($movement['warehouse_id'] ?? 0),
   877→   877→                    (int)($movement['product_id'] ?? 0)
   878→   878→                );
   879→   879→            }
   880→   880→            $total += $qty * $rate;
   881→   881→        }
   882→   882→
   883→   883→        return round($total, 2);
   884→   884→    }
