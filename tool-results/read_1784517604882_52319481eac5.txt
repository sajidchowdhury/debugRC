    12→function renderCustomerChips($from, $to, $branches, $salesmen, $cats, $minRev, $custTypes) {
    13→    $tags = [];
    14→    if ($from && $to) {
    15→        $tags[] = '<span class="filter-tag"><i class="fas fa-calendar me-1"></i> ' . htmlspecialchars($from) . ' → ' . htmlspecialchars($to) . '</span>';
    16→    }
    17→    if (!empty($branches)) {
    18→        $cnt = count($branches);
    19→        $tags[] = '<span class="filter-tag"><i class="fas fa-code-branch me-1"></i> ' . $cnt . ' branch' . ($cnt > 1 ? 'es' : '') . '</span>';
    20→    }
    21→    if (!empty($salesmen)) {
    22→        $cnt = count($salesmen);
    23→        $tags[] = '<span class="filter-tag"><i class="fas fa-user-tie me-1"></i> ' . $cnt . ' rep' . ($cnt > 1 ? 's' : '') . '</span>';
    24→    }
    25→    if (!empty($cats)) {
    26→        $cnt = count($cats);
    27→        $tags[] = '<span class="filter-tag"><i class="fas fa-tags me-1"></i> ' . $cnt . ' categor' . ($cnt > 1 ? 'ies' : 'y') . '</span>';
    28→    }
    29→    if ($minRev > 0) {
    30→        $tags[] = '<span class="filter-tag"><i class="fas fa-dollar-sign me-1"></i> Min Rev ' . number_format($minRev) . '</span>';
    31→    }
    32→    if (!empty($custTypes)) {
    33→        $tags[] = '<span class="filter-tag"><i class="fas fa-users me-1"></i> ' . implode(', ', (array)$custTypes) . '</span>';
    34→    }
    35→    if (empty($tags)) {
    36→        return '<span class="text-muted small">No filters active — open Filters to slice customer intelligence</span>';
    37→    }
    38→    $base = defined('BASE_URL') ? BASE_URL : '';
    39→    return implode(' ', $tags) . ' <button type="button" class="btn btn-link btn-sm p-0 ms-auto" onclick="window.location.href=\'' . $base . 'Report/customerPerformance\'">Clear all</button>';
    40→}
