    10→// Helper to render active filter chips (adapted)
    11→function renderRevenueChips($from, $to, $branches, $salesmen, $cats, $comparison) {
    12→    $tags = [];
    13→    if ($from && $to) {
    14→        $tags[] = '<span class="filter-tag"><i class="fas fa-calendar me-1"></i> ' . htmlspecialchars($from) . ' → ' . htmlspecialchars($to) . '</span>';
    15→    }
    16→    if (!empty($branches)) {
    17→        $cnt = count($branches);
    18→        $tags[] = '<span class="filter-tag"><i class="fas fa-code-branch me-1"></i> ' . $cnt . ' branch' . ($cnt > 1 ? 'es' : '') . '</span>';
    19→    }
    20→    if (!empty($salesmen)) {
    21→        $cnt = count($salesmen);
    22→        $tags[] = '<span class="filter-tag"><i class="fas fa-user-tie me-1"></i> ' . $cnt . ' rep' . ($cnt > 1 ? 's' : '') . '</span>';
    23→    }
    24→    if (!empty($cats)) {
    25→        $cnt = count($cats);
    26→        $tags[] = '<span class="filter-tag"><i class="fas fa-tags me-1"></i> ' . $cnt . ' categor' . ($cnt > 1 ? 'ies' : 'y') . '</span>';
    27→    }
    28→    $tags[] = '<span class="filter-tag"><i class="fas fa-balance-scale me-1"></i> vs ' . ucfirst(str_replace('_', ' ', $comparison)) . '</span>';
    29→    if (empty($tags)) {
    30→        return '<span class="text-muted small">No filters active — open Filters to slice the cockpit view</span>';
    31→    }
    32→    $base = defined('BASE_URL') ? BASE_URL : '';
    33→    return implode(' ', $tags) . ' <button type="button" class="btn btn-link btn-sm p-0 ms-auto" onclick="window.location.href=\'' . $base . 'Report/revenueOverview\'">Clear all</button>';
    34→}

    36→function renderKpiCard($label, $value, $sub = '', $trend = null, $color = 'primary', $sparkData = null) {
    37→    $trendHtml = '';
    38→    if ($trend !== null) {
    39→        $arrow = $trend >= 0 ? '↑' : '↓';
    40→        $tclass = $trend >= 0 ? 'text-success' : 'text-danger';
    41→        $trendHtml = '<div class="' . $tclass . ' small fw-semibold">' . $arrow . ' ' . abs($trend) . '%</div>';
    42→    }
    43→    $spark = '';
    44→    if ($sparkData) {
    45→        $spark = '<div class="mt-1" style="height:28px"><canvas class="sparkline" data-values="' . implode(',', $sparkData) . '"></canvas></div>';
    46→    }
    47→    return '
    48→    <div class="col-6 col-md-3 col-lg-2">
    49→        <div class="card border-0 shadow-sm h-100" style="border-radius:12px;">
    50→            <div class="card-body p-2">
    51→                <div class="text-muted small fw-semibold text-uppercase" style="letter-spacing:0.5px;">' . htmlspecialchars($label) . '</div>
    52→                <div class="display-6 fw-bold text-' . $color . ' mt-1" style="line-height:1.05;">' . $value . '</div>
    53→                ' . ($sub ? '<div class="small text-muted">' . $sub . '</div>' : '') . '
    54→                ' . $trendHtml . '
    55→                ' . $spark . '
    56→            </div>
    57→        </div>
    58→    </div>';
    59→}
