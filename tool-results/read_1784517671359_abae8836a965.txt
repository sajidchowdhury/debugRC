    38→public function createTransaction($post) {
    39→    $this->db->beginTransaction();
    40→    try {
    41→        $branch_id = self::sessionBranchId() ?: (int)($_SESSION['branch_id'] ?? 0);
    42→        if ($branch_id <= 0) {
    43→            $branch_id = (int)($_SESSION['branch_id'] ?? 1);
    44→        }
    45→        $user_id = (int)($_SESSION['user_id'] ?? 1);
    46→
    47→        $type   = $post['transaction_type'] ?? 'receive';
    48→        $amount = round((float)($post['amount'] ?? 0), 2);
    49→        $customer_id = (int)($post['customer_id'] ?? 0);
    50→        $mode = strtolower((string)($post['mode'] ?? 'cash'));
    51→
    52→        if ($amount <= 0 || $customer_id <= 0) {
    53→            throw new Exception('Invalid amount or customer');
    54→        }
    55→
    56→        $allowedTypes = ['receive', 'payment', 'discount', 'write_off'];
    57→        if (!in_array($type, $allowedTypes, true)) {
    58→            throw new Exception('Invalid transaction type');
    59→        }
    60→
    61→        if ($mode === 'bank' && empty($post['bank_id'])) {
    62→            throw new Exception('Select a bank account for bank mode');
    63→        }
    64→
    65→        $customer = $this->Get_Customer_By_Id($customer_id);
    66→        if (!$customer || empty($customer['is_active'])) {
    67→            throw new Exception('Customer not found or inactive');
    68→        }
    69→
    70→        if (in_array($type, ['discount', 'write_off'], true)) {
    71→            $due = $this->getCustomerDue($customer_id);
    72→            if ($amount > $due + 0.01) {
    73→                throw new Exception(
    74→                    'Amount cannot exceed customer due (Tk ' . number_format($due, 2) . ')'
    75→                );
    76→            }
    77→        }
    78→
    79→        $payment_code = $this->generateCustomerPaymentCode($branch_id);
    80→
    81→        // === 1. INSERT MAIN PAYMENT RECORD ===
    82→        $this->db->query("
    83→            INSERT INTO customer_payments 
    84→            (payment_code, payment_date, customer_id, amount, payment_mode, 
    85→             bank_id, remarks, created_by, branch_id, collected_by, transaction_type)
    86→            VALUES 
    87→            (:code, :date, :cid, :amt, :mode, :bank, :rem, :uid, :bid, :col, :type)
    88→        ");
    89→
    90→        $this->db->bind(':code', $payment_code);
    91→        $this->db->bind(':date', $post['transaction_date'] ?? date('Y-m-d'));
    92→        $this->db->bind(':cid', $customer_id);
    93→        $this->db->bind(':amt', $amount);
    94→        $this->db->bind(':mode', $post['mode'] ?? 'cash');
    95→        $this->db->bind(':bank', $post['bank_id'] ?? null);
    96→        $this->db->bind(':rem', $post['narration'] ?? '');
    97→        $this->db->bind(':uid', $user_id);
    98→        $this->db->bind(':bid', $branch_id);
    99→        $this->db->bind(':col', $post['collected_by'] ?? $user_id);
   100→        $this->db->bind(':type', $type);
   101→        $this->db->execute();
   102→
   103→        $payment_id = $this->db->lastInsertId();
   104→
   105→        // === 2. CUSTOMER LEDGER ===
   106→        $this->insertCustomerLedger($payment_id, $post, $amount, $branch_id, $user_id, $type);
   107→
   108→        // === 3. GL (all transaction types) ===
   109→        $paymentPayload = [
   110→            'payment_code'  => $payment_code,
   111→            'payment_date'  => $post['transaction_date'] ?? date('Y-m-d'),
   112→            'customer_id'   => $customer_id,
   113→            'amount'        => $amount,
   114→            'payment_mode'  => $post['mode'] ?? 'cash',
   115→            'bank_id'       => !empty($post['bank_id']) ? (int)$post['bank_id'] : null,
   116→            'branch_id'     => $branch_id,
   117→        ];
   118→
   119→        require_once __DIR__ . '/../services/Accounting/JournalPostingService.php';
   120→        $journalService = new JournalPostingService($this->db);
   121→        $journalResult = $journalService->postCustomerTransactionJournal($payment_id, $paymentPayload, $type);
   122→        if (($journalResult['status'] ?? '') === 'error') {
   123→            throw new Exception('Journal posting failed: ' . ($journalResult['message'] ?? 'unknown'));
   124→        }
   125→
   126→        $journalEntryId = null;
   127→        if (!empty($journalResult['journal_entry_id'])) {
   128→            $journalEntryId = (int)$journalResult['journal_entry_id'];
   129→            $this->setPaymentJournalEntryId($payment_id, $journalEntryId);
   130→        }
   131→
   132→        // === 5. Cash book (banks.balance) for bank mode receive/refund ===
   133→        $this->syncBankBookBalance(
   134→            !empty($post['bank_id']) ? (int)$post['bank_id'] : null,
   135→            $amount,
   136→            $type,
   137→            $post['mode'] ?? 'cash',
   138→            false
   139→        );
   140→
   141→        if ($type === 'receive') {
   142→            require_once __DIR__ . '/../services/Branch/BranchIntercompanyService.php';
   143→            $mode = strtolower((string)($post['mode'] ?? 'cash'));
   144→            $bankId = !empty($post['bank_id']) ? (int)$post['bank_id'] : null;
   145→            $paymentMode = ($mode === 'bank' || $bankId > 0) ? 'bank' : 'cash';
   146→            $settleResult = (new BranchIntercompanyService($this->db))->settleFromCustomerPayment($payment_id, [
   147→                'payment_code'  => $payment_code,
   148→                'payment_date'  => $post['transaction_date'] ?? date('Y-m-d'),
   149→                'amount'        => $amount,
   150→                'payment_mode'  => $paymentMode,
   151→                'bank_id'       => $bankId,
   152→                'branch_id'     => $branch_id,
   153→            ]);
   154→            if (($settleResult['status'] ?? '') === 'error') {
   155→                throw new Exception('Branch demand settlement failed: ' . ($settleResult['message'] ?? ''));
   156→            }
   157→        }
   158→
   159→        $this->db->commit();
   160→
   161→        return [
   162→            'status' => 'success',
   163→            'message' => 'Payment recorded — ledger and GL posted.',
   164→            'payment_code' => $payment_code,
   165→            'payment_id' => $payment_id,
   166→            'journal_entry_id' => $journalEntryId,
   167→            'total_amount' => $amount,
   168→        ];
   169→
   170→    } catch (Throwable $e) {
   171→        if ($this->db->inTransaction()) {
   172→            $this->db->rollback();
   173→        }
   174→        error_log('Customer Transaction Error: ' . $e->getMessage());
   175→        return ['status' => 'error', 'message' => $e->getMessage()];
   176→    }

   190→    /**
   191→     * Debit/credit sides for customer_ledger (positive running_balance = amount owed by customer).
   192→     * - receive/discount/write_off: credit (reduce AR/due)
   193→     * - payment (refund/advance out): debit (increase AR/due, matches Dr AR in GL)
   194→     */
   195→    private function getCustomerLedgerSides(string $type, float $amount): array
   196→    {
   197→        switch ($type) {
   198→            case 'receive':
   199→            case 'discount':
   200→            case 'write_off':
   201→                return ['debit' => 0.0, 'credit' => $amount];
   202→            case 'payment':
   203→                return ['debit' => $amount, 'credit' => 0.0];
   204→            default:
   205→                return ['debit' => 0.0, 'credit' => $amount];
   206→        }
   207→    }

   217→    /**
   218→     * Update banks.balance for bank-mode receive (in) or refund (out).
   219→     *
   220→     * @param bool $undo When true, reverses the balance movement (used on payment reversal).
   221→     */
   222→    private function syncBankBookBalance(?int $bankId, float $amount, string $transactionType, string $paymentMode, bool $undo = false): void
   223→    {
   224→        $paymentMode = strtolower(trim($paymentMode));
   225→        if ($paymentMode !== 'bank' || !$bankId || $amount <= 0) {
   226→            return;
   227→        }
   228→
   229→        if (!in_array($transactionType, ['receive', 'payment'], true)) {
   230→            return;
   231→        }
   232→
   233→        $bankModel = new BankModel($this->db);
   234→        $increaseBank = ($transactionType === 'receive');
   235→        if ($undo) {
   236→            $increaseBank = !$increaseBank;
   237→        }
   238→
   239→        $bankModel->updateBalance($bankId, $amount, $increaseBank ? 'credit' : 'debit');
   240→    }

   243→    private function insertCustomerLedger($payment_id, $post, $amount, $branch_id, $user_id, $type) {
   244→        $customer_id = (int)$post['customer_id'];
   245→        $current = $this->getCustomerRunningBalance($customer_id);
   246→        $sides = $this->getCustomerLedgerSides($type, $amount);
   247→        $new_balance = $current + $sides['debit'] - $sides['credit'];
   248→
   249→        $this->insertCustomerLedgerEntry([
   250→            'customer_id'      => $customer_id,
   251→            'reference_type'   => 'payment',
   252→            'reference_id'     => (int)$payment_id,
   253→            'debit'            => $sides['debit'],
   254→            'credit'           => $sides['credit'],
   255→            'running_balance'  => $new_balance,
   256→            'branch_id'        => (int)$branch_id,
   257→            'transaction_date' => $post['transaction_date'] ?? date('Y-m-d'),
   258→            'remarks'          => $post['narration'] ?? '',
   259→            'created_by'       => (int)$user_id,
   260→        ]);
   261→    }

   296→    public function reverseTransaction($id, $reason) {
   297→        $this->db->beginTransaction();
   298→        try {
   299→            $reason = trim($reason);
   300→            if (strlen($reason) < 3) {
   301→                throw new Exception('Reversal reason is required (min 3 characters)');
   302→            }
   303→
   304→            $user_id = (int)($_SESSION['user_id'] ?? 1);
   305→
   306→            $trans = $this->getTransactionById((int)$id);
   307→            if (!$trans || !empty($trans['is_reversed'])) {
   308→                throw new Exception('Transaction not found or already reversed');
   309→            }
   310→
   311→            if (!$this->canUserReversePayment($trans)) {
   312→                throw new Exception('You do not have access to reverse this payment');
   313→            }
   314→
   315→            $amount = (float)$trans['amount'];
   316→            $type   = $trans['transaction_type'] ?? 'receive';
   317→
   318→            if (!empty($trans['journal_entry_id'])) {
   319→                require_once __DIR__ . '/../services/Accounting/JournalPostingService.php';
   320→                $journalService = new JournalPostingService($this->db);
   321→                $rev = $journalService->reverseLinkedJournal(
   322→                    (int)$trans['journal_entry_id'],
   323→                    'Payment reversal: ' . ($trans['payment_code'] ?? $id) . ' — ' . $reason
   324→                );
   325→                if (($rev['status'] ?? '') === 'error') {
   326→                    throw new Exception('Failed to reverse payment journal: ' . ($rev['message'] ?? ''));
   327→                }
   328→            }
   329→
   330→            $bankId = !empty($trans['bank_id']) ? (int)$trans['bank_id'] : null;
   331→            $payMode = strtolower(trim((string)($trans['payment_mode'] ?? 'cash')));
   332→            if ($bankId && $payMode !== 'bank') {
   333→                $payMode = 'bank';
   334→            }
   335→            $this->syncBankBookBalance($bankId, $amount, $type, $payMode, true);
   336→
   337→            $orig = $this->getCustomerLedgerSides($type, $amount);
   338→            $debit = $orig['credit'];
   339→            $credit = $orig['debit'];
   340→
   341→            $current = $this->getCustomerRunningBalance((int)$trans['customer_id']);
   342→            $new_balance = $current + $debit - $credit;
   343→            $revBranchId = (int)($trans['branch_id'] ?? ($_SESSION['branch_id'] ?? 1));
   344→
   345→            $this->insertCustomerLedgerEntry([
   346→                'customer_id'      => (int)$trans['customer_id'],
   347→                'reference_type'   => 'reversal',
   348→                'reference_id'     => (int)$id,
   349→                'debit'            => $debit,
   350→                'credit'           => $credit,
   351→                'running_balance'  => $new_balance,
   352→                'branch_id'        => $revBranchId,
   353→                'transaction_date' => date('Y-m-d'),
   354→                'remarks'          => 'Reversal of #' . ($trans['payment_code'] ?? $id) . ' - ' . $reason,
   355→                'created_by'       => (int)$user_id,
   356→                'is_reversed'      => 0,
   357→            ]);
   358→
   359→            $this->db->query("
   360→                UPDATE customer_ledger
   361→                SET is_reversed = 1
   362→                WHERE reference_id = :pid
   363→                  AND reference_type = 'payment'
   364→                  AND COALESCE(is_reversed, 0) = 0
   365→            ");
   366→            $this->db->bind(':pid', $id);
   367→            $this->db->execute();
   368→
   369→            if ($type === 'receive') {
   370→                require_once __DIR__ . '/../services/Branch/BranchIntercompanyService.php';
   371→                $settleRev = (new BranchIntercompanyService($this->db))->reverseCustomerPaymentSettlements($id);
   372→                if (($settleRev['status'] ?? '') === 'error') {
   373→                    throw new Exception($settleRev['message'] ?? 'Failed to reverse branch demand settlements');
   374→                }
   375→            }
   376→
   377→            $this->db->query('UPDATE customer_payments SET is_reversed = 1, reversed_at = NOW(),
   378→                              reversed_by = :uid, reverse_reason = :reason WHERE id = :id');
   379→            $this->db->bind(':uid', $user_id);
   380→            $this->db->bind(':reason', $reason);
   381→            $this->db->bind(':id', $id);
   382→            $this->db->execute();
   383→
   384→            $this->db->commit();
   385→
   386→            $glNote = !empty($trans['journal_entry_id']) ? ' GL and bank (if any) reversed.' : '';
   387→
   388→            return [
   389→                'status'  => 'success',
   390→                'message' => 'Payment reversed. Customer ledger restored.' . $glNote,
   391→            ];
   392→
   393→        } catch (Throwable $e) {
   394→            if ($this->db->inTransaction()) {
   395→                $this->db->rollback();
   396→            }
   397→            error_log('Customer payment reverse: ' . $e->getMessage());
   398→
   399→            return ['status' => 'error', 'message' => $e->getMessage()];
   400→        }
   401→    }

   470→    /**
   471→     * Build WHERE clause fragments for payment list queries.
   472→     *
   473→     * @return array{0: array<int, string>, 1: array<string, mixed>}
   474→     */
   475→    private function buildPaymentListFilters(array $filters, ?int $branchId): array
   476→    {
   477→        $where = [];
   478→        $bindings = [];
   479→
   480→        if ($branchId > 0) {
   481→            $where[] = 'cp.branch_id = :branch_id';
   482→            $bindings[':branch_id'] = $branchId;
   483→        } elseif (!$this->canOverrideBranch() && self::sessionBranchId() > 0) {
   484→            $where[] = 'cp.branch_id = :branch_id';
   485→            $bindings[':branch_id'] = self::sessionBranchId();
   486→        }
   487→
   488→        $dateFrom = !empty($filters['date_from']) ? (string)$filters['date_from'] : null;
   489→        $dateTo = !empty($filters['date_to']) ? (string)$filters['date_to'] : null;
   490→
   491→        if ($dateFrom && $dateTo && $dateFrom === $dateTo) {
   492→            $where[] = 'cp.payment_date = :pay_date';
   493→            $bindings[':pay_date'] = $dateFrom;
   494→        } else {
   495→            if ($dateFrom) {
   496→                $where[] = 'cp.payment_date >= :date_from';
   497→                $bindings[':date_from'] = $dateFrom;
   498→            }
   499→            if ($dateTo) {
   500→                $where[] = 'cp.payment_date <= :date_to';
   501→                $bindings[':date_to'] = $dateTo;
   502→            }
   503→            if (!$dateFrom && !$dateTo) {
   504→                $where[] = 'cp.payment_date = CURRENT_DATE';
   505→            }
   506→        }
   507→
   508→        if (!empty($filters['transaction_type']) && $filters['transaction_type'] !== 'all') {
   509→            $where[] = 'cp.transaction_type = :ttype';
   510→            $bindings[':ttype'] = $filters['transaction_type'];
   511→        }
   512→
   513→        if (!empty($filters['status']) && $filters['status'] !== 'all') {
   514→            if ($filters['status'] === 'reversed') {
   515→                $where[] = 'cp.is_reversed = 1';
   516→            } else {
   517→                $where[] = 'COALESCE(cp.is_reversed, 0) = 0';
   518→            }
   519→        }
   520→
   521→        if (!empty($filters['payment_mode']) && $filters['payment_mode'] !== 'all') {
   522→            $where[] = 'LOWER(cp.payment_mode) = :pmode';
   523→            $bindings[':pmode'] = strtolower((string)$filters['payment_mode']);
   524→        }
   525→
   526→        if (!empty($filters['customer_id'])) {
   527→            $where[] = 'cp.customer_id = :cid';
   528→            $bindings[':cid'] = (int)$filters['customer_id'];
   529→        }
   530→
   531→        return [$where, $bindings];
   532→    }

   534→    private function paymentListFromSql(): string
   535→    {
   536→        return "
   537→            FROM customer_payments cp
   538→            JOIN customers c ON cp.customer_id = c.id
   539→            LEFT JOIN banks b ON cp.bank_id = b.id
   540→            LEFT JOIN users u ON cp.created_by = u.id
   541→            LEFT JOIN employees e ON cp.collected_by = e.id
   542→            LEFT JOIN branches br ON cp.branch_id = br.id
   543→        ";

   546→    /**
   547→     * Server-side DataTables for customer payment index.
   548→     */
   549→    public function getPaymentsForDataTable(array $params, array $filters, ?int $branchId): array
   550→    {
   551→        $start       = (int)($params['start'] ?? 0);
   552→        $length      = (int)($params['length'] ?? 25);
   553→        if ($length <= 0 || $length > 100) {
   554→            $length = 25;
   555→        }
   556→        $searchValue = trim($params['search']['value'] ?? '');
   557→        $reversedMode = $params['reversedMode'] ?? '';
   558→        $orderColumn = (int)($params['order'][0]['column'] ?? 0);
   559→        $orderDir    = strtolower($params['order'][0]['dir'] ?? 'desc') === 'asc' ? 'ASC' : 'DESC';
   560→
   561→        $columns = [
   562→            'cp.payment_date',
   563→            'cp.payment_code',
   564→            'c.shop_name',
   565→            'cp.transaction_type',
   566→            'cp.amount',
   567→            'cp.payment_mode',
   568→            'e.name',
   569→            'cp.is_reversed',
   570→        ];
   571→
   572→        $baseQuery = $this->paymentListFromSql();
   573→
   574→        [$where, $bindParams] = $this->buildPaymentListFilters($filters, $branchId);
   575→
   576→        if ($reversedMode === 'only_reversed') {
   577→            $where[] = 'cp.is_reversed = 1';
   578→        }
   579→
   580→        $branchOnlyWhere = [];
   581→        $branchOnlyBind = [];
   582→        if ($branchId > 0) {
   583→            $branchOnlyWhere[] = 'cp.branch_id = :branch_id';
   584→            $branchOnlyBind[':branch_id'] = $branchId;
   585→        } elseif (!$this->canOverrideBranch() && self::sessionBranchId() > 0) {
   586→            $branchOnlyWhere[] = 'cp.branch_id = :branch_id';
   587→            $branchOnlyBind[':branch_id'] = self::sessionBranchId();
   588→        }
   589→        $branchOnlySql = $branchOnlyWhere ? 'WHERE ' . implode(' AND ', $branchOnlyWhere) : '';
   590→
   591→        if ($searchValue !== '') {
   592→            $where[] = '(cp.payment_code LIKE :search
   593→                      OR c.shop_name LIKE :search
   594→                      OR c.customer_name LIKE :search
   595→                      OR c.mobile LIKE :search)';
   596→            $bindParams[':search'] = '%' . $searchValue . '%';
   597→        }
   598→
   599→        $whereSql = $where ? 'WHERE ' . implode(' AND ', $where) : '';
   600→
   601→        $totalQuery = "SELECT COUNT(cp.id) AS total {$baseQuery} {$branchOnlySql}";
   602→        $this->db->query($totalQuery);
   603→        foreach ($branchOnlyBind as $key => $val) {
   604→            $this->db->bind($key, $val);
   605→        }
   606→        $recordsTotal = (int)($this->db->single()['total'] ?? 0);
   607→
   608→        $filteredQuery = "SELECT COUNT(cp.id) AS total {$baseQuery} {$whereSql}";
   609→        $this->db->query($filteredQuery);
   610→        foreach ($bindParams as $key => $val) {
   611→            $this->db->bind($key, $val);
   612→        }
   613→        $recordsFiltered = (int)($this->db->single()['total'] ?? 0);
   614→
   615→        $orderBy = $columns[$orderColumn] ?? 'cp.payment_date';
   616→        $dataQuery = "
   617→            SELECT
   618→                cp.id,
   619→                cp.payment_date,
   620→                cp.payment_code,
   621→                cp.customer_id,
   622→                cp.transaction_type,
   623→                cp.amount,
   624→                cp.payment_mode,
   625→                cp.is_reversed,
   626→                cp.branch_id,
   627→                c.shop_name,
   628→                c.mobile,
   629→                e.name AS collected_by_name
   630→            {$baseQuery}
   631→            {$whereSql}
   632→            ORDER BY {$orderBy} {$orderDir}, cp.id DESC
   633→            LIMIT {$length} OFFSET {$start}
   634→        ";
   635→
   636→        $this->db->query($dataQuery);
   637→        foreach ($bindParams as $key => $val) {
   638→            $this->db->bind($key, $val);
   639→        }
   640→        $data = $this->db->resultSet() ?: [];
   641→
   642→        foreach ($data as &$row) {
   643→            $row['can_reverse'] = $this->canUserReversePayment($row) ? 1 : 0;
   644→        }
   645→        unset($row);
   646→
   647→        return [
   648→            'draw'            => (int)($params['draw'] ?? 1),
   649→            'recordsTotal'    => $recordsTotal,
   650→            'recordsFiltered' => $recordsFiltered,
   651→            'data'            => $data,
   652→        ];
   653→    }

   655→    /**
   656→     * Get all customer transactions for listing
   657→     */
   658→    public function getFilteredTransactions(array $filters = [], ?int $branchId = null): array
   659→    {
   660→        $sql = "
   661→            SELECT
   662→                cp.*,
   663→                c.shop_name,
   664→                c.mobile,
   665→                b.bank_name,
   666→                u.username AS created_by_name,
   667→                e.name AS collected_by_name,
   668→                br.branch_name
   669→            FROM customer_payments cp
   670→            JOIN customers c ON cp.customer_id = c.id
   671→            LEFT JOIN banks b ON cp.bank_id = b.id
   672→            LEFT JOIN users u ON cp.created_by = u.id
   673→            LEFT JOIN employees e ON cp.collected_by = e.id
   674→            LEFT JOIN branches br ON cp.branch_id = br.id
   675→        ";
   676→
   677→        [$where, $bindings] = $this->buildPaymentListFilters($filters, $branchId);
   678→
   679→        if (!empty($where)) {
   680→            $sql .= ' WHERE ' . implode(' AND ', $where);
   681→        }
   682→
   683→        $sql .= ' ORDER BY cp.payment_date DESC, cp.id DESC LIMIT 500';
   684→
   685→        $this->db->query($sql);
   686→        foreach ($bindings as $k => $v) {
   687→            $this->db->bind($k, $v);
   688→        }
   689→
   690→        return $this->db->resultSet() ?: [];
   691→    }

   705→public function getTransactionById($id) {
   706→    $this->db->query("
   707→        SELECT cp.*,
   708→               c.shop_name, c.customer_name, c.mobile, c.address AS customer_address,
   709→               e.name AS collected_by_name,
   710→               b.bank_name, b.account_number AS bank_account_number,
   711→               u.username AS created_by_name,
   712→               ru.username AS reversed_by_name,
   713→               br.branch_name, br.address AS branch_address, br.phone AS branch_phone
   714→        FROM customer_payments cp
   715→        JOIN customers c ON cp.customer_id = c.id
   716→        LEFT JOIN employees e ON cp.collected_by = e.id
   717→        LEFT JOIN banks b ON cp.bank_id = b.id
   718→        LEFT JOIN users u ON cp.created_by = u.id
   719→        LEFT JOIN users ru ON cp.reversed_by = ru.id
   720→        LEFT JOIN branches br ON cp.branch_id = br.id
   721→        WHERE cp.id = :id
   722→    ");
   723→    $this->db->bind(':id', $id);
   724→    return $this->db->single();
