   340→    // -------- AJAX base helpers --------

   350→    }
   351→    function ajaxGet(url, params) {
   352→        return $.ajax({
   353→            url: url,
   354→            method: 'GET',
   355→            dataType: 'json',
   356→            data: params || {},
   357→            headers: { 'X-Requested-With': 'XMLHttpRequest' }

   386→
   387→    // -------- Money / qty formatting --------
   388→    function fmtMoney(v) {
   389→        var n = parseFloat(v || 0);
   390→        return n.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 });
   391→    }
   392→    function fmtQty(v) {
   393→        var n = parseFloat(v || 0);
   394→        return n.toLocaleString(undefined, { minimumFractionDigits: 0, maximumFractionDigits: 4 });
   395→    }
   396→    function escHtml(s) {
   397→        if (s === null || s === undefined) return '';
   398→        return String(s).replace(/[&<>"']/g, function (c) {
   399→            return { '&':'&amp;', '<':'&lt;', '>':'&gt;', '"':'&quot;', "'":'&#39;' }[c];
   400→        });
   401→    }
   402→
   403→    // -------- Toast / alert helper --------
   404→    function toast(message, type) {
   405→        type = type || 'info';
   406→        Swal.fire({
   407→            toast: true,
   408→            position: 'top-end',
   409→            icon: type === 'error' ? 'error' : (type === 'success' ? 'success' : 'info'),
   410→            title: message,
   411→            showConfirmButton: false,
   412→            timer: 2500,
   413→            timerProgressBar: true
   414→        });
   415→    }

   421→    function renderAll() {

   427→        var hasItems = state.cart && state.cart.items && state.cart.items.length > 0;
   428→        var btnFinalize = document.getElementById('btnFinalize');
   429→        btnFinalize.disabled = !(hasItems && valid);
   430→    }
   431→
   432→    function renderCartTable() {
   433→        var $body = $('#cartItemsBody');
   434→        $body.empty();
   435→
   436→        var items = (state.cart && state.cart.items) ? state.cart.items : [];
   437→        $('#itemsCountBadge').text(items.length);
   438→        $('#cartEmptyRow').toggle(items.length === 0);
   439→
   440→        items.forEach(function (item) {
   441→            var productId = parseInt(item.product_id, 10);
   442→            var qty   = parseFloat(item.qty || 0);
   443→            var rate  = parseFloat(item.rate || 0);
   444→            var total = parseFloat(item.total || (qty * rate));
   445→            var avail = item.available_qty !== undefined ? parseFloat(item.available_qty) : null;
   446→            var minR  = item.min_rate !== null && item.min_rate !== undefined ? parseFloat(item.min_rate) : null;
   447→            var maxR  = item.max_rate !== null && item.max_rate !== undefined ? parseFloat(item.max_rate) : null;
   448→
   449→            var availClass = '';
   450→            if (avail !== null) {
   451→                if (qty > avail + 0.0001) availClass = 'text-danger fw-bold';
   452→                else if (qty > avail * 0.9) availClass = 'text-warning';
   453→                else availClass = 'text-success';
   454→            }
   455→
   456→            var row =
   457→                '<tr data-product-id="' + productId + '">' +
   458→                    '<td>' +
   459→                        '<div class="fw-semibold">' + escHtml(item.product_name) + '</div>' +
   460→                        '<div class="small text-muted">#' + productId + '</div>' +
   461→                    '</td>' +
   462→                    '<td>' +
   463→                        '<input type="number" class="form-control form-control-sm cart-qty" min="0.001" step="0.001" value="' + qty + '">' +
   464→                    '</td>' +
   465→                    '<td>' +
   466→                        '<input type="number" class="form-control form-control-sm cart-rate" min="0" step="0.01" value="' + rate.toFixed(2) + '">' +
   467→                        (minR !== null
   468→                            ? '<div class="form-text small text-muted">Min ' + fmtMoney(minR) + ' / Max ' + fmtMoney(maxR) + '</div>'
   469→                            : '') +
   470→                    '</td>' +

   926→        // ============================================================
   927→        // ============== FINALIZE INVOICE FLOW (P0-6) ================
   928→        // ============================================================
   929→        //
   930→        // Multi-step flow:
   931→        //   1. Pre-flight: cart must be non-empty + validated.
   932→        //   2. Open a SweetAlert "Finalize Invoice" dialog with editable
   933→        //      fields (invoice_date, discount, transport, notes, override).
   934→        //   3. On confirm: pre-check credit limit via GET /credit-check.
   935→        //      If exceeded and user didn't tick override → re-prompt.
   936→        //   4. POST /finalize with all fields.
   937→        //   5. On success: redirect to the new invoice's show page.
   938→        //      On error: show SweetAlert error, re-enable button.
   939→        //
   940→        // The button is disabled during the AJAX request to mitigate
   941→        // double-submit (P2-6 will add a proper idempotency token).
   942→
   943→        function finalizeInvoice() {
   944→            // --- Step 1: pre-flight checks ---
   945→            if (!state.cart || !state.cart.items || state.cart.items.length === 0) {
   946→                Swal.fire({
   947→                    icon: 'warning',
   948→                    title: 'Cart is empty',
   949→                    text: 'Add at least one product before finalizing.',
   950→                    confirmButtonColor: '#7c3aed'
   951→                });
   952→                return;
   953→            }
   954→
   955→            var valid = state.validation ? !!state.validation.valid : false;
   956→            if (!valid) {
   957→                Swal.fire({
   958→                    icon: 'warning',
   959→                    title: 'Cart validation failed',
   960→                    text: 'Click "Validate Cart" and resolve any stock/rate errors before finalizing.',
   961→                    confirmButtonColor: '#7c3aed'
   962→                });
   963→                return;
   964→            }
   965→
   966→            var customerId = state.customerId;
   967→            if (!customerId) {
   968→                Swal.fire({
   969→                    icon: 'warning',
   970→                    title: 'No customer selected',
   971→                    text: 'Please select a customer first.',
   972→                    confirmButtonColor: '#7c3aed'
   973→                });
   974→                return;
   975→            }
   976→
   977→            var subtotal = parseFloat((state.cart && state.cart.subtotal) || 0);
   978→            var today = new Date().toISOString().split('T')[0];
   979→
   980→            // P2-6: Generate idempotency token (UUID v4) for this finalize attempt.
   981→            // Prevents duplicate invoice creation on double-click or refresh-after-submit.
   982→            var idempotencyToken = (function() {
   983→                if (window.crypto && window.crypto.randomUUID) {
   984→                    return window.crypto.randomUUID();
   985→                }
   986→                // Fallback UUID v4 generator (older browsers).
   987→                return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, function(c) {
   988→                    var r = Math.random() * 16 | 0;
   989→                    var v = c === 'x' ? r : (r & 0x3 | 0x8);
   990→                    return v.toString(16);
   991→                });
   992→            })();
   993→
   994→            // --- Step 2: open the finalize dialog ---
   995→            Swal.fire({
   996→                title: '<i class="fas fa-file-invoice-dollar me-2"></i>Finalize Invoice',
   997→                html:
