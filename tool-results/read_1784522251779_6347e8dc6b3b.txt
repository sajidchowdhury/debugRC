    12→
    13→/**
    14→ * Sales Challan Service — Phase 8.3.
    15→ *
    16→ * Two-step flow (from Phase 8.2 draft invoice):
    17→ *   1. prepareGodown: assign warehouse_id to invoice items + dispatches.
    18→ *      Invoice status: draft → confirmed, is_godown_prepared=true.
    19→ *      NO stock movement, NO GL — just warehouse assignment.
    20→ *
    21→ *   2. issueChallan: stock OUT via StockService at avg_cost + GL Dr COGS / Cr Inventory.
    22→ *      Creates sales_challan, moves stock, posts COGS journal.
    23→ *      Invoice: is_challan_issued=true.
    24→ *
    25→ *   3. cancelChallan: reverse stock + GL (append-only reversal).
    26→ *
    27→ * Rate semantics (per avg_cost_rule.md §3):
    28→ *   Stock OUT at current avg_cost (cost flows out at average, avg unchanged on OUT).
    29→ *   The COGS = Σ (qty × avg_cost) for all dispatch lines.
    30→ *
    31→ * GL posting (re-derived from double-entry):
    32→ *   Dr COGS (nature: cogs)
    33→ *   Cr Inventory (nature: inventory)
    34→ */
    35→class SalesChallanService
    36→{
    37→    public function __construct(
    38→        private StockService $stockService,
    39→        private JournalPostingService $journalPosting,
    40→        private SalesAccess $salesAccess,
    41→        private SalesAuditLogger $auditLogger,
    42→        private StockAvailabilityService $availabilityService
    43→    ) {}
    44→
    45→    /**
    46→     * Step 1: Prepare godown — assign warehouse_id to invoice items + dispatches.
    47→     * No stock movement, no GL.
    48→     *
    49→     * @param int $invoiceId
    50→     * @param array $warehouseAssignments [product_id => warehouse_id]
    51→     * @param int $preparedBy
    52→     * @return SalesInvoice
    53→     * @throws \RuntimeException If invoice not draft or already godown-prepared.
    54→     */
    55→    public function prepareGodown(int $invoiceId, array $warehouseAssignments, int $preparedBy): SalesInvoice
    56→    {
    57→        return DB::transaction(function () use ($invoiceId, $warehouseAssignments, $preparedBy) {
    58→            $invoice = SalesInvoice::with('items', 'dispatches')->lockForUpdate()->find($invoiceId);
    59→
    60→            if (!$invoice) {
    61→                throw new \RuntimeException("Invoice {$invoiceId} not found.");
    62→            }
    63→            if (!$invoice->isDraft()) {
    64→                throw new \RuntimeException("Only draft invoices can be godown-prepared (current: {$invoice->status}).");
    65→            }
    66→
    67→            // Assign warehouse_id to each invoice item.
    68→            foreach ($invoice->items as $item) {
    69→                $wid = $warehouseAssignments[$item->product_id] ?? null;
    70→                if (!$wid) {
    71→                    throw new \RuntimeException("Warehouse not assigned for product {$item->product_id}.");
    72→                }
    73→
    74→                // Check availability in the assigned warehouse.
    75→                $available = $this->stockService->getWarehouseQty((int) $wid, $item->product_id);
    76→                if ((float) $item->qty > $available + 0.0001) {
    77→                    throw new \RuntimeException(
    78→                        "Insufficient stock in warehouse {$wid} for product {$item->product_id}: "
    79→                        . "available {$available}, required {$item->qty}"
    80→                    );
    81→                }
    82→
    83→                DB::table('sales_invoice_items')
    84→                    ->where('id', $item->id)
    85→                    ->update(['warehouse_id' => (int) $wid]);
    86→
    87→                DB::table('sales_invoice_dispatches')
    88→                    ->where('sales_invoice_id', $invoiceId)
    89→                    ->where('product_id', $item->product_id)
    90→                    ->update(['warehouse_id' => (int) $wid]);
    91→            }
    92→
    93→            // Update invoice status.
    94→            DB::table('sales_invoices')
    95→                ->where('id', $invoiceId)
    96→                ->update([
    97→                    'status' => 'confirmed',
    98→                    'is_godown_prepared' => true,
    99→                    'godown_prepared_at' => now(),
   100→                    'updated_at' => now(),
   101→                ]);
   102→
   103→            // P1-3: Audit log — godown_prepared.
   104→            $this->auditLogger->godownPrepared(
   105→                $preparedBy, $invoiceId, $invoice->invoice_code, (int) $invoice->branch_id
   106→            );
   107→
   108→            return SalesInvoice::with(['items.product', 'dispatches', 'customer', 'branch'])->find($invoiceId);
   109→        });
   110→    }
   111→
   112→    /**
   113→     * Step 2: Issue challan — stock OUT + GL Dr COGS / Cr Inventory.
   114→     *
   115→     * @param int $invoiceId
   116→     * @param array $data {
   117→     *     transport_name: string|null,
   118→     *     vehicle_number: string|null,
   119→     *     driver_name: string|null,
   120→     *     transport_cost: float,
   121→     *     notes: string|null,
   122→     *     created_by: int,
   123→     * }
   124→     * @return SalesChallan
   125→     * @throws \RuntimeException If not godown-prepared or already challan-issued.
   126→     */
   127→    public function issueChallan(int $invoiceId, array $data): SalesChallan
   128→    {
   129→        return DB::transaction(function () use ($invoiceId, $data) {
   130→            $invoice = SalesInvoice::with(['items', 'dispatches'])->lockForUpdate()->find($invoiceId);
   131→
   132→            if (!$invoice) {
   133→                throw new \RuntimeException("Invoice {$invoiceId} not found.");
   134→            }
   135→
   136→            // P0-8: Defense-in-depth branch isolation check.
   137→            $this->salesAccess->assertBranchAccessible((int) $invoice->branch_id);
   138→
   139→            if (!$invoice->is_godown_prepared) {
   140→                throw new \RuntimeException("Invoice must be godown-prepared before issuing challan.");
   141→            }
   142→            if ($invoice->is_challan_issued) {
   143→                throw new \RuntimeException("Challan already issued for this invoice.");
   144→            }
   145→
   146→            $challanCode = $this->generateChallanCode();
   147→            $challanDate = now()->format('Y-m-d');
   148→            $cogsTotal = 0.0;
   149→
   150→            // Create the challan header.
   151→            $challanId = DB::table('sales_challans')->insertGetId([
   152→                'challan_code' => $challanCode,
   153→                'challan_date' => $challanDate,
   154→                'sales_invoice_id' => $invoiceId,
   155→                'branch_id' => $invoice->branch_id,
   156→                'transport_name' => $data['transport_name'] ?? null,
   157→                'transport_phone' => $data['transport_phone'] ?? null,
   158→                'vehicle_number' => $data['vehicle_number'] ?? null,
   159→                'driver_name' => $data['driver_name'] ?? null,
   160→                'transport_cost' => $data['transport_cost'] ?? 0,
   161→                'is_reversed' => false,
   162→                'is_dispatch_soft_hold' => false,
   163→                'created_by' => $data['created_by'] ?? null,
   164→                'created_at' => now(),
   165→                'updated_at' => now(),
   166→            ]);
   167→
   168→            // Process each dispatch line: stock OUT at avg_cost.
   169→            foreach ($invoice->items as $item) {
   170→                $warehouseId = $item->warehouse_id;
   171→                $productId = $item->product_id;
   172→                $qty = (float) $item->qty;
   173→
   174→                if (!$warehouseId) {
   175→                    throw new \RuntimeException("Product {$productId} has no warehouse assigned. Run godown prep first.");
   176→                }
   177→
   178→                // Get current avg_cost (this is the COGS rate).
   179→                $avgCost = $this->stockService->getWarehouseAvgCost($warehouseId, $productId);
   180→                if ($avgCost <= 0) {
   181→                    throw new \RuntimeException(
   182→                        "Cannot issue challan: zero avg_cost for product {$productId} in warehouse {$warehouseId}. "
   183→                        . "Receive stock or set cost first."
   184→                    );
   185→                }
   186→
   187→                // Stock OUT via StockService (reference_type='sales_challan', rate=avg_cost).
   188→                $this->stockService->applyTransaction([
   189→                    'warehouse_id' => $warehouseId,
   190→                    'product_id' => $productId,
   191→                    'qty' => -$qty, // negative = OUT
   192→                    'rate' => $avgCost, // current avg_cost (cost flows out, avg unchanged on OUT)
   193→                    'reference_type' => 'sales_challan',
   194→                    'reference_id' => $challanId,
   195→                    'notes' => 'Sales Challan ' . $challanCode,
   196→                    'transaction_date' => $challanDate,
   197→                    'created_by' => $data['created_by'] ?? null,
   198→                ]);
   199→
   200→                // P0-5: Persist per-line issue cost snapshot (sales_challan_items).
   201→                // This denormalized record captures the avg_cost at the moment of stock OUT
   202→                // so that reversals and reports can use the ORIGINAL per-line rate.
   203→                DB::table('sales_challan_items')->insert([
   204→                    'sales_challan_id' => $challanId,
   205→                    'product_id' => $productId,
   206→                    'warehouse_id' => $warehouseId,
   207→                    'qty' => $qty,
   208→                    'issue_rate' => $avgCost,
   209→                    'cogs_amount' => round($qty * $avgCost, 2),
   210→                    'created_at' => now(),
   211→                ]);
   212→
   213→                // Update dispatch row: dispatched_qty = ordered_qty, qty mirrors for GENERATED amount.
   214→                DB::table('sales_invoice_dispatches')
   215→                    ->where('sales_invoice_id', $invoiceId)
   216→                    ->where('product_id', $productId)
   217→                    ->update([
   218→                        'qty' => DB::raw('ordered_qty'),
   219→                        'dispatched_qty' => DB::raw('ordered_qty'),
   220→                        'warehouse_id' => $warehouseId,
   221→                    ]);
   222→
   223→                $cogsTotal += $qty * $avgCost;
   224→            }
   225→
   226→            // Post GL: Dr COGS / Cr Inventory (single journal for all lines).
   227→            $journalEntryId = $this->postCogsGL($challanId, $challanCode, $challanDate, $invoice->branch_id, $cogsTotal, $data['created_by'] ?? null);
   228→
   229→            // Update challan with journal_entry_id + issue_cost.
   230→            DB::table('sales_challans')
   231→                ->where('id', $challanId)
   232→                ->update([
   233→                    'journal_entry_id' => $journalEntryId,
   234→                    'issue_cost' => round($cogsTotal, 2),
   235→                    'updated_at' => now(),
   236→                ]);
   237→
   238→            // P2-3: Transport snapshot + adjustment.
   239→            // If the challan form's transport_cost differs from the invoice's
   240→            // original transport_cost, snapshot the original values + post an
   241→            // adjustment (customer_ledger + GL) for the delta.
   242→            $newTransport = (float) ($data['transport_cost'] ?? 0);
   243→            $oldTransport = (float) $invoice->transport_cost;
   244→            $transportAdjustment = round($newTransport - $oldTransport, 2);
   245→            $adjustmentJournalEntryId = null;
   246→
   247→            if (abs($transportAdjustment) > 0.01) {
   248→                // Snapshot original values on the invoice.
   249→                DB::table('sales_invoices')
   250→                    ->where('id', $invoiceId)
   251→                    ->update([
   252→                        'pre_challan_transport' => $oldTransport,
   253→                        'pre_challan_total' => (float) $invoice->total_amount,
   254→                    ]);
   255→
   256→                // Update invoice transport_cost + total_amount + due_amount.
   257→                $newTotal = (float) $invoice->sub_total - (float) $invoice->discount_amount + $newTransport;
   258→                DB::table('sales_invoices')
   259→                    ->where('id', $invoiceId)
   260→                    ->update([
   261→                        'transport_cost' => round($newTransport, 2),
   262→                        'total_amount' => round($newTotal, 2),
   263→                        'due_amount' => round($newTotal - (float) $invoice->paid_amount, 2),
   264→                    ]);
   265→
   266→                // Post customer_ledger 'invoice_adjustment' entry for the delta.
   267→                $this->postTransportAdjustmentLedger($invoice, $transportAdjustment, $challanCode, $data['created_by'] ?? null);
   268→
   269→                // Post GL adjustment JE (Dr/Cr AR + Revenue, swapped by sign).
   270→                $adjustmentJournalEntryId = $this->postTransportAdjustmentGL(
   271→                    $invoiceId, $challanCode, $challanDate, (int) $invoice->branch_id,
   272→                    $transportAdjustment, (int) $invoice->customer_id,
   273→                    $data['created_by'] ?? null
   274→                );
   275→            }
   276→
   277→            // Store transport_adjustment + adjustment_journal_entry_id on the challan.
   278→            DB::table('sales_challans')
   279→                ->where('id', $challanId)
   280→                ->update([
   281→                    'transport_adjustment' => $transportAdjustment,
   282→                    'adjustment_journal_entry_id' => $adjustmentJournalEntryId,
   283→                ]);
   284→
   285→            // Update invoice: is_challan_issued=true.
   286→            DB::table('sales_invoices')
   287→                ->where('id', $invoiceId)
   288→                ->update([
   289→                    'is_challan_issued' => true,
   290→                    'challan_issued_at' => now(),
   291→                    'cogs_journal_entry_id' => $journalEntryId,
   292→                    'updated_at' => now(),
   293→                ]);
   294→
   295→            // P1-3: Audit log — challan_issued.
   296→            $this->auditLogger->challanIssued(
   297→                $data['created_by'] ?? auth()->id() ?? 0,
   298→                $challanId, $challanCode, $invoiceId, (int) $invoice->branch_id,
   299→                $cogsTotal, $journalEntryId
   300→            );
   301→
   302→            // P2-7: Invalidate pipeline cache (dispatched_qty changed).
   303→            $this->availabilityService->invalidatePipelineForInvoice($invoiceId);
   304→
   305→            return SalesChallan::with(['salesInvoice.items.product', 'items.product', 'items.warehouse', 'branch', 'journalEntry.lines.ledger'])
   306→                ->find($challanId);
   307→        });
   308→    }
   309→
   310→    /**
   311→     * Step 3: Cancel a challan — reverse stock + GL (append-only reversal).
   312→     *
   313→     * @param int $challanId
   314→     * @param int $cancelledBy
   315→     * @param string $reason
   316→     * @return SalesChallan
   317→     */
   318→    public function cancelChallan(int $challanId, int $cancelledBy, string $reason = ''): SalesChallan
   319→    {
   320→        return DB::transaction(function () use ($challanId, $cancelledBy, $reason) {
   321→            $challan = SalesChallan::with('salesInvoice.items')->lockForUpdate()->find($challanId);
   322→
   323→            if (!$challan) {
   324→                throw new \RuntimeException("Challan {$challanId} not found.");
   325→            }
   326→            if ($challan->is_reversed) {
   327→                throw new \RuntimeException("Challan is already cancelled.");
   328→            }
   329→
   330→            // Reverse GL (COGS journal).
   331→            if ($challan->journal_entry_id) {
   332→                $this->journalPosting->reverseJournalEntry(
   333→                    $challan->journal_entry_id, $cancelledBy,
   334→                    "Challan cancelled: {$reason}"
   335→                );
   336→            }
   337→
   338→            // Reverse each stock movement.
   339→            $stockTxs = DB::table('stock_transactions')
   340→                ->where('reference_type', 'sales_challan')
   341→                ->where('reference_id', $challanId)
   342→                ->where('is_reversed', false)
   343→                ->get();
   344→
   345→            foreach ($stockTxs as $tx) {
   346→                $this->stockService->reverseTransaction(
   347→                    $tx->id, $cancelledBy,
   348→                    "Challan cancelled: {$reason}"
   349→                );
   350→            }
   351→
   352→            // Reset dispatch rows: dispatched_qty = 0, qty mirrors for GENERATED amount.
   353→            DB::table('sales_invoice_dispatches')
   354→                ->where('sales_invoice_id', $challan->sales_invoice_id)
   355→                ->update([
   356→                    'qty' => 0,
   357→                    'dispatched_qty' => 0,
   358→                ]);
   359→
   360→            // Mark challan as reversed.
   361→            DB::table('sales_challans')
   362→                ->where('id', $challanId)
   363→                ->update([
   364→                    'is_reversed' => true,
   365→                    'reversed_at' => now(),
   366→                    'reversed_by' => $cancelledBy,
   367→                    'reverse_reason' => $reason,
   368→                ]);
   369→
   370→            // P2-3: Reverse transport adjustment (if any).
   371→            // If the challan had a transport_adjustment, reverse the adjustment
   372→            // GL JE + restore the invoice's original transport_cost + total_amount
   373→            // from the pre_challan snapshot.
   374→            if ($challan->adjustment_journal_entry_id) {
   375→                $this->journalPosting->reverseJournalEntry(
   376→                    $challan->adjustment_journal_entry_id, $cancelledBy,
   377→                    "Challan cancelled: transport adjustment reversed — {$reason}"
   378→                );
   379→            }
   380→
   381→            // Restore invoice transport snapshot (if snapshot exists).
   382→            $invoiceRow = DB::table('sales_invoices')
   383→                ->where('id', $challan->sales_invoice_id)
   384→                ->first(['pre_challan_transport', 'pre_challan_total', 'paid_amount']);
   385→
   386→            if ($invoiceRow && $invoiceRow->pre_challan_transport !== null) {
   387→                $restoredTotal = (float) $invoiceRow->pre_challan_total;
   388→                DB::table('sales_invoices')
   389→                    ->where('id', $challan->sales_invoice_id)
   390→                    ->update([
   391→                        'transport_cost' => (float) $invoiceRow->pre_challan_transport,
   392→                        'total_amount' => $restoredTotal,
   393→                        'due_amount' => round($restoredTotal - (float) $invoiceRow->paid_amount, 2),
   394→                        'pre_challan_transport' => null,
   395→                        'pre_challan_total' => null,
   396→                    ]);
   397→            }
   398→
   399→            // P2-2: Reset invoice back to DRAFT state (fully editable).
   400→            // Legacy reverseChallan resets invoice to 'godown_issued' state,
   401→            // but Laravel's edit flow requires status='draft'. Since cancelling
   402→            // a challan invalidates the godown assignment (warehouse_id on items
   403→            // was set during godown prep, and the items may need re-picking),
   404→            // we reset the invoice all the way back to draft.
   405→            DB::table('sales_invoices')
   406→                ->where('id', $challan->sales_invoice_id)
   407→                ->update([
   408→                    'status' => 'draft',
   409→                    'is_challan_issued' => false,
   410→                    'challan_issued_at' => null,
   411→                    'cogs_journal_entry_id' => null,
   412→                    'is_godown_prepared' => false,
   413→                    'godown_prepared_at' => null,
   414→                    'updated_at' => now(),
   415→                ]);
   416→
   417→            // P2-2: Reset warehouse_id on invoice items + dispatches (godown
   418→            // assignment is invalidated — user must re-run godown prep).
   419→            DB::table('sales_invoice_items')
   420→                ->where('sales_invoice_id', $challan->sales_invoice_id)
   421→                ->update(['warehouse_id' => null]);
   422→
   423→            DB::table('sales_invoice_dispatches')
   424→                ->where('sales_invoice_id', $challan->sales_invoice_id)
   425→                ->update(['warehouse_id' => null]);
   426→
   427→            // P1-3: Audit log — challan_reversed.
   428→            $this->auditLogger->challanReversed(
   429→                $cancelledBy,
   430→                $challanId, $challan->challan_code, $challan->sales_invoice_id,
   431→                (int) $challan->branch_id, $reason
   432→            );
   433→
   434→            // P2-7: Invalidate pipeline cache (dispatched_qty reset to 0).
   435→            $this->availabilityService->invalidatePipelineForInvoice($challan->sales_invoice_id);
   436→
   437→            return SalesChallan::find($challanId);
   438→        });
   439→    }

   441→    /**
   442→     * Post GL: Dr COGS / Cr Inventory.
   443→     *
   444→     * @return int journal_entry_id
   445→     */
   446→    private function postCogsGL(int $challanId, string $challanCode, string $challanDate, int $branchId, float $cogsAmount, ?int $createdBy): int
   447→    {
   448→        if ($cogsAmount < 0.01) {
   449→            return 0; // No COGS for zero-value
   450→        }
   451→
   452→        $cogsLedgerId = $this->journalPosting->lookupLedgerByNature('cogs');
   453→        $inventoryLedgerId = $this->journalPosting->lookupLedgerByNature('inventory');
   454→
   455→        if (!$cogsLedgerId) {
   456→            throw new \RuntimeException('COGS ledger not found (nature: cogs).');
   457→        }
   458→        if (!$inventoryLedgerId) {
   459→            throw new \RuntimeException('Inventory ledger not found (nature: inventory).');
   460→        }
   461→
   462→        return $this->journalPosting->createJournalEntry([
   463→            'entry_date' => $challanDate,
   464→            'reference_type' => 'sales_challan',
   465→            'reference_id' => $challanId,
   466→            'branch_id' => $branchId,
   467→            'description' => 'Sales Challan COGS ' . $challanCode,
   468→            'source' => 'sales_challan',
   469→            'created_by' => $createdBy,
   470→        ], [
   471→            [
   472→                'ledger_id' => $cogsLedgerId,
   473→                'debit' => $cogsAmount, 'credit' => 0,
   474→                'entity_type' => 'sales_challan', 'entity_id' => $challanId,
   475→                'memo' => 'Challan ' . $challanCode . ' — COGS',
   476→            ],
   477→            [
   478→                'ledger_id' => $inventoryLedgerId,
   479→                'debit' => 0, 'credit' => $cogsAmount,
   480→                'entity_type' => 'sales_challan', 'entity_id' => $challanId,
   481→                'memo' => 'Challan ' . $challanCode . ' — Inventory issued',
   482→            ],
   483→        ]);
   484→    }

   486→    /**
   487→     * Generate atomic challan code: CH-YYYYMMDD-NNNN.
   488→     */
   489→    private function generateChallanCode(): string
   490→    {
   491→        $datePart = now()->format('Ymd');
   492→        $periodKey = now()->format('Y-m');
   493→        $docType = 'sales_challan';
   494→
   495→        return DB::transaction(function () use ($docType, $periodKey, $datePart) {
   496→            $seqRow = DB::table('document_sequences')
   497→                ->where('doc_type', $docType)
   498→                ->where('branch_id', 0)
   499→                ->where('period_key', $periodKey)
   500→                ->lockForUpdate()
   501→                ->first();
   502→
   503→            $nextNumber = $seqRow ? ((int) $seqRow->last_number + 1) : 1;
   504→
   505→            if ($seqRow) {
   506→                DB::table('document_sequences')->where('id', $seqRow->id)
   507→                    ->update(['last_number' => $nextNumber, 'updated_at' => now()]);
   508→            } else {
   509→                DB::table('document_sequences')->insert([
   510→                    'doc_type' => $docType, 'branch_id' => 0,
   511→                    'period_key' => $periodKey, 'last_number' => $nextNumber,
   512→                    'updated_at' => now(),
   513→                ]);
   514→            }
   515→
   516→            return "CH-{$datePart}-" . str_pad((string) $nextNumber, 4, '0', STR_PAD_LEFT);
   517→        });
   518→    }

   554→    /**
   555→     * P2-3: Post customer_ledger 'invoice_adjustment' entry for transport delta.
   556→     *
   557→     * @param SalesInvoice $invoice
   558→     * @param float $adjustment Signed delta (positive = transport increased → customer owes more)
   559→     * @param string $challanCode
   560→     * @param int|null $createdBy
   561→     */
   562→    private function postTransportAdjustmentLedger($invoice, float $adjustment, string $challanCode, ?int $createdBy): void
   563→    {
   564→        $currentBalance = (float) DB::table('customer_ledger')
   565→            ->where('customer_id', $invoice->customer_id)
   566→            ->orderByDesc('id')
   567→            ->value('balance');
   568→
   569→        // Positive adjustment = customer owes more (debit); negative = owes less (credit).
   570→        $debit = $adjustment > 0 ? abs($adjustment) : 0;
   571→        $credit = $adjustment < 0 ? abs($adjustment) : 0;
   572→        $newBalance = $currentBalance + $adjustment;
   573→
   574→        DB::table('customer_ledger')->insert([
   575→            'customer_id' => $invoice->customer_id,
   576→            'branch_id' => $invoice->branch_id,
   577→            'transaction_date' => now()->format('Y-m-d'),
   578→            'transaction_type' => 'invoice_adjustment',
   579→            'reference_type' => 'sales_invoice',
   580→            'reference_id' => $invoice->id,
   581→            'debit' => $debit,
   582→            'credit' => $credit,
   583→            'balance' => $newBalance,
   584→            'description' => 'Challan ' . $challanCode . ' — transport adjustment ' . ($adjustment >= 0 ? '+' : '') . number_format($adjustment, 2),
   585→            'created_by' => $createdBy,
   586→            'created_at' => now(),
   587→        ]);
   588→    }

   590→    /**
   591→     * P2-3: Post GL adjustment journal for transport delta.
   592→     *
   593→     * If transport increased (positive adjustment):
   594→     *   Dr AR / Cr Transport Revenue
   595→     * If transport decreased (negative adjustment):
   596→     *   Dr Transport Revenue / Cr AR  (i.e., Cr AR / Dr Transport Revenue swapped)
   597→     *
   598→     * @return int|null journal_entry_id (null if no transport_revenue ledger configured)
   599→     */
   600→    private function postTransportAdjustmentGL(
   601→        int $invoiceId, string $challanCode, string $challanDate, int $branchId,
   602→        float $adjustment, int $customerId, ?int $createdBy
   603→    ): ?int {
   604→        $amount = abs($adjustment);
   605→        if ($amount < 0.01) {
   606→            return null;
   607→        }
   608→
   609→        $arLedgerId = $this->journalPosting->lookupLedgerByNature('ar');
   610→        if (!$arLedgerId) {
   611→            throw new \RuntimeException('Accounts Receivable ledger not found (nature: ar).');
   612→        }
   613→
   614→        $transportLedgerId = $this->journalPosting->lookupLedgerByNature('transport_revenue');
   615→        if (!$transportLedgerId) {
   616→            // Fall back to sales_revenue if transport_revenue not configured.
   617→            $transportLedgerId = $this->journalPosting->lookupLedgerByNature('sales_revenue');
   618→        }
   619→        if (!$transportLedgerId) {
   620→            throw new \RuntimeException('Transport Revenue / Sales Revenue ledger not found.');
   621→        }
   622→
   623→        // Positive adjustment: Dr AR / Cr Transport Revenue
   624→        // Negative adjustment: Dr Transport Revenue / Cr AR (swapped)
   625→        $isIncrease = $adjustment > 0;
   626→
   627→        $lines = [
   628→            [
   629→                'ledger_id' => $arLedgerId,
   630→                'debit' => $isIncrease ? $amount : 0,
   631→                'credit' => $isIncrease ? 0 : $amount,
   632→                'entity_type' => 'customer', 'entity_id' => $customerId,
   633→                'memo' => 'Challan ' . $challanCode . ' — transport adjustment AR',
   634→            ],
   635→            [
   636→                'ledger_id' => $transportLedgerId,
   637→                'debit' => $isIncrease ? 0 : $amount,
   638→                'credit' => $isIncrease ? $amount : 0,
   639→                'entity_type' => 'sales_invoice', 'entity_id' => $invoiceId,
   640→                'memo' => 'Challan ' . $challanCode . ' — transport adjustment revenue',
   641→            ],
   642→        ];
   643→
   644→        return $this->journalPosting->createJournalEntry([
   645→            'entry_date' => $challanDate,
   646→            'reference_type' => 'sales_invoice',
   647→            'reference_id' => $invoiceId,
   648→            'branch_id' => $branchId,
   649→            'description' => 'Transport adjustment — Challan ' . $challanCode,
   650→            'source' => 'sales_challan',
   651→            'created_by' => $createdBy,
   652→        ], $lines);
   653→    }
   654→}
