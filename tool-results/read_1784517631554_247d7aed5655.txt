   645→    public function deleteInvoice($invoice_id, ?string $deleteReason = null) { 
   646→
   647→    
   648→        $this->db->beginTransaction();
   649→        try {
   650→            // 1. Get invoice details
   651→            $this->db->query("SELECT * FROM sales_invoices WHERE id = :id");
   652→            $this->db->bind(':id', $invoice_id);
   653→            $invoice = $this->db->single();
   654→
   655→            if (!$invoice) {
   656→                throw new Exception("Invoice not found");
   657→            }
   658→
   659→            if ((int)($invoice['is_reversed'] ?? 0) === 1) {
   660→                throw new Exception('Invoice is already deleted.');
   661→            }
   662→
   663→            if ($invoice['status'] !== 'draft') {
   664→                throw new Exception(
   665→                    'Cannot delete invoice: status is "' . $invoice['status'] . '". '
   666→                    . 'Only draft invoices (before godown/challan) can be deleted.'
   667→                );
   668→            }
   669→
   670→            if (!empty($invoice['godown_issued_at'])) {
   671→                throw new Exception(
   672→                    'Cannot delete: godown has been prepared. Reverse or complete warehouse flow first.'
   673→                );
   674→            }
   675→
   676→            $this->assertInvoiceAccessible((int)$invoice['branch_id']);
   677→
   678→            if (!empty($invoice['journal_entry_id'])) {
   679→                require_once __DIR__ . '/../../Accounting/JournalPostingService.php';
   680→                $journalService = new JournalPostingService($this->db ?? null);
   681→                $rev = $journalService->reverseLinkedJournal(
   682→                    (int)$invoice['journal_entry_id'],
   683→                    'Invoice deleted: ' . ($invoice['invoice_code'] ?? $invoice_id)
   684→                );
   685→                if (($rev['status'] ?? '') === 'error') {
   686→                    throw new Exception('Failed to reverse invoice journal: ' . ($rev['message'] ?? ''));
   687→                }
   688→            }
   689→
   690→            $this->db->query("
   691→                SELECT id FROM sales_challans
   692→                WHERE sales_invoice_id = :id AND COALESCE(is_reversed, 0) = 0
   693→                LIMIT 1
   694→            ");
   695→            $this->db->bind(':id', $invoice_id);
   696→            if ($this->db->single()) {
   697→                throw new Exception('Cannot delete: a delivery challan exists. Reverse the challan first.');
   698→            }
   699→
   700→            $this->db->query("
   701→                SELECT COUNT(*) AS c FROM sales_invoice_dispatches
   702→                WHERE sales_invoice_id = :id AND COALESCE(dispatched_qty, 0) > 0
   703→            ");
   704→            $this->db->bind(':id', $invoice_id);
   705→            if ((int)($this->db->single()['c'] ?? 0) > 0) {
   706→                throw new Exception('Cannot delete: stock has already been dispatched for this invoice.');
   707→            }
   708→
   709→            // 2. Check if any payment has been made against this invoice
   710→            if ($this->invoiceHasPayments($invoice_id)) {
   711→                throw new Exception("Cannot delete invoice. Payments have already been received against it.");
   712→            }
   713→
   714→            // 3. Find and reverse the customer ledger entry
   715→            $this->db->query("SELECT id FROM customer_ledger 
   716→                              WHERE reference_type = 'invoice' 
   717→                                AND reference_id = :invoice_id");
   718→            $this->db->bind(':invoice_id', $invoice_id);
   719→            $ledger = $this->db->single();
   720→
   721→            if ($ledger) {
   722→                // Reverse the debit entry by creating a credit reversal
   723→                $this->db->query("SELECT running_balance FROM customer_ledger 
   724→                                  WHERE customer_id = :cid 
   725→                                  ORDER BY id DESC LIMIT 1");
   726→                $this->db->bind(':cid', $invoice['customer_id']);
   727→                $currentBalance = $this->db->single()['running_balance'] ?? 0;
   728→
   729→                $newBalance = $currentBalance - $invoice['total_amount'];
   730→
   731→                $this->insertCustomerLedgerEntry([
   732→                    'customer_id'      => (int)$invoice['customer_id'],
   733→                    'reference_type'   => 'reversal',
   734→                    'reference_id'     => $invoice_id,
   735→                    'debit'            => 0,
   736→                    'credit'           => (float)$invoice['total_amount'],
   737→                    'running_balance'  => $newBalance,
   738→                    'branch_id'        => (int)$invoice['branch_id'],
   739→                    'remarks'          => 'Reversed due to invoice deletion',
   740→                    'is_reversed'      => 1,
   741→                ]);
   742→            }
   743→
   744→            // 4. Delete invoice items      
   745→            $this->db->query("DELETE FROM sales_invoice_items WHERE sales_invoice_id = :id");
   746→            $this->db->bind(':id', $invoice_id);
   747→            $this->db->execute();
   748→
   749→            $reverseReason = ($deleteReason !== null && trim($deleteReason) !== '')
   750→                ? trim($deleteReason)
   751→                : 'Deleted by user';
   752→
   753→            // 5. Delete the invoice itself (soft delete using is_reversed)
   754→            $this->db->query("UPDATE sales_invoices 
   755→                              SET is_reversed = 1, 
   756→                                  reversed_at = NOW(), 
   757→                                  reversed_by = :reversed_by,
   758→                                  reverse_reason = :reverse_reason
   759→                              WHERE id = :id");
   760→            $this->db->bind(':id', $invoice_id);
   761→            $this->db->bind(':reversed_by', $_SESSION['user_id'] ?? 1);
   762→            $this->db->bind(':reverse_reason', $reverseReason);
   763→            $this->db->execute();
   764→
   765→            // 6. Remove soft dispatch reservations (invoice is reversed)
   766→            $this->db->query("DELETE FROM sales_invoice_dispatches WHERE sales_invoice_id = :id");
   767→            $this->db->bind(':id', $invoice_id);
   768→            $this->db->execute();
   769→
   770→            $this->db->commit();
   771→
   772→            return [
   773→                'status' => 'success',
   774→                'message' => 'Invoice #' . $invoice['invoice_code'] . ' has been deleted successfully.'
   775→            ];
   776→
   777→        } catch (Exception $e) {
   778→            $this->db->rollback();
   779→            return [
   780→                'status' => 'error',
   781→                'message' => $e->getMessage()
   782→            ];
   783→        }
   784→
   785→     }

   831→    private function buildTodayInvoiceWhere(array $filters, array &$bindings, ?string $dataTablesSearch = null): array
   832→    {
   833→        $where = [];
   834→        $where[] = "si.branch_id = :scope_branch_id";
   835→        $bindings[':scope_branch_id'] = self::sessionBranchId();
   836→
   837→        $where[] = "COALESCE(si.call_a_day, 0) = 0";
   838→
   839→        if (!$this->canSeeAllBranchInvoices()) {
   840→            $where[] = "si.created_by = :created_by_user";
   841→            $bindings[':created_by_user'] = (int)($_SESSION['user_id'] ?? 0);
   842→        }
   843→
   844→        $hasDateFilter = false;
   845→        if (!empty($filters['date_from'])) {
   846→            $where[] = "si.invoice_date >= :date_from";
   847→            $bindings[':date_from'] = $filters['date_from'];
   848→            $hasDateFilter = true;
   849→        }
   850→        if (!empty($filters['date_to'])) {
   851→            $where[] = "si.invoice_date <= :date_to";
   852→            $bindings[':date_to'] = $filters['date_to'];
   853→            $hasDateFilter = true;
   854→        }
   855→        if (!$hasDateFilter && empty($filters['skip_default_today'])) {
   856→            $where[] = "si.invoice_date = CURRENT_DATE";
   857→        }
   858→
   859→        $searchTerm = trim($dataTablesSearch ?? $filters['search'] ?? '');
   860→        if ($searchTerm !== '') {
   861→            $term = '%' . $searchTerm . '%';
   862→            $where[] = "(
   863→                si.invoice_code LIKE :t1
   864→                OR c.shop_name LIKE :t2
   865→                OR c.customer_name LIKE :t3
   866→                OR c.mobile LIKE :t4
   867→                OR b.branch_name LIKE :t5
   868→                OR e.name LIKE :t6
   869→                OR u.username LIKE :t7
   870→                OR EXISTS (
   871→                    SELECT 1 FROM sales_invoice_items sii
   872→                    JOIN products p ON sii.product_id = p.id
   873→                    WHERE sii.sales_invoice_id = si.id
   874→                    AND (p.product_name LIKE :t8 OR p.product_code LIKE :t9)
   875→                )
   876→            )";
   877→            for ($i = 1; $i <= 9; $i++) {
   878→                $bindings[":t$i"] = $term;
   879→            }
   880→        }
   881→
   882→        $this->applyTodayChallanStatusFilter((string)($filters['challan_status'] ?? 'all'), $where);
   883→
   884→        $where[] = "si.is_reversed = 0";
   885→        return $where;
   886→    }

   888→    private function applyTodayChallanStatusFilter(string $status, array &$where): void
   889→    {
   890→        $status = trim($status);
   891→        if ($status === '' || $status === 'all') {
   892→            return;
   893→        }
   894→
   895→        switch ($status) {
   896→            case 'open_pipeline':
   897→                $where[] = "si.status IN ('draft', 'godown_issued')";
   898→                return;
   899→            case 'pending':
   900→                $where[] = "si.status = 'draft'";
   901→                return;
   902→            case 'godown_copy':
   903→                $where[] = "si.status = 'godown_issued'";
   904→                return;
   905→            case 'challan_generated':
   906→                $where[] = "si.status = 'challan_completed'";
   907→                return;
   908→            case 'awaiting_payment':
   909→                $paid = $this->invoicePaidAmountSql();
   910→                $where[] = "(si.total_amount - {$paid}) > 0.009";
   911→                return;
   912→        }
   913→    }

   915→    /**
   916→     * Status counts for sales/today (date + search scope; ignores workflow chip filter).
   917→     */
   918→    public function getTodayFilterSummary(array $filters): array
   919→    {
   920→        $summaryFilters = $filters;
   921→        unset($summaryFilters['challan_status'], $summaryFilters['smart_sort']);
   922→        $summaryFilters['challan_status'] = 'all';
   923→        $summaryFilters['skip_default_today'] = true;
   924→
   925→        $search = trim($filters['search'] ?? '');
   926→        $bindings = [];
   927→        $where = $this->buildTodayInvoiceWhere(
   928→            $summaryFilters,
   929→            $bindings,
   930→            $search !== '' ? $search : null
   931→        );
   932→        $whereSql = empty($where) ? '' : ' WHERE ' . implode(' AND ', $where);
   933→
   934→        $this->db->query("
   935→            SELECT si.status, COUNT(*) AS cnt
   936→            " . self::TODAY_INVOICE_FROM . $whereSql . "
   937→            GROUP BY si.status
   938→        ");
   939→        foreach ($bindings as $param => $value) {
   940→            $this->db->bind($param, $value);
   941→        }
   942→        $rows = $this->db->resultSet();
   943→
   944→        $draft = 0;
   945→        $godown = 0;
   946→        $done = 0;
   947→        $awaitingPayment = 0;
   948→        $total = 0;
   949→        foreach ($rows as $row) {
   950→            $cnt = (int)($row['cnt'] ?? 0);
   951→            $total += $cnt;
   952→            switch ($row['status'] ?? '') {
   953→                case 'draft':
   954→                    $draft = $cnt;
   955→                    break;
   956→                case 'godown_issued':
   957→                    $godown = $cnt;
   958→                    break;
   959→                case 'challan_completed':
   960→                    $done = $cnt;
   961→                    break;
   962→            }
   963→        }
   964→
   965→        $paidSql = $this->invoicePaidAmountSql();
   966→        $awaitBindings = $bindings;
   967→        $awaitWhere = $this->buildTodayInvoiceWhere(
   968→            $summaryFilters,
   969→            $awaitBindings,
   970→            $search !== '' ? $search : null
   971→        );
   972→        $awaitWhere[] = "(si.total_amount - {$paidSql}) > 0.009";
   973→        $awaitWhereSql = ' WHERE ' . implode(' AND ', $awaitWhere);
   974→        $this->db->query('SELECT COUNT(*) AS c ' . self::TODAY_INVOICE_FROM . $awaitWhereSql);
   975→        foreach ($awaitBindings as $param => $value) {
   976→            $this->db->bind($param, $value);
   977→        }
   978→        $awaitingPayment = (int)($this->db->single()['c'] ?? 0);
   979→
   980→        return [
   981→            'total'             => $total,
   982→            'pending'           => $draft,
   983→            'godown_copy'       => $godown,
   984→            'challan_generated' => $done,
   985→            'open_pipeline'     => $draft + $godown,
   986→            'awaiting_payment'  => $awaitingPayment,
   987→        ];
   988→    }

  1190→    {

  1193→        $branchSql = '';

  1195→            $branchSql = ' AND si.branch_id = ' . (int)$branchId;

  1199→            SELECT si.id, si.invoice_code
  1200→            FROM sales_invoices si
  1201→            WHERE si.status = 'draft'
  1202→              AND COALESCE(si.is_reversed, 0) = 0

  1211→
  1212→        $cancelled = 0;

  1214→        $cancelReason = sprintf('Stale draft auto-cancel (>%d days)', $days);
