    59→    /**

    62→    public function runFullReport(?int $branchId = null, ?string $fromDate = null, ?string $toDate = null): array
    63→    {
    64→        $branchId = $branchId ?? self::sessionBranchId();
    65→        $branchFilter = ($branchId > 0) ? $branchId : null;
    66→        $fromDate = $fromDate ?: date('Y-m-01');
    67→        $toDate = $toDate ?: date('Y-m-d');
    68→
    69→        $ar = $this->buildArSection($branchFilter);
    70→        $ap = $this->buildApSection($branchFilter);
    71→        $employee = $this->buildEmployeeSection($branchFilter);
    72→        $cashBank = $this->buildCashBankSection($branchFilter);
    73→        $inventory = $this->buildInventorySection($branchFilter);
    74→        $cogs = $this->buildCogsSection($branchFilter, $fromDate, $toDate);
    75→
    76→        $issues = [];
    77→        if (!$ar['within_tolerance']) {
    78→            $issues[] = 'AR sub-ledger vs GL difference ' . number_format($ar['difference'], 2);
    79→        }
    80→        if ($ar['ledger_mismatch_count'] > 0) {
    81→            $issues[] = $ar['ledger_mismatch_count'] . ' customer ledger balance mismatch(es)';
    82→        }
    83→        if (!$ap['within_tolerance']) {
    84→            $issues[] = 'AP sub-ledger vs GL difference ' . number_format($ap['difference'], 2);
    85→        }
    86→        if ($ap['ledger_mismatch_count'] > 0) {
    87→            $issues[] = $ap['ledger_mismatch_count'] . ' supplier ledger balance mismatch(es)';
    88→        }
    89→        if (!$employee['within_tolerance']) {
    90→            $issues[] = 'Employee sub-ledger vs GL difference ' . number_format($employee['difference'], 2);
    91→        }
    92→        if ($employee['ledger_mismatch_count'] > 0) {
    93→            $issues[] = $employee['ledger_mismatch_count'] . ' employee ledger balance mismatch(es)';
    94→        }
    95→        if (!$cashBank['within_tolerance']) {
    96→            $issues[] = 'Cash/bank register vs GL difference ' . number_format($cashBank['difference'], 2);
    97→        }
    98→        if ($cashBank['mapping_mismatch_count'] > 0) {
    99→            $issues[] = $cashBank['mapping_mismatch_count'] . ' bank GL mapping mismatch(es)';
   100→        }
   101→        if (!$inventory['within_tolerance']) {
   102→            $issues[] = 'Inventory GL vs stock valuation difference ' . number_format($inventory['difference'], 2);
   103→        }
   104→        if (!$cogs['within_tolerance']) {
   105→            $issues[] = 'COGS GL vs stock OUT difference ' . number_format($cogs['difference'], 2) . ' (period)';
   106→        }
   107→
   108→        return [
   109→            'branch_id'   => $branchFilter,
   110→            'branch_name' => $this->resolveBranchName($branchFilter),
   111→            'from_date'   => $fromDate,
   112→            'to_date'     => $toDate,
   113→            'tolerance'   => $this->tolerance,
   114→            'ran_at'      => date('Y-m-d H:i:s'),
   115→            'ar'          => $ar,
   116→            'ap'          => $ap,
   117→            'employee'    => $employee,
   118→            'cash_bank'   => $cashBank,
   119→            'inventory'   => $inventory,
   120→            'cogs'        => $cogs,
   121→            'has_issues'  => $issues !== [],
   122→            'issues'      => $issues,
   123→        ];
   124→    }

   126→    private function buildArSection(?int $branchId): array
   127→    {
   128→        $customerNet = $this->sumCustomerLedgerNetBalances($branchId);
   129→        $glArNet = $this->sumGlArControlBalance($branchId);
   130→        $diff = round($customerNet - $glArNet, 2);
   131→        $mismatches = $this->getCustomerLedgerBalanceMismatches(
   132→            $this->tolerance,
   133→            $branchId,
   134→            self::PARTY_MISMATCH_LIMIT
   135→        );
   136→
   137→        return [
   138→            'branch_id'               => $branchId,
   139→            'branch_name'             => $this->resolveBranchName($branchId),
   140→            'customer_ledger_net'     => round($customerNet, 2),
   141→            'gl_ar_net'               => round($glArNet, 2),
   142→            'difference'              => $diff,
   143→            'within_tolerance'        => abs($diff) <= $this->tolerance,
   144→            'ledger_mismatch_count'   => count($mismatches),
   145→            'ledger_mismatches'       => $mismatches,
   146→            'null_branch_ledger_rows' => $this->countNullBranchLedgerRows($branchId),
   147→            'ar_ledger_ids'           => $this->getLedgerIdsByNature('customer_receivable'),
   148→        ];
   149→    }

   198→    private function buildCashBankSection(?int $branchId): array
   199→    {
   200→        $banksTotal = $this->sumActiveBankBalances();
   201→        $glCashNet = $this->sumGlControlByNature('cash_bank', $branchId);
   202→        $branchScoped = $branchId !== null && $branchId > 0;
   203→        $mappingMismatches = $this->getBankMappingMismatches(self::PARTY_MISMATCH_LIMIT);
   204→
   205→        $diff = $branchScoped ? 0.0 : round($banksTotal - $glCashNet, 2);
   206→        $withinTolerance = $branchScoped
   207→            ? ($mappingMismatches === [])
   208→            : (abs($diff) <= $this->tolerance && $mappingMismatches === []);
   209→
   210→        return [
   211→            'branch_id'               => $branchId,
   212→            'banks_total_balance'     => round($banksTotal, 2),
   213→            'gl_cash_bank_net'        => round($glCashNet, 2),
   214→            'difference'              => $diff,
   215→            'within_tolerance'        => $withinTolerance,
   216→            'mapping_mismatch_count'    => count($mappingMismatches),
   217→            'mapping_mismatches'      => $mappingMismatches,
   218→            'cash_bank_ledger_ids'    => $this->getLedgerIdsByNature('cash_bank'),
   219→            'branch_scoped_note'      => $branchScoped
   220→                ? 'Bank register is company-wide; GL cash_bank is branch-scoped — compare totals on combined (all branches) report.'
   221→                : null,
   222→        ];

   259→    private function sumWarehouseStockValue(?int $branchId): float
   260→    {
   261→        $branchSql = '';
   262→        if ($branchId !== null && $branchId > 0) {
   263→            $branchSql = ' AND w.branch_id = ' . (int)$branchId;
   264→        }
   265→
   266→        $this->db->query("
   267→            SELECT COALESCE(SUM(ws.qty * ws.avg_cost), 0) AS val
   268→            FROM warehouse_stock ws
   269→            INNER JOIN warehouses w ON w.id = ws.warehouse_id
   270→            WHERE w.is_active = 1 {$branchSql}
   271→        ");
   272→
   273→        return (float)($this->db->single()['val'] ?? 0);
   274→    }

   276→    private function sumGlControlByNature(string $nature, ?int $branchId): float
   277→    {
   278→        $ledgerIds = $this->getLedgerIdsByNature($nature);
   279→        if ($ledgerIds === []) {
   280→            return 0.0;
   281→        }
   282→
   283→        $idList = implode(',', array_map('intval', $ledgerIds));
   284→        $branchSql = '';
   285→        if ($branchId !== null && $branchId > 0) {
   286→            $branchSql = ' AND je.branch_id = ' . (int)$branchId;
   287→        }
   288→
   289→        $this->db->query("
   290→            SELECT
   291→                COALESCE(SUM(jl.debit), 0) AS total_debit,
   292→                COALESCE(SUM(jl.credit), 0) AS total_credit
   293→            FROM journal_lines jl
   294→            INNER JOIN journal_entries je ON je.id = jl.journal_entry_id
   295→            WHERE jl.ledger_id IN ({$idList})
   296→              AND COALESCE(je.is_reversed, 0) = 0
   297→              {$branchSql}
   298→        ");
   299→        $row = $this->db->single() ?: [];
   300→        $debit = (float)($row['total_debit'] ?? 0);
   301→        $credit = (float)($row['total_credit'] ?? 0);
   302→        $normal = self::NATURE_NORMAL[$nature] ?? 'debit';
   303→
   304→        return $normal === 'credit' ? ($credit - $debit) : ($debit - $credit);
   305→    }

   313→    private function sumChallanStockCogs(?int $branchId, string $fromDate, string $toDate): float
   314→    {
   315→        $branchSql = '';
   316→        if ($branchId !== null && $branchId > 0) {
   317→            $branchSql = ' AND si.branch_id = ' . (int)$branchId;
   318→        }
   319→
   320→        $this->db->query("
   321→            SELECT COALESCE(SUM(ABS(st.qty) * st.rate), 0) AS amt
   322→            FROM stock_transactions st
   323→            INNER JOIN sales_challans sc ON st.reference_type = 'sales_challan'
   324→                AND st.reference_id = sc.id
   325→            INNER JOIN sales_invoices si ON si.id = sc.sales_invoice_id
   326→            WHERE st.qty < -0.0001
   327→              AND COALESCE(sc.is_reversed, 0) = 0
   328→              AND sc.challan_date BETWEEN :from_d AND :to_d
   329→              {$branchSql}
   330→        ");
   331→        $this->db->bind(':from_d', $fromDate);
   332→        $this->db->bind(':to_d', $toDate);
   333→
   334→        return (float)($this->db->single()['amt'] ?? 0);
   335→    }

   379→    private function sumCustomerLedgerNetBalances(?int $branchId): float
   380→    {
   381→        $branchSql = '';
   382→        if ($branchId !== null && $branchId > 0) {
   383→            $branchSql = ' AND (cl.branch_id = ' . (int)$branchId . ' OR cl.branch_id IS NULL)';
   384→        }
   385→
   386→        $this->db->query("
   387→            SELECT COALESCE(SUM(lb.last_balance), 0) AS total
   388→            FROM (
   389→                SELECT cl1.customer_id, cl1.running_balance AS last_balance
   390→                FROM customer_ledger cl1
   391→                INNER JOIN (
   392→                    SELECT customer_id, MAX(id) AS max_id
   393→                    FROM customer_ledger cl
   394→                    WHERE COALESCE(cl.is_reversed, 0) = 0 {$branchSql}
   395→                    GROUP BY customer_id
   396→                ) latest ON cl1.id = latest.max_id
   397→                WHERE COALESCE(cl1.is_reversed, 0) = 0
   398→            ) lb
   399→        ");
   400→
   401→        return (float)($this->db->single()['total'] ?? 0);
   402→    }

   466→    /**
   467→     * @return array<int, array<string, mixed>>
   468→     */
   469→    private function getBankMappingMismatches(int $limit): array
   470→    {
   471→        $limit = max(1, min(100, $limit));
   472→
   473→        try {
   474→            $this->db->query("SELECT 1 FROM information_schema.tables WHERE table_schema = current_schema() AND table_name = 'bank_ledger_mappings'");
   475→            if (!$this->db->single()) {
   476→                return [];
   477→            }
   478→        } catch (Exception $e) {
   479→            return [];
   480→        }
   481→
   482→        $this->db->query("
   483→            SELECT
   484→                b.id AS bank_id,
   485→                b.bank_name,
   486→                COALESCE(b.balance, 0) AS bank_balance,
   487→                blm.ledger_id AS mapped_ledger_id,
   488→                l.ledger_name AS mapped_ledger_name,
   489→                COALESCE(gl.net_bal, 0) AS gl_net,
   490→                CASE
   491→                    WHEN blm.ledger_id IS NULL THEN ABS(COALESCE(b.balance, 0))
   492→                    ELSE ABS(COALESCE(b.balance, 0) - COALESCE(gl.net_bal, 0))
   493→                END AS difference,
   494→                CASE WHEN blm.ledger_id IS NULL THEN 1 ELSE 0 END AS is_unmapped
   495→            FROM banks b
   496→            LEFT JOIN bank_ledger_mappings blm ON blm.bank_id = b.id
   497→            LEFT JOIN ledgers l ON l.id = blm.ledger_id
   498→            LEFT JOIN (
   499→                SELECT jl.ledger_id,
   500→                       SUM(jl.debit) - SUM(jl.credit) AS net_bal
   501→                FROM journal_lines jl
   502→                INNER JOIN journal_entries je ON je.id = jl.journal_entry_id
   503→                WHERE COALESCE(je.is_reversed, 0) = 0
   504→                GROUP BY jl.ledger_id
   505→            ) gl ON gl.ledger_id = blm.ledger_id
   506→            WHERE COALESCE(b.is_active, 1) = 1
   507→            HAVING difference > :tol OR is_unmapped = 1
   508→            ORDER BY difference DESC
   509→            LIMIT {$limit}
   510→        ");
   511→        $this->db->bind(':tol', $this->tolerance);
   512→
   513→        return $this->db->resultSet() ?: [];
   514→    }

   558→    /**
   559→     * Run reconciliation for all active branches (or one branch / all combined).
   560→     *
   561→     * @return array<int, array>
   562→     */
   563→    public function runScheduledReconciliation(?int $branchFilter = null, ?string $fromDate = null, ?string $toDate = null): array
   564→    {
   565→        $fromDate = $fromDate ?: date('Y-m-01');
   566→        $toDate = $toDate ?: date('Y-m-d');
   567→        $reports = [];
   568→
   569→        if ($branchFilter !== null && $branchFilter > 0) {
   570→            $reports[] = $this->runFullReport($branchFilter, $fromDate, $toDate);
   571→            $this->emitAlertsForReport($reports[0]);
   572→            return $reports;
   573→        }
   574→
   575→        $reports[] = $this->runFullReport(null, $fromDate, $toDate);
   576→
   577→        $this->db->query('SELECT id FROM branches WHERE COALESCE(is_active, 1) = 1 ORDER BY id ASC');
   578→        foreach ($this->db->resultSet() as $row) {
   579→            $bid = (int)$row['id'];
   580→            if ($bid <= 0) {
   581→                continue;
   582→            }
   583→            $reports[] = $this->runFullReport($bid, $fromDate, $toDate);
   584→        }
   585→
   586→        foreach ($reports as $report) {
   587→            $this->emitAlertsForReport($report);
   588→        }
   589→
   590→        return $reports;
   591→    }

   633→    /**
   634→     * Write alert when audit/reconciliation finds failures (Phase 5.4).

   636→    public static function writeAlert(string $message, array $context = []): void
   637→    {
   638→        $path = self::alertLogPath();
   639→        $logDir = dirname($path);
   640→        if (!is_dir($logDir)) {
   641→            @mkdir($logDir, 0755, true);

   645→            'timestamp' => date('Y-m-d H:i:s'),

   648→        ], JSON_UNESCAPED_UNICODE);

   651→    }

   663→    /**
   664→     * Read the most recent alert log entries (newest first) for UI / CLI review.
   665→     *
   666→     * @return array<int, array{timestamp: string, message: string, context: array<string, mixed>}>
   667→     */
   668→    public static function readRecentAlerts(int $limit = 30): array
   669→    {
   670→        $limit = max(1, min(200, $limit));
   671→        $path = self::alertLogPath();
   672→        if (!is_readable($path)) {
   673→            return [];
   674→        }
   675→
   676→        $lines = @file($path, FILE_IGNORE_NEW_LINES | FILE_SKIP_EMPTY_LINES);
   677→        if ($lines === false || $lines === []) {
   678→            return [];
   679→        }
   680→
   681→        $entries = [];
   682→        for ($i = count($lines) - 1; $i >= 0 && count($entries) < $limit; $i--) {
   683→            $decoded = json_decode($lines[$i], true);
   684→            if (!is_array($decoded)) {
   685→                continue;
   686→            }
   687→
   688→            $entries[] = [
   689→                'timestamp' => (string)($decoded['timestamp'] ?? ''),
   690→                'message'   => (string)($decoded['message'] ?? ''),
   691→                'context'   => is_array($decoded['context'] ?? null) ? $decoded['context'] : [],
   692→            ];
   693→        }
   694→
   695→        return $entries;
   696→    }

   730→    /**
   731→     * @return 'ok'|'warn'|'fail'
   732→     */
   733→    public static function sectionStatus(string $key, array $report): string
   734→    {
   735→        switch ($key) {
   736→            case 'ar':
   737→                $s = $report['ar'] ?? [];
   738→                if (empty($s['within_tolerance'])) {
   739→                    return 'fail';
   740→                }
   741→                if (((int)($s['ledger_mismatch_count'] ?? 0)) > 0 || ((int)($s['null_branch_ledger_rows'] ?? 0)) > 0) {
   742→                    return 'warn';
   743→                }
   744→                return 'ok';
   745→            case 'ap':
   746→                $s = $report['ap'] ?? [];
   747→                if (empty($s['within_tolerance'])) {
   748→                    return 'fail';
   749→                }
   750→                if (((int)($s['ledger_mismatch_count'] ?? 0)) > 0) {
   751→                    return 'warn';
   752→                }
   753→                return 'ok';
   754→            case 'employee':
   755→                $s = $report['employee'] ?? [];
   756→                if (empty($s['within_tolerance'])) {
   757→                    return 'fail';
   758→                }
   759→                if (((int)($s['ledger_mismatch_count'] ?? 0)) > 0) {
   760→                    return 'warn';
   761→                }
   762→                return 'ok';
   763→            case 'cash_bank':
   764→                $s = $report['cash_bank'] ?? [];
   765→                if (((int)($s['mapping_mismatch_count'] ?? 0)) > 0) {
   766→                    return empty($s['within_tolerance']) ? 'fail' : 'warn';
   767→                }
   768→                return !empty($s['within_tolerance']) ? 'ok' : 'fail';
   769→            case 'inventory':
   770→                return !empty(($report['inventory'] ?? [])['within_tolerance']) ? 'ok' : 'fail';
   771→            case 'cogs':
   772→                return !empty(($report['cogs'] ?? [])['within_tolerance']) ? 'ok' : 'fail';
   773→            default:
   774→                return 'ok';
   775→        }
   776→    }

   791→    public static function overallStatus(array $report): string
