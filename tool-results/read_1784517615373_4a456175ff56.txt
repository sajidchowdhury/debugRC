    34→    34→    34→     * @return array<int, array{item_id:int, product_id:int, product_name:string, warehouse_id:int, demand_qty:float, dispatched_ctn:float}>

    36→    36→    36→    protected function resolveDispatchLinesForInvoice(

    41→    41→    41→    ): array

    52→    52→    52→                continue;
    53→    53→    53→            }

    56→    56→    56→                SELECT sii.id, sii.product_id, sii.qty, p.product_name
    57→    57→    57→                FROM sales_invoice_items sii

    59→    59→    59→                WHERE sii.id = :iid AND sii.sales_invoice_id = :inv
    60→    60→    60→            ");

    62→    62→    62→            $this->db->bind(':inv', $invoiceId);
    63→    63→    63→            $row = $this->db->single();
    64→    64→    64→            if (!$row) {
    65→    65→    65→                throw new Exception(
    66→    66→    66→                    'Invoice lines have changed. Refresh this page and try again.'

    71→    71→    71→            if ($demandQty <= 0) {

    73→    73→    73→            }

    77→    77→    77→                throw new Exception(
    78→    78→    78→                    'Dispatched quantity must match invoice demand ('

    81→    81→    81→                );
    82→    82→    82→            }
    83→    83→    83→
    84→    84→    84→            $wid = (int)($data['warehouse_id'][$i] ?? 0);
    85→    85→    85→            if ($wid <= 0) {
    86→    86→    86→                throw new Exception(
    87→    87→    87→                    'Select a warehouse for ' . ($row['product_name'] ?? 'each product') . '.'
    88→    88→    88→                );
    89→    89→    89→            }
    90→    90→    90→
    91→    91→    91→            $sessionBranchId = self::sessionBranchId();
    92→    92→    92→            if (!$this->warehouseBelongsToBranch($wid, $sessionBranchId)) {

    94→    94→    94→            }
    95→    95→    95→
    96→    96→    96→            $productId = (int)$row['product_id'];
    97→    97→    97→            $productName = (string)($row['product_name'] ?? 'product');
    98→    98→    98→
    99→    99→    99→            if ($validationMode === 'finalize') {
   100→   100→   100→                $this->db->query("
   101→   101→   101→                    SELECT warehouse_id, ordered_qty
   102→   102→   102→                    FROM sales_invoice_dispatches
   103→   103→   103→                    WHERE sales_invoice_id = :inv AND product_id = :pid
   104→   104→   104→                    LIMIT 1
   105→   105→   105→                ");

   110→   110→   110→                    throw new Exception(
   111→   111→   111→                        'Godown setup missing for ' . $productName . '. Save godown copy again.'
   112→   112→   112→                    );
   113→   113→   113→                }
   114→   114→   114→
   115→   115→   115→                $godownWh = (int)$godownRow['warehouse_id'];
   116→   116→   116→                if ($godownWh <= 0) {
   117→   117→   117→                    throw new Exception('Godown warehouse not set for ' . $productName . '.');
   118→   118→   118→                }
   119→   119→   119→                if ($wid !== $godownWh) {
   120→   120→   120→                    throw new Exception(
   121→   121→   121→                        'Warehouse for ' . $productName . ' was fixed at godown and cannot be changed.'
   122→   122→   122→                    );
   123→   123→   123→                }
   124→   124→   124→                $wid = $godownWh;
   125→   125→   125→
   126→   126→   126→                $godownOrdered = (float)$godownRow['ordered_qty'];

   129→   129→   129→                        'Invoice quantity changed since godown was saved. Refresh and try again.'
   130→   130→   130→                    );

   134→   134→   134→                if ($demandQty > $physical + 0.0001) {
   135→   135→   135→                    throw new Exception(
   136→   136→   136→                        'Not enough physical stock for ' . $productName
   137→   137→   137→                        . ' in the selected warehouse (need '
   138→   138→   138→                        . number_format($demandQty, 2) . ', on hand '
   139→   139→   139→                        . number_format($physical, 2)
   140→   140→   140→                        . '). Receive stock or reverse and re-assign godown.'
   141→   141→   141→                    );
   142→   142→   142→                }
   143→   143→   143→            } else {
   144→   144→   144→                $available = $this->Get_Warehouse_Available_Stock($productId, $wid, $invoiceId);
   145→   145→   145→                if ($demandQty > $available + 0.0001) {
   146→   146→   146→                    throw new Exception(
   147→   147→   147→                        'Insufficient stock for ' . $productName
   148→   148→   148→                        . '. Invoice demand: ' . number_format($demandQty, 2)
   149→   149→   149→                        . ', available in warehouse: ' . number_format($available, 2)
   150→   150→   150→                        . '. Choose another warehouse or receive stock first.'
   151→   151→   151→                    );
   152→   152→   152→                }
   153→   153→   153→            }
   154→   154→   154→
   155→   155→   155→            $lines[] = [
   156→   156→   156→                'item_id'         => $itemId,
   157→   157→   157→                'product_id'      => (int)$row['product_id'],
   158→   158→   158→                'product_name'    => (string)$row['product_name'],
   159→   159→   159→                'warehouse_id'    => $wid,
   160→   160→   160→                'demand_qty'      => $demandQty,
   161→   161→   161→                'dispatched_ctn'  => (float)($data['dispatched_ctn'][$i] ?? 0),
   162→   162→   162→            ];
   163→   163→   163→        }
   164→   164→   164→
   165→   165→   165→        if ($lines === []) {
   166→   166→   166→            throw new Exception('No invoice lines to dispatch.');
   167→   167→   167→        }
   168→   168→   168→
   169→   169→   169→        return $lines;
   170→   170→   170→    }

   210→   210→   210→    public function prepareGodown($data) {
   211→   211→   211→        $this->db->beginTransaction();
   212→   212→   212→        try {
   213→   213→   213→            $invoice_id = (int)$data['invoice_id'];
   214→   214→   214→
   215→   215→   215→            $this->db->query("
   216→   216→   216→                SELECT status, branch_id, customer_id, subtotal, discount, transport_cost, total_amount
   217→   217→   217→                FROM sales_invoices
   218→   218→   218→                WHERE id = :id AND is_reversed = 0
   219→   219→   219→                FOR UPDATE
   220→   220→   220→            ");
   221→   221→   221→            $this->db->bind(':id', $invoice_id);
   222→   222→   222→            $invoiceRow = $this->db->single();
   223→   223→   223→
   224→   224→   224→            if (!$invoiceRow) {
   225→   225→   225→                throw new Exception('Invoice not found.');
   226→   226→   226→            }
   227→   227→   227→
   228→   228→   228→            $this->assertInvoiceAccessible((int)$invoiceRow['branch_id']);
   229→   229→   229→            $invoiceBranchId = (int)$invoiceRow['branch_id'];
   230→   230→   230→
   231→   231→   231→            $status = $invoiceRow['status'] ?? '';
   232→   232→   232→            if (!in_array($status, ['draft', 'godown_issued'], true)) {
   233→   233→   233→                throw new Exception('Godown can only be prepared for draft or godown-issued invoices.');
   234→   234→   234→            }
   235→   235→   235→
   236→   236→   236→            $transportResult = $this->persistInvoiceTransportCost(
   237→   237→   237→                $data,
   238→   238→   238→                $invoiceRow,
   239→   239→   239→                $invoice_id,
   240→   240→   240→                'Transport/total updated at godown save'
   241→   241→   241→            );
   242→   242→   242→
   243→   243→   243→            $dispatchMode = ($status === 'godown_issued') ? 'finalize' : 'godown';
   244→   244→   244→            $dispatchLines = $this->resolveDispatchLinesForInvoice(
   245→   245→   245→                $data,
   246→   246→   246→                $invoice_id,
   247→   247→   247→                $invoiceBranchId,
   248→   248→   248→                $dispatchMode
   249→   249→   249→            );
   250→   250→   250→
   251→   251→   251→            foreach ($dispatchLines as $line) {
   252→   252→   252→                $this->db->query("UPDATE sales_invoice_items SET warehouse_id = :wid WHERE id = :id");
   253→   253→   253→                $this->db->bind(':wid', $line['warehouse_id']);
   254→   254→   254→                $this->db->bind(':id', $line['item_id']);
   255→   255→   255→                $this->db->execute();
   256→   256→   256→            }
   257→   257→   257→
   258→   258→   258→            // 2. Save dispatchers (junction table) - Clear & Insert
   259→   259→   259→            $this->db->query("DELETE FROM sales_invoice_dispatchers WHERE sales_invoice_id = :iid");
   260→   260→   260→            $this->db->bind(':iid', $invoice_id);
   261→   261→   261→            $this->db->execute();
   262→   262→   262→
   263→   263→   263→            if (!empty($data['dispatcher_id']) && is_array($data['dispatcher_id'])) {
   264→   264→   264→                foreach ($data['dispatcher_id'] as $emp_id) {
   265→   265→   265→                    if (!empty($emp_id)) {
   266→   266→   266→                        $this->db->query("INSERT INTO sales_invoice_dispatchers (sales_invoice_id, employee_id) 
   267→   267→   267→                                        VALUES (:iid, :eid)");
   268→   268→   268→                        $this->db->bind(':iid', $invoice_id);
   269→   269→   269→                        $this->db->bind(':eid', (int)$emp_id);
   270→   270→   270→                        $this->db->execute();
   271→   271→   271→                    }
   272→   272→   272→                }
   273→   273→   273→            }
   274→   274→   274→
   275→   275→   275→            // 3. Clear previous dispatches and INSERT fresh (This solves your duplicate problem)
   276→   276→   276→            $this->db->query("DELETE FROM sales_invoice_dispatches WHERE sales_invoice_id = :iid");
   277→   277→   277→            $this->db->bind(':iid', $invoice_id);
   278→   278→   278→            $this->db->execute();
   279→   279→   279→
   280→   280→   280→            foreach ($dispatchLines as $line) {
   281→   281→   281→                $this->db->query("
   282→   282→   282→                    INSERT INTO sales_invoice_dispatches
   283→   283→   283→                    (sales_invoice_id, product_id, warehouse_id, ordered_qty, dispatched_qty, dispatched_ctn)
   284→   284→   284→                    VALUES (:iid, :pid, :wid, :oqty, 0, :ctn)
   285→   285→   285→                ");
   286→   286→   286→                $this->db->bind(':iid', $invoice_id);
   287→   287→   287→                $this->db->bind(':pid', $line['product_id']);
   288→   288→   288→                $this->db->bind(':wid', $line['warehouse_id']);
   289→   289→   289→                $this->db->bind(':oqty', $line['demand_qty']);
   290→   290→   290→                $this->db->bind(':ctn', $line['dispatched_ctn']);
   291→   291→   291→                $this->db->execute();
   292→   292→   292→            }
   293→   293→   293→
   294→   294→   294→            // 4. Update invoice status
   295→   295→   295→            $this->db->query("
   296→   296→   296→                UPDATE sales_invoices
   297→   297→   297→                SET status = 'godown_issued',
   298→   298→   298→                    godown_issued_at = NOW()
   299→   299→   299→                WHERE id = :id
   300→   300→   300→            ");
   301→   301→   301→            $this->db->bind(':id', $invoice_id);
   302→   302→   302→            $this->db->execute();
   303→   303→   303→
   304→   304→   304→            $this->db->commit();
   305→   305→   305→
   306→   306→   306→            $message = 'Godown prepared successfully!';
   307→   307→   307→            if (abs($transportResult['total_diff']) > 0.0001) {
   308→   308→   308→                $message .= ' Invoice total updated to Tk '
   309→   309→   309→                    . number_format($transportResult['new_total'], 2) . '.';
   310→   310→   310→            }
   311→   311→   311→
   312→   312→   312→            return [
   313→   313→   313→                'status'       => 'success',
   314→   314→   314→                'message'      => $message,
   315→   315→   315→                'new_total'    => $transportResult['new_total'],
   316→   316→   316→                'transport'    => $transportResult['transport'],
   317→   317→   317→                'total_diff'   => $transportResult['total_diff'],
   318→   318→   318→            ];
   319→   319→   319→
   320→   320→   320→        } catch (Exception $e) {
   321→   321→   321→            $this->db->rollback();
   322→   322→   322→            return ['status' => 'error', 'message' => $e->getMessage()];
   323→   323→   323→        }
   324→   324→   324→    }

   327→   327→   327→    public function finalizeChallan($data) {
   328→   328→   328→        $this->db->beginTransaction();
   329→   329→   329→        try {
   330→   330→   330→            $invoice_id = (int)$data['invoice_id'];
   331→   331→   331→
   332→   332→   332→            // Double-finalization protection + branch scope
   333→   333→   333→            $this->db->query("
   334→   334→   334→                SELECT status, branch_id, customer_id, subtotal, discount, transport_cost, total_amount, invoice_code
   335→   335→   335→                FROM sales_invoices WHERE id = :id AND is_reversed = 0 FOR UPDATE
   336→   336→   336→            ");
   337→   337→   337→            $this->db->bind(':id', $invoice_id);
   338→   338→   338→            $invoiceRow = $this->db->single();
   339→   339→   339→            if (!$invoiceRow) {
   340→   340→   340→                throw new Exception('Invoice not found.');
   341→   341→   341→            }
   342→   342→   342→            $this->assertInvoiceAccessible((int)$invoiceRow['branch_id']);
   343→   343→   343→            $invoiceBranchId = (int)$invoiceRow['branch_id'];
   344→   344→   344→            $customerId = (int)$invoiceRow['customer_id'];
   345→   345→   345→            if (($invoiceRow['status'] ?? '') === 'challan_completed') {
   346→   346→   346→                throw new Exception("Challan already completed.");
   347→   347→   347→            }
   348→   348→   348→
   349→   349→   349→            $this->assertGodownPreparedForChallan($invoice_id);
   350→   350→   350→            $dispatchLines = $this->resolveDispatchLinesForInvoice(
   351→   351→   351→                $data,
   352→   352→   352→                $invoice_id,
   353→   353→   353→                $invoiceBranchId,
   354→   354→   354→                'finalize'
   355→   355→   355→            );
   356→   356→   356→
   357→   357→   357→            $challan_code = $this->generateSalesChallanCode();
   358→   358→   358→            $oldTransport = (float)$invoiceRow['transport_cost'];
   359→   359→   359→            $oldTotal = (float)$invoiceRow['total_amount'];
   360→   360→   360→
   361→   361→   361→            $transportResult = $this->persistInvoiceTransportCost(
   362→   362→   362→                $data,
   363→   363→   363→                $invoiceRow,
   364→   364→   364→                $invoice_id,
   365→   365→   365→                'Transport/total adjustment on challan #' . $challan_code
   366→   366→   366→            );
   367→   367→   367→            $newTotal = $transportResult['new_total'];
   368→   368→   368→            $totalDiff = $transportResult['total_diff'];
   369→   369→   369→
   370→   370→   370→            // Update Invoice status (transport/total already persisted above)
   371→   371→   371→            if ($this->invoiceHasPreChallanColumns()) {
   372→   372→   372→                $this->db->query("
   373→   373→   373→                    UPDATE sales_invoices
   374→   374→   374→                    SET status = 'challan_completed',
   375→   375→   375→                        challan_completed_at = NOW(),
   376→   376→   376→                        godown_issued_at = COALESCE(godown_issued_at, NOW()),
   377→   377→   377→                        pre_challan_transport = :pre_tc,
   378→   378→   378→                        pre_challan_total = :pre_total
   379→   379→   379→                    WHERE id = :id
   380→   380→   380→                ");
   381→   381→   381→                $this->db->bind(':pre_tc', $oldTransport);
   382→   382→   382→                $this->db->bind(':pre_total', $oldTotal);
   383→   383→   383→            } else {
   384→   384→   384→                $this->db->query("
   385→   385→   385→                    UPDATE sales_invoices
   386→   386→   386→                    SET status = 'challan_completed',
   387→   387→   387→                        challan_completed_at = NOW(),
   388→   388→   388→                        godown_issued_at = COALESCE(godown_issued_at, NOW())
   389→   389→   389→                    WHERE id = :id
   390→   390→   390→                ");
   391→   391→   391→            }

   522→   522→   522→    /**

   653→   653→   653→            ");
   654→   654→   654→            $this->db->bind(':uid', $userId);
   655→   655→   655→            $this->db->bind(':reason', $reason);
   656→   656→   656→            $this->db->bind(':id', $challanId);
   657→   657→   657→            $this->db->execute();
   658→   658→   658→

   714→   714→   714→    /**

   716→   716→   716→     */
   717→   717→   717→    protected function restoreStockFromChallanIssue(
   718→   718→   718→        int $challanId,
   719→   719→   719→        string $challanCode,
   720→   720→   720→        int $invoiceId,
   721→   721→   721→        int $branchId,

   733→   733→   733→                continue;
   734→   734→   734→            }
   735→   735→   735→

   748→   748→   748→            );
   749→   749→   749→
   750→   750→   750→            $this->stock->updateWarehouseStock($wid, $pid, $qty, $issueRate);
   751→   751→   751→            $this->stock->logMovement([
   752→   752→   752→                'product_id'     => $pid,
   753→   753→   753→                'warehouse_id'   => $wid,
   754→   754→   754→                'qty'            => $qty,
   755→   755→   755→                'rate'           => $issueRate,
   756→   756→   756→                'reference_type' => 'sales_challan_reversal',
   757→   757→   757→                'reference_id'   => $challanId,
   758→   758→   758→                'remarks'        => "Reversal of Challan #{$challanCode}: {$reason}",
   759→   759→   759→            ]);
