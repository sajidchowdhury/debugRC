    13→/**
    14→ * Sales Invoice Service — Phase 8.2.
    15→ *
    16→ * Finalizes a cart into a draft sales invoice:
    17→ *   1. Validate cart (stock availability + price range)
    18→ *   2. Check credit limit (with optional override)
    19→ *   3. Lock branch products FOR UPDATE (prevent race conditions)
    20→ *   4. Create sales_invoice (status=draft)
    21→ *   5. Create sales_invoice_items (from cart, warehouse_id=NULL)
    22→ *   6. Create sales_invoice_dispatches (ordered_qty, dispatched_qty=0)
    23→ *   7. Post customer_ledger debit (customer owes more)
    24→ *   8. Post GL: Dr Accounts Receivable / Cr Sales Revenue
    25→ *      (+ Dr Discount / Cr Transport if applicable)
    26→ *   9. Clear the cart
    27→ *
    28→ * The invoice is a DRAFT at this point — no stock movement yet.
    29→ * Stock moves on challan issue (Phase 8.3).
    30→ */
    31→class SalesInvoiceService
    32→{
    33→    public function __construct(
    34→        private SalesCartService $cartService,
    35→        private StockAvailabilityService $availabilityService,
    36→        private StockService $stockService,
    37→        private JournalPostingService $journalPosting,
    38→        private SalesAccess $salesAccess,
    39→        private SalesAuditLogger $auditLogger
    40→    ) {}
    41→
    42→    /**
    43→     * Finalize a cart into a draft sales invoice.
    44→     *
    45→     * @param array $data {
    46→     *     customer_id: int,
    47→     *     branch_id: int,
    48→     *     invoice_date: string (Y-m-d),
    49→     *     salesman_id: int|null,
    50→     *     sales_person: string|null,
    51→     *     discount_amount: float,
    52→     *     transport_cost: float,
    53→     *     notes: string|null,
    54→     *     is_soft_hold: bool,
    55→     *     credit_limit_override: bool,
    56→     *     override_reason: string|null,
    57→     *     created_by: int,
    58→     * }
    59→     * @return SalesInvoice
    60→     * @throws \RuntimeException If cart empty, stock insufficient, credit limit exceeded without override, or GL posting fails.
    61→     */
    62→    public function finalizeFromCart(array $data): SalesInvoice
    63→    {
    64→        $customerId = (int) $data['customer_id'];
    65→        $branchId = (int) $data['branch_id'];
    66→
    67→        if ($customerId <= 0) {
    68→            throw new \InvalidArgumentException('customer_id is required.');
    69→        }
    70→        if ($branchId <= 0) {
    71→            throw new \InvalidArgumentException('branch_id is required.');
    72→        }
    73→
    74→        // P0-8: Defense-in-depth branch isolation check.
    75→        $this->salesAccess->assertBranchAccessible($branchId);
    76→
    77→        // Step 1: Load + validate the cart.
    78→        $cartData = $this->cartService->getCart($data['created_by'] ?? auth()->id(), $customerId, $branchId);
    79→        $items = $cartData['items'];
    80→
    81→        if (empty($items)) {
    82→            throw new \RuntimeException('Cart is empty. Add products before finalizing.');
    83→        }
    84→
    85→        $validation = $cartData['validation'];
    86→        if (!$validation['valid']) {
    87→            throw new \RuntimeException('Cart validation failed: ' . $validation['message']);
    88→        }
    89→
    90→        // Step 2: Calculate totals.
    91→        $subTotal = collect($items)->sum('total');
    92→        $discount = (float) ($data['discount_amount'] ?? 0);
    93→        $transport = (float) ($data['transport_cost'] ?? 0);
    94→        $totalAmount = $subTotal + $transport - $discount;
    95→
    96→        // Step 3: Credit limit check.
    97→        $creditCheck = $this->checkCreditLimit($customerId, $totalAmount);
    98→        $isOverride = !empty($data['credit_limit_override']);
    99→        $overrideReason = trim($data['override_reason'] ?? '');
   100→
   101→        if ($creditCheck['exceeds'] && !$isOverride) {
   102→            throw new \RuntimeException(
   103→                "Credit limit exceeded. Customer balance: {$creditCheck['current_balance']}, "
   104→                . "Credit limit: {$creditCheck['credit_limit']}, "
   105→                . "This invoice: {$totalAmount}. "
   106→                . "Override with a reason to proceed."
   107→            );
   108→        }
   109→
   110→        if ($creditCheck['exceeds'] && $isOverride && strlen($overrideReason) < 10) {
   111→            throw new \RuntimeException('Override reason must be at least 10 characters when exceeding credit limit.');
   112→        }
   113→
   114→        // Step 4: Atomic finalize in a DB transaction.
   115→        return DB::transaction(function () use (
   116→            $data, $items, $customerId, $branchId, $subTotal, $discount, $transport, $totalAmount,
   117→            $creditCheck, $isOverride, $overrideReason
   118→        ) {
   119→            // Lock branch products FOR UPDATE (prevent race conditions on concurrent finalizes).
   120→            $productIds = collect($items)->pluck('product_id')->unique()->toArray();
   121→            $this->stockService->lockBranchProductsForUpdate($branchId, $productIds);
   122→
   123→            // Re-check availability after locking.
   124→            $qtyByProduct = [];
   125→            foreach ($items as $item) {
   126→                $pid = (int) $item['product_id'];
   127→                $qtyByProduct[$pid] = ($qtyByProduct[$pid] ?? 0) + (float) $item['qty'];
   128→            }
   129→
   130→            foreach ($qtyByProduct as $pid => $qty) {
   131→                $available = $this->availabilityService->getBranchAvailableQty($pid, $branchId);
   132→                if ($qty > $available + 0.0001) {
   133→                    throw new \RuntimeException(
   134→                        "Insufficient stock for product {$pid}: requested {$qty}, available {$available}"
   135→                    );
   136→                }
   137→            }
   138→
   139→            // Generate invoice code.
   140→            $invoiceCode = $this->generateInvoiceCode();
   141→
   142→            // Step 5: Create sales_invoice header.
   143→            $invoiceId = DB::table('sales_invoices')->insertGetId([
   144→                'invoice_code' => $invoiceCode,
   145→                'invoice_date' => $data['invoice_date'] ?? now()->format('Y-m-d'),
   146→                'customer_id' => $customerId,
   147→                'salesman_id' => $data['salesman_id'] ?? null,
   148→                'sales_person' => $data['sales_person'] ?? null,
   149→                'branch_id' => $branchId,
   150→                'sub_total' => round($subTotal, 2),
   151→                'discount_amount' => round($discount, 2),
   152→                'transport_cost' => round($transport, 2),
   153→                'total_amount' => round($totalAmount, 2),
   154→                'paid_amount' => 0,
   155→                'due_amount' => round($totalAmount, 2),
   156→                'payment_mode' => 'cash',
   157→                'status' => 'draft',
   158→                'is_godown_prepared' => false,
   159→                'is_challan_issued' => false,
   160→                'is_reversed' => false,
   161→                'is_soft_hold' => $data['is_soft_hold'] ?? false,
   162→                'notes' => $data['notes'] ?? null,
   163→                'created_by' => $data['created_by'] ?? null,
   164→                'created_at' => now(),
   165→                'updated_at' => now(),
   166→            ]);
   167→
   168→            // Step 6: Create sales_invoice_items (warehouse_id=NULL — assigned at godown).
   169→            $itemRows = [];
   170→            foreach ($items as $item) {
   171→                $itemRows[] = [
   172→                    'sales_invoice_id' => $invoiceId,
   173→                    'product_id' => (int) $item['product_id'],
   174→                    'warehouse_id' => null, // assigned at godown (Phase 8.3)
   175→                    'qty' => (float) $item['qty'],
   176→                    'rate' => (float) $item['rate'],
   177→                    'condition_state' => 'Good',
   178→                ];
   179→            }
   180→            DB::table('sales_invoice_items')->insert($itemRows);
   181→
   182→            // Step 7: Create sales_invoice_dispatches (pipeline tracking).
   183→            $dispatchRows = [];
   184→            foreach ($items as $item) {
   185→                $dispatchRows[] = [
   186→                    'sales_invoice_id' => $invoiceId,
   187→                    'product_id' => (int) $item['product_id'],
   188→                    'warehouse_id' => null, // assigned at godown
   189→                    'qty' => (float) $item['qty'], // mirrors ordered_qty for GENERATED amount
   190→                    'ordered_qty' => (float) $item['qty'],
   191→                    'dispatched_qty' => 0,
   192→                    'created_by' => $data['created_by'] ?? null,
   193→                ];
   194→            }
   195→            DB::table('sales_invoice_dispatches')->insert($dispatchRows);
   196→
   197→            // Step 8: Post customer_ledger debit (customer owes us more).
   198→            $this->postCustomerLedgerDebit(
   199→                $customerId, $branchId, $invoiceId, $invoiceCode,
   200→                $totalAmount, $data['invoice_date'] ?? now()->format('Y-m-d'),
   201→                $data['created_by'] ?? null
   202→            );
   203→
   204→            // Step 9: Post GL: Dr AR / Cr Sales Revenue (+ Discount + Transport).
   205→            $journalEntryId = $this->postInvoiceGL(
   206→                $invoiceId, $invoiceCode, $customerId, $branchId,
   207→                $subTotal, $discount, $transport, $totalAmount,
   208→                $data['invoice_date'] ?? now()->format('Y-m-d'),
   209→                $data['created_by'] ?? null
   210→            );
   211→
   212→            // Update invoice with journal_entry_id.
   213→            DB::table('sales_invoices')
   214→                ->where('id', $invoiceId)
   215→                ->update(['journal_entry_id' => $journalEntryId, 'updated_at' => now()]);
   216→
   217→            // Step 10: Clear the cart.
   218→            $this->cartService->clearCart($data['created_by'] ?? auth()->id(), $customerId);
   219→
   220→            // Audit log for credit limit override.
   221→            if ($isOverride && $creditCheck['exceeds']) {
   222→                DB::table('user_audit_log')->insert([
   223→                    'user_id' => $data['created_by'] ?? null,
   224→                    'action' => 'credit_limit_override',
   225→                    'target_user_id' => null,
   226→                    'branch_id' => $branchId,
   227→                    'details' => json_encode([
   228→                        'invoice_id' => $invoiceId,
   229→                        'invoice_code' => $invoiceCode,
   230→                        'customer_id' => $customerId,
   231→                        'total_amount' => $totalAmount,
   232→                        'credit_limit' => $creditCheck['credit_limit'],
   233→                        'current_balance' => $creditCheck['current_balance'],
   234→                        'override_reason' => $overrideReason,
   235→                    ]),
   236→                    'ip_address' => request()?->ip(),
   237→                    'user_agent' => request()?->userAgent() ? mb_substr(request()->userAgent(), 0, 255) : null,
   238→                    'created_at' => now(),
   239→                ]);
   240→            }
   241→
   242→            // P1-3: Audit log — sale_created.
   243→            $this->auditLogger->saleCreated(
   244→                $data['created_by'] ?? auth()->id() ?? 0,
   245→                $invoiceId, $invoiceCode, $customerId, $branchId,
   246→                $totalAmount, $data['salesman_id'] ?? null
   247→            );
   248→
   249→            // P2-7: Invalidate pipeline cache (new dispatches were added).
   250→            $this->availabilityService->invalidatePipelineForInvoice($invoiceId);
   251→
   252→            return SalesInvoice::with(['items.product', 'customer', 'branch', 'journalEntry.lines.ledger'])
   253→                ->find($invoiceId);
   254→        });
   255→    }

   257→    /**
   258→     * Cancel a draft invoice (reverse GL + customer_ledger + dispatches).
   259→     * Only draft invoices can be cancelled (before godown/challan).
   260→     *
   261→     * @param int $invoiceId
   262→     * @param int $cancelledBy
   263→     * @param string $reason
   264→     * @return SalesInvoice
   265→     */
   266→    public function cancelInvoice(int $invoiceId, int $cancelledBy, string $reason = ''): SalesInvoice
   267→    {
   268→        return DB::transaction(function () use ($invoiceId, $cancelledBy, $reason) {
   269→            $invoice = SalesInvoice::lockForUpdate()->find($invoiceId);
   270→
   271→            if (!$invoice) {
   272→                throw new \RuntimeException("Invoice {$invoiceId} not found.");
   273→            }
   274→
   275→            // P0-8: Defense-in-depth branch isolation check.
   276→            $this->salesAccess->assertBranchAccessible((int) $invoice->branch_id);
   277→
   278→            if (!$invoice->isDraft()) {
   279→                throw new \RuntimeException("Only draft invoices can be cancelled (current: {$invoice->status}). Cancel via reversal for confirmed invoices.");
   280→            }
   281→
   282→            // P2-2: Explicit guards with clear messages (legacy layered checks).
   283→            if ($this->hasActiveChallan($invoiceId)) {
   284→                throw new \RuntimeException("Cannot cancel: a delivery challan exists for this invoice. Reverse the challan first.");
   285→            }
   286→            if ($this->invoiceHasPayments($invoiceId)) {
   287→                throw new \RuntimeException("Cannot cancel: payments have been received against this invoice. Reverse the payments first.");
   288→            }
   289→
   290→            // Reverse GL.
   291→            if ($invoice->journal_entry_id) {
   292→                $this->journalPosting->reverseJournalEntry(
   293→                    $invoice->journal_entry_id, $cancelledBy,
   294→                    "Invoice cancelled: {$reason}"
   295→                );
   296→            }
   297→
   298→            // Reverse customer_ledger (credit entry to reduce what customer owes).
   299→            $this->reverseCustomerLedgerDebit($invoice, $cancelledBy, $reason);
   300→
   301→            // Mark invoice as cancelled.
   302→            DB::table('sales_invoices')
   303→                ->where('id', $invoiceId)
   304→                ->update([
   305→                    'status' => 'cancelled',
   306→                    'is_reversed' => true,
   307→                    'reversed_at' => now(),
   308→                    'reversed_by' => $cancelledBy,
   309→                    'reverse_reason' => $reason,
   310→                    'updated_at' => now(),
   311→                ]);
   312→
   313→            // P1-3: Audit log — sale_cancelled.
   314→            $this->auditLogger->saleCancelled(
   315→                $cancelledBy,
   316→                $invoiceId, $invoice->invoice_code, (int) $invoice->branch_id,
   317→                (float) $invoice->total_amount, $reason
   318→            );
   319→
   320→            // P2-7: Invalidate pipeline cache (dispatches were deleted).
   321→            $this->availabilityService->invalidatePipelineForInvoice($invoiceId);
   322→
   323→            return SalesInvoice::find($invoiceId);
   324→        });
   325→    }

   327→    /**
   328→     * Update an existing DRAFT invoice (P1-1).
   329→     *
   330→     * Restores the legacy updateExistingInvoice flow:
   331→     *   1. Assert editable (draft, no godown, no payments, not reversed)
   332→     *   2. Re-validate credit limit (using NET increase = max(0, newTotal - oldTotal))
   333→     *   3. Lock branch products FOR UPDATE + re-check availability
   334→     *   4. Reverse old customer_ledger debit (append-only credit entry)
   335→     *   5. Reverse old GL journal entry (append-only)
   336→     *   6. DELETE old items + dispatches + dispatchers
   337→     *   7. INSERT new items + dispatches (soft reservation, warehouse_id=NULL)
   338→     *   8. Post new customer_ledger debit (new total)
   339→     *   9. Post new GL journal entry (Dr AR / Cr Revenue + Discount + Transport)
   340→     *  10. UPDATE invoice header (sub_total, discount, transport, total, notes, etc.)
   341→     *
   342→     * @param int $invoiceId
   343→     * @param array $data {
   344→     *     items: array of {product_id, qty, rate},
   345→     *     invoice_date: string,
   346→     *     sales_person: string|null,
   347→     *     discount_amount: float,
   348→     *     transport_cost: float,
   349→     *     notes: string|null,
   350→     *     is_soft_hold: bool,
   351→     *     credit_limit_override: bool,
   352→     *     override_reason: string|null,
   353→     *     updated_by: int,
   354→     * }
   355→     * @return SalesInvoice
   356→     * @throws \RuntimeException If not editable, stock insufficient, credit exceeded, or GL fails.
   357→     */
   358→    public function updateInvoice(int $invoiceId, array $data): SalesInvoice
   359→    {
   360→        $items = $data['items'] ?? [];
   361→        $updatedBy = (int) ($data['updated_by'] ?? auth()->id() ?? 0);
   362→
   363→        if (empty($items)) {
   364→            throw new \RuntimeException('Cannot update: items list is empty.');
   365→        }
   366→
   367→        // --- Pre-flight: load invoice (without transaction yet) for validation ---
   368→        $invoice = SalesInvoice::find($invoiceId);
   369→        if (!$invoice) {
   370→            throw new \RuntimeException("Invoice {$invoiceId} not found.");
   371→        }
   372→
   373→        // P0-8: Defense-in-depth branch isolation check.
   374→        $this->salesAccess->assertBranchAccessible((int) $invoice->branch_id);
   375→
   376→        // Assert editable.
   377→        $this->assertEditable($invoice);
   378→
   379→        // Assert no payments.
   380→        if ($this->invoiceHasPayments($invoiceId)) {
   381→            throw new \RuntimeException('Cannot edit: payments have been received against this invoice. Reverse the payments first.');
   382→        }
   383→
   384→        $oldTotal = (float) $invoice->total_amount;
   385→        $customerId = (int) $invoice->customer_id;
   386→        $branchId = (int) $invoice->branch_id;
   387→        $oldJournalId = $invoice->journal_entry_id ? (int) $invoice->journal_entry_id : null;
   388→
   389→        // --- Calculate new totals ---
   390→        $subTotal = 0.0;
   391→        foreach ($items as $item) {
   392→            $subTotal += (float) $item['qty'] * (float) $item['rate'];
   393→        }
   394→        $discount = (float) ($data['discount_amount'] ?? 0);
   395→        $transport = (float) ($data['transport_cost'] ?? 0);
   396→        $newTotal = $subTotal + $transport - $discount;
   397→
   398→        // --- Credit limit check (NET increase, not full new total) ---
   399→        $netIncrease = max(0.0, $newTotal - $oldTotal);
   400→        $creditCheck = $this->checkCreditLimit($customerId, $netIncrease);
   401→        $isOverride = !empty($data['credit_limit_override']);
   402→        $overrideReason = trim($data['override_reason'] ?? '');
   403→
   404→        if ($creditCheck['exceeds'] && !$isOverride) {
   405→            throw new \RuntimeException(
   406→                "Updating this invoice would exceed the customer's credit limit. "
   407→                . "Current balance: {$creditCheck['current_balance']}, "
   408→                . "Credit limit: {$creditCheck['credit_limit']}, "
   409→                . "Net increase: {$netIncrease}. "
   410→                . "Override with a reason to proceed."
   411→            );
   412→        }
   413→
   414→        if ($creditCheck['exceeds'] && $isOverride && strlen($overrideReason) < 10) {
   415→            throw new \RuntimeException('Override reason must be at least 10 characters when exceeding credit limit.');
   416→        }
   417→
   418→        $invoiceDate = $data['invoice_date'] ?? $invoice->invoice_date;
   419→
   420→        // --- Atomic update in a DB transaction ---
   421→        return DB::transaction(function () use (
   422→            $invoiceId, $invoice, $items, $customerId, $branchId, $oldTotal, $newTotal,
   423→            $subTotal, $discount, $transport, $invoiceDate, $oldJournalId,
   424→            $creditCheck, $isOverride, $overrideReason, $updatedBy, $data
   425→        ) {
   426→            // Lock the invoice row FOR UPDATE.
   427→            $locked = SalesInvoice::lockForUpdate()->find($invoiceId);
   428→            if (!$locked) {
   429→                throw new \RuntimeException("Invoice {$invoiceId} not found (locked).");
   430→            }
   431→
   432→            // Re-assert editable after lock (race protection).
   433→            $this->assertEditable($locked);
   434→
   435→            // Lock branch products FOR UPDATE.
   436→            $productIds = collect($items)->pluck('product_id')->unique()->toArray();
   437→            $this->stockService->lockBranchProductsForUpdate($branchId, $productIds);
   438→
   439→            // Re-check stock availability (excluding this invoice's own pipeline).
   440→            $qtyByProduct = [];
   441→            foreach ($items as $item) {
   442→                $pid = (int) $item['product_id'];
   443→                $qtyByProduct[$pid] = ($qtyByProduct[$pid] ?? 0) + (float) $item['qty'];
   444→            }
   445→
   446→            foreach ($qtyByProduct as $pid => $qty) {
   447→                $available = $this->availabilityService->getBranchAvailableQty($pid, $branchId, $invoiceId);
   448→                if ($qty > $available + 0.0001) {
   449→                    throw new \RuntimeException(
   450→                        "Insufficient stock for product {$pid}: requested {$qty}, available {$available}"
   451→                    );
   452→                }
   453→            }
   454→
   455→            // Step 1: Reverse old customer_ledger debit (append-only credit entry).
   456→            $this->reverseCustomerLedgerDebit($locked, $updatedBy, 'Invoice edited');
   457→
   458→            // Step 2: Reverse old GL journal entry (append-only).
   459→            if ($oldJournalId) {
   460→                $this->journalPosting->reverseJournalEntry(
   461→                    $oldJournalId, $updatedBy,
   462→                    'Invoice edited: ' . $invoice->invoice_code
   463→                );
   464→            }
   465→
   466→            // Step 3: DELETE old items + dispatches + dispatchers.
   467→            DB::table('sales_invoice_items')->where('sales_invoice_id', $invoiceId)->delete();
   468→            DB::table('sales_invoice_dispatches')->where('sales_invoice_id', $invoiceId)->delete();
   469→            DB::table('sales_invoice_dispatchers')->where('sales_invoice_id', $invoiceId)->delete();
   470→
   471→            // Step 4: INSERT new items.
   472→            $itemRows = [];
   473→            foreach ($items as $item) {
   474→                $itemRows[] = [
   475→                    'sales_invoice_id' => $invoiceId,
   476→                    'product_id' => (int) $item['product_id'],
   477→                    'warehouse_id' => null, // reset to NULL — godown must re-assign after edit
   478→                    'qty' => (float) $item['qty'],
   479→                    'rate' => (float) $item['rate'],
   480→                    'condition_state' => $item['condition_state'] ?? 'Good',
   481→                ];
   482→            }
   483→            DB::table('sales_invoice_items')->insert($itemRows);
   484→
   485→            // Step 5: INSERT new dispatches (soft reservation, warehouse_id=NULL).
   486→            $dispatchRows = [];
   487→            foreach ($items as $item) {
   488→                $dispatchRows[] = [
   489→                    'sales_invoice_id' => $invoiceId,
   490→                    'product_id' => (int) $item['product_id'],
   491→                    'warehouse_id' => null,
   492→                    'qty' => (float) $item['qty'],
   493→                    'ordered_qty' => (float) $item['qty'],
   494→                    'dispatched_qty' => 0,
   495→                    'created_by' => $updatedBy,
   496→                ];
   497→            }
   498→            DB::table('sales_invoice_dispatches')->insert($dispatchRows);
   499→
   500→            // Step 6: Post new customer_ledger debit (new total).
   501→            $this->postCustomerLedgerDebit(
   502→                $customerId, $branchId, $invoiceId, $invoice->invoice_code,
   503→                $newTotal, $invoiceDate, $updatedBy
   504→            );
   505→
   506→            // Step 7: Post new GL: Dr AR / Cr Revenue (+ Discount + Transport).
   507→            $newJournalEntryId = $this->postInvoiceGL(
   508→                $invoiceId, $invoice->invoice_code, $customerId, $branchId,
   509→                $subTotal, $discount, $transport, $newTotal,
   510→                $invoiceDate, $updatedBy
   511→            );
   512→
   513→            // Step 8: UPDATE invoice header.
   514→            DB::table('sales_invoices')
   515→                ->where('id', $invoiceId)
   516→                ->update([
   517→                    'invoice_date' => $invoiceDate,
   518→                    'sales_person' => $data['sales_person'] ?? null,
   519→                    'sub_total' => round($subTotal, 2),
   520→                    'discount_amount' => round($discount, 2),
   521→                    'transport_cost' => round($transport, 2),
   522→                    'total_amount' => round($newTotal, 2),
   523→                    'due_amount' => round($newTotal, 2) - (float) $invoice->paid_amount,
   524→                    'is_soft_hold' => $data['is_soft_hold'] ?? false,
   525→                    'notes' => $data['notes'] ?? null,
   526→                    'journal_entry_id' => $newJournalEntryId,
   527→                    // Reset godown/challan flags (edit invalidates prior godown prep).
   528→                    'is_godown_prepared' => false,
   529→                    'godown_prepared_at' => null,
   530→                    'updated_at' => now(),
   531→                ]);
   532→
   533→            // Step 9: Audit log for credit limit override.
   534→            if ($isOverride && $creditCheck['exceeds']) {
   535→                DB::table('user_audit_log')->insert([
   536→                    'user_id' => $updatedBy,
   537→                    'action' => 'credit_limit_override',
   538→                    'target_user_id' => null,
   539→                    'branch_id' => $branchId,
   540→                    'details' => json_encode([
   541→                        'action' => 'invoice_edit',
   542→                        'invoice_id' => $invoiceId,
   543→                        'invoice_code' => $invoice->invoice_code,
   544→                        'customer_id' => $customerId,
   545→                        'old_total' => $oldTotal,
   546→                        'new_total' => $newTotal,
   547→                        'net_increase' => $netIncrease,
   548→                        'credit_limit' => $creditCheck['credit_limit'],
   549→                        'current_balance' => $creditCheck['current_balance'],
   550→                        'override_reason' => $overrideReason,
   551→                    ]),
   552→                    'ip_address' => request()?->ip(),
   553→                    'user_agent' => request()?->userAgent() ? mb_substr(request()->userAgent(), 0, 255) : null,
   554→                    'created_at' => now(),
   555→                ]);
   556→            }
   557→
   558→            // Audit log for the edit itself.
   559→            DB::table('user_audit_log')->insert([
   560→                'user_id' => $updatedBy,
   561→                'action' => 'sale_updated',
   562→                'target_user_id' => null,
   563→                'branch_id' => $branchId,
   564→                'details' => json_encode([
   565→                    'invoice_id' => $invoiceId,
   566→                    'invoice_code' => $invoice->invoice_code,
   567→                    'old_total' => $oldTotal,
   568→                    'new_total' => $newTotal,
   569→                    'items_count' => count($items),
   570→                    'credit_override' => $isOverride && $creditCheck['exceeds'],
   571→                ]),
   572→                'ip_address' => request()?->ip(),
   573→                'user_agent' => request()?->userAgent() ? mb_substr(request()->userAgent(), 0, 255) : null,
   574→                'created_at' => now(),
   575→            ]);
   576→
   577→            // P2-7: Invalidate pipeline cache (dispatches were deleted + re-inserted).
   578→            $this->availabilityService->invalidatePipelineForInvoice($invoiceId);
   579→
   580→            return SalesInvoice::with(['items.product', 'customer', 'branch', 'journalEntry.lines.ledger'])
   581→                ->find($invoiceId);
   582→        });
   583→    }

   585→    /**
   586→     * Assert an invoice is editable (P1-1).
   587→     * Must be: status='draft', not godown-prepared, not reversed.
   588→     */
   589→    private function assertEditable(SalesInvoice $invoice): void
   590→    {
   591→        if ($invoice->is_reversed) {
   592→            throw new \RuntimeException('Cannot edit: invoice is reversed.');
   593→        }
   594→        if ($invoice->status !== 'draft') {
   595→            throw new \RuntimeException("Cannot edit: invoice status is '{$invoice->status}'. Only draft invoices can be edited.");
   596→        }
   597→        if ($invoice->is_godown_prepared) {
   598→            throw new \RuntimeException('Cannot edit: godown has already been prepared for this invoice.');
   599→        }
   600→        if ($invoice->is_challan_issued) {
   601→            throw new \RuntimeException('Cannot edit: a delivery challan has already been issued for this invoice.');
   602→        }
   603→    }

   628→    /**
   629→     * Check if the invoice would exceed the customer's credit limit.
   630→     *
   631→     * @return array{ exceeds: bool, current_balance: float, credit_limit: float, new_balance: float }
   632→     */
   633→    private function checkCreditLimit(int $customerId, float $invoiceAmount): array
   634→    {
   635→        $customer = DB::table('customers')->where('id', $customerId)->first();
   636→
   637→        if (!$customer) {
   638→            return ['exceeds' => false, 'current_balance' => 0, 'credit_limit' => 0, 'new_balance' => 0];
   639→        }
   640→
   641→        $creditLimit = (float) ($customer->credit_limit ?? 0);
   642→
   643→        // Current AR balance from customer_ledger (sum of debit - credit).
   644→        $currentBalance = (float) DB::table('customer_ledger')
   645→            ->where('customer_id', $customerId)
   646→            ->where('is_reversed', false)
   647→            ->selectRaw('COALESCE(SUM(debit) - SUM(credit), 0) as balance')
   648→            ->value('balance');
   649→
   650→        $newBalance = $currentBalance + $invoiceAmount;
   651→        $exceeds = $creditLimit > 0 && $newBalance > $creditLimit + 0.01;
   652→
   653→        return [
   654→            'exceeds' => $exceeds,
   655→            'current_balance' => $currentBalance,
   656→            'credit_limit' => $creditLimit,
   657→            'new_balance' => $newBalance,
   658→        ];
   659→    }

   691→    /**
   692→     * Reverse customer_ledger (credit entry to reduce what customer owes).
   693→     */
   694→    private function reverseCustomerLedgerDebit(SalesInvoice $invoice, int $cancelledBy, string $reason): void
   695→    {
   696→        $amount = (float) $invoice->total_amount;
   697→        if ($amount < 0.01) return;
   698→
   699→        $currentBalance = (float) DB::table('customer_ledger')
   700→            ->where('customer_id', $invoice->customer_id)
   701→            ->orderByDesc('id')
   702→            ->value('balance');
   703→
   704→        $newBalance = $currentBalance - $amount;
   705→
   706→        DB::table('customer_ledger')->insert([
   707→            'customer_id' => $invoice->customer_id,
   708→            'branch_id' => $invoice->branch_id,
   709→            'transaction_date' => now()->format('Y-m-d'),
   710→            'transaction_type' => 'sales_invoice_reversal',
   711→            'reference_type' => 'sales_invoice',
   712→            'reference_id' => $invoice->id,
   713→            'debit' => 0,
   714→            'credit' => $amount,
   715→            'balance' => $newBalance,
   716→            'description' => 'Invoice reversal ' . $invoice->invoice_code . ": {$reason}",
   717→            'created_by' => $cancelledBy,
   718→            'created_at' => now(),
   719→        ]);
   720→    }

   722→    /**
   723→     * Post GL: Dr Accounts Receivable / Cr Sales Revenue.
   724→     * + Dr Discount (if discount > 0)
   725→     * + Cr Transport Revenue (if transport > 0)
   726→     *
   727→     * @return int journal_entry_id
   728→     */
   729→    private function postInvoiceGL(
   730→        int $invoiceId, string $invoiceCode, int $customerId, int $branchId,
   731→        float $subTotal, float $discount, float $transport, float $total,
   732→        string $invoiceDate, ?int $createdBy
   733→    ): int {
   734→        $arLedgerId = $this->journalPosting->lookupLedgerByNature('ar');
   735→        $revenueLedgerId = $this->journalPosting->lookupLedgerByNature('sales_revenue');
   736→
   737→        if (!$arLedgerId) {
   738→            throw new \RuntimeException('Accounts Receivable ledger not found (nature: ar).');
   739→        }
   740→        if (!$revenueLedgerId) {
   741→            throw new \RuntimeException('Sales Revenue ledger not found (nature: sales_revenue).');
   742→        }
   743→
   744→        $lines = [];
   745→
   746→        // Dr Accounts Receivable (total)
   747→        $lines[] = [
   748→            'ledger_id' => $arLedgerId,
   749→            'debit' => $total, 'credit' => 0,
   750→            'entity_type' => 'customer', 'entity_id' => $customerId,
   751→            'memo' => 'Invoice ' . $invoiceCode . ' — AR',
   752→        ];
   753→
   754→        // Cr Sales Revenue (subTotal - discount, or subTotal if no discount ledger)
   755→        $discountLedgerId = $discount > 0.01 ? $this->journalPosting->lookupLedgerByNature('sales_discount') : null;
   756→        $revenueAmount = $discountLedgerId ? $subTotal : max(0, $subTotal - $discount);
   757→
   758→        $lines[] = [
   759→            'ledger_id' => $revenueLedgerId,
   760→            'debit' => 0, 'credit' => $revenueAmount,
   761→            'entity_type' => 'sales_invoice', 'entity_id' => $invoiceId,
   762→            'memo' => 'Invoice ' . $invoiceCode . ' — Revenue',
   763→        ];
   764→
   765→        // Dr Discount (if applicable)
   766→        if ($discountLedgerId && $discount > 0.01) {
   767→            $lines[] = [
   768→                'ledger_id' => $discountLedgerId,
   769→                'debit' => $discount, 'credit' => 0,
   770→                'entity_type' => 'sales_invoice', 'entity_id' => $invoiceId,
   771→                'memo' => 'Invoice ' . $invoiceCode . ' — Discount allowed',
   772→            ];
   773→        }
   774→
   775→        // Cr Transport Revenue (if applicable)
   776→        if ($transport > 0.01) {
   777→            $transportLedgerId = $this->journalPosting->lookupLedgerByNature('transport_revenue');
   778→            if ($transportLedgerId) {
   779→                $lines[] = [
   780→                    'ledger_id' => $transportLedgerId,
   781→                    'debit' => 0, 'credit' => $transport,
   782→                    'entity_type' => 'sales_invoice', 'entity_id' => $invoiceId,
   783→                    'memo' => 'Invoice ' . $invoiceCode . ' — Transport',
   784→                ];
   785→            }
   786→        }
   787→
   788→        return $this->journalPosting->createJournalEntry([
   789→            'entry_date' => $invoiceDate,
   790→            'reference_type' => 'sales_invoice',
   791→            'reference_id' => $invoiceId,
   792→            'branch_id' => $branchId,
   793→            'description' => 'Sales Invoice ' . $invoiceCode,
   794→            'source' => 'sales_invoice',
   795→            'created_by' => $createdBy,
   796→        ], $lines);
   797→    }

   799→    /**
   800→     * Generate atomic invoice code: INV-YYYYMMDD-NNNN.
   801→     */
   802→    private function generateInvoiceCode(): string
   803→    {
   804→        $datePart = now()->format('Ymd');
   805→        $periodKey = now()->format('Y-m');
   806→        $docType = 'sales_invoice';
   807→
   808→        return DB::transaction(function () use ($docType, $periodKey, $datePart) {
   809→            $seqRow = DB::table('document_sequences')
   810→                ->where('doc_type', $docType)
   811→                ->where('branch_id', 0)
   812→                ->where('period_key', $periodKey)
   813→                ->lockForUpdate()
   814→                ->first();
   815→
   816→            $nextNumber = $seqRow ? ((int) $seqRow->last_number + 1) : 1;
   817→
   818→            if ($seqRow) {
   819→                DB::table('document_sequences')->where('id', $seqRow->id)
   820→                    ->update(['last_number' => $nextNumber, 'updated_at' => now()]);
   821→            } else {
   822→                DB::table('document_sequences')->insert([
   823→                    'doc_type' => $docType, 'branch_id' => 0,
   824→                    'period_key' => $periodKey, 'last_number' => $nextNumber,
   825→                    'updated_at' => now(),
   826→                ]);
   827→            }
   828→
   829→            return "INV-{$datePart}-" . str_pad((string) $nextNumber, 4, '0', STR_PAD_LEFT);
   830→        });
   831→    }
