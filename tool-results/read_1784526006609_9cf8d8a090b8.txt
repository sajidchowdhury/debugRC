    13→
    14→/**
    15→ * Sales Challan Service — Phase 8.3.
    16→ *
    17→ * Two-step flow (from Phase 8.2 draft invoice):
    18→ *   1. prepareGodown: assign warehouse_id to invoice items + dispatches.
    19→ *      Invoice status: draft → confirmed, is_godown_prepared=true.
    20→ *      NO stock movement, NO GL — just warehouse assignment.
    21→ *
    22→ *   2. issueChallan: stock OUT via StockService at avg_cost + GL Dr COGS / Cr Inventory.
    23→ *      Creates sales_challan, moves stock, posts COGS journal.
    24→ *      Invoice: is_challan_issued=true.
    25→ *
    26→ *   3. cancelChallan: reverse stock + GL (append-only reversal).
    27→ *
    28→ * Rate semantics (per avg_cost_rule.md §3):
    29→ *   Stock OUT at current avg_cost (cost flows out at average, avg unchanged on OUT).
    30→ *   The COGS = Σ (qty × avg_cost) for all dispatch lines.
    31→ *
    32→ * GL posting (re-derived from double-entry):
    33→ *   Dr COGS (nature: cogs)
    34→ *   Cr Inventory (nature: inventory)
    35→ */
    36→class SalesChallanService
    37→{
    38→    public function __construct(
    39→        private StockService $stockService,
    40→        private JournalPostingService $journalPosting,
    41→        private SubLedgerService $subLedger,
    42→        private SalesAccess $salesAccess,
    43→        private SalesAuditLogger $auditLogger,
    44→        private StockAvailabilityService $availabilityService
    45→    ) {}
    46→
    47→    /**
    48→     * Step 1: Prepare godown — assign warehouse_id to invoice items + dispatches.
    49→     * No stock movement, no GL.
    50→     *
    51→     * @param int $invoiceId
    52→     * @param array $warehouseAssignments [product_id => warehouse_id]
    53→     * @param int $preparedBy
    54→     * @return SalesInvoice
    55→     * @throws \RuntimeException If invoice not draft or already godown-prepared.
    56→     */
    57→    public function prepareGodown(int $invoiceId, array $warehouseAssignments, int $preparedBy): SalesInvoice
    58→    {
    59→        return DB::transaction(function () use ($invoiceId, $warehouseAssignments, $preparedBy) {
    60→            $invoice = SalesInvoice::with('items', 'dispatches')->lockForUpdate()->find($invoiceId);
    61→
    62→            if (!$invoice) {
    63→                throw new \RuntimeException("Invoice {$invoiceId} not found.");
    64→            }
    65→            if (!$invoice->isDraft()) {
    66→                throw new \RuntimeException("Only draft invoices can be godown-prepared (current: {$invoice->status}).");
    67→            }
    68→
    69→            // Assign warehouse_id to each invoice item.
    70→            foreach ($invoice->items as $item) {
    71→                $wid = $warehouseAssignments[$item->product_id] ?? null;
    72→                if (!$wid) {
    73→                    throw new \RuntimeException("Warehouse not assigned for product {$item->product_id}.");
    74→                }
    75→
    76→                // Check availability in the assigned warehouse.
    77→                $available = $this->stockService->getWarehouseQty((int) $wid, $item->product_id);
    78→                if ((float) $item->qty > $available + 0.0001) {
    79→                    throw new \RuntimeException(
    80→                        "Insufficient stock in warehouse {$wid} for product {$item->product_id}: "
    81→                        . "available {$available}, required {$item->qty}"
    82→                    );
    83→                }
    84→
    85→                DB::table('sales_invoice_items')
    86→                    ->where('id', $item->id)
    87→                    ->update(['warehouse_id' => (int) $wid]);
    88→
    89→                DB::table('sales_invoice_dispatches')
    90→                    ->where('sales_invoice_id', $invoiceId)
    91→                    ->where('product_id', $item->product_id)
    92→                    ->update(['warehouse_id' => (int) $wid]);
    93→            }
    94→
    95→            // Update invoice status.
    96→            DB::table('sales_invoices')
    97→                ->where('id', $invoiceId)
    98→                ->update([
    99→                    'status' => 'confirmed',
   100→                    'is_godown_prepared' => true,
   101→                    'godown_prepared_at' => now(),
   102→                    'updated_at' => now(),
   103→                ]);
   104→
   105→            // P1-3: Audit log — godown_prepared.
   106→            $this->auditLogger->godownPrepared(
   107→                $preparedBy, $invoiceId, $invoice->invoice_code, (int) $invoice->branch_id
   108→            );
   109→
   110→            return SalesInvoice::with(['items.product', 'dispatches', 'customer', 'branch'])->find($invoiceId);
   111→        });
   112→    }
   113→
   114→    /**
   115→     * Step 2: Issue challan — stock OUT + GL Dr COGS / Cr Inventory.
   116→     *
   117→     * @param int $invoiceId
   118→     * @param array $data {
   119→     *     transport_name: string|null,
   120→     *     vehicle_number: string|null,
   121→     *     driver_name: string|null,
   122→     *     transport_cost: float,
   123→     *     notes: string|null,
   124→     *     created_by: int,
   125→     * }
   126→     * @return SalesChallan
   127→     * @throws \RuntimeException If not godown-prepared or already challan-issued.
   128→     */
   129→    public function issueChallan(int $invoiceId, array $data): SalesChallan
   130→    {
   131→        return DB::transaction(function () use ($invoiceId, $data) {
   132→            $invoice = SalesInvoice::with(['items', 'dispatches'])->lockForUpdate()->find($invoiceId);
   133→
   134→            if (!$invoice) {
   135→                throw new \RuntimeException("Invoice {$invoiceId} not found.");
   136→            }
   137→
   138→            // P0-8: Defense-in-depth branch isolation check.
   139→            $this->salesAccess->assertBranchAccessible((int) $invoice->branch_id);
   140→
   141→            if (!$invoice->is_godown_prepared) {
   142→                throw new \RuntimeException("Invoice must be godown-prepared before issuing challan.");
   143→            }
   144→            if ($invoice->is_challan_issued) {
   145→                throw new \RuntimeException("Challan already issued for this invoice.");
   146→            }
   147→
   148→            $challanCode = $this->generateChallanCode();
   149→            $challanDate = now()->format('Y-m-d');
   150→            $cogsTotal = 0.0;
   151→
   152→            // Create the challan header.
   153→            $challanId = DB::table('sales_challans')->insertGetId([
   154→                'challan_code' => $challanCode,
   155→                'challan_date' => $challanDate,
   156→                'sales_invoice_id' => $invoiceId,
   157→                'branch_id' => $invoice->branch_id,
   158→                'transport_name' => $data['transport_name'] ?? null,
   159→                'transport_phone' => $data['transport_phone'] ?? null,
   160→                'vehicle_number' => $data['vehicle_number'] ?? null,
   161→                'driver_name' => $data['driver_name'] ?? null,
   162→                'transport_cost' => $data['transport_cost'] ?? 0,
   163→                'is_reversed' => false,
   164→                'is_dispatch_soft_hold' => false,
   165→                'created_by' => $data['created_by'] ?? null,
   166→                'created_at' => now(),
   167→                'updated_at' => now(),
   168→            ]);
   169→
   170→            // Process each dispatch line: stock OUT at avg_cost.
   171→            foreach ($invoice->items as $item) {
   172→                $warehouseId = $item->warehouse_id;
   173→                $productId = $item->product_id;
   174→                $qty = (float) $item->qty;
   175→
   176→                if (!$warehouseId) {
   177→                    throw new \RuntimeException("Product {$productId} has no warehouse assigned. Run godown prep first.");
   178→                }
   179→
   180→                // Get current avg_cost (this is the COGS rate).
   181→                $avgCost = $this->stockService->getWarehouseAvgCost($warehouseId, $productId);
   182→                if ($avgCost <= 0) {
   183→                    throw new \RuntimeException(
   184→                        "Cannot issue challan: zero avg_cost for product {$productId} in warehouse {$warehouseId}. "
   185→                        . "Receive stock or set cost first."
   186→                    );
   187→                }
   188→
   189→                // Stock OUT via StockService (reference_type='sales_challan', rate=avg_cost).
   190→                $this->stockService->applyTransaction([
   191→                    'warehouse_id' => $warehouseId,
   192→                    'product_id' => $productId,
   193→                    'qty' => -$qty, // negative = OUT
   194→                    'rate' => $avgCost, // current avg_cost (cost flows out, avg unchanged on OUT)
   195→                    'reference_type' => 'sales_challan',
   196→                    'reference_id' => $challanId,
   197→                    'notes' => 'Sales Challan ' . $challanCode,
   198→                    'transaction_date' => $challanDate,
   199→                    'created_by' => $data['created_by'] ?? null,
   200→                ]);
   201→
   202→                // P0-5: Persist per-line issue cost snapshot (sales_challan_items).
   203→                // This denormalized record captures the avg_cost at the moment of stock OUT
   204→                // so that reversals and reports can use the ORIGINAL per-line rate.
   205→                DB::table('sales_challan_items')->insert([
   206→                    'sales_challan_id' => $challanId,
   207→                    'product_id' => $productId,
   208→                    'warehouse_id' => $warehouseId,
   209→                    'qty' => $qty,
   210→                    'issue_rate' => $avgCost,
   211→                    'cogs_amount' => round($qty * $avgCost, 2),
   212→                    'created_at' => now(),
   213→                ]);
   214→
   215→                // Update dispatch row: dispatched_qty = ordered_qty, qty mirrors for GENERATED amount.
   216→                DB::table('sales_invoice_dispatches')
   217→                    ->where('sales_invoice_id', $invoiceId)
   218→                    ->where('product_id', $productId)
   219→                    ->update([
   220→                        'qty' => DB::raw('ordered_qty'),
   221→                        'dispatched_qty' => DB::raw('ordered_qty'),
   222→                        'warehouse_id' => $warehouseId,
   223→                    ]);
   224→
   225→                $cogsTotal += $qty * $avgCost;
   226→            }
   227→
   228→            // Post GL: Dr COGS / Cr Inventory (single journal for all lines).
   229→            $journalEntryId = $this->postCogsGL($challanId, $challanCode, $challanDate, $invoice->branch_id, $cogsTotal, $data['created_by'] ?? null);
   230→
   231→            // Update challan with journal_entry_id + issue_cost.
   232→            DB::table('sales_challans')
   233→                ->where('id', $challanId)
   234→                ->update([
   235→                    'journal_entry_id' => $journalEntryId,
   236→                    'issue_cost' => round($cogsTotal, 2),
   237→                    'updated_at' => now(),
   238→                ]);
   239→
   240→            // P2-3: Transport snapshot + adjustment.
   241→            // If the challan form's transport_cost differs from the invoice's
   242→            // original transport_cost, snapshot the original values + post an
   243→            // adjustment (customer_ledger + GL) for the delta.
   244→            $newTransport = (float) ($data['transport_cost'] ?? 0);
   245→            $oldTransport = (float) $invoice->transport_cost;
   246→            $transportAdjustment = round($newTransport - $oldTransport, 2);
   247→            $adjustmentJournalEntryId = null;
   248→
   249→            if (abs($transportAdjustment) > 0.01) {
   250→                // Snapshot original values on the invoice.
   251→                DB::table('sales_invoices')
   252→                    ->where('id', $invoiceId)
   253→                    ->update([
   254→                        'pre_challan_transport' => $oldTransport,
   255→                        'pre_challan_total' => (float) $invoice->total_amount,
   256→                    ]);
   257→
   258→                // Update invoice transport_cost + total_amount + due_amount.
   259→                $newTotal = (float) $invoice->sub_total - (float) $invoice->discount_amount + $newTransport;
   260→                DB::table('sales_invoices')
   261→                    ->where('id', $invoiceId)
   262→                    ->update([
   263→                        'transport_cost' => round($newTransport, 2),
   264→                        'total_amount' => round($newTotal, 2),
   265→                        'due_amount' => round($newTotal - (float) $invoice->paid_amount, 2),
   266→                    ]);
   267→
   268→                // Post GL adjustment JE FIRST (Dr/Cr AR + Revenue, swapped by sign) to get journal_entry_id.
   269→                $adjustmentJournalEntryId = $this->postTransportAdjustmentGL(
   270→                    $invoiceId, $challanCode, $challanDate, (int) $invoice->branch_id,
   271→                    $transportAdjustment, (int) $invoice->customer_id,
   272→                    $data['created_by'] ?? null
   273→                );
   274→
   275→                // Post customer_ledger 'invoice_adjustment' entry via SubLedgerService for the delta.
   276→                $debit = $transportAdjustment > 0 ? abs($transportAdjustment) : 0;
   277→                $credit = $transportAdjustment < 0 ? abs($transportAdjustment) : 0;
   278→                $this->subLedger->postCustomerLedgerEntry([
   279→                    'customer_id' => $invoice->customer_id,
   280→                    'branch_id' => $invoice->branch_id,
   281→                    'transaction_date' => now()->format('Y-m-d'),
   282→                    'transaction_type' => 'invoice_adjustment',
   283→                    'reference_type' => 'sales_invoice',
   284→                    'reference_id' => $invoice->id,
   285→                    'debit' => $debit,
   286→                    'credit' => $credit,
   287→                    'description' => 'Challan ' . $challanCode . ' — transport adjustment ' . ($transportAdjustment >= 0 ? '+' : '') . number_format($transportAdjustment, 2),
   288→                    'journal_entry_id' => $adjustmentJournalEntryId,
   289→                    'created_by' => $data['created_by'] ?? null,
   290→                ]);
   291→            }
   292→
   293→            // Store transport_adjustment + adjustment_journal_entry_id on the challan.
   294→            DB::table('sales_challans')
   295→                ->where('id', $challanId)
   296→                ->update([
   297→                    'transport_adjustment' => $transportAdjustment,
   298→                    'adjustment_journal_entry_id' => $adjustmentJournalEntryId,
   299→                ]);
   300→
   301→            // Update invoice: is_challan_issued=true.
   302→            DB::table('sales_invoices')
   303→                ->where('id', $invoiceId)
   304→                ->update([
   305→                    'is_challan_issued' => true,
   306→                    'challan_issued_at' => now(),
   307→                    'cogs_journal_entry_id' => $journalEntryId,
   308→                    'updated_at' => now(),
   309→                ]);
   310→
   311→            // P1-3: Audit log — challan_issued.
   312→            $this->auditLogger->challanIssued(
   313→                $data['created_by'] ?? auth()->id() ?? 0,
   314→                $challanId, $challanCode, $invoiceId, (int) $invoice->branch_id,
   315→                $cogsTotal, $journalEntryId
   316→            );
   317→
   318→            // P2-7: Invalidate pipeline cache (dispatched_qty changed).
   319→            $this->availabilityService->invalidatePipelineForInvoice($invoiceId);
   320→
   321→            return SalesChallan::with(['salesInvoice.items.product', 'items.product', 'items.warehouse', 'branch', 'journalEntry.lines.ledger'])
   322→                ->find($challanId);
   323→        });
   324→    }
   325→
   326→    /**
   327→     * Step 3: Cancel a challan — reverse stock + GL (append-only reversal).
   328→     *
   329→     * @param int $challanId
   330→     * @param int $cancelledBy
   331→     * @param string $reason
   332→     * @return SalesChallan
   333→     */
   334→    public function cancelChallan(int $challanId, int $cancelledBy, string $reason = ''): SalesChallan
   335→    {
   336→        return DB::transaction(function () use ($challanId, $cancelledBy, $reason) {
   337→            $challan = SalesChallan::with('salesInvoice.items')->lockForUpdate()->find($challanId);
   338→
   339→            if (!$challan) {
   340→                throw new \RuntimeException("Challan {$challanId} not found.");
   341→            }
   342→            if ($challan->is_reversed) {
   343→                throw new \RuntimeException("Challan is already cancelled.");
   344→            }
   345→
   346→            // Reverse GL (COGS journal).
   347→            if ($challan->journal_entry_id) {
   348→                $this->journalPosting->reverseJournalEntry(
   349→                    $challan->journal_entry_id, $cancelledBy,
   350→                    "Challan cancelled: {$reason}"
   351→                );
   352→            }
   353→
   354→            // Reverse each stock movement.
   355→            $stockTxs = DB::table('stock_transactions')
   356→                ->where('reference_type', 'sales_challan')
   357→                ->where('reference_id', $challanId)
   358→                ->where('is_reversed', false)
   359→                ->get();
   360→
   361→            foreach ($stockTxs as $tx) {
   362→                $this->stockService->reverseTransaction(
   363→                    $tx->id, $cancelledBy,
   364→                    "Challan cancelled: {$reason}"
   365→                );
   366→            }
   367→
   368→            // Reset dispatch rows: dispatched_qty = 0, qty mirrors for GENERATED amount.
   369→            DB::table('sales_invoice_dispatches')
   370→                ->where('sales_invoice_id', $challan->sales_invoice_id)
   371→                ->update([
   372→                    'qty' => 0,
   373→                    'dispatched_qty' => 0,
   374→                ]);
   375→
   376→            // Mark challan as reversed.
   377→            DB::table('sales_challans')
   378→                ->where('id', $challanId)
   379→                ->update([
   380→                    'is_reversed' => true,
   381→                    'reversed_at' => now(),
   382→                    'reversed_by' => $cancelledBy,
   383→                    'reverse_reason' => $reason,
   384→                ]);
   385→
   386→            // P2-3: Reverse transport adjustment (if any).
   387→            // If the challan had a transport_adjustment, reverse the adjustment
   388→            // GL JE + restore the invoice's original transport_cost + total_amount
   389→            // from the pre_challan snapshot.
   390→            if ($challan->adjustment_journal_entry_id) {
   391→                $this->journalPosting->reverseJournalEntry(
   392→                    $challan->adjustment_journal_entry_id, $cancelledBy,
   393→                    "Challan cancelled: transport adjustment reversed — {$reason}"
   394→                );
   395→            }
   396→
   397→            // Restore invoice transport snapshot (if snapshot exists).
   398→            $invoiceRow = DB::table('sales_invoices')
   399→                ->where('id', $challan->sales_invoice_id)
   400→                ->first(['pre_challan_transport', 'pre_challan_total', 'paid_amount']);
   401→
   402→            if ($invoiceRow && $invoiceRow->pre_challan_transport !== null) {
   403→                $restoredTotal = (float) $invoiceRow->pre_challan_total;
   404→                DB::table('sales_invoices')
   405→                    ->where('id', $challan->sales_invoice_id)
   406→                    ->update([
   407→                        'transport_cost' => (float) $invoiceRow->pre_challan_transport,
   408→                        'total_amount' => $restoredTotal,
   409→                        'due_amount' => round($restoredTotal - (float) $invoiceRow->paid_amount, 2),
   410→                        'pre_challan_transport' => null,
   411→                        'pre_challan_total' => null,
   412→                    ]);
   413→            }
   414→
   415→            // P2-2: Reset invoice back to DRAFT state (fully editable).
   416→            // Legacy reverseChallan resets invoice to 'godown_issued' state,
   417→            // but Laravel's edit flow requires status='draft'. Since cancelling
   418→            // a challan invalidates the godown assignment (warehouse_id on items
   419→            // was set during godown prep, and the items may need re-picking),
   420→            // we reset the invoice all the way back to draft.
   421→            DB::table('sales_invoices')
   422→                ->where('id', $challan->sales_invoice_id)
   423→                ->update([
   424→                    'status' => 'draft',
   425→                    'is_challan_issued' => false,
   426→                    'challan_issued_at' => null,
   427→                    'cogs_journal_entry_id' => null,
   428→                    'is_godown_prepared' => false,
   429→                    'godown_prepared_at' => null,
   430→                    'updated_at' => now(),
   431→                ]);
   432→
   433→            // P2-2: Reset warehouse_id on invoice items + dispatches (godown
   434→            // assignment is invalidated — user must re-run godown prep).
   435→            DB::table('sales_invoice_items')
   436→                ->where('sales_invoice_id', $challan->sales_invoice_id)
   437→                ->update(['warehouse_id' => null]);
   438→
   439→            DB::table('sales_invoice_dispatches')
   440→                ->where('sales_invoice_id', $challan->sales_invoice_id)
   441→                ->update(['warehouse_id' => null]);
   442→
   443→            // P1-3: Audit log — challan_reversed.
   444→            $this->auditLogger->challanReversed(
   445→                $cancelledBy,
   446→                $challanId, $challan->challan_code, $challan->sales_invoice_id,
   447→                (int) $challan->branch_id, $reason
   448→            );
   449→
   450→            // P2-7: Invalidate pipeline cache (dispatched_qty reset to 0).
   451→            $this->availabilityService->invalidatePipelineForInvoice($challan->sales_invoice_id);
   452→
   453→            return SalesChallan::find($challanId);
   454→        });
   455→    }

   457→    /**
   458→     * Post GL: Dr COGS / Cr Inventory.
   459→     *
   460→     * @return int journal_entry_id
   461→     */
   462→    private function postCogsGL(int $challanId, string $challanCode, string $challanDate, int $branchId, float $cogsAmount, ?int $createdBy): int
   463→    {
   464→        if ($cogsAmount < 0.01) {
   465→            return 0; // No COGS for zero-value
   466→        }
   467→
   468→        $cogsLedgerId = $this->journalPosting->lookupLedgerByNature('cogs');
   469→        $inventoryLedgerId = $this->journalPosting->lookupLedgerByNature('inventory');
   470→
   471→        if (!$cogsLedgerId) {
   472→            throw new \RuntimeException('COGS ledger not found (nature: cogs).');
   473→        }
   474→        if (!$inventoryLedgerId) {
   475→            throw new \RuntimeException('Inventory ledger not found (nature: inventory).');
   476→        }
   477→
   478→        return $this->journalPosting->createJournalEntry([
   479→            'entry_date' => $challanDate,
   480→            'reference_type' => 'sales_challan',
   481→            'reference_id' => $challanId,
   482→            'branch_id' => $branchId,
   483→            'description' => 'Sales Challan COGS ' . $challanCode,
   484→            'source' => 'sales_challan',
   485→            'created_by' => $createdBy,
   486→        ], [
   487→            [
   488→                'ledger_id' => $cogsLedgerId,
   489→                'debit' => $cogsAmount, 'credit' => 0,
   490→                'entity_type' => 'sales_challan', 'entity_id' => $challanId,
   491→                'memo' => 'Challan ' . $challanCode . ' — COGS',
   492→            ],
   493→            [
   494→                'ledger_id' => $inventoryLedgerId,
   495→                'debit' => 0, 'credit' => $cogsAmount,
   496→                'entity_type' => 'sales_challan', 'entity_id' => $challanId,
   497→                'memo' => 'Challan ' . $challanCode . ' — Inventory issued',
   498→            ],
   499→        ]);
   500→    }

   502→    /**
   503→     * Generate atomic challan code: CH-YYYYMMDD-NNNN.
   504→     */
   505→    private function generateChallanCode(): string
   506→    {
   507→        $datePart = now()->format('Ymd');
   508→        $periodKey = now()->format('Y-m');
   509→        $docType = 'sales_challan';
   510→
   511→        return DB::transaction(function () use ($docType, $periodKey, $datePart) {
   512→            $seqRow = DB::table('document_sequences')
   513→                ->where('doc_type', $docType)
   514→                ->where('branch_id', 0)
   515→                ->where('period_key', $periodKey)
   516→                ->lockForUpdate()
   517→                ->first();
   518→
   519→            $nextNumber = $seqRow ? ((int) $seqRow->last_number + 1) : 1;
   520→
   521→            if ($seqRow) {
   522→                DB::table('document_sequences')->where('id', $seqRow->id)
   523→                    ->update(['last_number' => $nextNumber, 'updated_at' => now()]);
   524→            } else {
   525→                DB::table('document_sequences')->insert([
   526→                    'doc_type' => $docType, 'branch_id' => 0,
   527→                    'period_key' => $periodKey, 'last_number' => $nextNumber,
   528→                    'updated_at' => now(),
   529→                ]);
   530→            }
   531→
   532→            return "CH-{$datePart}-" . str_pad((string) $nextNumber, 4, '0', STR_PAD_LEFT);
   533→        });
   534→    }

   548→    public function getChallanLineItems(int $challanId)

   570→    /**
   571→     * P2-3: Post GL adjustment journal for transport delta.
   572→     *
   573→     * If transport increased (positive adjustment):
   574→     *   Dr AR / Cr Transport Revenue
   575→     * If transport decreased (negative adjustment):
   576→     *   Dr Transport Revenue / Cr AR  (i.e., Cr AR / Dr Transport Revenue swapped)
   577→     *
   578→     * @return int|null journal_entry_id (null if no transport_revenue ledger configured)
   579→     */
   580→    private function postTransportAdjustmentGL(
   581→        int $invoiceId, string $challanCode, string $challanDate, int $branchId,
   582→        float $adjustment, int $customerId, ?int $createdBy
   583→    ): ?int {
   584→        $amount = abs($adjustment);
   585→        if ($amount < 0.01) {
   586→            return null;
   587→        }
   588→
   589→        $arLedgerId = $this->journalPosting->lookupLedgerByNature('ar');
   590→        if (!$arLedgerId) {
   591→            throw new \RuntimeException('Accounts Receivable ledger not found (nature: ar).');
   592→        }
   593→
   594→        $transportLedgerId = $this->journalPosting->lookupLedgerByNature('transport_revenue');
   595→        if (!$transportLedgerId) {
   596→            // Fall back to sales_revenue if transport_revenue not configured.
   597→            $transportLedgerId = $this->journalPosting->lookupLedgerByNature('sales_revenue');
   598→        }
   599→        if (!$transportLedgerId) {
   600→            throw new \RuntimeException('Transport Revenue / Sales Revenue ledger not found.');
   601→        }
   602→
   603→        // Positive adjustment: Dr AR / Cr Transport Revenue
   604→        // Negative adjustment: Dr Transport Revenue / Cr AR (swapped)
   605→        $isIncrease = $adjustment > 0;
   606→
   607→        $lines = [
   608→            [
   609→                'ledger_id' => $arLedgerId,
   610→                'debit' => $isIncrease ? $amount : 0,
   611→                'credit' => $isIncrease ? 0 : $amount,
   612→                'entity_type' => 'customer', 'entity_id' => $customerId,
   613→                'memo' => 'Challan ' . $challanCode . ' — transport adjustment AR',
   614→            ],
   615→            [
   616→                'ledger_id' => $transportLedgerId,
   617→                'debit' => $isIncrease ? 0 : $amount,
   618→                'credit' => $isIncrease ? $amount : 0,
   619→                'entity_type' => 'sales_invoice', 'entity_id' => $invoiceId,
   620→                'memo' => 'Challan ' . $challanCode . ' — transport adjustment revenue',
   621→            ],
   622→        ];
   623→
   624→        return $this->journalPosting->createJournalEntry([
   625→            'entry_date' => $challanDate,
   626→            'reference_type' => 'sales_invoice',
   627→            'reference_id' => $invoiceId,
   628→            'branch_id' => $branchId,
   629→            'description' => 'Transport adjustment — Challan ' . $challanCode,
   630→            'source' => 'sales_challan',
   631→            'created_by' => $createdBy,
   632→        ], $lines);
   633→    }
