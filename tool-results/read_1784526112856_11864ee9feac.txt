    14→/**
    15→ * Sales Invoice Service — Phase 8.2.
    16→ *
    17→ * Finalizes a cart into a draft sales invoice:
    18→ *   1. Validate cart (stock availability + price range)
    19→ *   2. Check credit limit (with optional override)
    20→ *   3. Lock branch products FOR UPDATE (prevent race conditions)
    21→ *   4. Create sales_invoice (status=draft)
    22→ *   5. Create sales_invoice_items (from cart, warehouse_id=NULL)
    23→ *   6. Create sales_invoice_dispatches (ordered_qty, dispatched_qty=0)
    24→ *   7. Post customer_ledger debit (customer owes more)
    25→ *   8. Post GL: Dr Accounts Receivable / Cr Sales Revenue
    26→ *      (+ Dr Discount / Cr Transport if applicable)
    27→ *   9. Clear the cart
    28→ *
    29→ * The invoice is a DRAFT at this point — no stock movement yet.
    30→ * Stock moves on challan issue (Phase 8.3).
    31→ */
    32→class SalesInvoiceService
    33→{
    34→    public function __construct(
    35→        private SalesCartService $cartService,
    36→        private StockAvailabilityService $availabilityService,
    37→        private StockService $stockService,
    38→        private JournalPostingService $journalPosting,
    39→        private SubLedgerService $subLedger,
    40→        private SalesAccess $salesAccess,
    41→        private SalesAuditLogger $auditLogger
    42→    ) {}
    43→
    44→    /**
    45→     * Finalize a cart into a draft sales invoice.
    46→     *
    47→     * @param array $data {
    48→     *     customer_id: int,
    49→     *     branch_id: int,
    50→     *     invoice_date: string (Y-m-d),
    51→     *     salesman_id: int|null,
    52→     *     sales_person: string|null,
    53→     *     discount_amount: float,
    54→     *     transport_cost: float,
    55→     *     notes: string|null,
    56→     *     is_soft_hold: bool,
    57→     *     credit_limit_override: bool,
    58→     *     override_reason: string|null,
    59→     *     created_by: int,
    60→     * }
    61→     * @return SalesInvoice
    62→     * @throws \RuntimeException If cart empty, stock insufficient, credit limit exceeded without override, or GL posting fails.
    63→     */
    64→    public function finalizeFromCart(array $data): SalesInvoice
    65→    {
    66→        $customerId = (int) $data['customer_id'];
    67→        $branchId = (int) $data['branch_id'];
    68→
    69→        if ($customerId <= 0) {
    70→            throw new \InvalidArgumentException('customer_id is required.');
    71→        }
    72→        if ($branchId <= 0) {
    73→            throw new \InvalidArgumentException('branch_id is required.');
    74→        }
    75→
    76→        // P0-8: Defense-in-depth branch isolation check.
    77→        $this->salesAccess->assertBranchAccessible($branchId);
    78→
    79→        // Step 1: Load + validate the cart.
    80→        $cartData = $this->cartService->getCart($data['created_by'] ?? auth()->id(), $customerId, $branchId);
    81→        $items = $cartData['items'];
    82→
    83→        if (empty($items)) {
    84→            throw new \RuntimeException('Cart is empty. Add products before finalizing.');
    85→        }
    86→
    87→        $validation = $cartData['validation'];
    88→        if (!$validation['valid']) {
    89→            throw new \RuntimeException('Cart validation failed: ' . $validation['message']);
    90→        }
    91→
    92→        // Step 2: Calculate totals.
    93→        $subTotal = collect($items)->sum('total');
    94→        $discount = (float) ($data['discount_amount'] ?? 0);
    95→        $transport = (float) ($data['transport_cost'] ?? 0);
    96→        $totalAmount = $subTotal + $transport - $discount;
    97→
    98→        // Step 3: Credit limit check.
    99→        $creditCheck = $this->checkCreditLimit($customerId, $totalAmount);
   100→        $isOverride = !empty($data['credit_limit_override']);
   101→        $overrideReason = trim($data['override_reason'] ?? '');
   102→
   103→        if ($creditCheck['exceeds'] && !$isOverride) {
   104→            throw new \RuntimeException(
   105→                "Credit limit exceeded. Customer balance: {$creditCheck['current_balance']}, "
   106→                . "Credit limit: {$creditCheck['credit_limit']}, "
   107→                . "This invoice: {$totalAmount}. "
   108→                . "Override with a reason to proceed."
   109→            );
   110→        }
   111→
   112→        if ($creditCheck['exceeds'] && $isOverride && strlen($overrideReason) < 10) {
   113→            throw new \RuntimeException('Override reason must be at least 10 characters when exceeding credit limit.');
   114→        }
   115→
   116→        // Step 4: Atomic finalize in a DB transaction.
   117→        return DB::transaction(function () use (
   118→            $data, $items, $customerId, $branchId, $subTotal, $discount, $transport, $totalAmount,
   119→            $creditCheck, $isOverride, $overrideReason
   120→        ) {
   121→            // Lock branch products FOR UPDATE (prevent race conditions on concurrent finalizes).
   122→            $productIds = collect($items)->pluck('product_id')->unique()->toArray();
   123→            $this->stockService->lockBranchProductsForUpdate($branchId, $productIds);
   124→
   125→            // Re-check availability after locking.
   126→            $qtyByProduct = [];
   127→            foreach ($items as $item) {
   128→                $pid = (int) $item['product_id'];
   129→                $qtyByProduct[$pid] = ($qtyByProduct[$pid] ?? 0) + (float) $item['qty'];
   130→            }
   131→
   132→            foreach ($qtyByProduct as $pid => $qty) {
   133→                $available = $this->availabilityService->getBranchAvailableQty($pid, $branchId);
   134→                if ($qty > $available + 0.0001) {
   135→                    throw new \RuntimeException(
   136→                        "Insufficient stock for product {$pid}: requested {$qty}, available {$available}"
   137→                    );
   138→                }
   139→            }
   140→
   141→            // Generate invoice code.
   142→            $invoiceCode = $this->generateInvoiceCode();
   143→
   144→            // Step 5: Create sales_invoice header.
   145→            $invoiceId = DB::table('sales_invoices')->insertGetId([
   146→                'invoice_code' => $invoiceCode,
   147→                'invoice_date' => $data['invoice_date'] ?? now()->format('Y-m-d'),
   148→                'customer_id' => $customerId,
   149→                'salesman_id' => $data['salesman_id'] ?? null,
   150→                'sales_person' => $data['sales_person'] ?? null,
   151→                'branch_id' => $branchId,
   152→                'sub_total' => round($subTotal, 2),
   153→                'discount_amount' => round($discount, 2),
   154→                'transport_cost' => round($transport, 2),
   155→                'total_amount' => round($totalAmount, 2),
   156→                'paid_amount' => 0,
   157→                'due_amount' => round($totalAmount, 2),
   158→                'payment_mode' => 'cash',
   159→                'status' => 'draft',
   160→                'is_godown_prepared' => false,
   161→                'is_challan_issued' => false,
   162→                'is_reversed' => false,
   163→                'is_soft_hold' => $data['is_soft_hold'] ?? false,
   164→                'notes' => $data['notes'] ?? null,
   165→                'created_by' => $data['created_by'] ?? null,
   166→                'created_at' => now(),
   167→                'updated_at' => now(),
   168→            ]);
   169→
   170→            // Step 6: Create sales_invoice_items (warehouse_id=NULL — assigned at godown).
   171→            $itemRows = [];
   172→            foreach ($items as $item) {
   173→                $itemRows[] = [
   174→                    'sales_invoice_id' => $invoiceId,
   175→                    'product_id' => (int) $item['product_id'],
   176→                    'warehouse_id' => null, // assigned at godown (Phase 8.3)
   177→                    'qty' => (float) $item['qty'],
   178→                    'rate' => (float) $item['rate'],
   179→                    'condition_state' => 'Good',
   180→                ];
   181→            }
   182→            DB::table('sales_invoice_items')->insert($itemRows);
   183→
   184→            // Step 7: Create sales_invoice_dispatches (pipeline tracking).
   185→            $dispatchRows = [];
   186→            foreach ($items as $item) {
   187→                $dispatchRows[] = [
   188→                    'sales_invoice_id' => $invoiceId,
   189→                    'product_id' => (int) $item['product_id'],
   190→                    'warehouse_id' => null, // assigned at godown
   191→                    'qty' => (float) $item['qty'], // mirrors ordered_qty for GENERATED amount
   192→                    'ordered_qty' => (float) $item['qty'],
   193→                    'dispatched_qty' => 0,
   194→                    'created_by' => $data['created_by'] ?? null,
   195→                ];
   196→            }
   197→            DB::table('sales_invoice_dispatches')->insert($dispatchRows);
   198→
   199→            // Step 8: Post GL FIRST to get journal_entry_id.
   200→            $journalEntryId = $this->postInvoiceGL(
   201→                $invoiceId, $invoiceCode, $customerId, $branchId,
   202→                $subTotal, $discount, $transport, $totalAmount,
   203→                $data['invoice_date'] ?? now()->format('Y-m-d'),
   204→                $data['created_by'] ?? null
   205→            );
   206→
   207→            // Step 9: Post customer_ledger debit via SubLedgerService (customer owes more).
   208→            $this->subLedger->postCustomerLedgerEntry([
   209→                'customer_id' => $customerId,
   210→                'branch_id' => $branchId,
   211→                'transaction_date' => $data['invoice_date'] ?? now()->format('Y-m-d'),
   212→                'transaction_type' => 'sales_invoice',
   213→                'reference_type' => 'sales_invoice',
   214→                'reference_id' => $invoiceId,
   215→                'debit' => $totalAmount,
   216→                'credit' => 0,
   217→                'description' => 'Invoice ' . $invoiceCode,
   218→                'journal_entry_id' => $journalEntryId,
   219→                'created_by' => $data['created_by'] ?? null,
   220→            ]);
   221→
   222→            // Update invoice with journal_entry_id.
   223→            DB::table('sales_invoices')
   224→                ->where('id', $invoiceId)
   225→                ->update(['journal_entry_id' => $journalEntryId, 'updated_at' => now()]);
   226→
   227→            // Step 10: Clear the cart.
   228→            $this->cartService->clearCart($data['created_by'] ?? auth()->id(), $customerId);
   229→
   230→            // Audit log for credit limit override.
   231→            if ($isOverride && $creditCheck['exceeds']) {
   232→                DB::table('user_audit_log')->insert([
   233→                    'user_id' => $data['created_by'] ?? null,
   234→                    'action' => 'credit_limit_override',
   235→                    'target_user_id' => null,
   236→                    'branch_id' => $branchId,
   237→                    'details' => json_encode([
   238→                        'invoice_id' => $invoiceId,
   239→                        'invoice_code' => $invoiceCode,
   240→                        'customer_id' => $customerId,
   241→                        'total_amount' => $totalAmount,
   242→                        'credit_limit' => $creditCheck['credit_limit'],
   243→                        'current_balance' => $creditCheck['current_balance'],
   244→                        'override_reason' => $overrideReason,
   245→                    ]),
   246→                    'ip_address' => request()?->ip(),
   247→                    'user_agent' => request()?->userAgent() ? mb_substr(request()->userAgent(), 0, 255) : null,
   248→                    'created_at' => now(),
   249→                ]);
   250→            }
   251→
   252→            // P1-3: Audit log — sale_created.
   253→            $this->auditLogger->saleCreated(
   254→                $data['created_by'] ?? auth()->id() ?? 0,
   255→                $invoiceId, $invoiceCode, $customerId, $branchId,
   256→                $totalAmount, $data['salesman_id'] ?? null
   257→            );
   258→
   259→            // P2-7: Invalidate pipeline cache (new dispatches were added).
   260→            $this->availabilityService->invalidatePipelineForInvoice($invoiceId);
   261→
   262→            return SalesInvoice::with(['items.product', 'customer', 'branch', 'journalEntry.lines.ledger'])
   263→                ->find($invoiceId);
   264→        });
   265→    }

   267→    /**
   268→     * Cancel a draft invoice (reverse GL + customer_ledger + dispatches).
   269→     * Only draft invoices can be cancelled (before godown/challan).
   270→     *
   271→     * @param int $invoiceId
   272→     * @param int $cancelledBy
   273→     * @param string $reason
   274→     * @return SalesInvoice
   275→     */
   276→    public function cancelInvoice(int $invoiceId, int $cancelledBy, string $reason = ''): SalesInvoice
   277→    {
   278→        return DB::transaction(function () use ($invoiceId, $cancelledBy, $reason) {
   279→            $invoice = SalesInvoice::lockForUpdate()->find($invoiceId);
   280→
   281→            if (!$invoice) {
   282→                throw new \RuntimeException("Invoice {$invoiceId} not found.");
   283→            }
   284→
   285→            // P0-8: Defense-in-depth branch isolation check.
   286→            $this->salesAccess->assertBranchAccessible((int) $invoice->branch_id);
   287→
   288→            if (!$invoice->isDraft()) {
   289→                throw new \RuntimeException("Only draft invoices can be cancelled (current: {$invoice->status}). Cancel via reversal for confirmed invoices.");
   290→            }
   291→
   292→            // P2-2: Explicit guards with clear messages (legacy layered checks).
   293→            if ($this->hasActiveChallan($invoiceId)) {
   294→                throw new \RuntimeException("Cannot cancel: a delivery challan exists for this invoice. Reverse the challan first.");
   295→            }
   296→            if ($this->invoiceHasPayments($invoiceId)) {
   297→                throw new \RuntimeException("Cannot cancel: payments have been received against this invoice. Reverse the payments first.");
   298→            }
   299→
   300→            // Reverse GL.
   301→            if ($invoice->journal_entry_id) {
   302→                $this->journalPosting->reverseJournalEntry(
   303→                    $invoice->journal_entry_id, $cancelledBy,
   304→                    "Invoice cancelled: {$reason}"
   305→                );
   306→            }
   307→
   308→            // Reverse customer_ledger via SubLedgerService (credit entry to reduce what customer owes).
   309→            $this->subLedger->postCustomerLedgerEntry([
   310→                'customer_id' => $invoice->customer_id,
   311→                'branch_id' => $invoice->branch_id,
   312→                'transaction_date' => now()->format('Y-m-d'),
   313→                'transaction_type' => 'sales_invoice_reversal',
   314→                'reference_type' => 'sales_invoice',
   315→                'reference_id' => $invoice->id,
   316→                'debit' => 0,
   317→                'credit' => (float) $invoice->total_amount,
   318→                'description' => 'Invoice reversal ' . $invoice->invoice_code . ": {$reason}",
   319→                'journal_entry_id' => $invoice->journal_entry_id,
   320→                'created_by' => $cancelledBy,
   321→            ]);
   322→
   323→            // Mark invoice as cancelled.
   324→            DB::table('sales_invoices')
   325→                ->where('id', $invoiceId)
   326→                ->update([
   327→                    'status' => 'cancelled',
   328→                    'is_reversed' => true,
   329→                    'reversed_at' => now(),
   330→                    'reversed_by' => $cancelledBy,
   331→                    'reverse_reason' => $reason,
   332→                    'updated_at' => now(),
   333→                ]);
   334→
   335→            // P1-3: Audit log — sale_cancelled.
   336→            $this->auditLogger->saleCancelled(
   337→                $cancelledBy,
   338→                $invoiceId, $invoice->invoice_code, (int) $invoice->branch_id,
   339→                (float) $invoice->total_amount, $reason
   340→            );
   341→
   342→            // P2-7: Invalidate pipeline cache (dispatches were deleted).
   343→            $this->availabilityService->invalidatePipelineForInvoice($invoiceId);
   344→
   345→            return SalesInvoice::find($invoiceId);
   346→        });
   347→    }

   349→    /**
   350→     * Update an existing DRAFT invoice (P1-1).
   351→     *
   352→     * Restores the legacy updateExistingInvoice flow:
   353→     *   1. Assert editable (draft, no godown, no payments, not reversed)
   354→     *   2. Re-validate credit limit (using NET increase = max(0, newTotal - oldTotal))
   355→     *   3. Lock branch products FOR UPDATE + re-check availability
   356→     *   4. Reverse old customer_ledger debit (append-only credit entry)
   357→     *   5. Reverse old GL journal entry (append-only)
   358→     *   6. DELETE old items + dispatches + dispatchers
   359→     *   7. INSERT new items + dispatches (soft reservation, warehouse_id=NULL)
   360→     *   8. Post new customer_ledger debit (new total)
   361→     *   9. Post new GL journal entry (Dr AR / Cr Revenue + Discount + Transport)
   362→     *  10. UPDATE invoice header (sub_total, discount, transport, total, notes, etc.)
   363→     *
   364→     * @param int $invoiceId
   365→     * @param array $data {
   366→     *     items: array of {product_id, qty, rate},
   367→     *     invoice_date: string,
   368→     *     sales_person: string|null,
   369→     *     discount_amount: float,
   370→     *     transport_cost: float,
   371→     *     notes: string|null,
   372→     *     is_soft_hold: bool,
   373→     *     credit_limit_override: bool,
   374→     *     override_reason: string|null,
   375→     *     updated_by: int,
   376→     * }
   377→     * @return SalesInvoice
   378→     * @throws \RuntimeException If not editable, stock insufficient, credit exceeded, or GL fails.
   379→     */
   380→    public function updateInvoice(int $invoiceId, array $data): SalesInvoice
   381→    {
   382→        $items = $data['items'] ?? [];
   383→        $updatedBy = (int) ($data['updated_by'] ?? auth()->id() ?? 0);
   384→
   385→        if (empty($items)) {
   386→            throw new \RuntimeException('Cannot update: items list is empty.');
   387→        }
   388→
   389→        // --- Pre-flight: load invoice (without transaction yet) for validation ---
   390→        $invoice = SalesInvoice::find($invoiceId);
   391→        if (!$invoice) {
   392→            throw new \RuntimeException("Invoice {$invoiceId} not found.");
   393→        }
   394→
   395→        // P0-8: Defense-in-depth branch isolation check.
   396→        $this->salesAccess->assertBranchAccessible((int) $invoice->branch_id);
   397→
   398→        // Assert editable.
   399→        $this->assertEditable($invoice);
   400→
   401→        // Assert no payments.
   402→        if ($this->invoiceHasPayments($invoiceId)) {
   403→            throw new \RuntimeException('Cannot edit: payments have been received against this invoice. Reverse the payments first.');
   404→        }
   405→
   406→        $oldTotal = (float) $invoice->total_amount;
   407→        $customerId = (int) $invoice->customer_id;
   408→        $branchId = (int) $invoice->branch_id;
   409→        $oldJournalId = $invoice->journal_entry_id ? (int) $invoice->journal_entry_id : null;
   410→
   411→        // --- Calculate new totals ---
   412→        $subTotal = 0.0;
   413→        foreach ($items as $item) {
   414→            $subTotal += (float) $item['qty'] * (float) $item['rate'];
   415→        }
   416→        $discount = (float) ($data['discount_amount'] ?? 0);
   417→        $transport = (float) ($data['transport_cost'] ?? 0);
   418→        $newTotal = $subTotal + $transport - $discount;
   419→
   420→        // --- Credit limit check (NET increase, not full new total) ---
   421→        $netIncrease = max(0.0, $newTotal - $oldTotal);
   422→        $creditCheck = $this->checkCreditLimit($customerId, $netIncrease);
   423→        $isOverride = !empty($data['credit_limit_override']);
   424→        $overrideReason = trim($data['override_reason'] ?? '');
   425→
   426→        if ($creditCheck['exceeds'] && !$isOverride) {
   427→            throw new \RuntimeException(
   428→                "Updating this invoice would exceed the customer's credit limit. "
   429→                . "Current balance: {$creditCheck['current_balance']}, "
   430→                . "Credit limit: {$creditCheck['credit_limit']}, "
   431→                . "Net increase: {$netIncrease}. "
   432→                . "Override with a reason to proceed."
   433→            );
   434→        }
   435→
   436→        if ($creditCheck['exceeds'] && $isOverride && strlen($overrideReason) < 10) {
   437→            throw new \RuntimeException('Override reason must be at least 10 characters when exceeding credit limit.');
   438→        }
   439→
   440→        $invoiceDate = $data['invoice_date'] ?? $invoice->invoice_date;
   441→
   442→        // --- Atomic update in a DB transaction ---
   443→        return DB::transaction(function () use (
   444→            $invoiceId, $invoice, $items, $customerId, $branchId, $oldTotal, $newTotal,
   445→            $subTotal, $discount, $transport, $invoiceDate, $oldJournalId,
   446→            $creditCheck, $isOverride, $overrideReason, $updatedBy, $data
   447→        ) {
   448→            // Lock the invoice row FOR UPDATE.
   449→            $locked = SalesInvoice::lockForUpdate()->find($invoiceId);
   450→            if (!$locked) {
   451→                throw new \RuntimeException("Invoice {$invoiceId} not found (locked).");
   452→            }
   453→
   454→            // Re-assert editable after lock (race protection).
   455→            $this->assertEditable($locked);
   456→
   457→            // Lock branch products FOR UPDATE.
   458→            $productIds = collect($items)->pluck('product_id')->unique()->toArray();
   459→            $this->stockService->lockBranchProductsForUpdate($branchId, $productIds);
   460→
   461→            // Re-check stock availability (excluding this invoice's own pipeline).
   462→            $qtyByProduct = [];
   463→            foreach ($items as $item) {
   464→                $pid = (int) $item['product_id'];
   465→                $qtyByProduct[$pid] = ($qtyByProduct[$pid] ?? 0) + (float) $item['qty'];
   466→            }
   467→
   468→            foreach ($qtyByProduct as $pid => $qty) {
   469→                $available = $this->availabilityService->getBranchAvailableQty($pid, $branchId, $invoiceId);
   470→                if ($qty > $available + 0.0001) {
   471→                    throw new \RuntimeException(
   472→                        "Insufficient stock for product {$pid}: requested {$qty}, available {$available}"
   473→                    );
   474→                }
   475→            }
   476→
   477→            // Step 1: Reverse old customer_ledger debit via SubLedgerService (append-only credit entry).
   478→            $this->subLedger->postCustomerLedgerEntry([
   479→                'customer_id' => $customerId,
   480→                'branch_id' => $branchId,
   481→                'transaction_date' => now()->format('Y-m-d'),
   482→                'transaction_type' => 'sales_invoice_reversal',
   483→                'reference_type' => 'sales_invoice',
   484→                'reference_id' => $invoiceId,
   485→                'debit' => 0,
   486→                'credit' => $oldTotal,
   487→                'description' => 'Invoice edit reversal ' . $locked->invoice_code,
   488→                'journal_entry_id' => $oldJournalId,
   489→                'created_by' => $updatedBy,
   490→            ]);
   491→
   492→            // Step 2: Reverse old GL journal entry (append-only).
   493→            if ($oldJournalId) {
   494→                $this->journalPosting->reverseJournalEntry(
   495→                    $oldJournalId, $updatedBy,
   496→                    'Invoice edited: ' . $invoice->invoice_code
   497→                );
   498→            }
   499→
   500→            // Step 3: DELETE old items + dispatches + dispatchers.
   501→            DB::table('sales_invoice_items')->where('sales_invoice_id', $invoiceId)->delete();
   502→            DB::table('sales_invoice_dispatches')->where('sales_invoice_id', $invoiceId)->delete();
   503→            DB::table('sales_invoice_dispatchers')->where('sales_invoice_id', $invoiceId)->delete();
   504→
   505→            // Step 4: INSERT new items.
   506→            $itemRows = [];
   507→            foreach ($items as $item) {
   508→                $itemRows[] = [
   509→                    'sales_invoice_id' => $invoiceId,
   510→                    'product_id' => (int) $item['product_id'],
   511→                    'warehouse_id' => null, // reset to NULL — godown must re-assign after edit
   512→                    'qty' => (float) $item['qty'],
   513→                    'rate' => (float) $item['rate'],
   514→                    'condition_state' => $item['condition_state'] ?? 'Good',
   515→                ];
   516→            }
   517→            DB::table('sales_invoice_items')->insert($itemRows);
   518→
   519→            // Step 5: INSERT new dispatches (soft reservation, warehouse_id=NULL).
   520→            $dispatchRows = [];
   521→            foreach ($items as $item) {
   522→                $dispatchRows[] = [
   523→                    'sales_invoice_id' => $invoiceId,
   524→                    'product_id' => (int) $item['product_id'],
   525→                    'warehouse_id' => null,
   526→                    'qty' => (float) $item['qty'],
   527→                    'ordered_qty' => (float) $item['qty'],
   528→                    'dispatched_qty' => 0,
   529→                    'created_by' => $updatedBy,
   530→                ];
   531→            }
   532→            DB::table('sales_invoice_dispatches')->insert($dispatchRows);
   533→
   534→            // Step 6: Post GL FIRST to get new journal_entry_id.
   535→            $newJournalEntryId = $this->postInvoiceGL(
   536→                $invoiceId, $invoice->invoice_code, $customerId, $branchId,
   537→                $subTotal, $discount, $transport, $newTotal,
   538→                $invoiceDate, $updatedBy
   539→            );
   540→
   541→            // Step 7: Post new customer_ledger debit via SubLedgerService.
   542→            $this->subLedger->postCustomerLedgerEntry([
   543→                'customer_id' => $customerId,
   544→                'branch_id' => $branchId,
   545→                'transaction_date' => $invoiceDate,
   546→                'transaction_type' => 'sales_invoice',
   547→                'reference_type' => 'sales_invoice',
   548→                'reference_id' => $invoiceId,
   549→                'debit' => $newTotal,
   550→                'credit' => 0,
   551→                'description' => 'Invoice ' . $invoice->invoice_code . ' (edited)',
   552→                'journal_entry_id' => $newJournalEntryId,
   553→                'created_by' => $updatedBy,
   554→            ]);
   555→
   556→            // Step 8: UPDATE invoice header.
   557→            DB::table('sales_invoices')
   558→                ->where('id', $invoiceId)
   559→                ->update([
   560→                    'invoice_date' => $invoiceDate,
   561→                    'sales_person' => $data['sales_person'] ?? null,
   562→                    'sub_total' => round($subTotal, 2),
   563→                    'discount_amount' => round($discount, 2),
   564→                    'transport_cost' => round($transport, 2),
   565→                    'total_amount' => round($newTotal, 2),
   566→                    'due_amount' => round($newTotal, 2) - (float) $invoice->paid_amount,
   567→                    'is_soft_hold' => $data['is_soft_hold'] ?? false,
   568→                    'notes' => $data['notes'] ?? null,
   569→                    'journal_entry_id' => $newJournalEntryId,
   570→                    // Reset godown/challan flags (edit invalidates prior godown prep).
   571→                    'is_godown_prepared' => false,
   572→                    'godown_prepared_at' => null,
   573→                    'updated_at' => now(),
   574→                ]);
   575→
   576→            // Step 9: Audit log for credit limit override.
   577→            if ($isOverride && $creditCheck['exceeds']) {
   578→                DB::table('user_audit_log')->insert([
   579→                    'user_id' => $updatedBy,
   580→                    'action' => 'credit_limit_override',
   581→                    'target_user_id' => null,
   582→                    'branch_id' => $branchId,
   583→                    'details' => json_encode([
   584→                        'action' => 'invoice_edit',
   585→                        'invoice_id' => $invoiceId,
   586→                        'invoice_code' => $invoice->invoice_code,
   587→                        'customer_id' => $customerId,
   588→                        'old_total' => $oldTotal,
   589→                        'new_total' => $newTotal,
   590→                        'net_increase' => $netIncrease,
   591→                        'credit_limit' => $creditCheck['credit_limit'],
   592→                        'current_balance' => $creditCheck['current_balance'],
   593→                        'override_reason' => $overrideReason,
   594→                    ]),
   595→                    'ip_address' => request()?->ip(),
   596→                    'user_agent' => request()?->userAgent() ? mb_substr(request()->userAgent(), 0, 255) : null,
   597→                    'created_at' => now(),
   598→                ]);
   599→            }
   600→
   601→            // Audit log for the edit itself.
   602→            DB::table('user_audit_log')->insert([
   603→                'user_id' => $updatedBy,
   604→                'action' => 'sale_updated',
   605→                'target_user_id' => null,
   606→                'branch_id' => $branchId,
   607→                'details' => json_encode([
   608→                    'invoice_id' => $invoiceId,
   609→                    'invoice_code' => $invoice->invoice_code,
   610→                    'old_total' => $oldTotal,
   611→                    'new_total' => $newTotal,
   612→                    'items_count' => count($items),
   613→                    'credit_override' => $isOverride && $creditCheck['exceeds'],
   614→                ]),
   615→                'ip_address' => request()?->ip(),
   616→                'user_agent' => request()?->userAgent() ? mb_substr(request()->userAgent(), 0, 255) : null,
   617→                'created_at' => now(),
   618→            ]);
   619→
   620→            // P2-7: Invalidate pipeline cache (dispatches were deleted + re-inserted).
   621→            $this->availabilityService->invalidatePipelineForInvoice($invoiceId);
   622→
   623→            return SalesInvoice::with(['items.product', 'customer', 'branch', 'journalEntry.lines.ledger'])
   624→                ->find($invoiceId);
   625→        });
   626→    }

   628→    /**
   629→     * Assert an invoice is editable (P1-1).
   630→     * Must be: status='draft', not godown-prepared, not reversed.
   631→     */
   632→    private function assertEditable(SalesInvoice $invoice): void
   633→    {
   634→        if ($invoice->is_reversed) {
   635→            throw new \RuntimeException('Cannot edit: invoice is reversed.');
   636→        }
   637→        if ($invoice->status !== 'draft') {
   638→            throw new \RuntimeException("Cannot edit: invoice status is '{$invoice->status}'. Only draft invoices can be edited.");
   639→        }
   640→        if ($invoice->is_godown_prepared) {
   641→            throw new \RuntimeException('Cannot edit: godown has already been prepared for this invoice.');
   642→        }
   643→        if ($invoice->is_challan_issued) {
   644→            throw new \RuntimeException('Cannot edit: a delivery challan has already been issued for this invoice.');
   645→        }
   646→    }

   651→    private function invoiceHasPayments(int $invoiceId): bool

   671→    /**
   672→     * Check if the invoice would exceed the customer's credit limit.
   673→     *
   674→     * @return array{ exceeds: bool, current_balance: float, credit_limit: float, new_balance: float }
   675→     */
   676→    private function checkCreditLimit(int $customerId, float $invoiceAmount): array
   677→    {
   678→        $customer = DB::table('customers')->where('id', $customerId)->first();
   679→
   680→        if (!$customer) {
   681→            return ['exceeds' => false, 'current_balance' => 0, 'credit_limit' => 0, 'new_balance' => 0];
   682→        }
   683→
   684→        $creditLimit = (float) ($customer->credit_limit ?? 0);
   685→
   686→        // Current AR balance from customer_ledger (sum of debit - credit).
   687→        $currentBalance = (float) DB::table('customer_ledger')
   688→            ->where('customer_id', $customerId)
   689→            ->where('is_reversed', false)
   690→            ->selectRaw('COALESCE(SUM(debit) - SUM(credit), 0) as balance')
   691→            ->value('balance');
   692→
   693→        $newBalance = $currentBalance + $invoiceAmount;
   694→        $exceeds = $creditLimit > 0 && $newBalance > $creditLimit + 0.01;
   695→
   696→        return [
   697→            'exceeds' => $exceeds,
   698→            'current_balance' => $currentBalance,
   699→            'credit_limit' => $creditLimit,
   700→            'new_balance' => $newBalance,
   701→        ];
   702→    }

   704→    /**
   705→     * Post GL: Dr Accounts Receivable / Cr Sales Revenue.
   706→     * + Dr Discount (if discount > 0)
   707→     * + Cr Transport Revenue (if transport > 0)
   708→     *
   709→     * @return int journal_entry_id
   710→     */
   711→    private function postInvoiceGL(
   712→        int $invoiceId, string $invoiceCode, int $customerId, int $branchId,
   713→        float $subTotal, float $discount, float $transport, float $total,
   714→        string $invoiceDate, ?int $createdBy
   715→    ): int {
   716→        $arLedgerId = $this->journalPosting->lookupLedgerByNature('ar');
   717→        $revenueLedgerId = $this->journalPosting->lookupLedgerByNature('sales_revenue');
   718→
   719→        if (!$arLedgerId) {
   720→            throw new \RuntimeException('Accounts Receivable ledger not found (nature: ar).');
   721→        }
   722→        if (!$revenueLedgerId) {
   723→            throw new \RuntimeException('Sales Revenue ledger not found (nature: sales_revenue).');
   724→        }
   725→
   726→        $lines = [];
   727→
   728→        // Dr Accounts Receivable (total)
   729→        $lines[] = [
   730→            'ledger_id' => $arLedgerId,
   731→            'debit' => $total, 'credit' => 0,
   732→            'entity_type' => 'customer', 'entity_id' => $customerId,
   733→            'memo' => 'Invoice ' . $invoiceCode . ' — AR',
   734→        ];
   735→
   736→        // Cr Sales Revenue (subTotal - discount, or subTotal if no discount ledger)
   737→        $discountLedgerId = $discount > 0.01 ? $this->journalPosting->lookupLedgerByNature('sales_discount') : null;
   738→        $revenueAmount = $discountLedgerId ? $subTotal : max(0, $subTotal - $discount);
   739→
   740→        $lines[] = [
   741→            'ledger_id' => $revenueLedgerId,
   742→            'debit' => 0, 'credit' => $revenueAmount,
   743→            'entity_type' => 'sales_invoice', 'entity_id' => $invoiceId,
   744→            'memo' => 'Invoice ' . $invoiceCode . ' — Revenue',
   745→        ];
   746→
   747→        // Dr Discount (if applicable)
   748→        if ($discountLedgerId && $discount > 0.01) {
   749→            $lines[] = [
   750→                'ledger_id' => $discountLedgerId,
   751→                'debit' => $discount, 'credit' => 0,
   752→                'entity_type' => 'sales_invoice', 'entity_id' => $invoiceId,
   753→                'memo' => 'Invoice ' . $invoiceCode . ' — Discount allowed',
   754→            ];
   755→        }
   756→
   757→        // Cr Transport Revenue (if applicable)
   758→        if ($transport > 0.01) {
   759→            $transportLedgerId = $this->journalPosting->lookupLedgerByNature('transport_revenue');
   760→            if ($transportLedgerId) {
   761→                $lines[] = [
   762→                    'ledger_id' => $transportLedgerId,
   763→                    'debit' => 0, 'credit' => $transport,
   764→                    'entity_type' => 'sales_invoice', 'entity_id' => $invoiceId,
   765→                    'memo' => 'Invoice ' . $invoiceCode . ' — Transport',
   766→                ];
   767→            }
   768→        }
   769→
   770→        return $this->journalPosting->createJournalEntry([
   771→            'entry_date' => $invoiceDate,
   772→            'reference_type' => 'sales_invoice',
   773→            'reference_id' => $invoiceId,
   774→            'branch_id' => $branchId,
   775→            'description' => 'Sales Invoice ' . $invoiceCode,
   776→            'source' => 'sales_invoice',
   777→            'created_by' => $createdBy,
   778→        ], $lines);
   779→    }

   781→    /**
   782→     * Generate atomic invoice code: INV-YYYYMMDD-NNNN.
   783→     */
   784→    private function generateInvoiceCode(): string
   785→    {
   786→        $datePart = now()->format('Ymd');
   787→        $periodKey = now()->format('Y-m');
   788→        $docType = 'sales_invoice';
   789→
   790→        return DB::transaction(function () use ($docType, $periodKey, $datePart) {
   791→            $seqRow = DB::table('document_sequences')
   792→                ->where('doc_type', $docType)
   793→                ->where('branch_id', 0)
   794→                ->where('period_key', $periodKey)
   795→                ->lockForUpdate()
   796→                ->first();
   797→
   798→            $nextNumber = $seqRow ? ((int) $seqRow->last_number + 1) : 1;
   799→
   800→            if ($seqRow) {
   801→                DB::table('document_sequences')->where('id', $seqRow->id)
   802→                    ->update(['last_number' => $nextNumber, 'updated_at' => now()]);
   803→            } else {
   804→                DB::table('document_sequences')->insert([
   805→                    'doc_type' => $docType, 'branch_id' => 0,
   806→                    'period_key' => $periodKey, 'last_number' => $nextNumber,
   807→                    'updated_at' => now(),
   808→                ]);
   809→            }
   810→
   811→            return "INV-{$datePart}-" . str_pad((string) $nextNumber, 4, '0', STR_PAD_LEFT);
   812→        });
   813→    }
