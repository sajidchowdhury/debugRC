    96→     * Link orphan return journals, backfill customer_ledger, post postSalesReturn from header + stock IN.
    97→     *
    98→     * @return array{linked: int, posted: int, ledgers_added: int, errors: string[]}

   100→    public function repairMissingSalesReturnJournals(): array
   101→    {
   102→        $result = ['linked' => 0, 'posted' => 0, 'ledgers_added' => 0, 'errors' => []];
   103→
   104→        try {
   105→            $branchSql = $this->branchId
   106→                ? ' AND EXISTS (
   107→                    SELECT 1 FROM sales_invoices si
   108→                    WHERE si.id = sr.sales_invoice_id AND si.branch_id = ' . (int)$this->branchId . '
   109→                )'
   110→                : '';
   111→
   112→            $this->db->query("
   113→                UPDATE sales_returns sr
   114→                INNER JOIN journal_entries je ON je.reference_type = 'sales_return'
   115→                    AND je.reference_id = sr.id
   116→                    AND COALESCE(je.is_reversed, 0) = 0
   117→                SET sr.journal_entry_id = je.id
   118→                WHERE sr.status = 'completed'
   119→                  AND COALESCE(sr.is_reversed, 0) = 0
   120→                  AND COALESCE(sr.journal_entry_id, 0) = 0
   121→                  {$branchSql}
   122→            ");
   123→            $this->db->execute();
   124→            $result['linked'] = $this->db->rowCount();
   125→
   126→            $this->db->query("
   127→                SELECT sr.id, sr.return_code, sr.return_date, sr.total_amount, sr.customer_id,
   128→                       si.branch_id,
   129→                       (
   130→                           SELECT COALESCE(SUM(st.qty * st.rate), 0)
   131→                           FROM stock_transactions st
   132→                           WHERE st.reference_type = 'sales_return'
   133→                             AND st.reference_id = sr.id
   134→                             AND st.qty > 0.0001
   135→                       ) AS cogs_amount
   136→                FROM sales_returns sr
   137→                INNER JOIN sales_invoices si ON si.id = sr.sales_invoice_id
   138→                WHERE sr.status = 'completed'
   139→                  AND COALESCE(sr.is_reversed, 0) = 0
   140→                  AND COALESCE(sr.journal_entry_id, 0) = 0
   141→                  AND sr.return_date >= (CURRENT_DATE - INTERVAL '365 days')
   142→                  AND sr.total_amount > 0
   143→                  {$branchSql}
   144→                ORDER BY sr.id ASC
   145→                LIMIT 50
   146→            ");
   147→            $returns = $this->db->resultSet() ?: [];
   148→
   149→            require_once __DIR__ . '/../services/Accounting/JournalPostingService.php';
   150→            $journalService = new JournalPostingService();
   151→
   152→            foreach ($returns as $sr) {
   153→                $returnId = (int)$sr['id'];
   154→                try {
   155→                    if (!$this->returnHasCustomerLedger($returnId)) {
   156→                        $this->insertCustomerLedgerForReturn($sr);
   157→                        $result['ledgers_added']++;
   158→                    }
   159→
   160→                    $revenueAmount = round((float)($sr['total_amount'] ?? 0), 2);
   161→                    $cogsAmount    = round((float)($sr['cogs_amount'] ?? 0), 2);
   162→
   163→                    if ($revenueAmount <= 0 && $cogsAmount <= 0) {
   164→                        continue;
   165→                    }
   166→
   167→                    $journalResult = $journalService->postSalesReturn($returnId, [
   168→                        'return_code'    => $sr['return_code'] ?? ('SR-' . $returnId),
   169→                        'return_date'    => $sr['return_date'] ?? date('Y-m-d'),
   170→                        'customer_id'    => (int)($sr['customer_id'] ?? 0),
   171→                        'branch_id'      => (int)($sr['branch_id'] ?? 1),
   172→                        'revenue_amount' => $revenueAmount,
   173→                        'cogs_amount'    => $cogsAmount,
   174→                    ]);
   175→
   176→                    if (($journalResult['status'] ?? '') === 'error') {
   177→                        $result['errors'][] = ($sr['return_code'] ?? $returnId) . ': ' . ($journalResult['message'] ?? 'journal error');
   178→                        continue;
   179→                    }
   180→
   181→                    if (!empty($journalResult['journal_entry_id'])) {
   182→                        $this->db->query('UPDATE sales_returns SET journal_entry_id = :jid WHERE id = :id');
   183→                        $this->db->bind(':jid', (int)$journalResult['journal_entry_id']);
   184→                        $this->db->bind(':id', $returnId);
   185→                        $this->db->execute();
   186→                        $result['posted']++;
   187→                    }
   188→                } catch (Exception $e) {
   189→                    $result['errors'][] = ($sr['return_code'] ?? $returnId) . ': ' . $e->getMessage();
   190→                }
   191→            }
   192→        } catch (Exception $e) {
   193→            error_log('SalesAuditModel::repairMissingSalesReturnJournals: ' . $e->getMessage());
   194→            $result['errors'][] = $e->getMessage();
   195→        }
   196→
   197→        return $result;
   198→    }

   200→    private function returnHasCustomerLedger(int $returnId): bool
   201→    {
   202→        $this->db->query("
   203→            SELECT 1 FROM customer_ledger
   204→            WHERE reference_type = 'sales_return' AND reference_id = :id
   205→            LIMIT 1
   206→        ");
   207→        $this->db->bind(':id', $returnId);
   208→        return (bool)$this->db->single();
   209→    }
   210→
   211→    private function insertCustomerLedgerForReturn(array $sr): void
   212→    {
   213→        $customerId = (int)($sr['customer_id'] ?? 0);
   214→        $returnId   = (int)($sr['id'] ?? 0);
   215→        $amount     = round((float)($sr['total_amount'] ?? 0), 2);
   216→        if ($customerId <= 0 || $returnId <= 0 || $amount <= 0) {
   217→            return;
   218→        }
   219→
   220→        $prev = $this->getCustomerRunningBalance($customerId);
   221→        $new  = $prev - $amount;
   222→        $txnDate = $sr['return_date'] ?? date('Y-m-d');
   223→
   224→        $branchId = (int)($sr['branch_id'] ?? 0);
   225→        $this->insertCustomerLedgerEntry([
   226→            'customer_id'      => $customerId,
   227→            'reference_type'   => 'sales_return',
   228→            'reference_id'     => $returnId,
   229→            'debit'            => 0,
   230→            'credit'           => $amount,
   231→            'running_balance'  => $new,
   232→            'branch_id'        => $branchId,
   233→            'transaction_date' => $txnDate,
   234→            'remarks'          => 'Sales Return #' . ($sr['return_code'] ?? $returnId),
   235→            'created_by'       => 1,
   236→        ]);
   237→    }

   239→    /**
   240→     * Link orphan payment journals and post missing customer_payment GL from payment header.
   241→     *
   242→     * @return array{linked: int, posted: int, errors: string[]}
   243→     */
   244→    public function repairMissingCustomerPaymentJournals(): array
   245→    {
   246→        $result = ['linked' => 0, 'posted' => 0, 'errors' => []];
   247→
   248→        try {
   249→            $branchSql = $this->branchFilter('cp.branch_id');
   250→
   251→            $this->db->query("
   252→                UPDATE customer_payments cp
   253→                INNER JOIN journal_entries je ON je.reference_type = 'customer_payment'
   254→                    AND je.reference_id = cp.id
   255→                    AND COALESCE(je.is_reversed, 0) = 0
   256→                SET cp.journal_entry_id = je.id
   257→                WHERE COALESCE(cp.is_reversed, 0) = 0
   258→                  AND COALESCE(cp.journal_entry_id, 0) = 0
   259→                  {$branchSql}
   260→            ");
   261→            $this->db->execute();
   262→            $result['linked'] = $this->db->rowCount();
   263→
   264→            $this->db->query("
   265→                SELECT cp.id, cp.payment_code, cp.payment_date, cp.amount, cp.payment_mode,
   266→                       cp.bank_id, cp.customer_id, cp.branch_id, cp.transaction_type
   267→                FROM customer_payments cp
   268→                WHERE COALESCE(cp.is_reversed, 0) = 0
   269→                  AND COALESCE(cp.journal_entry_id, 0) = 0
   270→                  AND COALESCE(cp.transaction_type, 'receive') IN ('receive', 'payment', 'discount', 'write_off')
   271→                  AND cp.payment_date >= (CURRENT_DATE - INTERVAL '365 days')
   272→                  AND cp.amount > 0
   273→                  {$branchSql}
   274→                ORDER BY cp.id ASC
   275→                LIMIT 50
   276→            ");
   277→            $payments = $this->db->resultSet() ?: [];
   278→
   279→            require_once __DIR__ . '/../services/Accounting/JournalPostingService.php';
   280→            $journalService = new JournalPostingService();
   281→
   282→            foreach ($payments as $cp) {
   283→                $paymentId = (int)$cp['id'];
   284→                $amount = round((float)($cp['amount'] ?? 0), 2);
   285→                if ($amount <= 0) {
   286→                    continue;
   287→                }
   288→
   289→                try {
   290→                    $txnType = $cp['transaction_type'] ?? 'receive';
   291→                    $journalResult = $journalService->postCustomerTransactionJournal($paymentId, [
   292→                        'payment_code'  => $cp['payment_code'] ?? ('PAY-' . $paymentId),
   293→                        'payment_date'  => $cp['payment_date'] ?? date('Y-m-d'),
   294→                        'customer_id'   => (int)($cp['customer_id'] ?? 0),
   295→                        'amount'        => $amount,
   296→                        'payment_mode'  => $cp['payment_mode'] ?? 'cash',
   297→                        'bank_id'       => !empty($cp['bank_id']) ? (int)$cp['bank_id'] : null,
   298→                        'branch_id'     => (int)($cp['branch_id'] ?? 1),
   299→                    ], $txnType);
   300→
   301→                    if (($journalResult['status'] ?? '') === 'error') {
   302→                        $result['errors'][] = ($cp['payment_code'] ?? $paymentId) . ': ' . ($journalResult['message'] ?? 'journal error');
   303→                        continue;
   304→                    }
   305→
   306→                    if (!empty($journalResult['journal_entry_id'])) {
   307→                        $this->db->query('UPDATE customer_payments SET journal_entry_id = :jid WHERE id = :id');
   308→                        $this->db->bind(':jid', (int)$journalResult['journal_entry_id']);
   309→                        $this->db->bind(':id', $paymentId);
   310→                        $this->db->execute();
   311→                        $result['posted']++;
   312→                    }
   313→                } catch (Exception $e) {
   314→                    $result['errors'][] = ($cp['payment_code'] ?? $paymentId) . ': ' . $e->getMessage();
   315→                }
   316→            }
   317→        } catch (Exception $e) {
   318→            error_log('SalesAuditModel::repairMissingCustomerPaymentJournals: ' . $e->getMessage());
   319→            $result['errors'][] = $e->getMessage();
   320→        }
   321→
   322→        return $result;
   323→    }

   330→    public function repairMissingChallanCogsJournals(): array
   331→    {
   332→        $result = ['linked' => 0, 'posted' => 0, 'skipped_no_cogs' => 0, 'errors' => []];
   333→
   334→        try {
   335→            $branchSql = $this->branchId
   336→                ? ' AND si.branch_id = ' . (int)$this->branchId
   337→                : '';
   338→
   339→            $this->db->query("
   340→                UPDATE sales_challans sc
   341→                INNER JOIN sales_invoices si ON si.id = sc.sales_invoice_id
   342→                INNER JOIN journal_entries je ON je.reference_type = 'sales_challan'
   343→                    AND je.reference_id = sc.id
   344→                    AND COALESCE(je.is_reversed, 0) = 0
   345→                SET sc.journal_entry_id = je.id
   346→                WHERE COALESCE(sc.is_reversed, 0) = 0
   347→                  AND COALESCE(sc.journal_entry_id, 0) = 0
   348→                  {$branchSql}
   349→            ");
   350→            $this->db->execute();
   351→            $result['linked'] = $this->db->rowCount();
   352→
   353→            $this->db->query("
   354→                SELECT sc.id, sc.challan_code, sc.challan_date, si.branch_id,
   355→                       (
   356→                           SELECT COALESCE(SUM(ABS(st.qty) * st.rate), 0)
   357→                           FROM stock_transactions st
   358→                           WHERE st.reference_type = 'sales_challan'
   359→                             AND st.reference_id = sc.id
   360→                             AND st.qty < -0.0001
   361→                       ) AS cogs_amount
   362→                FROM sales_challans sc
   363→                INNER JOIN sales_invoices si ON si.id = sc.sales_invoice_id
   364→                WHERE COALESCE(sc.is_reversed, 0) = 0
   365→                  AND si.status = 'challan_completed'
   366→                  AND COALESCE(sc.journal_entry_id, 0) = 0
   367→                  AND sc.challan_date >= (CURRENT_DATE - INTERVAL '365 days')
   368→                  {$branchSql}
   369→                ORDER BY sc.id ASC
   370→                LIMIT 50
   371→            ");
   372→            $challans = $this->db->resultSet() ?: [];

   449→    /**
   450→     * Link orphan GL entries, post missing sales_invoice journals, backfill customer_ledger when absent.
   451→     *
   452→     * @return array{linked: int, journals_posted: int, ledgers_added: int, errors: string[]}

   454→    public function repairMissingInvoiceAccounting(): array
   455→    {
   456→        $result = ['linked' => 0, 'journals_posted' => 0, 'ledgers_added' => 0, 'errors' => []];
   457→
   458→        try {
   459→            $branchSql = $this->branchId
   460→                ? ' AND si.branch_id = ' . (int)$this->branchId
   461→                : '';
   462→
   463→            $this->db->query("
   464→                UPDATE sales_invoices si
   465→                INNER JOIN journal_entries je ON je.reference_type = 'sales_invoice'
   466→                    AND je.reference_id = si.id
   467→                    AND COALESCE(je.is_reversed, 0) = 0
   468→                SET si.journal_entry_id = je.id
   469→                WHERE si.is_reversed = 0
   470→                  AND COALESCE(si.journal_entry_id, 0) = 0
   471→                  AND si.total_amount > 0
   472→                  {$branchSql}
   473→            ");
   474→            $this->db->execute();
   475→            $result['linked'] = $this->db->rowCount();
   476→
   477→            $this->db->query("
   478→                SELECT si.*
   479→                FROM sales_invoices si
   480→                WHERE si.is_reversed = 0
   481→                  AND si.total_amount > 0
   482→                  AND COALESCE(si.journal_entry_id, 0) = 0
   483→                  AND si.invoice_date >= (CURRENT_DATE - INTERVAL '365 days')
   484→                  {$branchSql}
   485→                ORDER BY si.id ASC
   486→                LIMIT 50
   487→            ");
   488→            $invoices = $this->db->resultSet() ?: [];
   489→
   490→            require_once __DIR__ . '/../services/Accounting/JournalPostingService.php';
   491→            $journalService = new JournalPostingService();
   492→
   493→            foreach ($invoices as $inv) {
   494→                $invoiceId = (int)$inv['id'];
   495→                try {
   496→                    if (!$this->invoiceHasCustomerLedger($invoiceId)) {
   497→                        $this->insertCustomerLedgerForInvoice($inv);
   498→                        $result['ledgers_added']++;
   499→                    }
   500→
   501→                    $journalResult = $journalService->postSalesInvoice($invoiceId, [
   502→                        'invoice_code'   => $inv['invoice_code'] ?? ('SI-' . $invoiceId),
   503→                        'invoice_date'   => $inv['invoice_date'] ?? date('Y-m-d'),
   504→                        'customer_id'    => (int)($inv['customer_id'] ?? 0),
   505→                        'branch_id'      => (int)($inv['branch_id'] ?? 1),
   506→                        'subtotal'       => (float)($inv['subtotal'] ?? 0),
   507→                        'discount'       => (float)($inv['discount'] ?? 0),
   508→                        'transport_cost' => (float)($inv['transport_cost'] ?? 0),
   509→                        'total_amount'   => (float)($inv['total_amount'] ?? 0),
   510→                    ]);
   511→
   512→                    if (($journalResult['status'] ?? '') === 'error') {
   513→                        $result['errors'][] = ($inv['invoice_code'] ?? $invoiceId) . ': ' . ($journalResult['message'] ?? 'journal error');
   514→                        continue;
   515→                    }
   516→
   517→                    if (!empty($journalResult['journal_entry_id'])) {
   518→                        $this->db->query('UPDATE sales_invoices SET journal_entry_id = :jid WHERE id = :id');
   519→                        $this->db->bind(':jid', (int)$journalResult['journal_entry_id']);
   520→                        $this->db->bind(':id', $invoiceId);
   521→                        $this->db->execute();
   522→                        $result['journals_posted']++;
   523→                    }
   524→                } catch (Exception $e) {
   525→                    $result['errors'][] = ($inv['invoice_code'] ?? $invoiceId) . ': ' . $e->getMessage();
   526→                }
   527→            }
   528→        } catch (Exception $e) {
   529→            error_log('SalesAuditModel::repairMissingInvoiceAccounting: ' . $e->getMessage());
   530→            $result['errors'][] = $e->getMessage();
   531→        }
   532→
   533→        return $result;
   534→    }

   616→    /**
   617→     * Backfill dispatch/item warehouse_id from sales_challan stock OUT when challan posted but row was left NULL.
   618→     */
   619→    public function repairDispatchWarehousesFromChallanStock(): int
   620→    {
   621→        try {
   622→            $branchSql = $this->branchId
   623→                ? ' AND si.branch_id = ' . (int)$this->branchId
   624→                : '';
   625→
   626→            $this->db->query("
   627→                UPDATE sales_invoice_dispatches sid
   628→                INNER JOIN sales_invoices si ON si.id = sid.sales_invoice_id
   629→                INNER JOIN sales_challans sc ON sc.sales_invoice_id = si.id AND COALESCE(sc.is_reversed, 0) = 0
   630→                INNER JOIN stock_transactions st ON st.reference_type = 'sales_challan'
   631→                    AND st.reference_id = sc.id
   632→                    AND st.product_id = sid.product_id
   633→                    AND st.qty < -0.0001
   634→                SET sid.warehouse_id = st.warehouse_id,
   635→                    sid.dispatched_at = COALESCE(sid.dispatched_at, st.transaction_date)
   636→                WHERE si.is_reversed = 0
   637→                  AND COALESCE(sid.warehouse_id, 0) = 0
   638→                  AND si.status IN ('godown_issued', 'challan_completed')
   639→                  {$branchSql}
   640→            ");
   641→            $this->db->execute();
   642→            $fixed = $this->db->rowCount();
   643→
   644→            $this->db->query("
   645→                UPDATE sales_invoice_items sii
   646→                INNER JOIN sales_invoices si ON si.id = sii.sales_invoice_id
   647→                INNER JOIN sales_invoice_dispatches sid ON sid.sales_invoice_id = si.id
   648→                    AND sid.product_id = sii.product_id
   649→                SET sii.warehouse_id = sid.warehouse_id
   650→                WHERE si.is_reversed = 0
   651→                  AND COALESCE(sii.warehouse_id, 0) = 0
   652→                  AND COALESCE(sid.warehouse_id, 0) > 0
   653→                  {$branchSql}
   654→            ");
   655→            $this->db->execute();
   656→
   657→            $this->db->query("
   658→                UPDATE sales_invoices si
   659→                SET godown_issued_at = COALESCE(godown_issued_at, challan_completed_at, NOW())
   660→                WHERE si.is_reversed = 0
   661→                  AND si.status = 'challan_completed'
   662→                  AND si.godown_issued_at IS NULL
   663→                  {$branchSql}
   664→            ");
   665→            $this->db->execute();
   666→
   667→            return $fixed;
   668→        } catch (Exception $e) {
   669→            error_log('SalesAuditModel::repairDispatchWarehousesFromChallanStock: ' . $e->getMessage());
   670→            return 0;
   671→        }
   672→    }
