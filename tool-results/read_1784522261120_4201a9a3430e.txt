    13→    13→/**
    14→    14→ * Sales Return Service — Phase 8.5.
    15→    15→ *
    16→    16→ * Two-phase: create → confirm → reverse.
    17→    17→ *
    18→    18→ * CRITICAL CORRECTNESS (per avg_cost_rule.md §3):
    19→    19→ *   Stock comes back IN at the ORIGINAL avg_cost from the challan's stock_transaction
    20→    20→ *   (NOT current avg_cost). This ensures:
    21→    21→ *   1. The COGS reversal matches the original COGS exactly.
    22→    22→ *   2. The avg_cost is restored to its pre-sale value.
    23→    23→ *
    24→    24→ * The original_cost is looked up from the stock_transaction for the challan
    25→    25→ * (reference_type='sales_challan', reference_id=challan_id, product_id).
    26→    26→ * This is snapshotted into sales_return_items.original_cost.
    27→    27→ *
    28→    28→ * On confirm (4 operations, all atomic):
    29→    29→ *   1. Stock IN via StockService at ORIGINAL avg_cost (ref='sales_return')
    30→    30→ *      → avg_cost recalculated (IN rule: weighted average with original cost)
    31→    31→ *   2. GL Revenue Reversal: Dr Sales Return / Cr AR (at sales rate)
    32→    32→ *   3. GL COGS Reversal: Dr Inventory / Cr COGS (at original avg_cost)
    33→    33→ *   4. Customer ledger: credit entry (customer owes less)
    34→    34→ */
    35→    35→class SalesReturnService
    36→    36→{
    37→    37→    public function __construct(
    38→    38→        private StockService $stockService,
    39→    39→        private JournalPostingService $journalPosting,
    40→    40→        private SalesAccess $salesAccess,
    41→    41→        private SalesAuditLogger $auditLogger,
    42→    42→        private DamageService $damageService,
    43→    43→        private NotificationService $notifications
    44→    44→    ) {}
    45→    45→
    46→    46→    /**
    47→    47→     * Phase 1: Create a sales return (no stock, no GL).
    48→    48→     * Looks up ORIGINAL avg_cost from the challan's stock_transaction.
    49→    49→     *
    50→    50→     * @param array $data {
    51→    51→     *     sales_invoice_id: int,
    52→    52→     *     return_date: string (Y-m-d),
    53→    53→     *     reason: string|null,
    54→    54→     *     created_by: int,
    55→    55→     *     items: array each { product_id, warehouse_id, qty, rate, sales_invoice_item_id }
    56→    56→     * }
    57→    57→     * @return SalesReturn
    58→    58→     */
    59→    59→    public function createReturn(array $data): SalesReturn
    60→    60→    {
    61→    61→        $invoiceId = (int) ($data['sales_invoice_id'] ?? 0);
    62→    62→        if ($invoiceId <= 0) {
    63→    63→            throw new \InvalidArgumentException('sales_invoice_id is required.');
    64→    64→        }
    65→    65→
    66→    66→        $invoice = DB::table('sales_invoices')->where('id', $invoiceId)->first();
    67→    67→        if (!$invoice) {
    68→    68→            throw new \InvalidArgumentException("Invoice {$invoiceId} not found.");
    69→    69→        }
    70→    70→        if ($invoice->is_reversed) {
    71→    71→            throw new \RuntimeException("Cannot return against a reversed invoice.");
    72→    72→        }
    73→    73→
    74→    74→        // P0-8: Defense-in-depth branch isolation check.
    75→    75→        $this->salesAccess->assertBranchAccessible((int) $invoice->branch_id);
    76→    76→
    77→    77→        // Find the challan for this invoice.
    78→    78→        $challan = DB::table('sales_challans')
    79→    79→            ->where('sales_invoice_id', $invoiceId)
    80→    80→            ->where('is_reversed', false)
    81→    81→            ->first();
    82→    82→        if (!$challan) {
    83→    83→            throw new \RuntimeException("No active challan found for invoice {$invoiceId}. Returns require a completed challan.");
    84→    84→        }
    85→    85→
    86→    86→        $items = $this->validateItems($data['items'], $invoiceId, $challan->id);
    87→    87→        $totalRevenue = collect($items)->sum(fn($i) => $i['qty'] * $i['rate']);
    88→    88→        $totalCogs = collect($items)->sum(fn($i) => $i['qty'] * $i['original_cost']);
    89→    89→
    90→    90→        $returnCode = $this->generateReturnCode();
    91→    91→        $customerId = (int) $invoice->customer_id;
    92→    92→        $branchId = (int) $invoice->branch_id;
    93→    93→
    94→    94→        return DB::transaction(function () use (
    95→    95→            $data, $items, $totalRevenue, $totalCogs, $returnCode, $invoiceId, $customerId, $branchId
    96→    96→        ) {
    97→    97→            $returnId = DB::table('sales_returns')->insertGetId([
    98→    98→                'return_code' => $returnCode,
    99→    99→                'return_date' => $data['return_date'] ?? now()->format('Y-m-d'),
   100→   100→                'sales_invoice_id' => $invoiceId,
   101→   101→                'customer_id' => $customerId,
   102→   102→                'branch_id' => $branchId,
   103→   103→                'total_amount' => round($totalRevenue, 2),
   104→   104→                'cogs_amount' => round($totalCogs, 2),
   105→   105→                'status' => 'created',
   106→   106→                'is_reversed' => false,
   107→   107→                'reason' => $data['reason'] ?? null,
   108→   108→                'created_by' => $data['created_by'] ?? null,
   109→   109→                'created_at' => now(),
   110→   110→                'updated_at' => now(),
   111→   111→            ]);
   112→   112→
   113→   113→            $itemRows = [];
   114→   114→            foreach ($items as $item) {
   115→   115→                $itemRows[] = [
   116→   116→                    'sales_return_id' => $returnId,
   117→   117→                    'sales_invoice_item_id' => $item['sales_invoice_item_id'],
   118→   118→                    'product_id' => $item['product_id'],
   119→   119→                    'warehouse_id' => $item['warehouse_id'],
   120→   120→                    'qty' => $item['qty'],
   121→   121→                    'rate' => $item['rate'],
   122→   122→                    'original_cost' => $item['original_cost'],
   123→   123→                ];
   124→   124→            }
   125→   125→            DB::table('sales_return_items')->insert($itemRows);
   126→   126→
   127→   127→            // P1-3: Audit log — return_created.
   128→   128→            $this->auditLogger->returnCreated(
   129→   129→                $data['created_by'] ?? auth()->id() ?? 0,
   130→   130→                $returnId, $returnCode, $invoiceId, $customerId, $branchId,
   131→   131→                $totalRevenue, $totalCogs
   132→   132→            );
   133→   133→
   134→   134→            // P1-7: Notify admins that a return was created.
   135→   135→            $this->notifications->dispatch(
   136→   136→                'return_created',
   137→   137→                "Return {$returnCode} created for Tk " . number_format($totalRevenue, 2)
   138→   138→                . " against invoice #{$invoiceId}.",
   139→   139→                'sales_return',
   140→   140→                $returnId
   141→   141→            );
   142→   142→
   143→   143→            return SalesReturn::with(['items.product', 'salesInvoice', 'customer', 'branch'])->find($returnId);
   144→   144→        });
   145→   145→    }
   146→   146→
   147→   147→    /**
   148→   148→     * Phase 2: Confirm a sales return — stock IN + GL + customer_ledger.
   149→   149→     *
   150→   150→     * @param int $returnId
   151→   151→     * @param int $confirmedBy
   152→   152→     * @return SalesReturn
   153→   153→     * @throws \RuntimeException If not 'created' status, or stock/GL fails.
   154→   154→     */
   155→   155→    public function confirmReturn(int $returnId, int $confirmedBy): SalesReturn
   156→   156→    {
   157→   157→        return DB::transaction(function () use ($returnId, $confirmedBy) {
   158→   158→            $return = SalesReturn::with('items')->lockForUpdate()->find($returnId);
   159→   159→
   160→   160→            if (!$return) {
   161→   161→                throw new \RuntimeException("Return {$returnId} not found.");
   162→   162→            }
   163→   163→            if (!$return->isCreated()) {
   164→   164→                throw new \RuntimeException("Only 'created' returns can be confirmed (current: {$return->status}).");
   165→   165→            }
   166→   166→
   167→   167→            $returnDate = $return->return_date->format('Y-m-d');
   168→   168→
   169→   169→            // 1. Stock IN at ORIGINAL avg_cost (CRITICAL: not current avg_cost).
   170→   170→            foreach ($return->items as $item) {
   171→   171→                $this->stockService->applyTransaction([
   172→   172→                    'warehouse_id' => $item->warehouse_id,
   173→   173→                    'product_id' => $item->product_id,
   174→   174→                    'qty' => (float) $item->qty, // positive = IN
   175→   175→                    'rate' => (float) $item->original_cost, // ORIGINAL avg_cost from challan
   176→   176→                    'reference_type' => 'sales_return',
   177→   177→                    'reference_id' => $return->id,
   178→   178→                    'notes' => 'Sales Return ' . $return->return_code . ' — stock restored at original cost',
   179→   179→                    'transaction_date' => $returnDate,
   180→   180→                    'created_by' => $confirmedBy,
   181→   181→                ]);
   182→   182→            }
   183→   183→
   184→   184→            // 1b. P1-5: Linked damage write-off for 'Damage' condition items.
   185→   185→            // Legacy: SalesReturnModel::createLinkedDamageWriteOff — for each
   186→   186→            // Damage-condition item, create a damage_invoices row + stock OUT
   187→   187→            // (immediately back OUT the damaged goods) + GL Dr damage_loss / Cr inventory.
   188→   188→            $linkedDamageIds = $this->createLinkedDamageWriteOffs($return, $confirmedBy);
   189→   189→
   190→   190→            // 2. GL Revenue Reversal: Dr Sales Return / Cr AR.
   191→   191→            $journalEntryId = $this->postRevenueReversalGL($return, $confirmedBy);
   192→   192→
   193→   193→            // 3. GL COGS Reversal: Dr Inventory / Cr COGS.
   194→   194→            $cogsJournalEntryId = $this->postCogsReversalGL($return, $confirmedBy);
   195→   195→
   196→   196→            // 4. Customer ledger: credit entry (customer owes less).
   197→   197→            $this->postCustomerLedgerCredit($return, $journalEntryId, $confirmedBy);
   198→   198→
   199→   199→            // 5. Update return status.
   200→   200→            DB::table('sales_returns')
   201→   201→                ->where('id', $returnId)
   202→   202→                ->update([
   203→   203→                    'status' => 'confirmed',
   204→   204→                    'journal_entry_id' => $journalEntryId,
   205→   205→                    'cogs_journal_entry_id' => $cogsJournalEntryId,
   206→   206→                    'updated_at' => now(),
   207→   207→                ]);
   208→   208→
   209→   209→            // P1-3: Audit log — return_confirmed.
   210→   210→            $this->auditLogger->returnConfirmed(
   211→   211→                $confirmedBy,
   212→   212→                $returnId, $return->return_code, (int) $return->branch_id,
   213→   213→                (float) $return->total_amount, (float) $return->cogs_amount,
   214→   214→                $journalEntryId
   215→   215→            );
   216→   216→
   217→   217→            // P1-7: Notify warehouse managers + accountants that a return was confirmed.
   218→   218→            $this->notifications->dispatch(
   219→   219→                'return_confirmed',
   220→   220→                "Return {$return->return_code} confirmed — stock restored, GL posted. "
   221→   221→                . "Total: Tk " . number_format((float) $return->total_amount, 2)
   222→   222→                . ", COGS reversed: Tk " . number_format((float) $return->cogs_amount, 2) . ".",
   223→   223→                'sales_return',
   224→   224→                $returnId
   225→   225→            );
   226→   226→
   227→   227→            return SalesReturn::with([
   228→   228→                'items.product', 'salesInvoice', 'customer', 'branch',
   229→   229→                'journalEntry.lines.ledger', 'cogsJournalEntry.lines.ledger',
   230→   230→            ])->find($returnId);
   231→   231→        });
   232→   232→    }
   233→   233→
   234→   234→    /**
   235→   235→     * Phase 3: Reverse a confirmed return.
   236→   236→     * Reverses stock + both GL journals + customer_ledger.
   237→   237→     *
   238→   238→     * @param int $returnId
   239→   239→     * @param int $reversedBy
   240→   240→     * @param string $reason
   241→   241→     * @return SalesReturn
   242→   242→     */
   243→   243→    public function reverseReturn(int $returnId, int $reversedBy, string $reason = ''): SalesReturn
   244→   244→    {
   245→   245→        return DB::transaction(function () use ($returnId, $reversedBy, $reason) {
   246→   246→            $return = SalesReturn::with('items')->lockForUpdate()->find($returnId);
   247→   247→
   248→   248→            if (!$return) {
   249→   249→                throw new \RuntimeException("Return {$returnId} not found.");
   250→   250→            }
   251→   251→            if (!$return->isConfirmed()) {
   252→   252→                throw new \RuntimeException("Only confirmed returns can be reversed (current: {$return->status}).");
   253→   253→            }
   254→   254→
   255→   255→            // Reverse both GL journals.
   256→   256→            if ($return->journal_entry_id) {
   257→   257→                $this->journalPosting->reverseJournalEntry(
   258→   258→                    $return->journal_entry_id, $reversedBy,
   259→   259→                    "Return reversed: {$reason}"
   260→   260→                );
   261→   261→            }
   262→   262→            if ($return->cogs_journal_entry_id) {
   263→   263→                $this->journalPosting->reverseJournalEntry(
   264→   264→                    $return->cogs_journal_entry_id, $reversedBy,
   265→   265→                    "Return reversed: {$reason}"
   266→   266→                );
   267→   267→            }
   268→   268→
   269→   269→            // Reverse customer_ledger.
   270→   270→            $this->reverseCustomerLedgerCredit($return, $reversedBy, $reason);
   271→   271→
   272→   272→            // P1-5: Reverse linked damage write-offs FIRST (before return stock reversal).
   273→   273→            // Each linked damage_invoice has its own stock OUT + GL that must be reversed.
   274→   274→            $this->reverseLinkedDamageForReturn($returnId, $reversedBy, $reason);
   275→   275→
   276→   276→            // Reverse each stock movement (stock OUT at original cost — append-only reversal).
   277→   277→            $stockTxs = DB::table('stock_transactions')
   278→   278→                ->where('reference_type', 'sales_return')
   279→   279→                ->where('reference_id', $returnId)
   280→   280→                ->where('is_reversed', false)
   281→   281→                ->get();
   282→   282→
   283→   283→            foreach ($stockTxs as $tx) {
   284→   284→                $this->stockService->reverseTransaction(
   285→   285→                    $tx->id, $reversedBy,
   286→   286→                    "Return reversed: {$reason}"
   287→   287→                );
   288→   288→            }
   289→   289→
   290→   290→            // Mark return as reversed.
   291→   291→            DB::table('sales_returns')
   292→   292→                ->where('id', $returnId)
   293→   293→                ->update([
   294→   294→                    'status' => 'reversed',
   295→   295→                    'is_reversed' => true,
   296→   296→                    'reversed_at' => now(),
   297→   297→                    'reversed_by' => $reversedBy,
   298→   298→                    'reverse_reason' => $reason,
   299→   299→                    'updated_at' => now(),
   300→   300→                ]);
   301→   301→
   302→   302→            // P1-3: Audit log — return_reversed.
   303→   303→            $this->auditLogger->returnReversed(
   304→   304→                $reversedBy,
   305→   305→                $returnId, $return->return_code, (int) $return->branch_id,
   306→   306→                (float) $return->total_amount, $reason
   307→   307→            );
   308→   308→
   309→   309→            // P1-7: Notify accountants + admins that a return was reversed.
   310→   310→            $this->notifications->dispatch(
   311→   311→                'return_reversed',
   312→   312→                "Return {$return->return_code} reversed — stock + GL + ledger rolled back. "
   313→   313→                . "Reason: {$reason}",
   314→   314→                'sales_return',
   315→   315→                $returnId
   316→   316→            );
   317→   317→
   318→   318→            return SalesReturn::find($returnId);
   319→   319→        });
   320→   320→    }

   322→   322→    /**
   323→   323→     * Post GL Revenue Reversal: Dr Sales Return / Cr AR.
   324→   324→     *
   325→   325→     * @return int journal_entry_id
   326→   326→     */
   327→   327→    private function postRevenueReversalGL(SalesReturn $return, int $createdBy): int
   328→   328→    {
   329→   329→        $amount = (float) $return->total_amount;
   330→   330→        if ($amount < 0.01) return 0;
   331→   331→
   332→   332→        $returnLedgerId = $this->journalPosting->lookupLedgerByNature('sales_return');

   341→   341→

   343→   343→            'entry_date' => $return->return_date->format('Y-m-d'),

   349→   349→            'created_by' => $createdBy,
   350→   350→        ], [
   351→   351→            [
   352→   352→                'ledger_id' => $returnLedgerId,
   353→   353→                'debit' => $amount, 'credit' => 0,
   354→   354→                'entity_type' => 'sales_return', 'entity_id' => $return->id,

   359→   359→                'debit' => 0, 'credit' => $amount,
   360→   360→                'entity_type' => 'customer', 'entity_id' => $return->customer_id,
   361→   361→                'memo' => 'Return ' . $return->return_code . ' — AR credit',
   362→   362→            ],
   363→   363→        ]);
   364→   364→    }

   366→   366→    /**
   367→   367→     * Post GL COGS Reversal: Dr Inventory / Cr COGS (at ORIGINAL avg_cost).

   371→   371→    private function postCogsReversalGL(SalesReturn $return, int $createdBy): int

   387→   387→            'entry_date' => $return->return_date->format('Y-m-d'),
   388→   388→            'reference_type' => 'sales_return',
   389→   389→            'reference_id' => $return->id,

   394→   394→        ], [

   401→   401→            [
   402→   402→                'ledger_id' => $cogsLedgerId,
   403→   403→                'debit' => 0, 'credit' => $cogsAmount,
   404→   404→                'entity_type' => 'sales_return', 'entity_id' => $return->id,
   405→   405→                'memo' => 'Return ' . $return->return_code . ' — COGS reversal',
   406→   406→            ],
   407→   407→        ]);
   408→   408→    }

   410→   410→    /**
   411→   411→     * Post customer_ledger credit (customer owes less).
   412→   412→     */
   413→   413→    private function postCustomerLedgerCredit(SalesReturn $return, ?int $journalEntryId, int $createdBy): void
   414→   414→    {
   415→   415→        $amount = (float) $return->total_amount;
   416→   416→        if ($amount < 0.01) return;
   417→   417→
   418→   418→        $currentBalance = (float) DB::table('customer_ledger')
   419→   419→            ->where('customer_id', $return->customer_id)
   420→   420→            ->orderByDesc('id')
   421→   421→            ->value('balance');
   422→   422→
   423→   423→        $newBalance = $currentBalance - $amount;
   424→   424→
   425→   425→        DB::table('customer_ledger')->insert([
   426→   426→            'customer_id' => $return->customer_id,
   427→   427→            'branch_id' => $return->branch_id,
   428→   428→            'transaction_date' => $return->return_date->format('Y-m-d'),
   429→   429→            'transaction_type' => 'sales_return',
   430→   430→            'reference_type' => 'sales_return',
   431→   431→            'reference_id' => $return->id,
   432→   432→            'debit' => 0,
   433→   433→            'credit' => $amount,
   434→   434→            'balance' => $newBalance,
   435→   435→            'description' => 'Sales Return ' . $return->return_code . ($return->reason ? ' — ' . $return->reason : ''),
   436→   436→            'journal_entry_id' => $journalEntryId,
   437→   437→            'created_by' => $createdBy,
   438→   438→            'created_at' => now(),
   439→   439→        ]);
   440→   440→    }
   441→   441→
   442→   442→    /**
   443→   443→     * Reverse customer_ledger (debit entry to restore what customer owes).
   444→   444→     */
   445→   445→    private function reverseCustomerLedgerCredit(SalesReturn $return, int $reversedBy, string $reason): void
   446→   446→    {
   447→   447→        $amount = (float) $return->total_amount;
   448→   448→        if ($amount < 0.01) return;
   449→   449→
   450→   450→        $currentBalance = (float) DB::table('customer_ledger')
   451→   451→            ->where('customer_id', $return->customer_id)
   452→   452→            ->orderByDesc('id')
   453→   453→            ->value('balance');
   454→   454→
   455→   455→        $newBalance = $currentBalance + $amount;
   456→   456→
   457→   457→        DB::table('customer_ledger')->insert([
   458→   458→            'customer_id' => $return->customer_id,
   459→   459→            'branch_id' => $return->branch_id,
   460→   460→            'transaction_date' => now()->format('Y-m-d'),
   461→   461→            'transaction_type' => 'sales_return_reversal',
   462→   462→            'reference_type' => 'sales_return',
   463→   463→            'reference_id' => $return->id,
   464→   464→            'debit' => $amount,
   465→   465→            'credit' => 0,
   466→   466→            'balance' => $newBalance,
   467→   467→            'description' => 'Return reversal ' . $return->return_code . ": {$reason}",
   468→   468→            'created_by' => $reversedBy,
   469→   469→            'created_at' => now(),
   470→   470→        ]);
   471→   471→    }

   473→   473→    /**
   474→   474→     * Generate atomic return code: SR-YYYYMMDD-NNNN.
   475→   475→     */
   476→   476→    private function generateReturnCode(): string
   477→   477→    {
   478→   478→        $datePart = now()->format('Ymd');
   479→   479→        $periodKey = now()->format('Y-m');
   480→   480→        $docType = 'sales_return';
   481→   481→
   482→   482→        return DB::transaction(function () use ($docType, $periodKey, $datePart) {
   483→   483→            $seqRow = DB::table('document_sequences')
   484→   484→                ->where('doc_type', $docType)
   485→   485→                ->where('branch_id', 0)
   486→   486→                ->where('period_key', $periodKey)
   487→   487→                ->lockForUpdate()
   488→   488→                ->first();
   489→   489→
   490→   490→            $nextNumber = $seqRow ? ((int) $seqRow->last_number + 1) : 1;
   491→   491→
   492→   492→            if ($seqRow) {
   493→   493→                DB::table('document_sequences')->where('id', $seqRow->id)
   494→   494→                    ->update(['last_number' => $nextNumber, 'updated_at' => now()]);
   495→   495→            } else {
   496→   496→                DB::table('document_sequences')->insert([
   497→   497→                    'doc_type' => $docType, 'branch_id' => 0,
   498→   498→                    'period_key' => $periodKey, 'last_number' => $nextNumber,
   499→   499→                    'updated_at' => now(),
   500→   500→                ]);
   501→   501→            }
   502→   502→
   503→   503→            return "SR-{$datePart}-" . str_pad((string) $nextNumber, 4, '0', STR_PAD_LEFT);
   504→   504→        });
   505→   505→    }

   507→   507→    /**
   508→   508→     * Validate items — look up ORIGINAL avg_cost from the challan's stock_transaction.
   509→   509→     *
   510→   510→     * @param array $items
   511→   511→     * @param int $invoiceId
   512→   512→     * @param int $challanId
   513→   513→     * @return array
   514→   514→     */
   515→   515→    private function validateItems(array $items, int $invoiceId, int $challanId): array
   516→   516→    {
   517→   517→        $validated = [];
   518→   518→
   519→   519→        foreach ($items as $item) {
   520→   520→            $productId = (int) ($item['product_id'] ?? 0);
   521→   521→            $qty = (float) ($item['qty'] ?? 0);
   522→   522→            $warehouseId = (int) ($item['warehouse_id'] ?? 0);
   523→   523→            $invoiceItemId = (int) ($item['sales_invoice_item_id'] ?? 0);
   524→   524→
   525→   525→            if ($productId <= 0 || $qty <= 0 || $warehouseId <= 0) continue;
   526→   526→
   527→   527→            // Check returnable_qty (invoice item qty - already returned).
   528→   528→            if ($invoiceItemId > 0) {
   529→   529→                $invoiceItem = DB::table('sales_invoice_items')->where('id', $invoiceItemId)->first();
   530→   530→                if ($invoiceItem) {
   531→   531→                    $alreadyReturned = (float) DB::table('sales_return_items as sri')
   532→   532→                        ->join('sales_returns as sr', 'sr.id', '=', 'sri.sales_return_id')
   533→   533→                        ->where('sri.sales_invoice_item_id', $invoiceItemId)
   534→   534→                        ->whereIn('sr.status', ['created', 'confirmed'])
   535→   535→                        ->where('sr.is_reversed', false)
   536→   536→                        ->sum('sri.qty');
   537→   537→
   538→   538→                    $returnable = (float) $invoiceItem->qty - $alreadyReturned;
   539→   539→                    if ($qty > $returnable + 0.0001) {
   540→   540→                        throw new \RuntimeException(
   541→   541→                            "Return qty {$qty} exceeds returnable qty {$returnable} for product {$productId}."
   542→   542→                        );
   543→   543→                    }
   544→   544→                }
   545→   545→            }
   546→   546→
   547→   547→            // CRITICAL: Look up ORIGINAL avg_cost from the challan's stock_transaction.
   548→   548→            $challanStockTx = DB::table('stock_transactions')
   549→   549→                ->where('reference_type', 'sales_challan')
   550→   550→                ->where('reference_id', $challanId)
   551→   551→                ->where('product_id', $productId)
   552→   552→                ->where('warehouse_id', $warehouseId)
   553→   553→                ->where('is_reversed', false)
   554→   554→                ->first();
   555→   555→
   556→   556→            $originalCost = $challanStockTx ? (float) $challanStockTx->rate : 0;
   557→   557→            if ($originalCost <= 0) {
   558→   558→                throw new \RuntimeException(
   559→   559→                    "Cannot determine original avg_cost for product {$productId} from challan. "
   560→   560→                    . "The challan stock transaction may be missing or reversed."
   561→   561→                );
   562→   562→            }
   563→   563→
   564→   564→            // Rate = sales rate from the invoice item (for revenue reversal).
   565→   565→            $rate = (float) ($item['rate'] ?? 0);
   566→   566→            if ($rate <= 0 && $invoiceItemId > 0) {
   567→   567→                $invoiceItem = DB::table('sales_invoice_items')->where('id', $invoiceItemId)->first();
   568→   568→                $rate = $invoiceItem ? (float) $invoiceItem->rate : 0;
   569→   569→            }
   570→   570→
   571→   571→            $validated[] = [
   572→   572→                'product_id' => $productId,
   573→   573→                'warehouse_id' => $warehouseId,
   574→   574→                'qty' => $qty,
   575→   575→                'rate' => $rate, // sales rate (for revenue reversal)
   576→   576→                'original_cost' => $originalCost, // ORIGINAL avg_cost (for COGS + stock IN)
   577→   577→                'sales_invoice_item_id' => $invoiceItemId > 0 ? $invoiceItemId : null,
   578→   578→            ];
   579→   579→        }
   580→   580→
   581→   581→        if (empty($validated)) {
   582→   582→            throw new \InvalidArgumentException('At least one valid item is required.');
   583→   583→        }
   584→   584→
   585→   585→        return $validated;
   586→   586→    }

   588→   588→    /**
   589→   589→     * P1-5: Create linked damage write-offs for 'Damage' condition return items.
   590→   590→     *
   591→   591→     * Mirrors legacy SalesReturnModel::createLinkedDamageWriteOff:1599.
   592→   592→     * For each Damage-condition item:
   593→   593→     *   1. Create a damage_invoices row (linked via sales_return_id)
   594→   594→     *   2. Create damage_invoice_items
   595→   595→     *   3. Stock OUT via DamageService (immediately back OUT the damaged goods)
   596→   596→     *   4. GL: Dr damage_loss / Cr inventory (via DamageService::confirmDamage)
   597→   597→     *   5. Link sales_return_items.damage_invoice_id → damage_invoices.id
   598→   598→     *
   599→   599→     * Groups damage items by warehouse (one damage_invoice per warehouse).
   600→   600→     *
   601→   601→     * @param SalesReturn $return
   602→   602→     * @param int $confirmedBy
   603→   603→     * @return array List of created damage_invoice IDs
   604→   604→     */
   605→   605→    private function createLinkedDamageWriteOffs(SalesReturn $return, int $confirmedBy): array
   606→   606→    {
   607→   607→        // Group Damage-condition items by warehouse.
   608→   608→        $damageByWarehouse = [];
   609→   609→        foreach ($return->items as $item) {
   610→   610→            if (($item->condition_state ?? 'Good') !== 'Damage') {
   611→   611→                continue;
   612→   612→            }
   613→   613→            $wid = (int) $item->warehouse_id;
   614→   614→            if ($wid <= 0) {
   615→   615→                continue;
   616→   616→            }
   617→   617→            $damageByWarehouse[$wid][] = $item;
   618→   618→        }
   619→   619→
   620→   620→        if (empty($damageByWarehouse)) {
   621→   621→            return []; // No damage items — nothing to write off.
   622→   622→        }
   623→   623→
   624→   624→        $linkedDamageIds = [];
   625→   625→        $returnDate = $return->return_date->format('Y-m-d');
   626→   626→
   627→   627→        foreach ($damageByWarehouse as $warehouseId => $damageItems) {
   628→   628→            // Build items array for DamageService::createDamage.
   629→   629→            $items = [];
   630→   630→            foreach ($damageItems as $item) {
   631→   631→                $items[] = [
   632→   632→                    'product_id' => (int) $item->product_id,
   633→   633→                    'qty' => (float) $item->qty,
   634→   634→                    'rate' => (float) $item->original_cost, // use original_cost (same as return stock IN)
   635→   635→                ];
   636→   636→            }
   637→   637→
   638→   638→            // Create the damage invoice (draft) via DamageService.
   639→   639→            $damage = $this->damageService->createDamage([
   640→   640→                'warehouse_id' => $warehouseId,
   641→   641→                'damage_date' => $returnDate,
   642→   642→                'reason' => 'Auto write-off for damaged sales return #' . $return->return_code,
   643→   643→                'created_by' => $confirmedBy,
   644→   644→                'items' => $items,
   645→   645→            ]);
   646→   646→
   647→   647→            // Link damage_invoices.sales_return_id → this return.
   648→   648→            DB::table('damage_invoices')
   649→   649→                ->where('id', $damage->id)
   650→   650→                ->update(['sales_return_id' => $return->id]);
   651→   651→
   652→   652→            // Confirm the damage (stock OUT + GL Dr damage_loss / Cr inventory).
   653→   653→            $damage = $this->damageService->confirmDamage($damage->id, $confirmedBy);
   654→   654→
   655→   655→            // Link each return item to the damage invoice.
   656→   656→            foreach ($damageItems as $item) {
   657→   657→                DB::table('sales_return_items')
   658→   658→                    ->where('id', $item->id)
   659→   659→                    ->update(['damage_invoice_id' => $damage->id]);
   660→   660→            }
   661→   661→
   662→   662→            $linkedDamageIds[] = $damage->id;
   663→   663→        }
   664→   664→
   665→   665→        return $linkedDamageIds;
   666→   666→    }

   668→   668→    /**
   669→   669→     * P1-5: Reverse linked damage write-offs when a return is reversed.
   670→   670→     *
   671→   671→     * Mirrors legacy SalesReturnModel::reverseLinkedDamageForReturn.
   672→   672→     * Finds all damage_invoices linked to this return (via sales_return_id)
   673→   673→     * and cancels each one via DamageService::cancelDamage (reverses stock + GL).
   674→   674→     *
   675→   675→     * Must be called BEFORE reversing the return's own stock movements
   676→   676→     * (so the damage stock OUT is reversed first, then the return stock IN).
   677→   677→     *
   678→   678→     * @param int $returnId
   679→   679→     * @param int $reversedBy
   680→   680→     * @param string $reason
   681→   681→     */
   682→   682→    private function reverseLinkedDamageForReturn(int $returnId, int $reversedBy, string $reason): void
   683→   683→    {
   684→   684→        $linkedDamages = DB::table('damage_invoices')
   685→   685→            ->where('sales_return_id', $returnId)
   686→   686→            ->where('is_reversed', false)
   687→   687→            ->get();
   688→   688→
   689→   689→        foreach ($linkedDamages as $damage) {
   690→   690→            // Cancel the damage (reverses stock OUT + GL Dr damage_loss / Cr inventory).
   691→   691→            $this->damageService->cancelDamage($damage->id, $reversedBy, "Return reversed: {$reason}");
   692→   692→
   693→   693→            // Clear the link on sales_return_items (set damage_invoice_id = null).
   694→   694→            DB::table('sales_return_items')
   695→   695→                ->where('damage_invoice_id', $damage->id)
   696→   696→                ->update(['damage_invoice_id' => null]);
   697→   697→        }
   698→   698→    }
