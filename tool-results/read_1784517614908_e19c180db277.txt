   226→   226→    public function uploadProductImage(array $file): ?string
   227→   227→    {
   228→   228→        if (empty($file['tmp_name']) || $file['error'] !== UPLOAD_ERR_OK) {
   229→   229→            return null;
   230→   230→        }
   231→   231→
   232→   232→        $maxSize = 2 * 1024 * 1024;
   233→   233→        if ($file['size'] > $maxSize) {
   234→   234→            return null;
   235→   235→        }
   236→   236→
   237→   237→        $finfo = finfo_open(FILEINFO_MIME_TYPE);
   238→   238→        $mime = finfo_file($finfo, $file['tmp_name']);
   239→   239→        finfo_close($finfo);
   240→   240→
   241→   241→        $allowedMimes = [
   242→   242→            'image/jpeg' => 'jpg',
   243→   243→            'image/png'  => 'png',
   244→   244→            'image/webp' => 'webp',
   245→   245→        ];
   246→   246→
   247→   247→        if (!isset($allowedMimes[$mime])) {
   248→   248→            return null;
   249→   249→        }
   250→   250→
   251→   251→        $uploadDir = __DIR__ . '/../../public/uploads/products/';
   252→   252→        if (!is_dir($uploadDir)) {
   253→   253→            mkdir($uploadDir, 0755, true);
   254→   254→        }
   255→   255→
   256→   256→        $extension = $allowedMimes[$mime];
   257→   257→        $filename = 'prod_' . time() . '_' . bin2hex(random_bytes(4)) . '.' . $extension;
   258→   258→        $destination = $uploadDir . $filename;
   259→   259→
   260→   260→        if (move_uploaded_file($file['tmp_name'], $destination)) {
   261→   261→            $imgInfo = @getimagesize($destination);
   262→   262→            if ($imgInfo === false) {
   263→   263→                @unlink($destination);
   264→   264→                return null;
   265→   265→            }
   266→   266→            return 'uploads/products/' . $filename;
   267→   267→        }
   268→   268→
   269→   269→        return null;

   465→   465→                ), 0) AS current_price,
   466→   466→                COALESCE((
   467→   467→                    SELECT lp.default_rate FROM product_price_history lp

   482→   482→                    LIMIT 1

   484→   484→            {$baseQuery}
   485→   485→            {$whereSql}

   487→   487→            LIMIT {$length} OFFSET {$start}

   755→   755→        $deletableIds = [];

   759→   759→            }
   760→   760→        }
   761→   761→
   762→   762→        if (empty($deletableIds)) {
   763→   763→            return false;
   764→   764→        }
   765→   765→
   766→   766→        $placeholders = implode(',', array_fill(0, count($deletableIds), '?'));
   767→   767→        $this->db->query("UPDATE products SET is_active = 0 WHERE id IN ($placeholders)");
   768→   768→
   769→   769→        foreach ($deletableIds as $i => $id) {
   770→   770→            $this->db->bind($i, $id);
