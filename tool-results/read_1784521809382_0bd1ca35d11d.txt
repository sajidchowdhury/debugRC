   108→    /**
   109→     * Load invoice for return with clear errors (branch + challan rules).
   110→     *
   111→     * @return array{status:string,message?:string,invoice?:array}
   112→     */
   113→    public function resolveInvoiceForReturn(string $invoiceCode): array
   114→    {
   115→        $invoiceCode = trim($invoiceCode);
   116→        if ($invoiceCode === '') {
   117→            return ['status' => 'error', 'message' => 'Invoice code is required.'];
   118→        }
   119→
   120→        $branchId = self::sessionBranchId();
   121→        $this->db->query("
   122→            SELECT si.*, c.shop_name, c.mobile, c.address
   123→            FROM sales_invoices si
   124→            JOIN customers c ON si.customer_id = c.id
   125→            WHERE si.invoice_code = :code
   126→              AND si.is_reversed = 0
   127→        ");
   128→        $this->db->bind(':code', $invoiceCode);
   129→        $row = $this->db->single();
   130→
   131→        if (!$row) {
   132→            return ['status' => 'error', 'message' => 'Invoice not found.'];
   133→        }
   134→
   135→        if ((int)$row['branch_id'] !== $branchId) {
   136→            return ['status' => 'error', 'message' => 'This invoice belongs to another branch. You can only return invoices from your branch.'];
   137→        }
   138→
   139→        $status = (string)($row['status'] ?? '');
   140→        if ($status === 'cancelled' || $status === 'reversed') {
   141→            return ['status' => 'error', 'message' => 'This invoice is cancelled or reversed.'];
   142→        }
   143→
   144→        if ($status !== self::RETURNABLE_INVOICE_STATUS) {
   145→            $label = str_replace('_', ' ', $status);
   146→            return [
   147→                'status'  => 'error',
   148→                'message' => 'Return is allowed only after challan is completed. This invoice status: ' . $label . '.',
   149→            ];
   150→        }
   151→
   152→        $invoice = $row;
   153→        $this->db->query("
   154→            SELECT sii.*, p.product_name, p.unit,
   155→                   COALESCE(sii.qty - (SELECT COALESCE(SUM(sri.return_qty),0)
   156→                                      FROM sales_return_items sri
   157→                                      JOIN sales_returns sr ON sr.id = sri.sales_return_id
   158→                                      WHERE sri.sales_invoice_item_id = sii.id
   159→                                        AND sr.status != 'reversed'
   160→                                        AND COALESCE(sr.is_reversed, 0) = 0), 0) as returnable_qty
   161→            FROM sales_invoice_items sii
   162→            JOIN products p ON sii.product_id = p.id
   163→            WHERE sii.sales_invoice_id = :invoice_id
   164→        ");
   165→        $this->db->bind(':invoice_id', $invoice['id']);
   166→        $invoice['items'] = $this->db->resultSet();
   167→
   168→        $hasReturnable = false;
   169→        foreach ($invoice['items'] as $line) {
   170→            if ((float)($line['returnable_qty'] ?? 0) > 0.0001) {
   171→                $hasReturnable = true;
   172→                break;
   173→            }
   174→        }
   175→        if (!$hasReturnable) {
   176→            return ['status' => 'error', 'message' => 'No returnable quantity left on this invoice.'];
   177→        }
   178→
   179→        return ['status' => 'success', 'invoice' => $invoice];
   180→    }

   194→    /**
   195→     * Max returnable qty for an invoice line (excludes other non-reversed returns).
   196→     */
   197→    public function getMaxReturnableQty(int $salesInvoiceItemId): float
   198→    {
   199→        if ($salesInvoiceItemId <= 0) {
   200→            return 0.0;
   201→        }
   202→
   203→        $this->db->query("
   204→            SELECT GREATEST(0,
   205→                COALESCE(sii.qty, 0) - COALESCE((
   206→                    SELECT SUM(sri.return_qty)
   207→                    FROM sales_return_items sri
   208→                    JOIN sales_returns sr ON sr.id = sri.sales_return_id
   209→                    WHERE sri.sales_invoice_item_id = sii.id
   210→                      AND sr.status != 'reversed'
   211→                      AND COALESCE(sr.is_reversed, 0) = 0
   212→                ), 0)
   213→            ) AS returnable_qty
   214→            FROM sales_invoice_items sii
   215→            WHERE sii.id = :id
   216→        ");
   217→        $this->db->bind(':id', $salesInvoiceItemId);
   218→        $row = $this->db->single();
   219→
   220→        return (float)($row['returnable_qty'] ?? 0);
   221→    }

   223→    // PHASE 1: Create Return (Sales Manager) — ledger credit posts on warehouse confirm (Phase 2)
   224→public function createReturn($data, $items) {
   225→    $this->lastError = '';
   226→    $this->db->beginTransaction();
   227→    try {
   228→        $invoiceId = (int)($data['sales_invoice_id'] ?? 0);
   229→        $this->db->query("SELECT branch_id, status FROM sales_invoices WHERE id = :id AND is_reversed = 0");
   230→        $this->db->bind(':id', $invoiceId);
   231→        $inv = $this->db->single();
   232→        if (!$inv) {
   233→            throw new Exception('Invoice not found.');
   234→        }
   235→        $this->assertInvoiceAccessible((int)$inv['branch_id']);
   236→        if (($inv['status'] ?? '') !== self::RETURNABLE_INVOICE_STATUS) {
   237→            throw new Exception('Return is allowed only after challan is completed for this invoice.');
   238→        }
   239→
   240→        $return_code = $this->generateSalesReturnCode();
   241→
   242→        $computedTotal = 0.0;
   243→        $validatedItems = [];
   244→
   245→        foreach ($items as $item) {
   246→            $return_qty = (float)($item['return_qty'] ?? 0);
   247→            if ($return_qty <= 0) {
   248→                continue;
   249→            }
   250→
   251→            $siiId = (int)($item['sales_invoice_item_id'] ?? 0);
   252→            $maxReturnable = $this->getMaxReturnableQty($siiId);
   253→
   254→            if ($return_qty > $maxReturnable + 0.0001) {
   255→                throw new Exception(
   256→                    "Return quantity exceeds returnable amount for invoice line #{$siiId}. "
   257→                    . "Max: " . number_format($maxReturnable, 2)
   258→                );
   259→            }
   260→
   261→            $rate = (float)($item['rate'] ?? $item['sales_rate'] ?? 0);
   262→            $amount = $return_qty * $rate;
   263→            $computedTotal += $amount;
   264→
   265→            $validatedItems[] = [
   266→                'sales_invoice_item_id' => $siiId,
   267→                'product_id'            => (int)($item['product_id'] ?? 0),
   268→                'return_qty'            => $return_qty,
   269→                'rate'                  => $rate,
   270→                'amount'                => $amount,
   271→                'condition'             => $item['condition'] ?? 'Good',
   272→            ];
   273→        }
   274→
   275→        if (empty($validatedItems)) {
   276→            throw new Exception('No valid return quantities entered.');
   277→        }
   278→
   279→        // Insert main return
   280→        $this->db->query("
   281→            INSERT INTO sales_returns 
   282→            (return_code, sales_invoice_id, customer_id, return_date, 
   283→             total_amount, reason, status, created_by)
   284→            VALUES (:code, :invoice_id, :customer_id, :return_date, 
   285→                    :total_amount, :reason, 'pending', :created_by)
   286→        ");
   287→
   288→        $this->db->bind(':code', $return_code);
   289→        $this->db->bind(':invoice_id', $data['sales_invoice_id']);
   290→        $this->db->bind(':customer_id', $data['customer_id']);
   291→        $this->db->bind(':return_date', $data['return_date'] ?? date('Y-m-d'));
   292→        $this->db->bind(':total_amount', $computedTotal);
   293→        $this->db->bind(':reason', $data['reason'] ?? '');
   294→        $this->db->bind(':created_by', $_SESSION['user_id'] ?? 1);
   295→        $this->db->execute();
   296→
   297→        $return_id = $this->db->lastInsertId();
   298→
   299→        foreach ($validatedItems as $item) {
   300→            $this->db->query("
   301→                INSERT INTO sales_return_items 
   302→                (sales_return_id, sales_invoice_item_id, product_id, 
   303→                 return_qty, rate, amount, condition)
   304→                VALUES (:rid, :sii, :pid, :rqty, :rate, :amt, :cond)
   305→            ");
   306→
   307→            $this->db->bind(':rid', $return_id);
   308→            $this->db->bind(':sii', $item['sales_invoice_item_id']);
   309→            $this->db->bind(':pid', $item['product_id']);
   310→            $this->db->bind(':rqty', $item['return_qty']);
   311→            $this->db->bind(':rate', $item['rate']);
   312→            $this->db->bind(':amt', $item['amount']);
   313→            $this->db->bind(':cond', $item['condition']);
   314→            $this->db->execute();
   315→        }
   316→
   317→        $this->db->commit();
   318→        return $return_id;
   319→
   320→    } catch (Exception $e) {
   321→        $this->db->rollback();
   322→        $this->lastError = $e->getMessage();
   323→        error_log("Create Sales Return Error: " . $e->getMessage());
   324→        return false;
   325→    }

   428→public function confirmReturn($return_id, $items) {
   429→    $this->lastError = '';
   430→    $return_id = (int)$return_id;
   431→
   432→    if ($return_id <= 0) {
   433→        $this->lastError = 'Invalid Return ID';
   434→        return ['success' => false, 'message' => $this->lastError];
   435→    }
   436→
   437→    $this->db->beginTransaction();
   438→    try {
   439→        $this->db->query("
   440→            SELECT * FROM sales_returns
   441→            WHERE id = :id AND status = 'pending' AND COALESCE(is_reversed, 0) = 0
   442→            FOR UPDATE
   443→        ");
   444→        $this->db->bind(':id', $return_id);
   445→        $returnHeader = $this->db->single();
   446→
   447→        if (!$returnHeader) {
   448→            throw new Exception('Return not found or already processed.');
   449→        }
   450→
   451→        $this->db->query("SELECT branch_id FROM sales_invoices WHERE id = :id");
   452→        $this->db->bind(':id', (int)$returnHeader['sales_invoice_id']);
   453→        $invBranch = $this->db->single();
   454→        $branchId = (int)($invBranch['branch_id'] ?? self::sessionBranchId());
   455→        $this->assertInvoiceAccessible($branchId);
   456→
   457→        $cogsTotal = 0.0;
   458→        $damageByWarehouse = [];
   459→
   460→        foreach ($items as $item) {
   461→            $return_item_id = (int)$item['return_item_id'];
   462→            $product_id     = (int)$item['product_id'];
   463→            $warehouse_id   = (int)$item['warehouse_id'];
   464→            $rate           = (float)($item['rate'] ?? 0);
   465→            $condition      = $this->normalizeReturnCondition($item['condition'] ?? 'Good');
   466→            $return_qty     = (float)($item['return_qty'] ?? 0);
   467→
   468→            if ($warehouse_id <= 0) {
   469→                throw new Exception("Warehouse is required");
   470→            }
   471→
   472→            if (!$this->warehouseBelongsToBranch($warehouse_id, $branchId)) {
   473→                throw new Exception('Selected warehouse does not belong to your branch.');
   474→            }
   475→
   476→            $this->db->query("
   477→                SELECT id, product_id, return_qty, rate
   478→                FROM sales_return_items
   479→                WHERE id = :item_id AND sales_return_id = :return_id
   480→                FOR UPDATE
   481→            ");
   482→            $this->db->bind(':item_id', $return_item_id);
   483→            $this->db->bind(':return_id', $return_id);
   484→            $lineRow = $this->db->single();
   485→
   486→            if (!$lineRow) {
   487→                throw new Exception("Return line #{$return_item_id} not found on this return.");
   488→            }
   489→
   490→            if ((int)$lineRow['product_id'] !== $product_id) {
   491→                throw new Exception("Product mismatch on return line #{$return_item_id}.");
   492→            }
   493→
   494→            $maxQty = (float)$lineRow['return_qty'];
   495→            if ($return_qty <= 0) {
   496→                throw new Exception("Return quantity must be greater than zero for line #{$return_item_id}.");
   497→            }
   498→            if ($return_qty > $maxQty + 0.0001) {
   499→                throw new Exception(
   500→                    "Confirmed quantity cannot exceed requested return qty ("
   501→                    . number_format($maxQty, 2) . ") on line #{$return_item_id}."
   502→                );
   503→            }
   504→
   505→            if ($rate <= 0) {
   506→                $rate = (float)$lineRow['rate'];
   507→            }
   508→
   509→            $lineAmount = round($return_qty * $rate, 2);
   510→
   511→            $this->db->query("
   512→                UPDATE sales_return_items
   513→                SET warehouse_id = :wid,
   514→                    condition = :cond,
   515→                    return_qty = :rqty,
   516→                    rate = :rate,
   517→                    amount = :amt,
   518→                    confirmed_at = NOW()
   519→                WHERE id = :item_id
   520→                  AND sales_return_id = :return_id
   521→            ");
   522→            $this->db->bind(':wid', $warehouse_id);
   523→            $this->db->bind(':cond', $condition);
   524→            $this->db->bind(':rqty', $return_qty);
   525→            $this->db->bind(':rate', $rate);
   526→            $this->db->bind(':amt', $lineAmount);
   527→            $this->db->bind(':item_id', $return_item_id);
   528→            $this->db->bind(':return_id', $return_id);
   529→            $this->db->execute();
   530→
   531→            // === Stock Logic (matches persisted condition) ===
   532→            if ($condition === 'Good' && $return_qty > 0) {
   533→                $avgCost = $this->stock->getWarehouseAvgCost($warehouse_id, $product_id);
   534→                if ($avgCost <= 0) {
   535→                    $avgCost = $rate;
   536→                }
   537→
   538→                $this->stock->updateWarehouseStock(
   539→                    $warehouse_id,
   540→                    $product_id,
   541→                    $return_qty,
   542→                    $avgCost
   543→                );
   544→
   545→                $this->stock->logMovement([
   546→                    'product_id'     => $product_id,
   547→                    'warehouse_id'   => $warehouse_id,
   548→                    'qty'            => $return_qty,

   552→                    'remarks'        => "Sales Return - Good Condition"
   553→                ]);
   554→                $cogsTotal += $return_qty * $avgCost;
   555→            } elseif ($condition === 'Damage' && $return_qty > 0) {
   556→                $avgCost = $this->stock->getWarehouseAvgCost($warehouse_id, $product_id);

   562→                    $warehouse_id,

   580→                }

   615→            SELECT COALESCE(SUM(amount), 0) AS total
   616→            FROM sales_return_items

   879→            }
   880→            $total += $qty * $rate;
   881→        }
   882→
   883→        return round($total, 2);

   910→    /**
   911→     * Lines that will be removed from stock on reversal (for UI + pre-check).
   912→     */
   913→    protected function buildStockReversalPreview(int $returnId): array
   914→    {

   926→            }
   927→            $warehouseId = (int)($movement['warehouse_id'] ?? 0);
   928→            $productId = (int)($movement['product_id'] ?? 0);

   930→                continue;
   931→            }

   942→                SELECT COALESCE(qty, 0) AS qty

   944→                WHERE warehouse_id = :wid AND product_id = :pid

   968→            LEFT JOIN warehouses w ON w.id = sri.warehouse_id

   970→              AND sri.condition = 'Good'

   981→                SELECT COALESCE(qty, 0) AS qty FROM warehouse_stock

   991→                'qty'            => $qty,
   992→                'physical_qty'   => (float)($this->db->single()['qty'] ?? 0),
   993→            ];
   994→        }
   995→
   996→        return $lines;

   999→// UNDO / REVERSAL - Full Reversal (stock, customer ledger, GL)
  1000→public function reverseReturn($return_id, $reason = '') {
  1001→    $reason = trim($reason);
  1002→    if (strlen($reason) < 5) {
  1003→        return ['status' => 'error', 'message' => 'Reversal reason is required (minimum 5 characters).'];
  1004→    }
  1005→
  1006→    $return_id = (int)$return_id;
  1007→    $this->db->beginTransaction();
  1008→
  1009→    try {
  1010→        $this->db->query("
  1011→            SELECT * FROM sales_returns
  1012→            WHERE id = :id AND COALESCE(is_reversed, 0) = 0
  1013→            FOR UPDATE
  1014→        ");
  1015→        $this->db->bind(':id', $return_id);
  1016→        $return = $this->db->single();
  1017→
  1018→        if (!$return) {
  1019→            throw new Exception('Return not found or already reversed.');
  1020→        }
  1021→
  1022→        $priorStatus = (string)($return['status'] ?? 'pending');
  1023→        $wasCompleted = $priorStatus === 'completed';
  1024→
  1025→        $this->db->query("SELECT branch_id FROM sales_invoices WHERE id = :id");
  1026→        $this->db->bind(':id', (int)$return['sales_invoice_id']);
  1027→        $invBr = $this->db->single();
  1028→        $returnBranchId = (int)($invBr['branch_id'] ?? 0);
  1029→        $this->assertInvoiceAccessible($returnBranchId);
  1030→
  1031→        if (!empty($this->stock->getByReference('sales_return_reversal', $return_id))) {
  1032→            throw new Exception('This return was already reversed (stock log exists).');
  1033→        }
  1034→
  1035→        $customer_id = (int)$return['customer_id'];
  1036→        $return_code = (string)($return['return_code'] ?? '');
  1037→        $journalReversedId = null;
  1038→        $stockLinesReversed = 0;
  1039→
  1040→        if ($wasCompleted) {
  1041→            $previewLines = $this->buildStockReversalPreview($return_id);
  1042→            $stockBlock = $this->getStockReversalBlockReason($previewLines);
  1043→            if ($stockBlock !== null) {
  1044→                throw new Exception($stockBlock);
  1045→            }
  1046→
  1047→            $this->reverseLinkedDamageForReturn($return_id, $return_code, $reason);
  1048→            $this->reverseConfirmedReturnStock($return_id, $return_code, $reason);
  1049→            $stockLinesReversed = count($this->stock->getByReference('sales_return_reversal', $return_id));
  1050→
  1051→            $ledgerAmount = $this->getReturnLedgerCreditAmount($return_id);
  1052→            if ($ledgerAmount > 0.0001) {
  1053→                $this->db->query("
  1054→                    SELECT COALESCE(running_balance, 0) AS balance
  1055→                    FROM customer_ledger
  1056→                    WHERE customer_id = :cid
  1057→                    ORDER BY id DESC
  1058→                    LIMIT 1
  1059→                ");
  1060→                $this->db->bind(':cid', $customer_id);
  1061→                $prevBalance = (float)($this->db->single()['balance'] ?? 0);
  1062→                $newBalance = $prevBalance + $ledgerAmount;
  1063→
  1064→                $this->insertCustomerLedgerEntry([
  1065→                    'customer_id'      => $customer_id,
  1066→                    'reference_type'   => 'reversal',
  1067→                    'reference_id'     => $return_id,
  1068→                    'debit'            => $ledgerAmount,
  1069→                    'credit'           => 0,
  1070→                    'running_balance'  => $newBalance,
  1071→                    'branch_id'        => $returnBranchId,
  1072→                    'remarks'          => "Reversal of Sales Return #{$return_code} — {$reason}",
  1073→                    'is_reversed'      => 1,
  1074→                ]);
  1075→
  1076→                $this->db->query("
  1077→                    UPDATE customer_ledger
  1078→                    SET is_reversed = 1
  1079→                    WHERE reference_type = 'sales_return'
  1080→                      AND reference_id = :rid
  1081→                      AND COALESCE(is_reversed, 0) = 0
  1082→                ");
  1083→                $this->db->bind(':rid', $return_id);
  1084→                $this->db->execute();
  1085→            }
  1086→
  1087→            require_once __DIR__ . '/../services/Accounting/JournalPostingService.php';
  1088→            $journalService = new JournalPostingService();

  1734→    /**
  1735→     * Undo linked auto damage write-offs before reversing return stock receive legs.
  1736→     */
  1737→    private function reverseLinkedDamageForReturn(int $returnId, string $returnCode, string $reason): void
  1738→    {
  1739→        $this->db->query('
  1740→            SELECT id, damage_code, journal_entry_id
  1741→            FROM damage_invoices
  1742→            WHERE sales_return_id = :rid
  1743→              AND COALESCE(is_reversed, 0) = 0
  1744→            FOR UPDATE
  1745→        ');
  1746→        $this->db->bind(':rid', $returnId);
  1747→        $damageRows = $this->db->resultSet() ?: [];
  1748→
  1749→        if ($damageRows === []) {
  1750→            return;
  1751→        }
  1752→
  1753→        require_once __DIR__ . '/../services/Accounting/JournalPostingService.php';
  1754→        $journalService = new JournalPostingService();
  1755→        $movementReason = 'Reversal of return #' . $returnCode . ' linked damage: ' . $reason;
  1756→
  1757→        foreach ($damageRows as $damage) {
  1758→            $damageId = (int)($damage['id'] ?? 0);
  1759→            if ($damageId <= 0) {
  1760→                continue;
  1761→            }
  1762→
  1763→            $movements = $this->stock->getByReference('damage', $damageId);
  1764→            $reversed = 0;
  1765→            foreach ($movements as $movement) {
  1766→                if (!empty($movement['is_reversed'])) {
  1767→                    continue;
  1768→                }
  1769→                $qty = (float)($movement['qty'] ?? 0);
  1770→                if (abs($qty) < 0.0001) {
  1771→                    continue;
  1772→                }
  1773→                try {
  1774→                    $ok = $this->stock->transactions()->reverseTransaction(
  1775→                        (int)$movement['id'],
  1776→                        $movementReason
  1777→                    );
  1778→                } catch (RuntimeException $e) {
  1779→                    throw new Exception($e->getMessage());
  1780→                }
  1781→                if ($ok) {
  1782→                    $reversed++;
  1783→                }
  1784→            }
  1785→
  1786→            if ($reversed === 0 && $movements !== []) {
  1787→                throw new Exception('Could not reverse linked damage stock for #' . ($damage['damage_code'] ?? $damageId));
  1788→            }
  1789→
  1790→            $journalId = (int)($damage['journal_entry_id'] ?? 0);
  1791→            if ($journalId > 0) {
  1792→                $rev = $journalService->reverseLinkedJournal($journalId, $movementReason);
  1793→                if (($rev['status'] ?? '') === 'error') {
  1794→                    throw new Exception('Failed to reverse linked damage GL: ' . ($rev['message'] ?? ''));
  1795→                }
  1796→            }
  1797→
  1798→            $this->db->query('
  1799→                UPDATE damage_invoices
  1800→                SET is_reversed = 1,
  1801→                    reversed_at = NOW(),
  1802→                    reversed_by = :uid,
  1803→                    reverse_reason = :reason
  1804→                WHERE id = :id
  1805→            ');
  1806→            $this->db->bind(':uid', (int)($_SESSION['user_id'] ?? 1));
  1807→            $this->db->bind(':reason', $reason);
  1808→            $this->db->bind(':id', $damageId);
  1809→            $this->db->execute();
  1810→        }
  1811→    }
