   226→    public function uploadProductImage(array $file): ?string
   227→    {
   228→        if (empty($file['tmp_name']) || $file['error'] !== UPLOAD_ERR_OK) {
   229→            return null;
   230→        }
   231→
   232→        $maxSize = 2 * 1024 * 1024;
   233→        if ($file['size'] > $maxSize) {
   234→            return null;
   235→        }
   236→
   237→        $finfo = finfo_open(FILEINFO_MIME_TYPE);
   238→        $mime = finfo_file($finfo, $file['tmp_name']);
   239→        finfo_close($finfo);
   240→
   241→        $allowedMimes = [
   242→            'image/jpeg' => 'jpg',
   243→            'image/png'  => 'png',
   244→            'image/webp' => 'webp',
   245→        ];
   246→
   247→        if (!isset($allowedMimes[$mime])) {
   248→            return null;
   249→        }
   250→
   251→        $uploadDir = __DIR__ . '/../../public/uploads/products/';
   252→        if (!is_dir($uploadDir)) {
   253→            mkdir($uploadDir, 0755, true);
   254→        }
   255→
   256→        $extension = $allowedMimes[$mime];
   257→        $filename = 'prod_' . time() . '_' . bin2hex(random_bytes(4)) . '.' . $extension;
   258→        $destination = $uploadDir . $filename;
   259→
   260→        if (move_uploaded_file($file['tmp_name'], $destination)) {
   261→            $imgInfo = @getimagesize($destination);
   262→            if ($imgInfo === false) {
   263→                @unlink($destination);
   264→                return null;
   265→            }
   266→            return 'uploads/products/' . $filename;
   267→        }
   268→
   269→        return null;
   270→    }

   365→    public function getProductsForDataTable(array $params): array
   366→    {
   367→        $start          = (int)($params['start'] ?? 0);
   368→        $length         = (int)($params['length'] ?? 25);
   369→        $searchValue    = trim($params['search']['value'] ?? '');
   370→        $orderColumn    = (int)($params['order'][0]['column'] ?? 0);
   371→        $orderDir       = $this->sanitizeOrderDir($params['order'][0]['dir'] ?? 'asc');
   372→
   373→        $filterCategory = $params['filterCategory'] ?? '';
   374→        $filterUnit     = $params['filterUnit'] ?? '';
   375→        $filterGroup    = $params['filterGroup'] ?? '';
   376→        $includeDeleted = !empty($params['includeDeleted']);
   377→
   378→        $columns = [
   379→            'p.product_code',
   380→            'p.product_name',
   381→            'g.group_name',
   382→            'c.category_name',
   383→            'p.unit',
   384→            'total_stock',
   385→            'current_default_rate',
   386→        ];
   387→
   388→        $baseQuery = "
   389→            FROM products p
   390→            LEFT JOIN product_categories c ON p.category_id = c.id
   391→            LEFT JOIN product_groups g ON p.group_id = g.id
   392→        ";
   393→
   394→        $where = [];
   395→        $bindParams = [];
   396→
   397→        if (!$includeDeleted) {
   398→            $where[] = "p.is_active = 1";
   399→        }
   400→
   401→        if ($searchValue !== '') {
   402→            $where[] = "(p.product_name LIKE :search OR p.product_code LIKE :search)";
   403→            $bindParams[':search'] = "%{$searchValue}%";
   404→        }
   405→
   406→        if ($filterCategory) {
   407→            $where[] = "p.category_id = :category";
   408→            $bindParams[':category'] = $filterCategory;
   409→        }
   410→
   411→        if ($filterGroup) {
   412→            $where[] = "p.group_id = :grp";
   413→            $bindParams[':grp'] = $filterGroup;
   414→        }
   415→
   416→        if ($filterUnit) {
   417→            $where[] = "p.unit = :unit";
   418→            $bindParams[':unit'] = $filterUnit;
   419→        }
   420→
   421→        $whereSql = $where ? 'WHERE ' . implode(' AND ', $where) : '';
   422→
   423→        $totalQuery = "SELECT COUNT(p.id) as total {$baseQuery}";
   424→        if (!$includeDeleted) {
   425→            $totalQuery .= " WHERE p.is_active = 1";
   426→        }
   427→        $this->db->query($totalQuery);
   428→        $recordsTotal = $this->db->single()['total'] ?? 0;
   429→
   430→        $filteredQuery = "SELECT COUNT(p.id) as total {$baseQuery} {$whereSql}";
   431→        $this->db->query($filteredQuery);
   432→        foreach ($bindParams as $key => $val) {
   433→            $this->db->bind($key, $val);
   434→        }
   435→        $recordsFiltered = $this->db->single()['total'] ?? 0;
   436→
   437→        $orderBy = $columns[$orderColumn] ?? 'p.product_name';
   438→
   439→        $latestPriceSub = "
   440→            SELECT ph2.min_rate, ph2.max_rate, ph2.default_rate
   441→            FROM product_price_history ph2
   442→            WHERE ph2.product_id = p.id
   443→            ORDER BY ph2.effective_from DESC, ph2.created_at DESC, ph2.id DESC
   444→            LIMIT 1
   445→        ";
   446→
   447→        $dataQuery = "
   448→            SELECT 
   449→                p.id, 
   450→                p.product_code, 
   451→                p.product_name, 
   452→                p.unit, 
   453→                p.is_active,
   454→                p.image,
   455→                c.category_name,
   456→                g.group_name,
   457→                COALESCE((
   458→                    SELECT SUM(ws.qty) FROM warehouse_stock ws WHERE ws.product_id = p.id
   459→                ), 0) AS total_stock,
   460→                COALESCE((
   461→                    SELECT lp.default_rate FROM product_price_history lp
   462→                    WHERE lp.product_id = p.id
   463→                    ORDER BY lp.effective_from DESC, lp.created_at DESC, lp.id DESC
   464→                    LIMIT 1
   465→                ), 0) AS current_price,
   466→                COALESCE((
   467→                    SELECT lp.default_rate FROM product_price_history lp
   468→                    WHERE lp.product_id = p.id
   469→                    ORDER BY lp.effective_from DESC, lp.created_at DESC, lp.id DESC
   470→                    LIMIT 1
   471→                ), 0) AS current_default_rate,
   472→                COALESCE((
   473→                    SELECT lp.min_rate FROM product_price_history lp
   474→                    WHERE lp.product_id = p.id
   475→                    ORDER BY lp.effective_from DESC, lp.created_at DESC, lp.id DESC
   476→                    LIMIT 1
   477→                ), 0) AS min_rate,
   478→                COALESCE((
   479→                    SELECT lp.max_rate FROM product_price_history lp
   480→                    WHERE lp.product_id = p.id
   481→                    ORDER BY lp.effective_from DESC, lp.created_at DESC, lp.id DESC
   482→                    LIMIT 1
   483→                ), 0) AS max_rate
   484→            {$baseQuery}
   485→            {$whereSql}
   486→            ORDER BY {$orderBy} {$orderDir}
   487→            LIMIT {$length} OFFSET {$start}
   488→        ";
   489→
   490→        $this->db->query($dataQuery);
   491→        foreach ($bindParams as $key => $val) {
   492→            $this->db->bind($key, $val);
   493→        }
   494→        $data = $this->db->resultSet();
   495→
   496→        return [
   497→            'draw'            => (int)($params['draw'] ?? 1),
   498→            'recordsTotal'    => $recordsTotal,
   499→            'recordsFiltered' => $recordsFiltered,
   500→            'data'            => $data,
   501→        ];
   502→    }

   613→    public function getGroupsForDataTable(array $params): array
   614→    {
   615→        $start       = (int)($params['start'] ?? 0);
   616→        $length      = (int)($params['length'] ?? 25);
   617→        $searchValue = trim($params['search']['value'] ?? '');
   618→        $orderColumn = (int)($params['order'][0]['column'] ?? 0);
   619→        $orderDir    = $this->sanitizeOrderDir($params['order'][0]['dir'] ?? 'asc');
   620→        $includeDeleted = !empty($params['includeDeleted']);
   621→
   622→        $columns = ['g.group_name'];
   623→        $baseQuery = "FROM product_groups g";
   624→        $where = [];
   625→        $bindParams = [];
   626→
   627→        if (!$includeDeleted) {
   628→            $where[] = "g.is_active = 1";
   629→        }
   630→
   631→        if ($searchValue !== '') {
   632→            $where[] = "g.group_name LIKE :search";
   633→            $bindParams[':search'] = "%{$searchValue}%";
   634→        }
   635→
   636→        $whereSql = $where ? 'WHERE ' . implode(' AND ', $where) : '';
   637→
   638→        $totalQuery = "SELECT COUNT(g.id) AS total {$baseQuery}";
   639→        if (!$includeDeleted) {
   640→            $totalQuery .= " WHERE g.is_active = 1";
   641→        }
   642→        $this->db->query($totalQuery);
   643→        $recordsTotal = $this->db->single()['total'] ?? 0;
   644→
   645→        $filteredQuery = "SELECT COUNT(g.id) AS total {$baseQuery} {$whereSql}";
   646→        $this->db->query($filteredQuery);
   647→        foreach ($bindParams as $key => $val) {
   648→            $this->db->bind($key, $val);
   649→        }
   650→        $recordsFiltered = $this->db->single()['total'] ?? 0;
   651→
   652→        $orderBy = $columns[$orderColumn] ?? 'g.group_name';
   653→        $dataQuery = "
   654→            SELECT g.id, g.group_name, g.is_active,
   655→                (SELECT COUNT(*) FROM products WHERE group_id = g.id AND is_active = 1) AS product_count
   656→            {$baseQuery} {$whereSql}
   657→            ORDER BY {$orderBy} {$orderDir}
   658→            LIMIT {$length} OFFSET {$start}
   659→        ";
   660→
   661→        $this->db->query($dataQuery);
   662→        foreach ($bindParams as $key => $val) {
   663→            $this->db->bind($key, $val);
   664→        }
   665→
   666→        return [
   667→            'draw'            => (int)($params['draw'] ?? 1),
   668→            'recordsTotal'    => $recordsTotal,
   669→            'recordsFiltered' => $recordsFiltered,
   670→            'data'            => $this->db->resultSet(),
   671→        ];
   672→    }

   750→    public function bulkDeactivate(array $ids) {
   751→        if (empty($ids)) {
   752→            return false;
   753→        }
   754→
   755→        $deletableIds = [];
   756→        foreach ($ids as $id) {
   757→            if ($this->canDelete($id)) {
   758→                $deletableIds[] = (int)$id;
   759→            }
   760→        }
   761→
   762→        if (empty($deletableIds)) {
   763→            return false;
   764→        }
   765→
   766→        $placeholders = implode(',', array_fill(0, count($deletableIds), '?'));
   767→        $this->db->query("UPDATE products SET is_active = 0 WHERE id IN ($placeholders)");
   768→
   769→        foreach ($deletableIds as $i => $id) {
   770→            $this->db->bind($i, $id);
   771→        }
   772→
   773→        return $this->db->execute();
   774→    }
