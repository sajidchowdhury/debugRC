   645→   645→    public function deleteInvoice($invoice_id, ?string $deleteReason = null) { 
   646→   646→
   647→   647→    
   648→   648→        $this->db->beginTransaction();
   649→   649→        try {
   650→   650→            // 1. Get invoice details
   651→   651→            $this->db->query("SELECT * FROM sales_invoices WHERE id = :id");
   652→   652→            $this->db->bind(':id', $invoice_id);
   653→   653→            $invoice = $this->db->single();
   654→   654→
   655→   655→            if (!$invoice) {
   656→   656→                throw new Exception("Invoice not found");
   657→   657→            }
   658→   658→
   659→   659→            if ((int)($invoice['is_reversed'] ?? 0) === 1) {
   660→   660→                throw new Exception('Invoice is already deleted.');
   661→   661→            }
   662→   662→
   663→   663→            if ($invoice['status'] !== 'draft') {
   664→   664→                throw new Exception(
   665→   665→                    'Cannot delete invoice: status is "' . $invoice['status'] . '". '
   666→   666→                    . 'Only draft invoices (before godown/challan) can be deleted.'
   667→   667→                );
   668→   668→            }
   669→   669→
   670→   670→            if (!empty($invoice['godown_issued_at'])) {
   671→   671→                throw new Exception(
   672→   672→                    'Cannot delete: godown has been prepared. Reverse or complete warehouse flow first.'
   673→   673→                );
   674→   674→            }
   675→   675→
   676→   676→            $this->assertInvoiceAccessible((int)$invoice['branch_id']);
   677→   677→
   678→   678→            if (!empty($invoice['journal_entry_id'])) {
   679→   679→                require_once __DIR__ . '/../../Accounting/JournalPostingService.php';
   680→   680→                $journalService = new JournalPostingService($this->db ?? null);
   681→   681→                $rev = $journalService->reverseLinkedJournal(
   682→   682→                    (int)$invoice['journal_entry_id'],
   683→   683→                    'Invoice deleted: ' . ($invoice['invoice_code'] ?? $invoice_id)
   684→   684→                );
   685→   685→                if (($rev['status'] ?? '') === 'error') {
   686→   686→                    throw new Exception('Failed to reverse invoice journal: ' . ($rev['message'] ?? ''));
   687→   687→                }
   688→   688→            }
   689→   689→
   690→   690→            $this->db->query("
   691→   691→                SELECT id FROM sales_challans
   692→   692→                WHERE sales_invoice_id = :id AND COALESCE(is_reversed, 0) = 0
   693→   693→                LIMIT 1
   694→   694→            ");
   695→   695→            $this->db->bind(':id', $invoice_id);
   696→   696→            if ($this->db->single()) {
   697→   697→                throw new Exception('Cannot delete: a delivery challan exists. Reverse the challan first.');
   698→   698→            }
   699→   699→
   700→   700→            $this->db->query("
   701→   701→                SELECT COUNT(*) AS c FROM sales_invoice_dispatches
   702→   702→                WHERE sales_invoice_id = :id AND COALESCE(dispatched_qty, 0) > 0
   703→   703→            ");
   704→   704→            $this->db->bind(':id', $invoice_id);
   705→   705→            if ((int)($this->db->single()['c'] ?? 0) > 0) {
   706→   706→                throw new Exception('Cannot delete: stock has already been dispatched for this invoice.');
   707→   707→            }
   708→   708→
   709→   709→            // 2. Check if any payment has been made against this invoice
   710→   710→            if ($this->invoiceHasPayments($invoice_id)) {
   711→   711→                throw new Exception("Cannot delete invoice. Payments have already been received against it.");
   712→   712→            }
   713→   713→
   714→   714→            // 3. Find and reverse the customer ledger entry
   715→   715→            $this->db->query("SELECT id FROM customer_ledger 
   716→   716→                              WHERE reference_type = 'invoice' 
   717→   717→                                AND reference_id = :invoice_id");
   718→   718→            $this->db->bind(':invoice_id', $invoice_id);
   719→   719→            $ledger = $this->db->single();
   720→   720→
   721→   721→            if ($ledger) {
   722→   722→                // Reverse the debit entry by creating a credit reversal
   723→   723→                $this->db->query("SELECT running_balance FROM customer_ledger 
   724→   724→                                  WHERE customer_id = :cid 
   725→   725→                                  ORDER BY id DESC LIMIT 1");
   726→   726→                $this->db->bind(':cid', $invoice['customer_id']);
   727→   727→                $currentBalance = $this->db->single()['running_balance'] ?? 0;
   728→   728→
   729→   729→                $newBalance = $currentBalance - $invoice['total_amount'];
   730→   730→
   731→   731→                $this->insertCustomerLedgerEntry([
   732→   732→                    'customer_id'      => (int)$invoice['customer_id'],
   733→   733→                    'reference_type'   => 'reversal',
   734→   734→                    'reference_id'     => $invoice_id,
   735→   735→                    'debit'            => 0,
   736→   736→                    'credit'           => (float)$invoice['total_amount'],
   737→   737→                    'running_balance'  => $newBalance,
   738→   738→                    'branch_id'        => (int)$invoice['branch_id'],
   739→   739→                    'remarks'          => 'Reversed due to invoice deletion',
   740→   740→                    'is_reversed'      => 1,
   741→   741→                ]);
   742→   742→            }
   743→   743→
   744→   744→            // 4. Delete invoice items      
   745→   745→            $this->db->query("DELETE FROM sales_invoice_items WHERE sales_invoice_id = :id");
   746→   746→            $this->db->bind(':id', $invoice_id);
   747→   747→            $this->db->execute();
   748→   748→
   749→   749→            $reverseReason = ($deleteReason !== null && trim($deleteReason) !== '')
   750→   750→                ? trim($deleteReason)
   751→   751→                : 'Deleted by user';
   752→   752→
   753→   753→            // 5. Delete the invoice itself (soft delete using is_reversed)
   754→   754→            $this->db->query("UPDATE sales_invoices 
   755→   755→                              SET is_reversed = 1, 
   756→   756→                                  reversed_at = NOW(), 
   757→   757→                                  reversed_by = :reversed_by,
   758→   758→                                  reverse_reason = :reverse_reason
   759→   759→                              WHERE id = :id");
   760→   760→            $this->db->bind(':id', $invoice_id);
   761→   761→            $this->db->bind(':reversed_by', $_SESSION['user_id'] ?? 1);
   762→   762→            $this->db->bind(':reverse_reason', $reverseReason);
   763→   763→            $this->db->execute();
   764→   764→
   765→   765→            // 6. Remove soft dispatch reservations (invoice is reversed)
   766→   766→            $this->db->query("DELETE FROM sales_invoice_dispatches WHERE sales_invoice_id = :id");
   767→   767→            $this->db->bind(':id', $invoice_id);
   768→   768→            $this->db->execute();
   769→   769→
   770→   770→            $this->db->commit();
   771→   771→
   772→   772→            return [
   773→   773→                'status' => 'success',
   774→   774→                'message' => 'Invoice #' . $invoice['invoice_code'] . ' has been deleted successfully.'
   775→   775→            ];
   776→   776→
   777→   777→        } catch (Exception $e) {
   778→   778→            $this->db->rollback();
   779→   779→            return [
   780→   780→                'status' => 'error',
   781→   781→                'message' => $e->getMessage()
   782→   782→            ];
   783→   783→        }
   784→   784→
