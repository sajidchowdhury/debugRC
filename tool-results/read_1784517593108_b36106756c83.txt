    38→    38→public function createTransaction($post) {
    39→    39→    $this->db->beginTransaction();
    40→    40→    try {
    41→    41→        $branch_id = self::sessionBranchId() ?: (int)($_SESSION['branch_id'] ?? 0);
    42→    42→        if ($branch_id <= 0) {
    43→    43→            $branch_id = (int)($_SESSION['branch_id'] ?? 1);
    44→    44→        }
    45→    45→        $user_id = (int)($_SESSION['user_id'] ?? 1);
    46→    46→
    47→    47→        $type   = $post['transaction_type'] ?? 'receive';
    48→    48→        $amount = round((float)($post['amount'] ?? 0), 2);
    49→    49→        $customer_id = (int)($post['customer_id'] ?? 0);
    50→    50→        $mode = strtolower((string)($post['mode'] ?? 'cash'));
    51→    51→
    52→    52→        if ($amount <= 0 || $customer_id <= 0) {
    53→    53→            throw new Exception('Invalid amount or customer');
    54→    54→        }
    55→    55→
    56→    56→        $allowedTypes = ['receive', 'payment', 'discount', 'write_off'];
    57→    57→        if (!in_array($type, $allowedTypes, true)) {
    58→    58→            throw new Exception('Invalid transaction type');
    59→    59→        }
    60→    60→
    61→    61→        if ($mode === 'bank' && empty($post['bank_id'])) {
    62→    62→            throw new Exception('Select a bank account for bank mode');
    63→    63→        }
    64→    64→
    65→    65→        $customer = $this->Get_Customer_By_Id($customer_id);
    66→    66→        if (!$customer || empty($customer['is_active'])) {
    67→    67→            throw new Exception('Customer not found or inactive');
    68→    68→        }
    69→    69→
    70→    70→        if (in_array($type, ['discount', 'write_off'], true)) {
    71→    71→            $due = $this->getCustomerDue($customer_id);
    72→    72→            if ($amount > $due + 0.01) {
    73→    73→                throw new Exception(
    74→    74→                    'Amount cannot exceed customer due (Tk ' . number_format($due, 2) . ')'
    75→    75→                );
    76→    76→            }
    77→    77→        }
    78→    78→
    79→    79→        $payment_code = $this->generateCustomerPaymentCode($branch_id);
    80→    80→
    81→    81→        // === 1. INSERT MAIN PAYMENT RECORD ===
    82→    82→        $this->db->query("
    83→    83→            INSERT INTO customer_payments 
    84→    84→            (payment_code, payment_date, customer_id, amount, payment_mode, 
    85→    85→             bank_id, remarks, created_by, branch_id, collected_by, transaction_type)
    86→    86→            VALUES 
    87→    87→            (:code, :date, :cid, :amt, :mode, :bank, :rem, :uid, :bid, :col, :type)
    88→    88→        ");
    89→    89→
    90→    90→        $this->db->bind(':code', $payment_code);
    91→    91→        $this->db->bind(':date', $post['transaction_date'] ?? date('Y-m-d'));
    92→    92→        $this->db->bind(':cid', $customer_id);
    93→    93→        $this->db->bind(':amt', $amount);
    94→    94→        $this->db->bind(':mode', $post['mode'] ?? 'cash');
    95→    95→        $this->db->bind(':bank', $post['bank_id'] ?? null);
    96→    96→        $this->db->bind(':rem', $post['narration'] ?? '');
    97→    97→        $this->db->bind(':uid', $user_id);
    98→    98→        $this->db->bind(':bid', $branch_id);
    99→    99→        $this->db->bind(':col', $post['collected_by'] ?? $user_id);
   100→   100→        $this->db->bind(':type', $type);
   101→   101→        $this->db->execute();
   102→   102→
   103→   103→        $payment_id = $this->db->lastInsertId();
   104→   104→
   105→   105→        // === 2. CUSTOMER LEDGER ===
   106→   106→        $this->insertCustomerLedger($payment_id, $post, $amount, $branch_id, $user_id, $type);
   107→   107→
   108→   108→        // === 3. GL (all transaction types) ===
   109→   109→        $paymentPayload = [
   110→   110→            'payment_code'  => $payment_code,
   111→   111→            'payment_date'  => $post['transaction_date'] ?? date('Y-m-d'),
   112→   112→            'customer_id'   => $customer_id,
   113→   113→            'amount'        => $amount,
   114→   114→            'payment_mode'  => $post['mode'] ?? 'cash',
   115→   115→            'bank_id'       => !empty($post['bank_id']) ? (int)$post['bank_id'] : null,
   116→   116→            'branch_id'     => $branch_id,
   117→   117→        ];
   118→   118→
   119→   119→        require_once __DIR__ . '/../services/Accounting/JournalPostingService.php';
   120→   120→        $journalService = new JournalPostingService($this->db);
   121→   121→        $journalResult = $journalService->postCustomerTransactionJournal($payment_id, $paymentPayload, $type);
   122→   122→        if (($journalResult['status'] ?? '') === 'error') {
   123→   123→            throw new Exception('Journal posting failed: ' . ($journalResult['message'] ?? 'unknown'));
   124→   124→        }
   125→   125→
   126→   126→        $journalEntryId = null;
   127→   127→        if (!empty($journalResult['journal_entry_id'])) {
   128→   128→            $journalEntryId = (int)$journalResult['journal_entry_id'];
   129→   129→            $this->setPaymentJournalEntryId($payment_id, $journalEntryId);
   130→   130→        }
   131→   131→
   132→   132→        // === 5. Cash book (banks.balance) for bank mode receive/refund ===
   133→   133→        $this->syncBankBookBalance(
   134→   134→            !empty($post['bank_id']) ? (int)$post['bank_id'] : null,
   135→   135→            $amount,
   136→   136→            $type,
   137→   137→            $post['mode'] ?? 'cash',
   138→   138→            false
   139→   139→        );
   140→   140→
   141→   141→        if ($type === 'receive') {
   142→   142→            require_once __DIR__ . '/../services/Branch/BranchIntercompanyService.php';
   143→   143→            $mode = strtolower((string)($post['mode'] ?? 'cash'));
   144→   144→            $bankId = !empty($post['bank_id']) ? (int)$post['bank_id'] : null;
   145→   145→            $paymentMode = ($mode === 'bank' || $bankId > 0) ? 'bank' : 'cash';
   146→   146→            $settleResult = (new BranchIntercompanyService($this->db))->settleFromCustomerPayment($payment_id, [
   147→   147→                'payment_code'  => $payment_code,
   148→   148→                'payment_date'  => $post['transaction_date'] ?? date('Y-m-d'),
   149→   149→                'amount'        => $amount,
   150→   150→                'payment_mode'  => $paymentMode,
   151→   151→                'bank_id'       => $bankId,
   152→   152→                'branch_id'     => $branch_id,
   153→   153→            ]);
   154→   154→            if (($settleResult['status'] ?? '') === 'error') {
   155→   155→                throw new Exception('Branch demand settlement failed: ' . ($settleResult['message'] ?? ''));
   156→   156→            }
   157→   157→        }
   158→   158→
   159→   159→        $this->db->commit();
   160→   160→
   161→   161→        return [
   162→   162→            'status' => 'success',
   163→   163→            'message' => 'Payment recorded — ledger and GL posted.',
   164→   164→            'payment_code' => $payment_code,
   165→   165→            'payment_id' => $payment_id,
   166→   166→            'journal_entry_id' => $journalEntryId,
   167→   167→            'total_amount' => $amount,
   168→   168→        ];
   169→   169→
   170→   170→    } catch (Throwable $e) {
   171→   171→        if ($this->db->inTransaction()) {
   172→   172→            $this->db->rollback();
   173→   173→        }
   174→   174→        error_log('Customer Transaction Error: ' . $e->getMessage());
   175→   175→        return ['status' => 'error', 'message' => $e->getMessage()];
   176→   176→    }
   177→   177→}

   190→   190→    /**
   191→   191→     * Debit/credit sides for customer_ledger (positive running_balance = amount owed by customer).
   192→   192→     * - receive/discount/write_off: credit (reduce AR/due)
   193→   193→     * - payment (refund/advance out): debit (increase AR/due, matches Dr AR in GL)
   194→   194→     */
   195→   195→    private function getCustomerLedgerSides(string $type, float $amount): array
   196→   196→    {
   197→   197→        switch ($type) {
   198→   198→            case 'receive':
   199→   199→            case 'discount':
   200→   200→            case 'write_off':
   201→   201→                return ['debit' => 0.0, 'credit' => $amount];
   202→   202→            case 'payment':
   203→   203→                return ['debit' => $amount, 'credit' => 0.0];
   204→   204→            default:
   205→   205→                return ['debit' => 0.0, 'credit' => $amount];
   206→   206→        }
   207→   207→    }

   218→   218→     * Update banks.balance for bank-mode receive (in) or refund (out).
   219→   219→     *
   220→   220→     * @param bool $undo When true, reverses the balance movement (used on payment reversal).
   221→   221→     */
   222→   222→    private function syncBankBookBalance(?int $bankId, float $amount, string $transactionType, string $paymentMode, bool $undo = false): void
   223→   223→    {
   224→   224→        $paymentMode = strtolower(trim($paymentMode));
   225→   225→        if ($paymentMode !== 'bank' || !$bankId || $amount <= 0) {
   226→   226→            return;
   227→   227→        }
   228→   228→
   229→   229→        if (!in_array($transactionType, ['receive', 'payment'], true)) {
   230→   230→            return;
   231→   231→        }
   232→   232→
   233→   233→        $bankModel = new BankModel($this->db);
   234→   234→        $increaseBank = ($transactionType === 'receive');
   235→   235→        if ($undo) {
   236→   236→            $increaseBank = !$increaseBank;
   237→   237→        }
   238→   238→
   239→   239→        $bankModel->updateBalance($bankId, $amount, $increaseBank ? 'credit' : 'debit');
   240→   240→    }

   243→   243→    private function insertCustomerLedger($payment_id, $post, $amount, $branch_id, $user_id, $type) {
   244→   244→        $customer_id = (int)$post['customer_id'];
   245→   245→        $current = $this->getCustomerRunningBalance($customer_id);
   246→   246→        $sides = $this->getCustomerLedgerSides($type, $amount);
   247→   247→        $new_balance = $current + $sides['debit'] - $sides['credit'];
   248→   248→
   249→   249→        $this->insertCustomerLedgerEntry([
   250→   250→            'customer_id'      => $customer_id,
   251→   251→            'reference_type'   => 'payment',
   252→   252→            'reference_id'     => (int)$payment_id,
   253→   253→            'debit'            => $sides['debit'],
   254→   254→            'credit'           => $sides['credit'],
   255→   255→            'running_balance'  => $new_balance,
   256→   256→            'branch_id'        => (int)$branch_id,
   257→   257→            'transaction_date' => $post['transaction_date'] ?? date('Y-m-d'),
   258→   258→            'remarks'          => $post['narration'] ?? '',
   259→   259→            'created_by'       => (int)$user_id,
   260→   260→        ]);
   261→   261→    }

   296→   296→    public function reverseTransaction($id, $reason) {
   297→   297→        $this->db->beginTransaction();
   298→   298→        try {
   299→   299→            $reason = trim($reason);
   300→   300→            if (strlen($reason) < 3) {
   301→   301→                throw new Exception('Reversal reason is required (min 3 characters)');
   302→   302→            }
   303→   303→
   304→   304→            $user_id = (int)($_SESSION['user_id'] ?? 1);
   305→   305→
   306→   306→            $trans = $this->getTransactionById((int)$id);
   307→   307→            if (!$trans || !empty($trans['is_reversed'])) {
   308→   308→                throw new Exception('Transaction not found or already reversed');
   309→   309→            }
   310→   310→
   311→   311→            if (!$this->canUserReversePayment($trans)) {
   312→   312→                throw new Exception('You do not have access to reverse this payment');
   313→   313→            }
   314→   314→
   315→   315→            $amount = (float)$trans['amount'];
   316→   316→            $type   = $trans['transaction_type'] ?? 'receive';
   317→   317→
   318→   318→            if (!empty($trans['journal_entry_id'])) {
   319→   319→                require_once __DIR__ . '/../services/Accounting/JournalPostingService.php';
   320→   320→                $journalService = new JournalPostingService($this->db);
   321→   321→                $rev = $journalService->reverseLinkedJournal(
   322→   322→                    (int)$trans['journal_entry_id'],
   323→   323→                    'Payment reversal: ' . ($trans['payment_code'] ?? $id) . ' — ' . $reason
   324→   324→                );
   325→   325→                if (($rev['status'] ?? '') === 'error') {
   326→   326→                    throw new Exception('Failed to reverse payment journal: ' . ($rev['message'] ?? ''));
   327→   327→                }
   328→   328→            }
   329→   329→
   330→   330→            $bankId = !empty($trans['bank_id']) ? (int)$trans['bank_id'] : null;
   331→   331→            $payMode = strtolower(trim((string)($trans['payment_mode'] ?? 'cash')));
   332→   332→            if ($bankId && $payMode !== 'bank') {
   333→   333→                $payMode = 'bank';
   334→   334→            }
   335→   335→            $this->syncBankBookBalance($bankId, $amount, $type, $payMode, true);
   336→   336→
   337→   337→            $orig = $this->getCustomerLedgerSides($type, $amount);
   338→   338→            $debit = $orig['credit'];
   339→   339→            $credit = $orig['debit'];
   340→   340→
   341→   341→            $current = $this->getCustomerRunningBalance((int)$trans['customer_id']);
   342→   342→            $new_balance = $current + $debit - $credit;
   343→   343→            $revBranchId = (int)($trans['branch_id'] ?? ($_SESSION['branch_id'] ?? 1));
   344→   344→
   345→   345→            $this->insertCustomerLedgerEntry([
   346→   346→                'customer_id'      => (int)$trans['customer_id'],
   347→   347→                'reference_type'   => 'reversal',
   348→   348→                'reference_id'     => (int)$id,
   349→   349→                'debit'            => $debit,
   350→   350→                'credit'           => $credit,
   351→   351→                'running_balance'  => $new_balance,
   352→   352→                'branch_id'        => $revBranchId,
   353→   353→                'transaction_date' => date('Y-m-d'),
   354→   354→                'remarks'          => 'Reversal of #' . ($trans['payment_code'] ?? $id) . ' - ' . $reason,
   355→   355→                'created_by'       => (int)$user_id,
   356→   356→                'is_reversed'      => 0,
   357→   357→            ]);
   358→   358→
   359→   359→            $this->db->query("
   360→   360→                UPDATE customer_ledger
   361→   361→                SET is_reversed = 1
   362→   362→                WHERE reference_id = :pid
   363→   363→                  AND reference_type = 'payment'
   364→   364→                  AND COALESCE(is_reversed, 0) = 0
   365→   365→            ");
   366→   366→            $this->db->bind(':pid', $id);
   367→   367→            $this->db->execute();
   368→   368→
   369→   369→            if ($type === 'receive') {
   370→   370→                require_once __DIR__ . '/../services/Branch/BranchIntercompanyService.php';
   371→   371→                $settleRev = (new BranchIntercompanyService($this->db))->reverseCustomerPaymentSettlements($id);
   372→   372→                if (($settleRev['status'] ?? '') === 'error') {
   373→   373→                    throw new Exception($settleRev['message'] ?? 'Failed to reverse branch demand settlements');
   374→   374→                }
   375→   375→            }
   376→   376→
   377→   377→            $this->db->query('UPDATE customer_payments SET is_reversed = 1, reversed_at = NOW(),
   378→   378→                              reversed_by = :uid, reverse_reason = :reason WHERE id = :id');
   379→   379→            $this->db->bind(':uid', $user_id);
   380→   380→            $this->db->bind(':reason', $reason);
   381→   381→            $this->db->bind(':id', $id);
   382→   382→            $this->db->execute();
   383→   383→
   384→   384→            $this->db->commit();
   385→   385→
   386→   386→            $glNote = !empty($trans['journal_entry_id']) ? ' GL and bank (if any) reversed.' : '';
   387→   387→
   388→   388→            return [
   389→   389→                'status'  => 'success',
   390→   390→                'message' => 'Payment reversed. Customer ledger restored.' . $glNote,
   391→   391→            ];
   392→   392→
   393→   393→        } catch (Throwable $e) {
   394→   394→            if ($this->db->inTransaction()) {
   395→   395→                $this->db->rollback();
   396→   396→            }
   397→   397→            error_log('Customer payment reverse: ' . $e->getMessage());
   398→   398→
   399→   399→            return ['status' => 'error', 'message' => $e->getMessage()];
   400→   400→        }
   401→   401→    }

   455→   455→            SELECT COALESCE(SUM(amount), 0) AS total
   456→   456→            FROM customer_payments
   457→   457→            WHERE is_reversed = 0
   458→   458→              AND COALESCE(transaction_type, 'receive') = 'receive'
   459→   459→              AND payment_date >= to_char(CURRENT_DATE, 'YYYY-MM-01')::date
   460→   460→              {$branchSql}
   461→   461→        ");
   462→   462→        foreach ($bind as $k => $v) {
   463→   463→            $this->db->bind($k, $v);
   464→   464→        }
   465→   465→        $stats['received_month'] = (float)($this->db->single()['total'] ?? 0);
   466→   466→
   467→   467→        return $stats;
   468→   468→    }

   470→   470→    /**
   471→   471→     * Build WHERE clause fragments for payment list queries.
   472→   472→     *
   473→   473→     * @return array{0: array<int, string>, 1: array<string, mixed>}
   474→   474→     */
   475→   475→    private function buildPaymentListFilters(array $filters, ?int $branchId): array
   476→   476→    {
   477→   477→        $where = [];
   478→   478→        $bindings = [];
   479→   479→
   480→   480→        if ($branchId > 0) {
   481→   481→            $where[] = 'cp.branch_id = :branch_id';
   482→   482→            $bindings[':branch_id'] = $branchId;
   483→   483→        } elseif (!$this->canOverrideBranch() && self::sessionBranchId() > 0) {
   484→   484→            $where[] = 'cp.branch_id = :branch_id';
   485→   485→            $bindings[':branch_id'] = self::sessionBranchId();
   486→   486→        }
   487→   487→
   488→   488→        $dateFrom = !empty($filters['date_from']) ? (string)$filters['date_from'] : null;
   489→   489→        $dateTo = !empty($filters['date_to']) ? (string)$filters['date_to'] : null;
   490→   490→
   491→   491→        if ($dateFrom && $dateTo && $dateFrom === $dateTo) {
   492→   492→            $where[] = 'cp.payment_date = :pay_date';
   493→   493→            $bindings[':pay_date'] = $dateFrom;
   494→   494→        } else {
   495→   495→            if ($dateFrom) {
   496→   496→                $where[] = 'cp.payment_date >= :date_from';
   497→   497→                $bindings[':date_from'] = $dateFrom;
   498→   498→            }
   499→   499→            if ($dateTo) {
   500→   500→                $where[] = 'cp.payment_date <= :date_to';
   501→   501→                $bindings[':date_to'] = $dateTo;
   502→   502→            }
   503→   503→            if (!$dateFrom && !$dateTo) {
   504→   504→                $where[] = 'cp.payment_date = CURRENT_DATE';
   505→   505→            }
   506→   506→        }
   507→   507→
   508→   508→        if (!empty($filters['transaction_type']) && $filters['transaction_type'] !== 'all') {
   509→   509→            $where[] = 'cp.transaction_type = :ttype';
   510→   510→            $bindings[':ttype'] = $filters['transaction_type'];
   511→   511→        }
   512→   512→
   513→   513→        if (!empty($filters['status']) && $filters['status'] !== 'all') {
   514→   514→            if ($filters['status'] === 'reversed') {
   515→   515→                $where[] = 'cp.is_reversed = 1';
   516→   516→            } else {
   517→   517→                $where[] = 'COALESCE(cp.is_reversed, 0) = 0';
   518→   518→            }
   519→   519→        }
   520→   520→
   521→   521→        if (!empty($filters['payment_mode']) && $filters['payment_mode'] !== 'all') {
   522→   522→            $where[] = 'LOWER(cp.payment_mode) = :pmode';
   523→   523→            $bindings[':pmode'] = strtolower((string)$filters['payment_mode']);
   524→   524→        }
   525→   525→
   526→   526→        if (!empty($filters['customer_id'])) {
   527→   527→            $where[] = 'cp.customer_id = :cid';
   528→   528→            $bindings[':cid'] = (int)$filters['customer_id'];
   529→   529→        }
   530→   530→
   531→   531→        return [$where, $bindings];
   532→   532→    }

   549→   549→    public function getPaymentsForDataTable(array $params, array $filters, ?int $branchId): array
   550→   550→    {
   551→   551→        $start       = (int)($params['start'] ?? 0);
   552→   552→        $length      = (int)($params['length'] ?? 25);
   553→   553→        if ($length <= 0 || $length > 100) {
   554→   554→            $length = 25;
   555→   555→        }
   556→   556→        $searchValue = trim($params['search']['value'] ?? '');
   557→   557→        $reversedMode = $params['reversedMode'] ?? '';
   558→   558→        $orderColumn = (int)($params['order'][0]['column'] ?? 0);
   559→   559→        $orderDir    = strtolower($params['order'][0]['dir'] ?? 'desc') === 'asc' ? 'ASC' : 'DESC';
   560→   560→
   561→   561→        $columns = [
   562→   562→            'cp.payment_date',
   563→   563→            'cp.payment_code',
   564→   564→            'c.shop_name',
   565→   565→            'cp.transaction_type',
   566→   566→            'cp.amount',
   567→   567→            'cp.payment_mode',
   568→   568→            'e.name',
   569→   569→            'cp.is_reversed',
   570→   570→        ];
   571→   571→
   572→   572→        $baseQuery = $this->paymentListFromSql();
   573→   573→
   574→   574→        [$where, $bindParams] = $this->buildPaymentListFilters($filters, $branchId);
   575→   575→
   576→   576→        if ($reversedMode === 'only_reversed') {
   577→   577→            $where[] = 'cp.is_reversed = 1';
   578→   578→        }
   579→   579→
   580→   580→        $branchOnlyWhere = [];
   581→   581→        $branchOnlyBind = [];
   582→   582→        if ($branchId > 0) {
   583→   583→            $branchOnlyWhere[] = 'cp.branch_id = :branch_id';
   584→   584→            $branchOnlyBind[':branch_id'] = $branchId;
   585→   585→        } elseif (!$this->canOverrideBranch() && self::sessionBranchId() > 0) {
   586→   586→            $branchOnlyWhere[] = 'cp.branch_id = :branch_id';
   587→   587→            $branchOnlyBind[':branch_id'] = self::sessionBranchId();
   588→   588→        }
   589→   589→        $branchOnlySql = $branchOnlyWhere ? 'WHERE ' . implode(' AND ', $branchOnlyWhere) : '';
   590→   590→
   591→   591→        if ($searchValue !== '') {
   592→   592→            $where[] = '(cp.payment_code LIKE :search
   593→   593→                      OR c.shop_name LIKE :search
   594→   594→                      OR c.customer_name LIKE :search
   595→   595→                      OR c.mobile LIKE :search)';
   596→   596→            $bindParams[':search'] = '%' . $searchValue . '%';
   597→   597→        }
   598→   598→
   599→   599→        $whereSql = $where ? 'WHERE ' . implode(' AND ', $where) : '';
   600→   600→
   601→   601→        $totalQuery = "SELECT COUNT(cp.id) AS total {$baseQuery} {$branchOnlySql}";
   602→   602→        $this->db->query($totalQuery);
   603→   603→        foreach ($branchOnlyBind as $key => $val) {
   604→   604→            $this->db->bind($key, $val);
   605→   605→        }
   606→   606→        $recordsTotal = (int)($this->db->single()['total'] ?? 0);
   607→   607→
   608→   608→        $filteredQuery = "SELECT COUNT(cp.id) AS total {$baseQuery} {$whereSql}";
   609→   609→        $this->db->query($filteredQuery);
   610→   610→        foreach ($bindParams as $key => $val) {
   611→   611→            $this->db->bind($key, $val);
   612→   612→        }
   613→   613→        $recordsFiltered = (int)($this->db->single()['total'] ?? 0);
   614→   614→
   615→   615→        $orderBy = $columns[$orderColumn] ?? 'cp.payment_date';
   616→   616→        $dataQuery = "
   617→   617→            SELECT
   618→   618→                cp.id,
   619→   619→                cp.payment_date,
   620→   620→                cp.payment_code,
   621→   621→                cp.customer_id,
   622→   622→                cp.transaction_type,
   623→   623→                cp.amount,
   624→   624→                cp.payment_mode,
   625→   625→                cp.is_reversed,
   626→   626→                cp.branch_id,
   627→   627→                c.shop_name,
   628→   628→                c.mobile,
   629→   629→                e.name AS collected_by_name
   630→   630→            {$baseQuery}
   631→   631→            {$whereSql}
   632→   632→            ORDER BY {$orderBy} {$orderDir}, cp.id DESC
   633→   633→            LIMIT {$length} OFFSET {$start}
   634→   634→        ";
   635→   635→
   636→   636→        $this->db->query($dataQuery);
   637→   637→        foreach ($bindParams as $key => $val) {
   638→   638→            $this->db->bind($key, $val);
   639→   639→        }
   640→   640→        $data = $this->db->resultSet() ?: [];
   641→   641→
   642→   642→        foreach ($data as &$row) {

   644→   644→        }
   645→   645→        unset($row);
   646→   646→
   647→   647→        return [
   648→   648→            'draw'            => (int)($params['draw'] ?? 1),
   649→   649→            'recordsTotal'    => $recordsTotal,
   650→   650→            'recordsFiltered' => $recordsFiltered,
   651→   651→            'data'            => $data,
   652→   652→        ];
   653→   653→    }
