    44→    private function filterSalesAuditLogs(int $limit = 300): array
    45→    {
    46→        $all = $this->userAudit->getRecentLogs(500);
    47→        $prefixes = ['sale_', 'godown_', 'challan_', 'payment_'];
    48→        $filtered = [];
    49→        foreach ($all as $entry) {
    50→            $action = $entry['action'] ?? '';
    51→            foreach ($prefixes as $p) {
    52→                if (stripos($action, $p) === 0) {
    53→                    $filtered[] = $entry;
    54→                    break;
    55→                }
    56→            }
    57→            if (count($filtered) >= $limit) {
    58→                break;
    59→            }
    60→        }
    61→        return $filtered;
    62→    }

   302→                    'total_amount'    => $_POST['total_amount'] ?? 0,

   572→    public function save_payment() {
   573→        $this->requireRouteAccess();
   574→        try {
   575→            $this->validateCSRF();
   576→            $result = $this->paymentService->recordCustomerPayment($_POST);
   577→            if (($result['status'] ?? '') === 'success') {
   578→                $this->auditLog('payment_received', (int)($result['payment_id'] ?? 0), [
   579→                    'payment_code'     => $result['payment_code'] ?? null,
   580→                    'invoice_id'       => (int)($_POST['invoice_id'] ?? 0),
   581→                    'customer_id'      => (int)($_POST['customer_id'] ?? 0),
   582→                    'amount'           => (float)($_POST['receive_amount'] ?? 0),
   583→                    'payment_mode'     => $_POST['payment_mode'] ?? null,
   584→                    'journal_entry_id' => $result['journal_entry_id'] ?? null,
   585→                ]);
   586→
   587→                $paymentId = (int)($result['payment_id'] ?? 0);
   588→                $invoiceId = (int)($result['invoice_id'] ?? $_POST['invoice_id'] ?? 0);
   589→                SalesTelegramNotifier::safe(function () use ($paymentId, $invoiceId) {
   590→                    (new SalesTelegramNotifier())->notifyTodayInvoicePayment($paymentId, $invoiceId);
   591→                });
   592→
   593→                $result['success'] = true;
   594→            }
   595→            $this->sendJson($result);
   596→        } catch (Throwable $e) {
   597→            Logger::error('save_payment failed', ['message' => $e->getMessage()]);
   598→            $this->sendJson([
   599→                'status'  => 'error',
   600→                'message' => $this->safeClientMessage($e, 'Payment could not be saved.'),
   601→            ], 500);
   602→        }
   603→    }

   617→        if (($result['status'] ?? '') === 'success') {
   618→            $this->auditLog('payment_reversed', $payment_id, [
   619→                'payment_code' => $result['payment_code'] ?? null,
   620→                'amount'       => $result['amount'] ?? null,
   621→                'invoice_ids'  => $result['invoice_ids'] ?? [],
   622→                'reason'       => $reason,
   623→            ]);
   624→            $result['success'] = true;
   625→        }

   627→    }

   756→    public function export() {
   757→    $this->requireRouteAccess();
   758→    $filters = [
   759→        'date_from'       => $_GET['date_from'] ?? null,
   760→        'date_to'         => $_GET['date_to'] ?? null,
   761→        'search'          => trim($_GET['search'] ?? ''),
   762→        'challan_status'  => $_GET['challan_status'] ?? 'all',
   763→        'smart_sort'      => ($_GET['smart_sort'] ?? '1') !== '0',
   764→        'skip_default_today' => true,
   765→    ];
   766→
   767→    $invoices = $this->invoiceService->getFilteredTodayInvoices($filters);
   768→
   769→    if (empty($invoices)) {
   770→        $_SESSION['error'] = "No records found!";
   771→        $this->redirect('sales/today');
   772→        return;
   773→    }
   774→
   775→    header('Content-Type: text/csv; charset=utf-8');
   776→    header('Content-Disposition: attachment; filename="Today_Invoices_' . date('Y-m-d_H-i') . '.csv"');
   777→    header('Pragma: no-cache');
   778→
   779→    $output = fopen('php://output', 'w');
   780→    fprintf($output, chr(0xEF).chr(0xBB).chr(0xBF));
   781→
   782→    fputcsv($output, [
   783→        'Invoice Code', 'Date', 'Customer', 'Mobile', 'Branch', 
   784→        'Sales Person', 'Total Amount', 'Status', 'Challan Status'
   785→    ]);
   786→
   787→    foreach ($invoices as $inv) {
   788→        fputcsv($output, [
   789→            $inv['invoice_code'],
   790→            $inv['invoice_date'],
   791→            $inv['shop_name'] ?? $inv['customer_name'],
   792→            $inv['mobile'],
   793→            $inv['branch_name'],
   794→            $inv['salesman_name'],
   795→            $inv['total_amount'],
   796→            $inv['status'],
   797→            $inv['call_a_day'] ? 'Called' : 'Today'
   798→        ]);
   799→    }
   800→
   801→    fclose($output);
   802→    exit;
   803→}
