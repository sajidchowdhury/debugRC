    29→    /**
    30→     * Resolve each dispatch line: demand qty = invoice line qty (cannot be reduced).
    31→     * Validates warehouse, stock, and optional posted qty tampering.
    32→     *
    33→     * @param string $validationMode godown = pick warehouse (net available); finalize = trust godown + physical qty
    34→     * @return array<int, array{item_id:int, product_id:int, product_name:string, warehouse_id:int, demand_qty:float, dispatched_ctn:float}>
    35→     */
    36→    protected function resolveDispatchLinesForInvoice(
    37→        array $data,
    38→        int $invoiceId,
    39→        int $invoiceBranchId,
    40→        string $validationMode = 'godown'
    41→    ): array
    42→    {
    43→        $validationMode = $validationMode === 'finalize' ? 'finalize' : 'godown';
    44→        if (empty($data['item_id']) || !is_array($data['item_id'])) {
    45→            throw new Exception('No invoice lines to dispatch.');
    46→        }
    47→
    48→        $lines = [];
    49→        foreach ($data['item_id'] as $i => $rawItemId) {
    50→            $itemId = (int)$rawItemId;
    51→            if ($itemId <= 0) {
    52→                continue;
    53→            }
    54→
    55→            $this->db->query("
    56→                SELECT sii.id, sii.product_id, sii.qty, p.product_name
    57→                FROM sales_invoice_items sii
    58→                JOIN products p ON p.id = sii.product_id
    59→                WHERE sii.id = :iid AND sii.sales_invoice_id = :inv
    60→            ");
    61→            $this->db->bind(':iid', $itemId);
    62→            $this->db->bind(':inv', $invoiceId);
    63→            $row = $this->db->single();
    64→            if (!$row) {
    65→                throw new Exception(
    66→                    'Invoice lines have changed. Refresh this page and try again.'
    67→                );
    68→            }
    69→
    70→            $demandQty = (float)$row['qty'];
    71→            if ($demandQty <= 0) {
    72→                throw new Exception('Invalid invoice quantity on line #' . ($i + 1) . '.');
    73→            }
    74→
    75→            $postedQty = (float)($data['dispatched_qty'][$i] ?? $demandQty);
    76→            if (abs($postedQty - $demandQty) > 0.0001) {
    77→                throw new Exception(
    78→                    'Dispatched quantity must match invoice demand ('
    79→                    . number_format($demandQty, 2) . '). Partial dispatch is not allowed for '
    80→                    . ($row['product_name'] ?? 'product') . '.'
    81→                );
    82→            }
    83→
    84→            $wid = (int)($data['warehouse_id'][$i] ?? 0);
    85→            if ($wid <= 0) {
    86→                throw new Exception(
    87→                    'Select a warehouse for ' . ($row['product_name'] ?? 'each product') . '.'
    88→                );
    89→            }
    90→
    91→            $sessionBranchId = self::sessionBranchId();
    92→            if (!$this->warehouseBelongsToBranch($wid, $sessionBranchId)) {
    93→                throw new Exception('Warehouse does not belong to your branch.');
    94→            }
    95→
    96→            $productId = (int)$row['product_id'];
    97→            $productName = (string)($row['product_name'] ?? 'product');
    98→
    99→            if ($validationMode === 'finalize') {
   100→                $this->db->query("
   101→                    SELECT warehouse_id, ordered_qty
   102→                    FROM sales_invoice_dispatches
   103→                    WHERE sales_invoice_id = :inv AND product_id = :pid
   104→                    LIMIT 1
   105→                ");
   106→                $this->db->bind(':inv', $invoiceId);
   107→                $this->db->bind(':pid', $productId);
   108→                $godownRow = $this->db->single();
   109→                if (!$godownRow) {
   110→                    throw new Exception(
   111→                        'Godown setup missing for ' . $productName . '. Save godown copy again.'
   112→                    );
   113→                }
   114→
   115→                $godownWh = (int)$godownRow['warehouse_id'];
   116→                if ($godownWh <= 0) {
   117→                    throw new Exception('Godown warehouse not set for ' . $productName . '.');
   118→                }
   119→                if ($wid !== $godownWh) {
   120→                    throw new Exception(
   121→                        'Warehouse for ' . $productName . ' was fixed at godown and cannot be changed.'
   122→                    );
   123→                }
   124→                $wid = $godownWh;
   125→
   126→                $godownOrdered = (float)$godownRow['ordered_qty'];
   127→                if (abs($godownOrdered - $demandQty) > 0.0001) {
   128→                    throw new Exception(
   129→                        'Invoice quantity changed since godown was saved. Refresh and try again.'
   130→                    );
   131→                }
   132→
   133→                $physical = $this->getWarehousePhysicalStock($productId, $wid);
   134→                if ($demandQty > $physical + 0.0001) {
   135→                    throw new Exception(
   136→                        'Not enough physical stock for ' . $productName
   137→                        . ' in the selected warehouse (need '
   138→                        . number_format($demandQty, 2) . ', on hand '
   139→                        . number_format($physical, 2)
   140→                        . '). Receive stock or reverse and re-assign godown.'
   141→                    );
   142→                }
   143→            } else {
   144→                $available = $this->Get_Warehouse_Available_Stock($productId, $wid, $invoiceId);
   145→                if ($demandQty > $available + 0.0001) {
   146→                    throw new Exception(
   147→                        'Insufficient stock for ' . $productName
   148→                        . '. Invoice demand: ' . number_format($demandQty, 2)
   149→                        . ', available in warehouse: ' . number_format($available, 2)
   150→                        . '. Choose another warehouse or receive stock first.'
   151→                    );
   152→                }
   153→            }
   154→
   155→            $lines[] = [
   156→                'item_id'         => $itemId,
   157→                'product_id'      => (int)$row['product_id'],
   158→                'product_name'    => (string)$row['product_name'],
   159→                'warehouse_id'    => $wid,
   160→                'demand_qty'      => $demandQty,
   161→                'dispatched_ctn'  => (float)($data['dispatched_ctn'][$i] ?? 0),
   162→            ];
   163→        }
   164→
   165→        if ($lines === []) {
   166→            throw new Exception('No invoice lines to dispatch.');
   167→        }
   168→
   169→        return $lines;
   170→    }

   172→    protected function assertGodownPreparedForChallan(int $invoiceId): void
   173→    {
   174→        $this->db->query("
   175→            SELECT status FROM sales_invoices
   176→            WHERE id = :id AND is_reversed = 0
   177→        ");
   178→        $this->db->bind(':id', $invoiceId);
   179→        $row = $this->db->single();
   180→        if (!$row) {
   181→            throw new Exception('Invoice not found.');
   182→        }
   183→        if (($row['status'] ?? '') !== 'godown_issued') {
   184→            throw new Exception(
   185→                'Save godown copy first. Challan can only be created after godown setup is saved.'
   186→            );
   187→        }
   188→
   189→        $this->db->query("
   190→            SELECT COUNT(*) AS c FROM sales_invoice_items WHERE sales_invoice_id = :id
   191→        ");
   192→        $this->db->bind(':id', $invoiceId);
   193→        $itemCount = (int)($this->db->single()['c'] ?? 0);
   194→
   195→        $this->db->query("
   196→            SELECT COUNT(*) AS c FROM sales_invoice_dispatches WHERE sales_invoice_id = :id
   197→        ");
   198→        $this->db->bind(':id', $invoiceId);
   199→        $dispatchCount = (int)($this->db->single()['c'] ?? 0);
   200→
   201→        if ($itemCount <= 0 || $dispatchCount < $itemCount) {
   202→            throw new Exception(
   203→                'Godown dispatch records are missing. Save godown copy again before creating challan.'
   204→            );
   205→        }
   206→    }

   209→       // ================= PREPARE GODOWN (Step 1) =================
   210→    public function prepareGodown($data) {
   211→        $this->db->beginTransaction();
   212→        try {
   213→            $invoice_id = (int)$data['invoice_id'];
   214→
   215→            $this->db->query("
   216→                SELECT status, branch_id, customer_id, subtotal, discount, transport_cost, total_amount
   217→                FROM sales_invoices
   218→                WHERE id = :id AND is_reversed = 0
   219→                FOR UPDATE
   220→            ");
   221→            $this->db->bind(':id', $invoice_id);
   222→            $invoiceRow = $this->db->single();
   223→
   224→            if (!$invoiceRow) {
   225→                throw new Exception('Invoice not found.');
   226→            }
   227→
   228→            $this->assertInvoiceAccessible((int)$invoiceRow['branch_id']);
   229→            $invoiceBranchId = (int)$invoiceRow['branch_id'];
   230→
   231→            $status = $invoiceRow['status'] ?? '';
   232→            if (!in_array($status, ['draft', 'godown_issued'], true)) {
   233→                throw new Exception('Godown can only be prepared for draft or godown-issued invoices.');
   234→            }
   235→
   236→            $transportResult = $this->persistInvoiceTransportCost(
   237→                $data,
   238→                $invoiceRow,
   239→                $invoice_id,
   240→                'Transport/total updated at godown save'
   241→            );
   242→
   243→            $dispatchMode = ($status === 'godown_issued') ? 'finalize' : 'godown';
   244→            $dispatchLines = $this->resolveDispatchLinesForInvoice(
   245→                $data,
   246→                $invoice_id,
   247→                $invoiceBranchId,
   248→                $dispatchMode
   249→            );
   250→
   251→            foreach ($dispatchLines as $line) {
   252→                $this->db->query("UPDATE sales_invoice_items SET warehouse_id = :wid WHERE id = :id");
   253→                $this->db->bind(':wid', $line['warehouse_id']);
   254→                $this->db->bind(':id', $line['item_id']);
   255→                $this->db->execute();
   256→            }
   257→
   258→            // 2. Save dispatchers (junction table) - Clear & Insert
   259→            $this->db->query("DELETE FROM sales_invoice_dispatchers WHERE sales_invoice_id = :iid");
   260→            $this->db->bind(':iid', $invoice_id);
   261→            $this->db->execute();
   262→
   263→            if (!empty($data['dispatcher_id']) && is_array($data['dispatcher_id'])) {
   264→                foreach ($data['dispatcher_id'] as $emp_id) {
   265→                    if (!empty($emp_id)) {
   266→                        $this->db->query("INSERT INTO sales_invoice_dispatchers (sales_invoice_id, employee_id) 
   267→                                        VALUES (:iid, :eid)");
   268→                        $this->db->bind(':iid', $invoice_id);
   269→                        $this->db->bind(':eid', (int)$emp_id);
   270→                        $this->db->execute();
   271→                    }
   272→                }
   273→            }
   274→
   275→            // 3. Clear previous dispatches and INSERT fresh (This solves your duplicate problem)
   276→            $this->db->query("DELETE FROM sales_invoice_dispatches WHERE sales_invoice_id = :iid");
   277→            $this->db->bind(':iid', $invoice_id);
   278→            $this->db->execute();
   279→
   280→            foreach ($dispatchLines as $line) {
   281→                $this->db->query("
   282→                    INSERT INTO sales_invoice_dispatches
   283→                    (sales_invoice_id, product_id, warehouse_id, ordered_qty, dispatched_qty, dispatched_ctn)
   284→                    VALUES (:iid, :pid, :wid, :oqty, 0, :ctn)
   285→                ");
   286→                $this->db->bind(':iid', $invoice_id);
   287→                $this->db->bind(':pid', $line['product_id']);
   288→                $this->db->bind(':wid', $line['warehouse_id']);
   289→                $this->db->bind(':oqty', $line['demand_qty']);
   290→                $this->db->bind(':ctn', $line['dispatched_ctn']);
   291→                $this->db->execute();
   292→            }
   293→
   294→            // 4. Update invoice status
   295→            $this->db->query("
   296→                UPDATE sales_invoices
   297→                SET status = 'godown_issued',
   298→                    godown_issued_at = NOW()
   299→                WHERE id = :id
   300→            ");
   301→            $this->db->bind(':id', $invoice_id);
   302→            $this->db->execute();
   303→
   304→            $this->db->commit();
   305→
   306→            $message = 'Godown prepared successfully!';
   307→            if (abs($transportResult['total_diff']) > 0.0001) {
   308→                $message .= ' Invoice total updated to Tk '
   309→                    . number_format($transportResult['new_total'], 2) . '.';
   310→            }
   311→
   312→            return [
   313→                'status'       => 'success',
   314→                'message'      => $message,
   315→                'new_total'    => $transportResult['new_total'],
   316→                'transport'    => $transportResult['transport'],
   317→                'total_diff'   => $transportResult['total_diff'],
   318→            ];
   319→
   320→        } catch (Exception $e) {
   321→            $this->db->rollback();
   322→            return ['status' => 'error', 'message' => $e->getMessage()];
   323→        }
   324→    }

   327→    public function finalizeChallan($data) {
   328→        $this->db->beginTransaction();
   329→        try {
   330→            $invoice_id = (int)$data['invoice_id'];
   331→
   332→            // Double-finalization protection + branch scope
   333→            $this->db->query("
   334→                SELECT status, branch_id, customer_id, subtotal, discount, transport_cost, total_amount, invoice_code
   335→                FROM sales_invoices WHERE id = :id AND is_reversed = 0 FOR UPDATE
   336→            ");
   337→            $this->db->bind(':id', $invoice_id);
   338→            $invoiceRow = $this->db->single();
   339→            if (!$invoiceRow) {
   340→                throw new Exception('Invoice not found.');
   341→            }
   342→            $this->assertInvoiceAccessible((int)$invoiceRow['branch_id']);
   343→            $invoiceBranchId = (int)$invoiceRow['branch_id'];
   344→            $customerId = (int)$invoiceRow['customer_id'];
   345→            if (($invoiceRow['status'] ?? '') === 'challan_completed') {
   346→                throw new Exception("Challan already completed.");
   347→            }
   348→
   349→            $this->assertGodownPreparedForChallan($invoice_id);
   350→            $dispatchLines = $this->resolveDispatchLinesForInvoice(
   351→                $data,
   352→                $invoice_id,
   353→                $invoiceBranchId,
   354→                'finalize'
   355→            );
   356→
   357→            $challan_code = $this->generateSalesChallanCode();
   358→            $oldTransport = (float)$invoiceRow['transport_cost'];
   359→            $oldTotal = (float)$invoiceRow['total_amount'];
   360→
   361→            $transportResult = $this->persistInvoiceTransportCost(
   362→                $data,
   363→                $invoiceRow,
   364→                $invoice_id,
   365→                'Transport/total adjustment on challan #' . $challan_code
   366→            );
   367→            $newTotal = $transportResult['new_total'];
   368→            $totalDiff = $transportResult['total_diff'];
   369→
   370→            // Update Invoice status (transport/total already persisted above)
   371→            if ($this->invoiceHasPreChallanColumns()) {
   372→                $this->db->query("
   373→                    UPDATE sales_invoices
   374→                    SET status = 'challan_completed',
   375→                        challan_completed_at = NOW(),
   376→                        godown_issued_at = COALESCE(godown_issued_at, NOW()),
   377→                        pre_challan_transport = :pre_tc,
   378→                        pre_challan_total = :pre_total
   379→                    WHERE id = :id
   380→                ");
   381→                $this->db->bind(':pre_tc', $oldTransport);
   382→                $this->db->bind(':pre_total', $oldTotal);
   383→            } else {
   384→                $this->db->query("
   385→                    UPDATE sales_invoices
   386→                    SET status = 'challan_completed',
   387→                        challan_completed_at = NOW(),
   388→                        godown_issued_at = COALESCE(godown_issued_at, NOW())
   389→                    WHERE id = :id
   390→                ");
   391→            }
   392→            $this->db->bind(':id', $invoice_id);
   393→            $this->db->execute();
   394→
   395→            // Create Challan
   396→            $this->db->query("INSERT INTO sales_challans (sales_invoice_id, challan_code, challan_date, created_by) VALUES (:iid, :code, CURRENT_DATE, :user)");
   397→            $this->db->bind(':iid', $invoice_id);
   398→            $this->db->bind(':code', $challan_code);
   399→            $this->db->bind(':user', $_SESSION['user_id'] ?? null);
   400→            $this->db->execute();
   401→            $challan_id = $this->db->lastInsertId();
   402→            $cogsTotal = 0.0;
   403→
   404→            foreach ($dispatchLines as $line) {
   405→                $qty = $line['demand_qty'];
   406→                $pid = $line['product_id'];
   407→                $wid = $line['warehouse_id'];
   408→
   409→                $this->db->query("SELECT avg_cost FROM warehouse_stock WHERE product_id = :pid AND warehouse_id = :wid");
   410→                $this->db->bind(':pid', $pid);
   411→                $this->db->bind(':wid', $wid);
   412→                $avg_cost = round((float)($this->db->single()['avg_cost'] ?? 0), 4);
   413→                if ($avg_cost <= 0) {
   414→                    throw new Exception(
   415→                        'Cannot complete challan: zero cost for product #' . $pid . ' in warehouse #' . $wid . '. Receive stock or set cost first.'
   416→                    );
   417→                }
   418→
   419→                $this->db->query("
   420→                    UPDATE sales_invoice_dispatches
   421→                    SET dispatched_qty = :dqty, dispatched_ctn = :ctn, dispatched_at = NOW(),
   422→                        warehouse_id = :wid
   423→                    WHERE sales_invoice_id = :iid AND product_id = :pid
   424→                ");
   425→                $this->db->bind(':dqty', $qty);
   426→                $this->db->bind(':ctn', $line['dispatched_ctn']);
   427→                $this->db->bind(':wid', $wid);
   428→                $this->db->bind(':iid', $invoice_id);
   429→                $this->db->bind(':pid', $pid);
   430→                $this->db->execute();
   431→                if ($this->db->rowCount() === 0) {
   432→                    throw new Exception(
   433→                        'Could not update dispatch row for product #' . $pid . '. Save godown copy again before challan.'
   434→                    );
   435→                }
   436→
   437→                $this->db->query("
   438→                    UPDATE sales_invoice_items
   439→                    SET warehouse_id = :wid
   440→                    WHERE id = :item_id AND sales_invoice_id = :iid
   441→                ");
   442→                $this->db->bind(':wid', $wid);
   443→                $this->db->bind(':item_id', $line['item_id']);
   444→                $this->db->bind(':iid', $invoice_id);
   445→                $this->db->execute();
   446→
   447→                $this->stock->updateWarehouseStock($wid, $pid, -$qty, $avg_cost);
   448→
   449→                $this->stock->logMovement([
   450→                    'product_id'     => $pid,
   451→                    'warehouse_id'   => $wid,
   452→                    'qty'            => -$qty,
   453→                    'rate'           => $avg_cost,
   454→                    'reference_type' => 'sales_challan',
   455→                    'reference_id'   => $challan_id,
   456→                    'remarks'        => 'Sales Challan #' . $challan_code,
   457→                ]);
   458→
   459→                $this->saveChallanIssueLine($challan_id, $pid, $wid, $qty, $avg_cost);
   460→
   461→                $cogsTotal += $qty * $avg_cost;
   462→            }
   463→
   464→            $journalService = new JournalPostingService($this->db);
   465→            $adjustmentJournalId = null;
   466→
   467→            if (abs($totalDiff) > 0.0001) {
   468→                $adjResult = $journalService->postSalesInvoiceTotalAdjustment($invoice_id, $totalDiff, [
   469→                    'invoice_code' => $invoiceRow['invoice_code'] ?? null,
   470→                    'customer_id'  => $customerId,
   471→                    'branch_id'    => $invoiceBranchId,
   472→                    'entry_date'   => date('Y-m-d'),
   473→                    'challan_code' => $challan_code,
   474→                    'challan_id'   => (int)$challan_id,
   475→                ]);
   476→                if (($adjResult['status'] ?? '') === 'error') {
   477→                    throw new Exception('Transport/total GL adjustment failed: ' . ($adjResult['message'] ?? ''));
   478→                }
   479→                $adjustmentJournalId = !empty($adjResult['journal_entry_id'])
   480→                    ? (int)$adjResult['journal_entry_id']
   481→                    : null;
   482→            }
   483→
   484→            $this->updateChallanTransportMeta($challan_id, $totalDiff, $adjustmentJournalId);
   485→
   486→            $cogsResult = $journalService->postSalesChallanCOGS($challan_id, [
   487→                'challan_code'  => $challan_code,
   488→                'challan_date'  => date('Y-m-d'),
   489→                'cogs_amount'   => round($cogsTotal, 2),
   490→                'branch_id'     => $invoiceBranchId,
   491→            ]);
   492→            if (($cogsResult['status'] ?? '') === 'error') {
   493→                throw new Exception('COGS journal posting failed: ' . ($cogsResult['message'] ?? ''));
   494→            }
   495→            $cogsPosted = round($cogsTotal, 2);
   496→            if ($cogsPosted > 0.0001 && empty($cogsResult['journal_entry_id'])) {
   497→                throw new Exception('COGS journal was not created although stock was issued (amount ' . $cogsPosted . ').');
   498→            }
   499→            if (!empty($cogsResult['journal_entry_id'])) {
   500→                $this->setChallanJournalEntryId($challan_id, (int)$cogsResult['journal_entry_id']);
   501→            }
   502→
   503→            $this->db->commit();
   504→            return [
   505→                'status' => 'success',
   506→                'message' => 'Challan completed! No: ' . $challan_code,
   507→                'challan_code' => $challan_code,
   508→                'challan_id' => $challan_id,
   509→                'invoice_id' => $invoice_id,
   510→                'transport_adjustment' => $totalDiff,
   511→                'new_total' => $newTotal,
   512→                'journal_entry_id' => $cogsResult['journal_entry_id'] ?? null,
   513→                'cogs_amount' => round($cogsTotal, 2),
   514→            ];
   515→
   516→        } catch (Exception $e) {
   517→            $this->db->rollback();
   518→            return ['status' => 'error', 'message' => $e->getMessage()];
   519→        }
   520→    }

   522→    /**
   523→     * Phase 4: Reverse a completed sales challan — restore stock, rollback invoice status.
   524→     * Phase 5: reversing journal when journal_entry_id is set on challan.
   525→     */
   526→    public function reverseChallan(int $invoiceId, string $reason): array
   527→    {
   528→        $reason = trim($reason);
   529→        if ($invoiceId <= 0 || strlen($reason) < 5) {
   530→            return ['status' => 'error', 'message' => 'Invoice ID and reversal reason (min 5 chars) are required.'];
   531→        }
   532→
   533→        $this->db->beginTransaction();
   534→        try {
   535→            $userId = (int)($_SESSION['user_id'] ?? 1);
   536→
   537→            $this->db->query("
   538→                SELECT si.* FROM sales_invoices si
   539→                WHERE si.id = :id AND si.is_reversed = 0 FOR UPDATE
   540→            ");
   541→            $this->db->bind(':id', $invoiceId);
   542→            $invoice = $this->db->single();
   543→
   544→            if (!$invoice) {
   545→                throw new Exception('Invoice not found.');
   546→            }
   547→            $this->assertInvoiceAccessible((int)$invoice['branch_id']);
   548→
   549→            if (($invoice['status'] ?? '') !== 'challan_completed') {
   550→                throw new Exception('Only completed challans can be reversed.');
   551→            }
   552→
   553→            $this->db->query("
   554→                SELECT COUNT(*) AS c FROM sales_returns
   555→                WHERE sales_invoice_id = :iid AND status = 'completed' AND COALESCE(is_reversed, 0) = 0
   556→            ");
   557→            $this->db->bind(':iid', $invoiceId);
   558→            if ((int)($this->db->single()['c'] ?? 0) > 0) {
   559→                throw new Exception('Cannot reverse challan: completed sales returns exist for this invoice.');
   560→            }
   561→
   562→            $this->db->query("
   563→                SELECT * FROM sales_challans
   564→                WHERE sales_invoice_id = :iid AND COALESCE(is_reversed, 0) = 0
   565→                ORDER BY id DESC LIMIT 1
   566→            ");
   567→            $this->db->bind(':iid', $invoiceId);
   568→            $challan = $this->db->single();
   569→            if (!$challan) {
   570→                throw new Exception('Active challan record not found.');
   571→            }
   572→
   573→            $challanId = (int)$challan['id'];
   574→            $challanCode = $challan['challan_code'];
   575→            $transportAdjustment = (float)($challan['transport_adjustment'] ?? 0);
   576→            if (abs($transportAdjustment) < 0.0001) {
   577→                $transportAdjustment = round(
   578→                    (float)$invoice['total_amount'] - (float)($invoice['pre_challan_total'] ?? 0),
   579→                    2
   580→                );
   581→            }
   582→
   583→            $ledgerAdjustment = $this->sumChallanInvoiceAdjustments(
   584→                (int)$invoice['customer_id'],
   585→                $invoiceId,
   586→                $challanCode
   587→            );
   588→            if (abs($ledgerAdjustment) > 0.0001) {
   589→                $transportAdjustment = $ledgerAdjustment;
   590→            }
   591→
   592→            $itemsReversed = $this->restoreStockFromChallanIssue(
   593→                $challanId,
   594→                $challanCode,
   595→                $invoiceId,
   596→                (int)$invoice['branch_id'],
   597→                $reason,
   598→                (int)($challan['journal_entry_id'] ?? 0)
   599→            );
   600→
   601→            $journalService = new JournalPostingService($this->db);
   602→
   603→            if (abs($transportAdjustment) > 0.0001) {
   604→                $this->reverseChallanTransportLedger(
   605→                    (int)$invoice['customer_id'],
   606→                    $invoiceId,
   607→                    $transportAdjustment,
   608→                    (int)$invoice['branch_id'],
   609→                    $challanCode,
   610→                    $reason
   611→                );
   612→
   613→                $adjJournalId = (int)($challan['adjustment_journal_entry_id'] ?? 0);
   614→                if ($adjJournalId <= 0) {
   615→                    $adjJournalId = (int)($this->findChallanAdjustmentJournalId(
   616→                        $invoiceId,
   617→                        $challanCode,
   618→                        $journalService
   619→                    ) ?? 0);
   620→                }
   621→                if ($adjJournalId > 0) {
   622→                    $adjRev = $journalService->reverseLinkedJournal(
   623→                        $adjJournalId,
   624→                        'Challan reversal transport adjustment: ' . $reason
   625→                    );
   626→                    if (($adjRev['status'] ?? '') === 'error') {
   627→                        throw new Exception(
   628→                            'Failed to reverse transport adjustment journal: ' . ($adjRev['message'] ?? '')
   629→                        );
   630→                    }
   631→                }
   632→            }
   633→
   634→            $restoreTransport = $invoice['pre_challan_transport'] ?? null;
   635→            $restoreTotal = $invoice['pre_challan_total'] ?? null;
   636→            if ($restoreTransport === null || $restoreTotal === null) {
   637→                $subtotal = (float)($invoice['subtotal'] ?? 0);
   638→                $discount = (float)($invoice['discount'] ?? 0);
   639→                $restoreTransport = round((float)($invoice['transport_cost'] ?? 0) - $transportAdjustment, 2);
   640→                $restoreTotal = round($subtotal + (float)$restoreTransport - $discount, 2);
   641→            } else {
   642→                $restoreTransport = (float)$restoreTransport;
   643→                $restoreTotal = (float)$restoreTotal;
   644→            }
   645→
   646→            $this->db->query("
   647→                UPDATE sales_challans
   648→                SET is_reversed = 1,
   649→                    reversed_at = NOW(),
   650→                    reversed_by = :uid,
   651→                    reverse_reason = :reason
   652→                WHERE id = :id
   653→            ");
   654→            $this->db->bind(':uid', $userId);
   655→            $this->db->bind(':reason', $reason);
   656→            $this->db->bind(':id', $challanId);
   657→            $this->db->execute();
   658→
   659→            if ($this->invoiceHasPreChallanColumns()) {
   660→                $this->db->query("
   661→                    UPDATE sales_invoices
   662→                    SET status = 'godown_issued',
   663→                        challan_completed_at = NULL,
   664→                        transport_cost = :tc,
   665→                        total_amount = :total,
   666→                        pre_challan_transport = NULL,
   667→                        pre_challan_total = NULL
   668→                    WHERE id = :id
   669→                ");
   670→            } else {
   671→                $this->db->query("
   672→                    UPDATE sales_invoices
   673→                    SET status = 'godown_issued',
   674→                        challan_completed_at = NULL,
   675→                        transport_cost = :tc,
   676→                        total_amount = :total
   677→                    WHERE id = :id
   678→                ");
   679→            }
   680→            $this->db->bind(':tc', (float)$restoreTransport);
   681→            $this->db->bind(':total', (float)$restoreTotal);
   682→            $this->db->bind(':id', $invoiceId);
   683→            $this->db->execute();
   684→
   685→            $journalEntryId = null;
   686→            if (!empty($challan['journal_entry_id'])) {
   687→                $rev = $journalService->reverseLinkedJournal((int)$challan['journal_entry_id'], $reason);
   688→                if (!empty($rev['journal_entry_id'])) {
   689→                    $journalEntryId = (int)$rev['journal_entry_id'];
   690→                }
   691→            }
   692→
   693→            $this->db->commit();
   694→
   695→            return [
   696→                'status'                 => 'success',
   697→                'message'                => 'Challan reversed. Stock, transport/AR, and COGS rolled back; invoice returned to godown issued.',
   698→                'invoice_id'             => $invoiceId,
   699→                'invoice_code'           => $invoice['invoice_code'],
   700→                'challan_id'             => $challanId,
   701→                'challan_code'           => $challanCode,
   702→                'items_reversed'         => $itemsReversed,
   703→                'journal_entry_id'       => $journalEntryId,
   704→                'transport_restored'   => (float)$restoreTransport,
   705→                'total_restored'       => (float)$restoreTotal,
   706→                'transport_adjustment' => $transportAdjustment,
   707→            ];
   708→        } catch (Exception $e) {
   709→            $this->db->rollback();
   710→            return ['status' => 'error', 'message' => $e->getMessage()];
   711→        }
   712→    }

   714→    /**
   715→     * Restore stock at original challan issue rates (sales_challan_items or stock_transactions).
   716→     */
   717→    protected function restoreStockFromChallanIssue(
   718→        int $challanId,
   719→        string $challanCode,
   720→        int $invoiceId,
   721→        int $branchId,
   722→        string $reason,
   723→        int $cogsJournalId = 0
   724→    ): int {
   725→        $issueLines = $this->loadChallanIssueLinesForRestore($challanId, $invoiceId, $cogsJournalId);
   726→        $itemsReversed = 0;
   727→
   728→        foreach ($issueLines as $line) {
   729→            $pid = (int)($line['product_id'] ?? 0);
   730→            $wid = (int)($line['warehouse_id'] ?? 0);
   731→            $qty = (float)($line['qty'] ?? 0);
   732→            if ($qty <= 0 || $wid <= 0 || $pid <= 0) {
   733→                continue;
   734→            }
   735→
   736→            if (!$this->warehouseBelongsToBranch($wid, $branchId)) {
   737→                throw new Exception('Warehouse branch mismatch during challan reversal.');
   738→            }
   739→
   740→            $issueRate = $this->resolveChallanIssueRate(
   741→                $challanId,
   742→                $invoiceId,
   743→                $pid,
   744→                $wid,
   745→                $qty,
   746→                $cogsJournalId,
   747→                isset($line['issue_rate']) ? (float)$line['issue_rate'] : null
   748→            );
   749→
   750→            $this->stock->updateWarehouseStock($wid, $pid, $qty, $issueRate);
   751→            $this->stock->logMovement([
   752→                'product_id'     => $pid,
   753→                'warehouse_id'   => $wid,
   754→                'qty'            => $qty,
   755→                'rate'           => $issueRate,
   756→                'reference_type' => 'sales_challan_reversal',
   757→                'reference_id'   => $challanId,
   758→                'remarks'        => "Reversal of Challan #{$challanCode}: {$reason}",
   759→            ]);
   760→            $itemsReversed++;
   761→
   762→            $this->db->query("
   763→                UPDATE sales_invoice_dispatches
   764→                SET dispatched_qty = 0, dispatched_ctn = 0, dispatched_at = NULL
   765→                WHERE sales_invoice_id = :iid AND product_id = :pid
   766→            ");
   767→            $this->db->bind(':iid', $invoiceId);
   768→            $this->db->bind(':pid', $pid);
   769→            $this->db->execute();
   770→        }
   771→
   772→        return $itemsReversed;
   773→    }

   775→    /**
   776→     * @return list<array{product_id:int, warehouse_id:int, qty:float, issue_rate?:float}>
   777→     */
   778→    protected function loadChallanIssueLinesForRestore(int $challanId, int $invoiceId, int $cogsJournalId): array
   779→    {
   780→        if ($this->hasChallanIssueItemsTable()) {
   781→            $items = $this->fetchChallanIssueItems($challanId);
   782→            if ($items !== []) {
   783→                return $items;
   784→            }
   785→        }
   786→
   787→        $fromTx = $this->fetchIssueLinesFromStockTransactions($challanId);
   788→        if ($fromTx !== []) {
   789→            return $fromTx;
   790→        }
   791→
   792→        return $this->fetchLegacyDispatchIssueLines($challanId, $invoiceId, $cogsJournalId);
   793→    }

   848→    /**
   849→     * @return list<array{product_id:int, warehouse_id:int, qty:float}>
   850→     */
   851→    protected function fetchLegacyDispatchIssueLines(int $challanId, int $invoiceId, int $cogsJournalId): array
   852→    {
   853→        $this->db->query("
   854→            SELECT product_id, warehouse_id, dispatched_qty AS qty
   855→            FROM sales_invoice_dispatches
   856→            WHERE sales_invoice_id = :iid AND COALESCE(dispatched_qty, 0) > 0
   857→        ");
   858→        $this->db->bind(':iid', $invoiceId);
   859→        $rows = $this->db->resultSet() ?: [];
   860→        $lines = [];
   861→
   862→        foreach ($rows as $row) {
   863→            $qty = (float)($row['qty'] ?? 0);
   864→            if ($qty <= 0) {
   865→                continue;
   866→            }
   867→            $lines[] = [
   868→                'product_id'   => (int)$row['product_id'],
   869→                'warehouse_id' => (int)$row['warehouse_id'],
   870→                'qty'          => $qty,
   871→            ];
   872→        }
   873→
   874→        if ($lines === [] && $cogsJournalId > 0) {
   875→            throw new Exception(
   876→                'Cannot reverse challan: no issue lines found. Restore stock_transactions or run migration 040_sales_challan_issue_cost.sql.'
   877→            );
   878→        }
   879→
   880→        return $lines;
   881→    }

   883→    protected function resolveChallanIssueRate(
   884→        int $challanId,
   885→        int $invoiceId,
   886→        int $productId,
   887→        int $warehouseId,
   888→        float $qty,
   889→        int $cogsJournalId,
   890→        ?float $knownRate = null
   891→    ): float {
   892→        if ($knownRate !== null && $knownRate > 0) {
   893→            return round($knownRate, 4);
   894→        }
   895→
   896→        if ($this->hasChallanIssueItemsTable()) {
   897→            $this->db->query("
   898→                SELECT issue_rate FROM sales_challan_items
   899→                WHERE sales_challan_id = :cid
   900→                  AND product_id = :pid
   901→                  AND warehouse_id = :wid
   902→                ORDER BY id DESC
   903→                LIMIT 1
   904→            ");
   905→            $this->db->bind(':cid', $challanId);
   906→            $this->db->bind(':pid', $productId);
   907→            $this->db->bind(':wid', $warehouseId);
   908→            $stored = (float)($this->db->single()['issue_rate'] ?? 0);
   909→            if ($stored > 0) {
   910→                return round($stored, 4);
   911→            }
   912→        }
   913→
   914→        $this->db->query("
   915→            SELECT rate FROM stock_transactions
   916→            WHERE reference_type = 'sales_challan'
   917→              AND reference_id = :cid
   918→              AND product_id = :pid
   919→              AND warehouse_id = :wid
   920→              AND qty < -0.0001
   921→            ORDER BY id DESC
   922→            LIMIT 1
   923→        ");
   924→        $this->db->bind(':cid', $challanId);
   925→        $this->db->bind(':pid', $productId);
   926→        $this->db->bind(':wid', $warehouseId);
   927→        $txRate = (float)($this->db->single()['rate'] ?? 0);
   928→        if ($txRate > 0) {
   929→            return round($txRate, 4);
   930→        }
   931→
   932→        if ($cogsJournalId > 0) {
   933→            $lineCount = count($this->fetchChallanIssueItems($challanId))
   934→                + count($this->fetchIssueLinesFromStockTransactions($challanId));
   935→            if ($lineCount <= 1) {
   936→                $cogsAmount = $this->sumJournalCogsDebit($cogsJournalId);
   937→                if ($cogsAmount > 0 && $qty > 0) {
   938→                    return round($cogsAmount / $qty, 4);
   939→                }
   940→            }
   941→        }
   942→
   943→        throw new Exception(
   944→            'Cannot resolve original issue cost for product #' . $productId
   945→            . ' (challan #' . $challanId . '). Re-post from stock history or run migration 040.'
   946→        );
   947→    }

   949→    protected function sumJournalCogsDebit(int $journalEntryId): float
   950→    {
   951→        $this->db->query("
   952→            SELECT COALESCE(SUM(jel.debit - jel.credit), 0) AS cogs
   953→            FROM journal_entry_lines jel
   954→            INNER JOIN ledgers l ON l.id = jel.ledger_id
   955→            WHERE jel.journal_entry_id = :jid
   956→              AND l.ledger_nature = 'cogs'
   957→        ");
   958→        $this->db->bind(':jid', $journalEntryId);
   959→        $row = $this->db->single();
   960→
   961→        return max(0.0, round((float)($row['cogs'] ?? 0), 2));
   962→    }

   964→    protected function saveChallanIssueLine(
   965→        int $challanId,
   966→        int $productId,
   967→        int $warehouseId,

   984→        $this->db->bind(':pid', $productId);
   985→        $this->db->bind(':wid', $warehouseId);
   986→        $this->db->bind(':qty', $qty);
   987→        $this->db->bind(':rate', $issueRate);
   988→        $this->db->bind(':cogs', $cogsAmount);

   990→    }

  1009→    /**
  1010→     * Undo customer_ledger transport/total adjustment posted at challan finalize only.
  1011→     * Godown transport adjustments remain active (invoice stays at godown totals).
  1012→     */
  1013→    protected function reverseChallanTransportLedger(
  1014→        int $customerId,
  1015→        int $invoiceId,
  1016→        float $adjustmentDelta,
  1017→        int $branchId,
  1018→        string $challanCode,
  1019→        string $reason
  1020→    ): void {
  1021→        if (abs($adjustmentDelta) < 0.0001) {
  1022→            return;
  1023→        }
  1024→
  1025→        $prevBalance = $this->getCustomerRunningBalance($customerId);
  1026→        $newBalance = $prevBalance - $adjustmentDelta;
  1027→
  1028→        $this->insertCustomerLedgerEntry([
  1029→            'customer_id'      => $customerId,
  1030→            'reference_type'   => 'reversal',
  1031→            'reference_id'     => $invoiceId,
  1032→            'debit'            => $adjustmentDelta < 0 ? abs($adjustmentDelta) : 0,
  1033→            'credit'           => $adjustmentDelta > 0 ? $adjustmentDelta : 0,
  1034→            'running_balance'  => $newBalance,
  1035→            'branch_id'        => $branchId,
  1036→            'remarks'          => "Reversal of transport/total on challan #{$challanCode} — {$reason}",
  1037→            'is_reversed'      => 1,
  1038→        ]);
  1039→
  1040→        $needle = '%adjustment on challan #' . $challanCode . '%';
  1041→        $this->db->query("
  1042→            UPDATE customer_ledger
  1043→            SET is_reversed = 1
  1044→            WHERE customer_id = :cid
  1045→              AND reference_type = 'invoice_adjustment'
  1046→              AND reference_id = :iid
  1047→              AND COALESCE(is_reversed, 0) = 0
  1048→              AND remarks LIKE :needle
  1049→        ");
  1050→        $this->db->bind(':cid', $customerId);
  1051→        $this->db->bind(':iid', $invoiceId);
  1052→        $this->db->bind(':needle', $needle);
  1053→        $this->db->execute();
  1054→    }

  1056→    /**
  1057→     * Net AR delta from challan-time invoice_adjustment rows (excludes godown saves).
  1058→     */
  1059→    protected function sumChallanInvoiceAdjustments(int $customerId, int $invoiceId, string $challanCode): float
  1060→    {
  1061→        $needle = '%adjustment on challan #' . $challanCode . '%';
  1062→        $this->db->query("
  1063→            SELECT COALESCE(SUM(debit - credit), 0) AS delta
  1064→            FROM customer_ledger
  1065→            WHERE customer_id = :cid
  1066→              AND reference_type = 'invoice_adjustment'
  1067→              AND reference_id = :iid
  1068→              AND COALESCE(is_reversed, 0) = 0
  1069→              AND remarks LIKE :needle
  1070→        ");
  1071→        $this->db->bind(':cid', $customerId);
  1072→        $this->db->bind(':iid', $invoiceId);
  1073→        $this->db->bind(':needle', $needle);
  1074→
  1075→        return round((float)($this->db->single()['delta'] ?? 0), 2);
  1076→    }

  1169→        array $invoiceRow,
  1170→        int $invoiceId,
  1171→        string $ledgerRemark
  1172→    ): array {
  1173→        $newTransport = max(0, (float)($data['transport_cost'] ?? 0));
  1174→        $subtotal = (float)($invoiceRow['subtotal'] ?? 0);
  1175→        $discount = (float)($invoiceRow['discount'] ?? 0);
  1176→        $oldTotal = (float)($invoiceRow['total_amount'] ?? 0);

  1186→        $this->db->bind(':tc', $newTransport);
  1187→        $this->db->bind(':total', $newTotal);
  1188→        $this->db->bind(':id', $invoiceId);
  1189→        $this->db->execute();
  1190→
  1191→        if (abs($totalDiff) > 0.0001) {
  1192→            $this->applyCustomerLedgerDelta(
  1193→                (int)($invoiceRow['customer_id'] ?? 0),
  1194→                $invoiceId,
  1195→                $totalDiff,
  1196→                $ledgerRemark,
  1197→                (int)($invoiceRow['branch_id'] ?? 0)
  1198→            );
  1199→        }
  1200→
  1201→        return [
  1202→            'transport'   => $newTransport,
  1203→            'new_total'   => $newTotal,
  1204→            'total_diff'  => $totalDiff,
