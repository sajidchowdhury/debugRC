    23→     * Post Other Income using proper double-entry
    24→     */
    25→    public function postOtherIncome($incomeId, array $incomeData): array
    26→    {
    27→        $amount = (float)$incomeData['amount'];
    28→        $narration = $incomeData['narration'] ?? 'Other Income Received';
    29→
    30→        $lines = [];
    31→
    32→        // Credit → Income Ledger (e.g. "Other Income - Rent Received")
    33→        $lines[] = [
    34→            'ledger_id'   => (int)$incomeData['ledger_id'],
    35→            'debit'       => 0,
    36→            'credit'      => $amount,
    37→            'description' => $narration,
    38→        ];
    39→
    40→        // Debit → Bank or Cash
    41→        if ($incomeData['payment_mode'] === 'bank' && !empty($incomeData['bank_id'])) {
    42→            $bankLedgerId = $this->getLedgerForBank((int)$incomeData['bank_id']);
    43→
    44→            $lines[] = [
    45→                'ledger_id'   => $bankLedgerId,
    46→                'debit'       => $amount,
    47→                'credit'      => 0,
    48→                'description' => 'Bank Receipt',
    49→                'entity_type' => 'bank',
    50→                'entity_id'   => (int)$incomeData['bank_id'],
    51→            ];
    52→        } else {
    53→            // Cash
    54→            $cashLedgerId = $this->getCashLedgerId();
    55→
    56→            $lines[] = [
    57→                'ledger_id'   => $cashLedgerId,
    58→                'debit'       => $amount,
    59→                'credit'      => 0,
    60→                'description' => 'Cash Receipt',
    61→            ];
    62→        }
    63→
    64→        $header = [
    65→            'entry_date'     => $incomeData['income_date'] ?? date('Y-m-d'),
    66→            'description'    => 'Other Income - ' . ($incomeData['income_code'] ?? ''),
    67→            'reference_type' => 'other_income',
    68→            'reference_id'   => $incomeId,
    69→            'branch_id'      => $incomeData['branch_id'] ?? ($_SESSION['branch_id'] ?? 1),
    70→        ];
    71→
    72→        return $this->journalModel->createEntry($header, $lines);
    73→    }

   129→    /**
   130→     * Get the main Bank Control Ledger (used for most bank transactions)
   131→     */
   132→    private function getBankControlLedgerId(): ?int
   133→    {
   134→        $ledgers = $this->ledgerModel->getLedgersByNature('cash_bank');
   135→
   136→        // Prefer a ledger that explicitly says "Bank" in the name
   137→        foreach ($ledgers as $ledger) {
   138→            if (stripos($ledger['ledger_name'], 'bank') !== false) {
   139→                return $ledger['id'];
   140→            }
   141→        }
   142→
   143→        // Fallback to first cash_bank ledger
   144→        return $ledgers[0]['id'] ?? null;
   145→    }

   147→    /**
   148→     * Get Cash ledger (usually "Cash in Hand" or similar)
   149→     */
   150→    private function getCashLedgerId(): ?int
   151→    {
   152→        $ledgers = $this->ledgerModel->getLedgersByNature('cash_bank');
   153→
   154→        foreach ($ledgers as $ledger) {
   155→            $name = strtolower($ledger['ledger_name']);
   156→            if (str_contains($name, 'cash') && !str_contains($name, 'bank')) {
   157→                return $ledger['id'];
   158→            }
   159→        }
   160→
   161→        // Fallback
   162→        return $ledgers[0]['id'] ?? null;
   163→    }

   238→     * Post Money Transfer using proper double-entry accounting

   240→    public function postMoneyTransfer($transferId, array $transferData): array
   241→    {
   242→        $amount = (float)$transferData['amount'];
   243→        $type = $transferData['transfer_type'];
   244→        $narration = $transferData['narration'] ?? 'Money Transfer';
   245→
   246→        $lines = [];
   247→        $description = "Money Transfer #{$transferData['transfer_code']} - {$type}";
   248→
   249→        switch ($type) {
   250→            case 'cash_to_bank':
   251→                // Debit Bank, Credit Cash
   252→                $bankLedgerId = $this->getLedgerForBank($transferData['to_bank_id'] ?? 0);
   253→                $cashLedgerId = $this->getCashLedgerId();
   254→
   255→                $lines[] = [
   256→                    'ledger_id'   => $bankLedgerId,
   257→                    'debit'       => $amount,
   258→                    'credit'      => 0,
   259→                    'description' => 'Cash deposited to Bank',
   260→                    'entity_type' => 'bank',
   261→                    'entity_id'   => $transferData['to_bank_id'] ?? null,
   262→                ];
   263→                $lines[] = [
   264→                    'ledger_id'   => $cashLedgerId,
   265→                    'debit'       => 0,
   266→                    'credit'      => $amount,
   267→                    'description' => 'Cash out for Bank deposit',
   268→                ];
   269→                break;
   270→
   271→            case 'bank_to_cash':
   272→                // Debit Cash, Credit Bank
   273→                $bankLedgerId = $this->getLedgerForBank($transferData['from_bank_id'] ?? 0);
   274→                $cashLedgerId = $this->getCashLedgerId();
   275→
   276→                $lines[] = [
   277→                    'ledger_id'   => $cashLedgerId,
   278→                    'debit'       => $amount,
   279→                    'credit'      => 0,
   280→                    'description' => 'Cash withdrawn from Bank',
   281→                ];
   282→                $lines[] = [
   283→                    'ledger_id'   => $bankLedgerId,
   284→                    'debit'       => 0,
   285→                    'credit'      => $amount,
   286→                    'description' => 'Bank withdrawal for Cash',
   287→                    'entity_type' => 'bank',
   288→                    'entity_id'   => $transferData['from_bank_id'] ?? null,
   289→                ];
   290→                break;
   291→
   292→            case 'cash_to_cash':
   293→                // Inter-branch cash movement
   294→                // Debit Receiving Branch Cash, Credit Sending Branch Cash
   295→                $cashLedgerId = $this->getCashLedgerId();
   296→
   297→                // Credit Sending
   298→                $lines[] = [
   299→                    'ledger_id'   => $cashLedgerId,
   300→                    'debit'       => 0,
   301→                    'credit'      => $amount,
   302→                    'description' => 'Cash sent to another branch',
   303→                    'entity_type' => 'branch',
   304→                    'entity_id'   => $transferData['from_branch_id'] ?? null,
   305→                ];
   306→                // Debit Receiving
   307→                $lines[] = [
   308→                    'ledger_id'   => $cashLedgerId,
   309→                    'debit'       => $amount,
   310→                    'credit'      => 0,
   311→                    'description' => 'Cash received from another branch',
   312→                    'entity_type' => 'branch',
   313→                    'entity_id'   => $transferData['to_branch_id'] ?? null,
   314→                ];
   315→                break;
   316→
   317→            case 'bank_to_bank':
   318→                // Bank to Bank transfer
   319→                $fromBankLedger = $this->getLedgerForBank($transferData['from_bank_id'] ?? 0);
   320→                $toBankLedger   = $this->getLedgerForBank($transferData['to_bank_id'] ?? 0);
   321→
   322→                $lines[] = [
   323→                    'ledger_id'   => $toBankLedger,
   324→                    'debit'       => $amount,
   325→                    'credit'      => 0,
   326→                    'description' => 'Received from another bank',
   327→                    'entity_type' => 'bank',
   328→                    'entity_id'   => $transferData['to_bank_id'] ?? null,
   329→                ];
   330→                $lines[] = [
   331→                    'ledger_id'   => $fromBankLedger,
   332→                    'debit'       => 0,
   333→                    'credit'      => $amount,
   334→                    'description' => 'Transferred to another bank',
   335→                    'entity_type' => 'bank',
   336→                    'entity_id'   => $transferData['from_bank_id'] ?? null,
   337→                ];
   338→                break;
   339→
   340→            default:
   341→                return ['status' => 'error', 'message' => 'Unknown transfer type for journal posting'];
   342→        }
   343→
   344→        $header = [
   345→            'entry_date'     => $transferData['transfer_date'] ?? date('Y-m-d'),
   346→            'description'    => $description,
   347→            'reference_type' => 'money_transfer',
   348→            'reference_id'   => $transferId,
   349→            'branch_id'      => $transferData['from_branch_id'] ?? ($_SESSION['branch_id'] ?? 1),
   350→        ];
   351→
   352→        return $this->journalModel->createEntry($header, $lines);
   353→    }

   380→    /**
   381→     * Post Purchase Receive (GRN) using double-entry.
   382→     *
   383→     * Decision (Phase 2): Credit AP on Goods Receipt (not PO).
   384→     * Dr Inventory (at actual purchase cost from GRN total)
   385→     * Cr Supplier Payable (control account)
   386→     *
   387→     * Amount uses the receive's total_amount (actual cost). Inventory avg cost
   388→     * is handled separately by StockTransactionModel for valuation/COGS.
   389→     */
   390→    public function postPurchaseReceive($receiveId, array $receiveData): array
   391→    {
   392→        $amount = (float)$receiveData['total_amount'];
   393→        if ($amount <= 0) {
   394→            return ['status' => 'error', 'message' => 'Invalid receive amount for journal'];
   395→        }
   396→
   397→        $narration = $receiveData['remarks'] ?? 'Goods Received';
   398→        $receiveCode = $receiveData['receive_code'] ?? ('GRN-' . $receiveId);
   399→
   400→        $lines = [];
   401→
   402→        // Debit Inventory (asset increase at cost)
   403→        $inventoryLedgerId = $this->getInventoryLedgerId();
   404→        if (!$inventoryLedgerId) {
   405→            return ['status' => 'error', 'message' => 'Inventory ledger not found (nature: inventory)'];
   406→        }
   407→
   408→        $lines[] = [
   409→            'ledger_id'   => $inventoryLedgerId,
   410→            'debit'       => $amount,
   411→            'credit'      => 0,
   412→            'description' => 'GRN #' . $receiveCode . ' - Inventory received @ purchase cost',
   413→            'entity_type' => 'purchase_receive',
   414→            'entity_id'   => (int)$receiveId,
   415→        ];
   416→
   417→        // Credit Supplier Payable (liability)
   418→        $supplierPayableLedgerId = $this->getSupplierPayableLedgerId();
   419→        if (!$supplierPayableLedgerId) {
   420→            return ['status' => 'error', 'message' => 'Supplier Payable ledger not found (nature: supplier_payable)'];
   421→        }
   422→
   423→        $lines[] = [
   424→            'ledger_id'   => $supplierPayableLedgerId,
   425→            'debit'       => 0,
   426→            'credit'      => $amount,
   427→            'description' => 'GRN #' . $receiveCode . ' - Payable to supplier',
   428→            'entity_type' => 'supplier',
   429→            'entity_id'   => $receiveData['supplier_id'] ?? null,
   430→        ];
   431→
   432→        $header = [
   433→            'entry_date'     => $receiveData['receive_date'] ?? date('Y-m-d'),
   434→            'description'    => 'Purchase Receive - ' . $receiveCode,
   435→            'reference_type' => 'purchase_receive',
   436→            'reference_id'   => (int)$receiveId,
   437→            'branch_id'      => $receiveData['branch_id'] ?? ($_SESSION['branch_id'] ?? 1),
   438→        ];
   439→
   440→        $result = $this->journalModel->createEntry($header, $lines);
   441→
   442→        // Return enhanced result so caller can easily get the id
   443→        if (!empty($result['journal_entry_id'])) {
   444→            $result['journal_entry_id'] = (int)$result['journal_entry_id'];
   445→        }
   446→
   447→        return $result;
   448→    }

   450→    /**
   451→     * Post Purchase Return using double-entry (reverses the GRN impact).
   452→     *
   453→     * Dr Supplier Payable
   454→     * Cr Inventory (at the cost recorded on the return)

   467→
   468→        // Debit Supplier Payable (reduce liability)

   470→        if (!$supplierPayableLedgerId) {
   471→            return ['status' => 'error', 'message' => 'Supplier Payable ledger not found'];

   474→        $lines[] = [
   475→            'ledger_id'   => $supplierPayableLedgerId,

   477→            'credit'      => 0,
   478→            'description' => 'Return #' . $returnCode . ' - AP reduced',
   479→            'entity_type' => 'purchase_return',
   480→            'entity_id'   => (int)$returnId,
   481→        ];
   482→
   483→        // Credit Inventory (asset reduction)
   484→        $inventoryLedgerId = $this->getInventoryLedgerId();
   485→        if (!$inventoryLedgerId) {
   486→            return ['status' => 'error', 'message' => 'Inventory ledger not found'];

   488→
   489→        $lines[] = [
   490→            'ledger_id'   => $inventoryLedgerId,
   491→            'debit'       => 0,

   493→            'description' => 'Return #' . $returnCode . ' - Inventory out (cost)',
   494→            'entity_type' => 'purchase_return',
   495→            'entity_id'   => (int)$returnId,
   496→        ];
   497→
   498→        $header = [

   501→            'reference_type' => 'purchase_return',
   502→            'reference_id'   => (int)$returnId,
   503→            'branch_id'      => $returnData['branch_id'] ?? ($_SESSION['branch_id'] ?? 1),
   504→        ];
   505→
   506→        $result = $this->journalModel->createEntry($header, $lines);
   507→
   508→        if (!empty($result['journal_entry_id'])) {
   509→            $result['journal_entry_id'] = (int)$result['journal_entry_id'];

   511→
   512→        return $result;

   540→    /**
   541→     * Post sales invoice — Dr AR / Cr revenue (and transport portion).
   542→     */
   543→    public function postSalesInvoice(int $invoiceId, array $invoiceData): array
   544→    {
   545→        $total = (float)($invoiceData['total_amount'] ?? 0);
   546→        $subtotal = (float)($invoiceData['subtotal'] ?? 0);
   547→        $discount = (float)($invoiceData['discount'] ?? 0);
   548→        $transport = (float)($invoiceData['transport_cost'] ?? 0);
   549→
   550→        if ($total <= 0) {
   551→            return ['status' => 'error', 'message' => 'Invalid invoice amount for journal'];
   552→        }
   553→
   554→        $discount = max(0, round($discount, 2));
   555→        $transport = max(0, round($transport, 2));
   556→        $grossRevenue = max(0, round($subtotal, 2));
   557→        $discountLedgerId = $discount > 0.0001 ? $this->getSalesDiscountLedgerId() : null;
   558→        $revenueBase = $discountLedgerId
   559→            ? $grossRevenue
   560→            : max(0, round($subtotal - $discount, 2));
   561→
   562→        if (round($revenueBase + $transport - ($discountLedgerId ? $discount : 0), 2) !== round($total, 2)) {
   563→            $revenueBase = max(0, round($total - $transport, 2));
   564→        }
   565→
   566→        $arLedgerId = $this->getCustomerReceivableLedgerId();
   567→        $revenueLedgerId = $this->getSalesRevenueLedgerId();
   568→        if (!$arLedgerId) {
   569→            return ['status' => 'error', 'message' => 'Accounts Receivable ledger not found (nature: customer_receivable)'];
   570→        }
   571→        if (!$revenueLedgerId) {
   572→            return ['status' => 'error', 'message' => 'Sales Revenue ledger not found (nature: sales_revenue)'];
   573→        }
   574→
   575→        $invoiceCode = $invoiceData['invoice_code'] ?? ('SI-' . $invoiceId);
   576→        $lines = [];
   577→
   578→        $lines[] = [
   579→            'ledger_id'   => $arLedgerId,
   580→            'debit'       => $total,
   581→            'credit'      => 0,
   582→            'description' => 'Invoice #' . $invoiceCode . ' — AR',
   583→            'entity_type' => 'customer',
   584→            'entity_id'   => $invoiceData['customer_id'] ?? null,
   585→        ];
   586→
   587→        if ($discountLedgerId && $discount > 0) {
   588→            $lines[] = [
   589→                'ledger_id'   => $discountLedgerId,
   590→                'debit'       => $discount,
   591→                'credit'      => 0,
   592→                'description' => 'Invoice #' . $invoiceCode . ' — Discount allowed',
   593→                'entity_type' => 'sales_invoice',
   594→                'entity_id'   => $invoiceId,
   595→            ];
   596→        }
   597→
   598→        if ($revenueBase > 0) {
   599→            $lines[] = [
   600→                'ledger_id'   => $revenueLedgerId,
   601→                'debit'       => 0,
   602→                'credit'      => $revenueBase,
   603→                'description' => 'Invoice #' . $invoiceCode . ' — Sales revenue' . ($discountLedgerId ? ' (gross)' : ''),
   604→                'entity_type' => 'sales_invoice',
   605→                'entity_id'   => $invoiceId,
   606→            ];
   607→        }
   608→
   609→        if ($transport > 0) {
   610→            $lines[] = [
   611→                'ledger_id'   => $revenueLedgerId,
   612→                'debit'       => 0,
   613→                'credit'      => $transport,
   614→                'description' => 'Invoice #' . $invoiceCode . ' — Transport charges',
   615→                'entity_type' => 'sales_invoice',
   616→                'entity_id'   => $invoiceId,
   617→            ];
   618→        }
   619→
   620→        $creditSum = $revenueBase + $transport;
   621→        if ($creditSum < 0.0001 && $total > 0) {
   622→            $lines[] = [
   623→                'ledger_id'   => $revenueLedgerId,
   624→                'debit'       => 0,
   625→                'credit'      => $total,
   626→                'description' => 'Invoice #' . $invoiceCode . ' — Sales',
   627→                'entity_type' => 'sales_invoice',
   628→                'entity_id'   => $invoiceId,
   629→            ];
   630→        }
   631→
   632→        $header = [
   633→            'entry_date'     => $invoiceData['invoice_date'] ?? date('Y-m-d'),
   634→            'description'    => 'Sales Invoice - ' . $invoiceCode,
   635→            'reference_type' => 'sales_invoice',
   636→            'reference_id'   => $invoiceId,
   637→            'branch_id'      => $invoiceData['branch_id'] ?? ($_SESSION['branch_id'] ?? 1),
   638→        ];
   639→
   640→        return $this->finalizeJournalResult($this->journalModel->createEntry($header, $lines));
   641→    }

  1062→    {

  1073→            $name = strtolower($ledger['ledger_name'] ?? '');
