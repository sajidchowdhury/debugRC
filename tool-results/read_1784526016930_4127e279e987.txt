     1→     1→<?php
     2→     2→
     3→     3→namespace App\Services\Sales;
     4→     4→
     5→     5→use App\Models\SalesInvoice;
     6→     6→use App\Models\SalesChallan;
     7→     7→use App\Services\Stock\StockService;
     8→     8→use App\Services\Stock\StockAvailabilityService;
     9→     9→use App\Services\Accounting\JournalPostingService;
    10→    10→use App\Services\Accounting\SubLedgerService;
    11→    11→use Illuminate\Support\Facades\DB;
    12→    12→use Illuminate\Support\Facades\Log;
    13→    13→
    14→    14→/**
    15→    15→ * Sales Challan Service — Phase 8.3.
    16→    16→ *
    17→    17→ * Two-step flow (from Phase 8.2 draft invoice):
    18→    18→ *   1. prepareGodown: assign warehouse_id to invoice items + dispatches.
    19→    19→ *      Invoice status: draft → confirmed, is_godown_prepared=true.
    20→    20→ *      NO stock movement, NO GL — just warehouse assignment.
    21→    21→ *
    22→    22→ *   2. issueChallan: stock OUT via StockService at avg_cost + GL Dr COGS / Cr Inventory.
    23→    23→ *      Creates sales_challan, moves stock, posts COGS journal.
    24→    24→ *      Invoice: is_challan_issued=true.
    25→    25→ *
    26→    26→ *   3. cancelChallan: reverse stock + GL (append-only reversal).
    27→    27→ *
    28→    28→ * Rate semantics (per avg_cost_rule.md §3):
    29→    29→ *   Stock OUT at current avg_cost (cost flows out at average, avg unchanged on OUT).
    30→    30→ *   The COGS = Σ (qty × avg_cost) for all dispatch lines.
    31→    31→ *
    32→    32→ * GL posting (re-derived from double-entry):
    33→    33→ *   Dr COGS (nature: cogs)
    34→    34→ *   Cr Inventory (nature: inventory)
    35→    35→ */
    36→    36→class SalesChallanService
    37→    37→{
    38→    38→    public function __construct(
    39→    39→        private StockService $stockService,
    40→    40→        private JournalPostingService $journalPosting,
    41→    41→        private SubLedgerService $subLedger,
    42→    42→        private SalesAccess $salesAccess,
    43→    43→        private SalesAuditLogger $auditLogger,
    44→    44→        private StockAvailabilityService $availabilityService
    45→    45→    ) {}
    46→    46→
    47→    47→    /**
    48→    48→     * Step 1: Prepare godown — assign warehouse_id to invoice items + dispatches.
    49→    49→     * No stock movement, no GL.
    50→    50→     *
    51→    51→     * @param int $invoiceId
    52→    52→     * @param array $warehouseAssignments [product_id => warehouse_id]
    53→    53→     * @param int $preparedBy
    54→    54→     * @return SalesInvoice
    55→    55→     * @throws \RuntimeException If invoice not draft or already godown-prepared.
    56→    56→     */
    57→    57→    public function prepareGodown(int $invoiceId, array $warehouseAssignments, int $preparedBy): SalesInvoice
    58→    58→    {
    59→    59→        return DB::transaction(function () use ($invoiceId, $warehouseAssignments, $preparedBy) {
    60→    60→            $invoice = SalesInvoice::with('items', 'dispatches')->lockForUpdate()->find($invoiceId);
    61→    61→
    62→    62→            if (!$invoice) {
    63→    63→                throw new \RuntimeException("Invoice {$invoiceId} not found.");
    64→    64→            }
    65→    65→            if (!$invoice->isDraft()) {
    66→    66→                throw new \RuntimeException("Only draft invoices can be godown-prepared (current: {$invoice->status}).");
    67→    67→            }
    68→    68→
    69→    69→            // Assign warehouse_id to each invoice item.
    70→    70→            foreach ($invoice->items as $item) {
    71→    71→                $wid = $warehouseAssignments[$item->product_id] ?? null;
    72→    72→                if (!$wid) {
    73→    73→                    throw new \RuntimeException("Warehouse not assigned for product {$item->product_id}.");
    74→    74→                }
    75→    75→
    76→    76→                // Check availability in the assigned warehouse.
    77→    77→                $available = $this->stockService->getWarehouseQty((int) $wid, $item->product_id);
    78→    78→                if ((float) $item->qty > $available + 0.0001) {
    79→    79→                    throw new \RuntimeException(
    80→    80→                        "Insufficient stock in warehouse {$wid} for product {$item->product_id}: "
    81→    81→                        . "available {$available}, required {$item->qty}"
    82→    82→                    );
    83→    83→                }
    84→    84→
    85→    85→                DB::table('sales_invoice_items')
    86→    86→                    ->where('id', $item->id)
    87→    87→                    ->update(['warehouse_id' => (int) $wid]);
    88→    88→
    89→    89→                DB::table('sales_invoice_dispatches')
    90→    90→                    ->where('sales_invoice_id', $invoiceId)
    91→    91→                    ->where('product_id', $item->product_id)
    92→    92→                    ->update(['warehouse_id' => (int) $wid]);
    93→    93→            }
    94→    94→
    95→    95→            // Update invoice status.
    96→    96→            DB::table('sales_invoices')
    97→    97→                ->where('id', $invoiceId)
    98→    98→                ->update([
    99→    99→                    'status' => 'confirmed',
   100→   100→                    'is_godown_prepared' => true,
   101→   101→                    'godown_prepared_at' => now(),
   102→   102→                    'updated_at' => now(),
   103→   103→                ]);
   104→   104→
   105→   105→            // P1-3: Audit log — godown_prepared.
   106→   106→            $this->auditLogger->godownPrepared(
   107→   107→                $preparedBy, $invoiceId, $invoice->invoice_code, (int) $invoice->branch_id
   108→   108→            );
   109→   109→
   110→   110→            return SalesInvoice::with(['items.product', 'dispatches', 'customer', 'branch'])->find($invoiceId);
   111→   111→        });
   112→   112→    }
   113→   113→
   114→   114→    /**
   115→   115→     * Step 2: Issue challan — stock OUT + GL Dr COGS / Cr Inventory.
   116→   116→     *
   117→   117→     * @param int $invoiceId
   118→   118→     * @param array $data {
   119→   119→     *     transport_name: string|null,
   120→   120→     *     vehicle_number: string|null,
   121→   121→     *     driver_name: string|null,
   122→   122→     *     transport_cost: float,
   123→   123→     *     notes: string|null,
   124→   124→     *     created_by: int,
   125→   125→     * }
   126→   126→     * @return SalesChallan
   127→   127→     * @throws \RuntimeException If not godown-prepared or already challan-issued.
   128→   128→     */
   129→   129→    public function issueChallan(int $invoiceId, array $data): SalesChallan
   130→   130→    {
   131→   131→        return DB::transaction(function () use ($invoiceId, $data) {
   132→   132→            $invoice = SalesInvoice::with(['items', 'dispatches'])->lockForUpdate()->find($invoiceId);
   133→   133→
   134→   134→            if (!$invoice) {
   135→   135→                throw new \RuntimeException("Invoice {$invoiceId} not found.");
   136→   136→            }
   137→   137→
   138→   138→            // P0-8: Defense-in-depth branch isolation check.
   139→   139→            $this->salesAccess->assertBranchAccessible((int) $invoice->branch_id);
   140→   140→
   141→   141→            if (!$invoice->is_godown_prepared) {
   142→   142→                throw new \RuntimeException("Invoice must be godown-prepared before issuing challan.");
   143→   143→            }
   144→   144→            if ($invoice->is_challan_issued) {
   145→   145→                throw new \RuntimeException("Challan already issued for this invoice.");
   146→   146→            }
   147→   147→
   148→   148→            $challanCode = $this->generateChallanCode();
   149→   149→            $challanDate = now()->format('Y-m-d');
   150→   150→            $cogsTotal = 0.0;
   151→   151→
   152→   152→            // Create the challan header.
   153→   153→            $challanId = DB::table('sales_challans')->insertGetId([
   154→   154→                'challan_code' => $challanCode,
   155→   155→                'challan_date' => $challanDate,
   156→   156→                'sales_invoice_id' => $invoiceId,
   157→   157→                'branch_id' => $invoice->branch_id,
   158→   158→                'transport_name' => $data['transport_name'] ?? null,
   159→   159→                'transport_phone' => $data['transport_phone'] ?? null,
   160→   160→                'vehicle_number' => $data['vehicle_number'] ?? null,
   161→   161→                'driver_name' => $data['driver_name'] ?? null,
   162→   162→                'transport_cost' => $data['transport_cost'] ?? 0,
   163→   163→                'is_reversed' => false,
   164→   164→                'is_dispatch_soft_hold' => false,
   165→   165→                'created_by' => $data['created_by'] ?? null,
   166→   166→                'created_at' => now(),
   167→   167→                'updated_at' => now(),
   168→   168→            ]);
   169→   169→
   170→   170→            // Process each dispatch line: stock OUT at avg_cost.
   171→   171→            foreach ($invoice->items as $item) {
   172→   172→                $warehouseId = $item->warehouse_id;
   173→   173→                $productId = $item->product_id;
   174→   174→                $qty = (float) $item->qty;
   175→   175→
   176→   176→                if (!$warehouseId) {
   177→   177→                    throw new \RuntimeException("Product {$productId} has no warehouse assigned. Run godown prep first.");
   178→   178→                }
   179→   179→
   180→   180→                // Get current avg_cost (this is the COGS rate).
   181→   181→                $avgCost = $this->stockService->getWarehouseAvgCost($warehouseId, $productId);
   182→   182→                if ($avgCost <= 0) {
   183→   183→                    throw new \RuntimeException(
   184→   184→                        "Cannot issue challan: zero avg_cost for product {$productId} in warehouse {$warehouseId}. "
   185→   185→                        . "Receive stock or set cost first."
   186→   186→                    );
   187→   187→                }
   188→   188→
   189→   189→                // Stock OUT via StockService (reference_type='sales_challan', rate=avg_cost).
   190→   190→                $this->stockService->applyTransaction([
   191→   191→                    'warehouse_id' => $warehouseId,
   192→   192→                    'product_id' => $productId,
   193→   193→                    'qty' => -$qty, // negative = OUT
   194→   194→                    'rate' => $avgCost, // current avg_cost (cost flows out, avg unchanged on OUT)
   195→   195→                    'reference_type' => 'sales_challan',
   196→   196→                    'reference_id' => $challanId,
   197→   197→                    'notes' => 'Sales Challan ' . $challanCode,
   198→   198→                    'transaction_date' => $challanDate,
   199→   199→                    'created_by' => $data['created_by'] ?? null,
   200→   200→                ]);
   201→   201→
   202→   202→                // P0-5: Persist per-line issue cost snapshot (sales_challan_items).
   203→   203→                // This denormalized record captures the avg_cost at the moment of stock OUT
   204→   204→                // so that reversals and reports can use the ORIGINAL per-line rate.
   205→   205→                DB::table('sales_challan_items')->insert([
   206→   206→                    'sales_challan_id' => $challanId,
   207→   207→                    'product_id' => $productId,
   208→   208→                    'warehouse_id' => $warehouseId,
   209→   209→                    'qty' => $qty,
   210→   210→                    'issue_rate' => $avgCost,
   211→   211→                    'cogs_amount' => round($qty * $avgCost, 2),
   212→   212→                    'created_at' => now(),
   213→   213→                ]);
   214→   214→
   215→   215→                // Update dispatch row: dispatched_qty = ordered_qty, qty mirrors for GENERATED amount.
   216→   216→                DB::table('sales_invoice_dispatches')
   217→   217→                    ->where('sales_invoice_id', $invoiceId)
   218→   218→                    ->where('product_id', $productId)
   219→   219→                    ->update([
   220→   220→                        'qty' => DB::raw('ordered_qty'),
   221→   221→                        'dispatched_qty' => DB::raw('ordered_qty'),
   222→   222→                        'warehouse_id' => $warehouseId,
   223→   223→                    ]);
   224→   224→
   225→   225→                $cogsTotal += $qty * $avgCost;
   226→   226→            }
   227→   227→
   228→   228→            // Post GL: Dr COGS / Cr Inventory (single journal for all lines).
   229→   229→            $journalEntryId = $this->postCogsGL($challanId, $challanCode, $challanDate, $invoice->branch_id, $cogsTotal, $data['created_by'] ?? null);
   230→   230→
   231→   231→            // Update challan with journal_entry_id + issue_cost.
   232→   232→            DB::table('sales_challans')
   233→   233→                ->where('id', $challanId)
   234→   234→                ->update([
   235→   235→                    'journal_entry_id' => $journalEntryId,
   236→   236→                    'issue_cost' => round($cogsTotal, 2),
   237→   237→                    'updated_at' => now(),
   238→   238→                ]);
   239→   239→
   240→   240→            // P2-3: Transport snapshot + adjustment.
   241→   241→            // If the challan form's transport_cost differs from the invoice's
   242→   242→            // original transport_cost, snapshot the original values + post an
   243→   243→            // adjustment (customer_ledger + GL) for the delta.
   244→   244→            $newTransport = (float) ($data['transport_cost'] ?? 0);
   245→   245→            $oldTransport = (float) $invoice->transport_cost;
   246→   246→            $transportAdjustment = round($newTransport - $oldTransport, 2);
   247→   247→            $adjustmentJournalEntryId = null;
   248→   248→
   249→   249→            if (abs($transportAdjustment) > 0.01) {
   250→   250→                // Snapshot original values on the invoice.
   251→   251→                DB::table('sales_invoices')
   252→   252→                    ->where('id', $invoiceId)
   253→   253→                    ->update([
   254→   254→                        'pre_challan_transport' => $oldTransport,
   255→   255→                        'pre_challan_total' => (float) $invoice->total_amount,
   256→   256→                    ]);
   257→   257→
   258→   258→                // Update invoice transport_cost + total_amount + due_amount.
   259→   259→                $newTotal = (float) $invoice->sub_total - (float) $invoice->discount_amount + $newTransport;
   260→   260→                DB::table('sales_invoices')
   261→   261→                    ->where('id', $invoiceId)
   262→   262→                    ->update([
   263→   263→                        'transport_cost' => round($newTransport, 2),
   264→   264→                        'total_amount' => round($newTotal, 2),
   265→   265→                        'due_amount' => round($newTotal - (float) $invoice->paid_amount, 2),
   266→   266→                    ]);
   267→   267→
   268→   268→                // Post GL adjustment JE FIRST (Dr/Cr AR + Revenue, swapped by sign) to get journal_entry_id.
   269→   269→                $adjustmentJournalEntryId = $this->postTransportAdjustmentGL(
   270→   270→                    $invoiceId, $challanCode, $challanDate, (int) $invoice->branch_id,
   271→   271→                    $transportAdjustment, (int) $invoice->customer_id,
   272→   272→                    $data['created_by'] ?? null
   273→   273→                );
   274→   274→
   275→   275→                // Post customer_ledger 'invoice_adjustment' entry via SubLedgerService for the delta.
   276→   276→                $debit = $transportAdjustment > 0 ? abs($transportAdjustment) : 0;
   277→   277→                $credit = $transportAdjustment < 0 ? abs($transportAdjustment) : 0;
   278→   278→                $this->subLedger->postCustomerLedgerEntry([
   279→   279→                    'customer_id' => $invoice->customer_id,
   280→   280→                    'branch_id' => $invoice->branch_id,
   281→   281→                    'transaction_date' => now()->format('Y-m-d'),
   282→   282→                    'transaction_type' => 'invoice_adjustment',
   283→   283→                    'reference_type' => 'sales_invoice',
   284→   284→                    'reference_id' => $invoice->id,
   285→   285→                    'debit' => $debit,
   286→   286→                    'credit' => $credit,
   287→   287→                    'description' => 'Challan ' . $challanCode . ' — transport adjustment ' . ($transportAdjustment >= 0 ? '+' : '') . number_format($transportAdjustment, 2),
   288→   288→                    'journal_entry_id' => $adjustmentJournalEntryId,
   289→   289→                    'created_by' => $data['created_by'] ?? null,
   290→   290→                ]);
   291→   291→            }
   292→   292→
   293→   293→            // Store transport_adjustment + adjustment_journal_entry_id on the challan.
   294→   294→            DB::table('sales_challans')
   295→   295→                ->where('id', $challanId)
   296→   296→                ->update([
   297→   297→                    'transport_adjustment' => $transportAdjustment,
   298→   298→                    'adjustment_journal_entry_id' => $adjustmentJournalEntryId,
   299→   299→                ]);
   300→   300→
   301→   301→            // Update invoice: is_challan_issued=true.
   302→   302→            DB::table('sales_invoices')
   303→   303→                ->where('id', $invoiceId)
   304→   304→                ->update([
   305→   305→                    'is_challan_issued' => true,
   306→   306→                    'challan_issued_at' => now(),
   307→   307→                    'cogs_journal_entry_id' => $journalEntryId,
   308→   308→                    'updated_at' => now(),
   309→   309→                ]);
   310→   310→
   311→   311→            // P1-3: Audit log — challan_issued.
   312→   312→            $this->auditLogger->challanIssued(
   313→   313→                $data['created_by'] ?? auth()->id() ?? 0,
   314→   314→                $challanId, $challanCode, $invoiceId, (int) $invoice->branch_id,
   315→   315→                $cogsTotal, $journalEntryId
   316→   316→            );
   317→   317→
   318→   318→            // P2-7: Invalidate pipeline cache (dispatched_qty changed).
   319→   319→            $this->availabilityService->invalidatePipelineForInvoice($invoiceId);
   320→   320→
   321→   321→            return SalesChallan::with(['salesInvoice.items.product', 'items.product', 'items.warehouse', 'branch', 'journalEntry.lines.ledger'])
   322→   322→                ->find($challanId);
   323→   323→        });
   324→   324→    }
   325→   325→
   326→   326→    /**
   327→   327→     * Step 3: Cancel a challan — reverse stock + GL (append-only reversal).
   328→   328→     *
   329→   329→     * @param int $challanId
   330→   330→     * @param int $cancelledBy
   331→   331→     * @param string $reason
   332→   332→     * @return SalesChallan
   333→   333→     */
   334→   334→    public function cancelChallan(int $challanId, int $cancelledBy, string $reason = ''): SalesChallan
   335→   335→    {
   336→   336→        return DB::transaction(function () use ($challanId, $cancelledBy, $reason) {
   337→   337→            $challan = SalesChallan::with('salesInvoice.items')->lockForUpdate()->find($challanId);
   338→   338→
   339→   339→            if (!$challan) {
   340→   340→                throw new \RuntimeException("Challan {$challanId} not found.");
   341→   341→            }
   342→   342→            if ($challan->is_reversed) {
   343→   343→                throw new \RuntimeException("Challan is already cancelled.");
   344→   344→            }
   345→   345→
   346→   346→            // Reverse GL (COGS journal).
   347→   347→            if ($challan->journal_entry_id) {
   348→   348→                $this->journalPosting->reverseJournalEntry(
   349→   349→                    $challan->journal_entry_id, $cancelledBy,
   350→   350→                    "Challan cancelled: {$reason}"
   351→   351→                );
   352→   352→            }
   353→   353→
   354→   354→            // Reverse each stock movement.
   355→   355→            $stockTxs = DB::table('stock_transactions')
   356→   356→                ->where('reference_type', 'sales_challan')
   357→   357→                ->where('reference_id', $challanId)
   358→   358→                ->where('is_reversed', false)
   359→   359→                ->get();
   360→   360→
   361→   361→            foreach ($stockTxs as $tx) {
   362→   362→                $this->stockService->reverseTransaction(
   363→   363→                    $tx->id, $cancelledBy,
   364→   364→                    "Challan cancelled: {$reason}"
   365→   365→                );
   366→   366→            }
   367→   367→
   368→   368→            // Reset dispatch rows: dispatched_qty = 0, qty mirrors for GENERATED amount.
   369→   369→            DB::table('sales_invoice_dispatches')
   370→   370→                ->where('sales_invoice_id', $challan->sales_invoice_id)
   371→   371→                ->update([
   372→   372→                    'qty' => 0,
   373→   373→                    'dispatched_qty' => 0,
   374→   374→                ]);
   375→   375→
   376→   376→            // Mark challan as reversed.
   377→   377→            DB::table('sales_challans')
   378→   378→                ->where('id', $challanId)
   379→   379→                ->update([
   380→   380→                    'is_reversed' => true,
   381→   381→                    'reversed_at' => now(),
   382→   382→                    'reversed_by' => $cancelledBy,
   383→   383→                    'reverse_reason' => $reason,
   384→   384→                ]);
   385→   385→
   386→   386→            // P2-3: Reverse transport adjustment (if any).
   387→   387→            // If the challan had a transport_adjustment, reverse the adjustment
   388→   388→            // GL JE + restore the invoice's original transport_cost + total_amount
   389→   389→            // from the pre_challan snapshot.
   390→   390→            if ($challan->adjustment_journal_entry_id) {
   391→   391→                $this->journalPosting->reverseJournalEntry(
   392→   392→                    $challan->adjustment_journal_entry_id, $cancelledBy,
   393→   393→                    "Challan cancelled: transport adjustment reversed — {$reason}"
   394→   394→                );
   395→   395→            }
   396→   396→
   397→   397→            // Restore invoice transport snapshot (if snapshot exists).
   398→   398→            $invoiceRow = DB::table('sales_invoices')
   399→   399→                ->where('id', $challan->sales_invoice_id)
   400→   400→                ->first(['pre_challan_transport', 'pre_challan_total', 'paid_amount']);
   401→   401→
   402→   402→            if ($invoiceRow && $invoiceRow->pre_challan_transport !== null) {
   403→   403→                $restoredTotal = (float) $invoiceRow->pre_challan_total;
   404→   404→                DB::table('sales_invoices')
   405→   405→                    ->where('id', $challan->sales_invoice_id)
   406→   406→                    ->update([
   407→   407→                        'transport_cost' => (float) $invoiceRow->pre_challan_transport,
   408→   408→                        'total_amount' => $restoredTotal,
   409→   409→                        'due_amount' => round($restoredTotal - (float) $invoiceRow->paid_amount, 2),
   410→   410→                        'pre_challan_transport' => null,
   411→   411→                        'pre_challan_total' => null,
   412→   412→                    ]);
   413→   413→            }
   414→   414→
   415→   415→            // P2-2: Reset invoice back to DRAFT state (fully editable).
   416→   416→            // Legacy reverseChallan resets invoice to 'godown_issued' state,
   417→   417→            // but Laravel's edit flow requires status='draft'. Since cancelling
   418→   418→            // a challan invalidates the godown assignment (warehouse_id on items
   419→   419→            // was set during godown prep, and the items may need re-picking),
   420→   420→            // we reset the invoice all the way back to draft.
   421→   421→            DB::table('sales_invoices')
   422→   422→                ->where('id', $challan->sales_invoice_id)
   423→   423→                ->update([
   424→   424→                    'status' => 'draft',
   425→   425→                    'is_challan_issued' => false,
   426→   426→                    'challan_issued_at' => null,
   427→   427→                    'cogs_journal_entry_id' => null,
   428→   428→                    'is_godown_prepared' => false,
   429→   429→                    'godown_prepared_at' => null,
   430→   430→                    'updated_at' => now(),
   431→   431→                ]);
   432→   432→
   433→   433→            // P2-2: Reset warehouse_id on invoice items + dispatches (godown
   434→   434→            // assignment is invalidated — user must re-run godown prep).
   435→   435→            DB::table('sales_invoice_items')
   436→   436→                ->where('sales_invoice_id', $challan->sales_invoice_id)
   437→   437→                ->update(['warehouse_id' => null]);
   438→   438→
   439→   439→            DB::table('sales_invoice_dispatches')
   440→   440→                ->where('sales_invoice_id', $challan->sales_invoice_id)
   441→   441→                ->update(['warehouse_id' => null]);
   442→   442→
   443→   443→            // P1-3: Audit log — challan_reversed.
   444→   444→            $this->auditLogger->challanReversed(
   445→   445→                $cancelledBy,
   446→   446→                $challanId, $challan->challan_code, $challan->sales_invoice_id,
   447→   447→                (int) $challan->branch_id, $reason
   448→   448→            );
   449→   449→
   450→   450→            // P2-7: Invalidate pipeline cache (dispatched_qty reset to 0).
   451→   451→            $this->availabilityService->invalidatePipelineForInvoice($challan->sales_invoice_id);
   452→   452→
   453→   453→            return SalesChallan::find($challanId);
   454→   454→        });
   455→   455→    }

   502→   502→    /**
   503→   503→     * Generate atomic challan code: CH-YYYYMMDD-NNNN.
   504→   504→     */
   505→   505→    private function generateChallanCode(): string
   506→   506→    {
   507→   507→        $datePart = now()->format('Ymd');
   508→   508→        $periodKey = now()->format('Y-m');
   509→   509→        $docType = 'sales_challan';
   510→   510→
   511→   511→        return DB::transaction(function () use ($docType, $periodKey, $datePart) {
   512→   512→            $seqRow = DB::table('document_sequences')
   513→   513→                ->where('doc_type', $docType)
   514→   514→                ->where('branch_id', 0)
   515→   515→                ->where('period_key', $periodKey)
   516→   516→                ->lockForUpdate()
   517→   517→                ->first();
   518→   518→
   519→   519→            $nextNumber = $seqRow ? ((int) $seqRow->last_number + 1) : 1;
   520→   520→
   521→   521→            if ($seqRow) {
   522→   522→                DB::table('document_sequences')->where('id', $seqRow->id)
   523→   523→                    ->update(['last_number' => $nextNumber, 'updated_at' => now()]);
   524→   524→            } else {
   525→   525→                DB::table('document_sequences')->insert([
   526→   526→                    'doc_type' => $docType, 'branch_id' => 0,
   527→   527→                    'period_key' => $periodKey, 'last_number' => $nextNumber,
   528→   528→                    'updated_at' => now(),
   529→   529→                ]);
   530→   530→            }
   531→   531→
   532→   532→            return "CH-{$datePart}-" . str_pad((string) $nextNumber, 4, '0', STR_PAD_LEFT);
   533→   533→        });
   534→   534→    }

   570→   570→    /**
   571→   571→     * P2-3: Post GL adjustment journal for transport delta.
   572→   572→     *
   573→   573→     * If transport increased (positive adjustment):
   574→   574→     *   Dr AR / Cr Transport Revenue
   575→   575→     * If transport decreased (negative adjustment):
   576→   576→     *   Dr Transport Revenue / Cr AR  (i.e., Cr AR / Dr Transport Revenue swapped)
   577→   577→     *
   578→   578→     * @return int|null journal_entry_id (null if no transport_revenue ledger configured)
   579→   579→     */
   580→   580→    private function postTransportAdjustmentGL(
   581→   581→        int $invoiceId, string $challanCode, string $challanDate, int $branchId,
   582→   582→        float $adjustment, int $customerId, ?int $createdBy
   583→   583→    ): ?int {
   584→   584→        $amount = abs($adjustment);
   585→   585→        if ($amount < 0.01) {
   586→   586→            return null;
   587→   587→        }
   588→   588→
   589→   589→        $arLedgerId = $this->journalPosting->lookupLedgerByNature('ar');
   590→   590→        if (!$arLedgerId) {
   591→   591→            throw new \RuntimeException('Accounts Receivable ledger not found (nature: ar).');
   592→   592→        }
   593→   593→
   594→   594→        $transportLedgerId = $this->journalPosting->lookupLedgerByNature('transport_revenue');
   595→   595→        if (!$transportLedgerId) {
   596→   596→            // Fall back to sales_revenue if transport_revenue not configured.
   597→   597→            $transportLedgerId = $this->journalPosting->lookupLedgerByNature('sales_revenue');
   598→   598→        }
   599→   599→        if (!$transportLedgerId) {
   600→   600→            throw new \RuntimeException('Transport Revenue / Sales Revenue ledger not found.');
   601→   601→        }
   602→   602→
   603→   603→        // Positive adjustment: Dr AR / Cr Transport Revenue
   604→   604→        // Negative adjustment: Dr Transport Revenue / Cr AR (swapped)
   605→   605→        $isIncrease = $adjustment > 0;
   606→   606→
   607→   607→        $lines = [
   608→   608→            [
   609→   609→                'ledger_id' => $arLedgerId,
   610→   610→                'debit' => $isIncrease ? $amount : 0,
   611→   611→                'credit' => $isIncrease ? 0 : $amount,
   612→   612→                'entity_type' => 'customer', 'entity_id' => $customerId,
   613→   613→                'memo' => 'Challan ' . $challanCode . ' — transport adjustment AR',
   614→   614→            ],
   615→   615→            [
   616→   616→                'ledger_id' => $transportLedgerId,
   617→   617→                'debit' => $isIncrease ? 0 : $amount,
   618→   618→                'credit' => $isIncrease ? $amount : 0,
   619→   619→                'entity_type' => 'sales_invoice', 'entity_id' => $invoiceId,
   620→   620→                'memo' => 'Challan ' . $challanCode . ' — transport adjustment revenue',
   621→   621→            ],
   622→   622→        ];
   623→   623→
   624→   624→        return $this->journalPosting->createJournalEntry([
   625→   625→            'entry_date' => $challanDate,
   626→   626→            'reference_type' => 'sales_invoice',
   627→   627→            'reference_id' => $invoiceId,
   628→   628→            'branch_id' => $branchId,
   629→   629→            'description' => 'Transport adjustment — Challan ' . $challanCode,
   630→   630→            'source' => 'sales_challan',
   631→   631→            'created_by' => $createdBy,
   632→   632→        ], $lines);
   633→   633→    }
