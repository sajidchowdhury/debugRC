   117→   117→   117→                SET sr.journal_entry_id = je.id

   119→   119→   119→                  AND COALESCE(sr.is_reversed, 0) = 0

   192→   192→   192→        } catch (Exception $e) {

   211→   211→   211→    private function insertCustomerLedgerForReturn(array $sr): void
   212→   212→   212→    {
   213→   213→   213→        $customerId = (int)($sr['customer_id'] ?? 0);
   214→   214→   214→        $returnId   = (int)($sr['id'] ?? 0);
   215→   215→   215→        $amount     = round((float)($sr['total_amount'] ?? 0), 2);

   217→   217→   217→            return;
   218→   218→   218→        }
   219→   219→   219→
   220→   220→   220→        $prev = $this->getCustomerRunningBalance($customerId);

   222→   222→   222→        $txnDate = $sr['return_date'] ?? date('Y-m-d');
   223→   223→   223→
   224→   224→   224→        $branchId = (int)($sr['branch_id'] ?? 0);
   225→   225→   225→        $this->insertCustomerLedgerEntry([
   226→   226→   226→            'customer_id'      => $customerId,
   227→   227→   227→            'reference_type'   => 'sales_return',
   228→   228→   228→            'reference_id'     => $returnId,
   229→   229→   229→            'debit'            => 0,
   230→   230→   230→            'credit'           => $amount,
   231→   231→   231→            'running_balance'  => $new,
   232→   232→   232→            'branch_id'        => $branchId,
   233→   233→   233→            'transaction_date' => $txnDate,
   234→   234→   234→            'remarks'          => 'Sales Return #' . ($sr['return_code'] ?? $returnId),
   235→   235→   235→            'created_by'       => 1,
   236→   236→   236→        ]);
   237→   237→   237→    }
