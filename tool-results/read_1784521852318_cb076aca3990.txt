    31→    public function ProductStockAnalysis()
    32→    {
    33→        $a = new Helper();
    34→
    35→        $data['branches']   = $a->Get_All_Active_Branches();
    36→        $data['categories'] = $a->Get_Active_Categories();
    37→        $data['products']   = $a->Get_All_Active_Product();
    38→        $data['warehouses'] = $a->Get_All_Active_Warehouses();
    39→
    40→        // Support both GET (initial) and POST (AJAX) - POST takes precedence for large filter sets
    41→        $input = array_merge($_GET, $_POST);
    42→
    43→        $range = $this->resolveReportDates(
    44→            $input['from_date'] ?? null,
    45→            $input['to_date'] ?? null,
    46→            date('Y-m-d', strtotime('-30 days')),
    47→            date('Y-m-d')
    48→        );
    49→        $from_date = $range['from'];
    50→        $to_date = $range['to'];
    51→
    52→        $branch_ids = [];
    53→        if (isset($input['branch_ids'])) {
    54→            $branch_ids = is_array($input['branch_ids']) ? $input['branch_ids'] : explode(',', $input['branch_ids']);
    55→            $branch_ids = array_filter(array_map('intval', $branch_ids));
    56→        }
    57→
    58→        $warehouse_ids = [];
    59→        if (isset($input['warehouse_ids'])) {
    60→            $warehouse_ids = is_array($input['warehouse_ids']) ? $input['warehouse_ids'] : explode(',', $input['warehouse_ids']);
    61→            $warehouse_ids = array_filter(array_map('intval', $warehouse_ids));
    62→        }
    63→
    64→        $category_ids = [];
    65→        if (isset($input['category_ids'])) {
    66→            $category_ids = is_array($input['category_ids']) ? $input['category_ids'] : explode(',', $input['category_ids']);
    67→            $category_ids = array_filter(array_map('intval', $category_ids));
    68→        }
    69→
    70→        $product_ids = [];
    71→        if (isset($input['product_ids'])) {
    72→            $product_ids = is_array($input['product_ids']) ? $input['product_ids'] : explode(',', $input['product_ids']);
    73→            $product_ids = array_filter(array_map('intval', $product_ids));
    74→        }
    75→
    76→        // Pass to view for sticky filters (used in offcanvas)
    77→        $data['from_date']     = $from_date;
    78→        $data['to_date']       = $to_date;
    79→        $this->attachReportPeriodMeta($data, $range);
    80→        $data['branch_ids']    = $branch_ids;
    81→        $data['warehouse_ids'] = $warehouse_ids;
    82→        $data['category_ids']  = $category_ids;
    83→        $data['product_ids']   = $product_ids;
    84→
    85→        $isAjax = !empty($_SERVER['HTTP_X_REQUESTED_WITH']) && 
    86→                  strtolower($_SERVER['HTTP_X_REQUESTED_WITH']) === 'xmlhttprequest';
    87→
    88→        $searched = isset($input['search']) || isset($input['export']);
    89→
    90→        if ($searched) {
    91→            require_once '../app/models/Reports/ProductStockAnalysisReport.php';
    92→            $report = new ProductStockAnalysisReport();
    93→
    94→            $result = $report->getStockAnalysis(
    95→                $from_date, 
    96→                $to_date, 
    97→                $branch_ids, 
    98→                $warehouse_ids, 
    99→                $category_ids, 
   100→                $product_ids
   101→            );
   102→
   103→            if (isset($input['export'])) {
   104→                $report->exportStockAnalysis($result, $from_date, $to_date);
   105→                return; // already exited inside export usually
   106→            }
   107→
   108→            if ($isAjax || isset($input['ajax'])) {
   109→                // Compute summary for KPIs
   110→                $total = count($result);
   111→                $sumOpen = 0; $sumRec = 0; $sumIss = 0; $sumVal = 0;
   112→                foreach ($result as $r) {
   113→                    $sumOpen += (float)($r['opening_qty'] ?? 0);
   114→                    $sumRec  += (float)($r['receipt_qty'] ?? 0);
   115→                    $sumIss  += (float)($r['issue_qty'] ?? 0);
   116→                    $sumVal  += (float)($r['closing_value'] ?? 0);
   117→                }
   118→
   119→                header('Content-Type: application/json');
   120→                echo json_encode([
   121→                    'status' => 'success',
   122→                    'data' => $result,
   123→                    'summary' => [
   124→                        'total_rows'   => $total,
   125→                        'sum_opening'  => round($sumOpen, 2),
   126→                        'sum_receipt'  => round($sumRec, 2),
   127→                        'sum_issue'    => round($sumIss, 2),
   128→                        'sum_value'    => round($sumVal, 2),
   129→                        'net_movement' => round($sumRec - $sumIss, 2)
   130→                    ],
   131→                    'count' => $total
   132→                ]);
   133→                exit;
   134→            }
   135→
   136→            $data['stock_data'] = $result;
   137→        }
   138→
   139→        $data['searched'] = $searched;
   140→        $this->view('report/ProductStockAnalysis', $data);
   141→    }

   363→/**
   364→ * Profit & Loss — Income − Expense by ledger_nature groups (Phase 7A).
   365→ */
   366→public function ProfitAndLoss()
   367→{
   368→    require_once '../app/models/Reports/ProfitAndLossReport.php';
   369→    $helper = new Helper();
   370→    $range = $this->resolveReportDates(
   371→        $_GET['from_date'] ?? null,
   372→        $_GET['to_date'] ?? null,
   373→        date('Y-m-01'),
   374→        date('Y-m-d')
   375→    );
   376→    $from_date = $range['from'];
   377→    $to_date = $range['to'];
   378→
   379→    $branch_id = isset($_GET['branch_id']) && $_GET['branch_id'] !== ''
   380→        ? (int)$_GET['branch_id']
   381→        : null;
   382→
   383→    if (!$helper->canOverrideBranch()) {
   384→        $branch_id = Helper::sessionBranchId() ?: null;
   385→    }
   386→
   387→    $include_zero = !empty($_GET['include_zero']);
   388→    $compare_prior = !empty($_GET['compare_prior']);
   389→    $compare_from = trim((string)($_GET['compare_from_date'] ?? ''));
   390→    $compare_to = trim((string)($_GET['compare_to_date'] ?? ''));
   391→
   392→    if ($compare_prior) {
   393→        $prior = ProfitAndLossReport::resolvePriorPeriod($from_date, $to_date);
   394→        if ($prior) {
   395→            $compare_from = $prior['from'];
   396→            $compare_to = $prior['to'];
   397→        }
   398→    }
   399→
   400→    $hasCompare = $compare_from !== '' && $compare_to !== ''
   401→        && preg_match('/^\d{4}-\d{2}-\d{2}$/', $compare_from)
   402→        && preg_match('/^\d{4}-\d{2}-\d{2}$/', $compare_to);
   403→
   404→    $data = [
   405→        'title'               => 'Profit & Loss',
   406→        'from_date'           => $from_date,
   407→        'to_date'             => $to_date,
   408→        'branch_id'           => $branch_id,
   409→        'include_zero'        => $include_zero,
   410→        'compare_prior'       => $compare_prior,
   411→        'compare_from_date'   => $hasCompare ? $compare_from : '',
   412→        'compare_to_date'     => $hasCompare ? $compare_to : '',
   413→        'branches'            => $helper->Get_All_Active_Branches(),
   414→        'can_override_branch' => $helper->canOverrideBranch(),
   415→    ];
   416→    $this->attachReportPeriodMeta($data, $range);
   417→
   418→    if (isset($_GET['search']) || isset($_GET['export'])) {
   419→        $report = new ProfitAndLossReport();
   420→        $result = $report->getProfitAndLoss(
   421→            $from_date,
   422→            $to_date,
   423→            $branch_id,
   424→            $include_zero,
   425→            $hasCompare ? $compare_from : null,
   426→            $hasCompare ? $compare_to : null
   427→        );
   428→
   429→        $export = $_GET['export'] ?? '';
   430→        if ($export === 'csv' || $export === '1') {
   431→            $report->exportToCsv($result, 'Profit_And_Loss');
   432→        }
   433→        if ($export === 'pdf') {
   434→            $report->exportToPdfHtml($result);
   435→        }
   436→
   437→        $data['profit_and_loss'] = $result;
   438→    }
   439→
   440→    $this->view('report/ProfitAndLoss', $data);
   441→}

  1044→    private function computePeriodRevenue($from, $to, $branchIds = [], $salesmanIds = [], $catIds = [])
  1045→    {
  1046→        $db = (new SalesModel())->getDatabase();
  1047→        $params = [':from' => $from, ':to' => $to];
  1048→        $where = "si.status = 'challan_completed' AND COALESCE(si.is_reversed,0) = 0 AND si.invoice_date BETWEEN :from AND :to";
  1049→
  1050→        if (!empty($branchIds)) {
  1051→            $ph = []; foreach ($branchIds as $i => $id) { $k=":b$i"; $ph[]=$k; $params[$k]=$id; }
  1052→            $where .= " AND si.branch_id IN (" . implode(',', $ph) . ")";
  1053→        }
  1054→        if (!empty($salesmanIds)) {
  1055→            $ph = []; foreach ($salesmanIds as $i => $id) { $k=":s$i"; $ph[]=$k; $params[$k]=$id; }
  1056→            $where .= " AND (si.salesman_id IN (" . implode(',', $ph) . ") OR si.sales_person IN (" . implode(',', $ph) . "))";
  1057→        }
  1058→        if (!empty($catIds)) {
  1059→            $ph = []; foreach ($catIds as $i => $id) { $k=":c$i"; $ph[]=$k; $params[$k]=$id; }
  1060→            $where .= " AND EXISTS (SELECT 1 FROM sales_invoice_items sii JOIN products p ON p.id = sii.product_id WHERE sii.sales_invoice_id = si.id AND p.category_id IN (" . implode(',', $ph) . ")) ";
