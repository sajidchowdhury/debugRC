    14→    14→/**
    15→    15→ * Sales Return Service — Phase 8.5.
    16→    16→ *
    17→    17→ * Two-phase: create → confirm → reverse.
    18→    18→ *
    19→    19→ * CRITICAL CORRECTNESS (per avg_cost_rule.md §3):
    20→    20→ *   Stock comes back IN at the ORIGINAL avg_cost from the challan's stock_transaction
    21→    21→ *   (NOT current avg_cost). This ensures:
    22→    22→ *   1. The COGS reversal matches the original COGS exactly.
    23→    23→ *   2. The avg_cost is restored to its pre-sale value.
    24→    24→ *
    25→    25→ * The original_cost is looked up from the stock_transaction for the challan
    26→    26→ * (reference_type='sales_challan', reference_id=challan_id, product_id).
    27→    27→ * This is snapshotted into sales_return_items.original_cost.
    28→    28→ *
    29→    29→ * On confirm (4 operations, all atomic):
    30→    30→ *   1. Stock IN via StockService at ORIGINAL avg_cost (ref='sales_return')
    31→    31→ *      → avg_cost recalculated (IN rule: weighted average with original cost)
    32→    32→ *   2. GL Revenue Reversal: Dr Sales Return / Cr AR (at sales rate)
    33→    33→ *   3. GL COGS Reversal: Dr Inventory / Cr COGS (at original avg_cost)
    34→    34→ *   4. Customer ledger: credit entry (customer owes less)
    35→    35→ */
    36→    36→class SalesReturnService
    37→    37→{
    38→    38→    public function __construct(
    39→    39→        private StockService $stockService,
    40→    40→        private JournalPostingService $journalPosting,
    41→    41→        private SubLedgerService $subLedger,
    42→    42→        private SalesAccess $salesAccess,
    43→    43→        private SalesAuditLogger $auditLogger,
    44→    44→        private DamageService $damageService,
    45→    45→        private NotificationService $notifications
    46→    46→    ) {}
    47→    47→
    48→    48→    /**
    49→    49→     * Phase 1: Create a sales return (no stock, no GL).
    50→    50→     * Looks up ORIGINAL avg_cost from the challan's stock_transaction.
    51→    51→     *
    52→    52→     * @param array $data {
    53→    53→     *     sales_invoice_id: int,
    54→    54→     *     return_date: string (Y-m-d),
    55→    55→     *     reason: string|null,
    56→    56→     *     created_by: int,
    57→    57→     *     items: array each { product_id, warehouse_id, qty, rate, sales_invoice_item_id }
    58→    58→     * }
    59→    59→     * @return SalesReturn
    60→    60→     */
    61→    61→    public function createReturn(array $data): SalesReturn
    62→    62→    {
    63→    63→        $invoiceId = (int) ($data['sales_invoice_id'] ?? 0);
    64→    64→        if ($invoiceId <= 0) {
    65→    65→            throw new \InvalidArgumentException('sales_invoice_id is required.');
    66→    66→        }
    67→    67→
    68→    68→        $invoice = DB::table('sales_invoices')->where('id', $invoiceId)->first();
    69→    69→        if (!$invoice) {
    70→    70→            throw new \InvalidArgumentException("Invoice {$invoiceId} not found.");
    71→    71→        }
    72→    72→        if ($invoice->is_reversed) {
    73→    73→            throw new \RuntimeException("Cannot return against a reversed invoice.");
    74→    74→        }
    75→    75→
    76→    76→        // P0-8: Defense-in-depth branch isolation check.
    77→    77→        $this->salesAccess->assertBranchAccessible((int) $invoice->branch_id);
    78→    78→
    79→    79→        // Find the challan for this invoice.
    80→    80→        $challan = DB::table('sales_challans')
    81→    81→            ->where('sales_invoice_id', $invoiceId)
    82→    82→            ->where('is_reversed', false)
    83→    83→            ->first();
    84→    84→        if (!$challan) {
    85→    85→            throw new \RuntimeException("No active challan found for invoice {$invoiceId}. Returns require a completed challan.");
    86→    86→        }
    87→    87→
    88→    88→        $items = $this->validateItems($data['items'], $invoiceId, $challan->id);
    89→    89→        $totalRevenue = collect($items)->sum(fn($i) => $i['qty'] * $i['rate']);
    90→    90→        $totalCogs = collect($items)->sum(fn($i) => $i['qty'] * $i['original_cost']);
    91→    91→
    92→    92→        $returnCode = $this->generateReturnCode();
    93→    93→        $customerId = (int) $invoice->customer_id;
    94→    94→        $branchId = (int) $invoice->branch_id;
    95→    95→
    96→    96→        return DB::transaction(function () use (
    97→    97→            $data, $items, $totalRevenue, $totalCogs, $returnCode, $invoiceId, $customerId, $branchId
    98→    98→        ) {
    99→    99→            $returnId = DB::table('sales_returns')->insertGetId([
   100→   100→                'return_code' => $returnCode,
   101→   101→                'return_date' => $data['return_date'] ?? now()->format('Y-m-d'),
   102→   102→                'sales_invoice_id' => $invoiceId,
   103→   103→                'customer_id' => $customerId,
   104→   104→                'branch_id' => $branchId,
   105→   105→                'total_amount' => round($totalRevenue, 2),
   106→   106→                'cogs_amount' => round($totalCogs, 2),
   107→   107→                'status' => 'created',
   108→   108→                'is_reversed' => false,
   109→   109→                'reason' => $data['reason'] ?? null,
   110→   110→                'created_by' => $data['created_by'] ?? null,
   111→   111→                'created_at' => now(),
   112→   112→                'updated_at' => now(),
   113→   113→            ]);
   114→   114→
   115→   115→            $itemRows = [];
   116→   116→            foreach ($items as $item) {
   117→   117→                $itemRows[] = [
   118→   118→                    'sales_return_id' => $returnId,
   119→   119→                    'sales_invoice_item_id' => $item['sales_invoice_item_id'],
   120→   120→                    'product_id' => $item['product_id'],
   121→   121→                    'warehouse_id' => $item['warehouse_id'],
   122→   122→                    'qty' => $item['qty'],
   123→   123→                    'rate' => $item['rate'],
   124→   124→                    'original_cost' => $item['original_cost'],
   125→   125→                ];
   126→   126→            }
   127→   127→            DB::table('sales_return_items')->insert($itemRows);
   128→   128→
   129→   129→            // P1-3: Audit log — return_created.
   130→   130→            $this->auditLogger->returnCreated(
   131→   131→                $data['created_by'] ?? auth()->id() ?? 0,
   132→   132→                $returnId, $returnCode, $invoiceId, $customerId, $branchId,
   133→   133→                $totalRevenue, $totalCogs
   134→   134→            );
   135→   135→
   136→   136→            // P1-7: Notify admins that a return was created.
   137→   137→            $this->notifications->dispatch(
   138→   138→                'return_created',
   139→   139→                "Return {$returnCode} created for Tk " . number_format($totalRevenue, 2)
   140→   140→                . " against invoice #{$invoiceId}.",
   141→   141→                'sales_return',
   142→   142→                $returnId
   143→   143→            );
   144→   144→
   145→   145→            return SalesReturn::with(['items.product', 'salesInvoice', 'customer', 'branch'])->find($returnId);
   146→   146→        });
   147→   147→    }
   148→   148→
   149→   149→    /**
   150→   150→     * Phase 2: Confirm a sales return — stock IN + GL + customer_ledger.
   151→   151→     *
   152→   152→     * @param int $returnId
   153→   153→     * @param int $confirmedBy
   154→   154→     * @return SalesReturn
   155→   155→     * @throws \RuntimeException If not 'created' status, or stock/GL fails.
   156→   156→     */
   157→   157→    public function confirmReturn(int $returnId, int $confirmedBy): SalesReturn
   158→   158→    {
   159→   159→        return DB::transaction(function () use ($returnId, $confirmedBy) {
   160→   160→            $return = SalesReturn::with('items')->lockForUpdate()->find($returnId);
   161→   161→
   162→   162→            if (!$return) {
   163→   163→                throw new \RuntimeException("Return {$returnId} not found.");
   164→   164→            }
   165→   165→            if (!$return->isCreated()) {
   166→   166→                throw new \RuntimeException("Only 'created' returns can be confirmed (current: {$return->status}).");
   167→   167→            }
   168→   168→
   169→   169→            $returnDate = $return->return_date->format('Y-m-d');
   170→   170→
   171→   171→            // 1. Stock IN at ORIGINAL avg_cost (CRITICAL: not current avg_cost).
   172→   172→            foreach ($return->items as $item) {
   173→   173→                $this->stockService->applyTransaction([
   174→   174→                    'warehouse_id' => $item->warehouse_id,
   175→   175→                    'product_id' => $item->product_id,
   176→   176→                    'qty' => (float) $item->qty, // positive = IN
   177→   177→                    'rate' => (float) $item->original_cost, // ORIGINAL avg_cost from challan
   178→   178→                    'reference_type' => 'sales_return',
   179→   179→                    'reference_id' => $return->id,
   180→   180→                    'notes' => 'Sales Return ' . $return->return_code . ' — stock restored at original cost',
   181→   181→                    'transaction_date' => $returnDate,
   182→   182→                    'created_by' => $confirmedBy,
   183→   183→                ]);
   184→   184→            }
   185→   185→
   186→   186→            // 1b. P1-5: Linked damage write-off for 'Damage' condition items.
   187→   187→            // Legacy: SalesReturnModel::createLinkedDamageWriteOff — for each
   188→   188→            // Damage-condition item, create a damage_invoices row + stock OUT
   189→   189→            // (immediately back OUT the damaged goods) + GL Dr damage_loss / Cr inventory.
   190→   190→            $linkedDamageIds = $this->createLinkedDamageWriteOffs($return, $confirmedBy);
   191→   191→
   192→   192→            // 2. GL Revenue Reversal: Dr Sales Return / Cr AR.
   193→   193→            $journalEntryId = $this->postRevenueReversalGL($return, $confirmedBy);
   194→   194→
   195→   195→            // 3. GL COGS Reversal: Dr Inventory / Cr COGS.
   196→   196→            $cogsJournalEntryId = $this->postCogsReversalGL($return, $confirmedBy);
   197→   197→
   198→   198→            // 4. Customer ledger via SubLedgerService: credit entry (customer owes less).
   199→   199→            $this->subLedger->postCustomerLedgerEntry([
   200→   200→                'customer_id' => $return->customer_id,
   201→   201→                'branch_id' => $return->branch_id,
   202→   202→                'transaction_date' => $return->return_date->format('Y-m-d'),
   203→   203→                'transaction_type' => 'sales_return',
   204→   204→                'reference_type' => 'sales_return',
   205→   205→                'reference_id' => $return->id,
   206→   206→                'debit' => 0,
   207→   207→                'credit' => (float) $return->total_amount,
   208→   208→                'description' => 'Sales Return ' . $return->return_code . ($return->reason ? ' — ' . $return->reason : ''),
   209→   209→                'journal_entry_id' => $journalEntryId,
   210→   210→                'created_by' => $confirmedBy,
   211→   211→            ]);
   212→   212→
   213→   213→            // 5. Update return status.
   214→   214→            DB::table('sales_returns')
   215→   215→                ->where('id', $returnId)
   216→   216→                ->update([
   217→   217→                    'status' => 'confirmed',
   218→   218→                    'journal_entry_id' => $journalEntryId,
   219→   219→                    'cogs_journal_entry_id' => $cogsJournalEntryId,
   220→   220→                    'updated_at' => now(),
   221→   221→                ]);
   222→   222→
   223→   223→            // P1-3: Audit log — return_confirmed.
   224→   224→            $this->auditLogger->returnConfirmed(
   225→   225→                $confirmedBy,
   226→   226→                $returnId, $return->return_code, (int) $return->branch_id,
   227→   227→                (float) $return->total_amount, (float) $return->cogs_amount,
   228→   228→                $journalEntryId
   229→   229→            );
   230→   230→
   231→   231→            // P1-7: Notify warehouse managers + accountants that a return was confirmed.
   232→   232→            $this->notifications->dispatch(
   233→   233→                'return_confirmed',
   234→   234→                "Return {$return->return_code} confirmed — stock restored, GL posted. "
   235→   235→                . "Total: Tk " . number_format((float) $return->total_amount, 2)
   236→   236→                . ", COGS reversed: Tk " . number_format((float) $return->cogs_amount, 2) . ".",
   237→   237→                'sales_return',
   238→   238→                $returnId
   239→   239→            );
   240→   240→
   241→   241→            return SalesReturn::with([
   242→   242→                'items.product', 'salesInvoice', 'customer', 'branch',
   243→   243→                'journalEntry.lines.ledger', 'cogsJournalEntry.lines.ledger',
   244→   244→            ])->find($returnId);
   245→   245→        });
   246→   246→    }
   247→   247→
   248→   248→    /**
   249→   249→     * Phase 3: Reverse a confirmed return.
   250→   250→     * Reverses stock + both GL journals + customer_ledger.
   251→   251→     *
   252→   252→     * @param int $returnId
   253→   253→     * @param int $reversedBy
   254→   254→     * @param string $reason
   255→   255→     * @return SalesReturn
   256→   256→     */
   257→   257→    public function reverseReturn(int $returnId, int $reversedBy, string $reason = ''): SalesReturn
   258→   258→    {
   259→   259→        return DB::transaction(function () use ($returnId, $reversedBy, $reason) {
   260→   260→            $return = SalesReturn::with('items')->lockForUpdate()->find($returnId);
   261→   261→
   262→   262→            if (!$return) {
   263→   263→                throw new \RuntimeException("Return {$returnId} not found.");
   264→   264→            }
   265→   265→            if (!$return->isConfirmed()) {
   266→   266→                throw new \RuntimeException("Only confirmed returns can be reversed (current: {$return->status}).");
   267→   267→            }
   268→   268→
   269→   269→            // Reverse both GL journals.
   270→   270→            if ($return->journal_entry_id) {
   271→   271→                $this->journalPosting->reverseJournalEntry(
   272→   272→                    $return->journal_entry_id, $reversedBy,
   273→   273→                    "Return reversed: {$reason}"
   274→   274→                );
   275→   275→            }
   276→   276→            if ($return->cogs_journal_entry_id) {
   277→   277→                $this->journalPosting->reverseJournalEntry(
   278→   278→                    $return->cogs_journal_entry_id, $reversedBy,
   279→   279→                    "Return reversed: {$reason}"
   280→   280→                );
   281→   281→            }
   282→   282→
   283→   283→            // Reverse customer_ledger via SubLedgerService (debit entry to restore what customer owes).
   284→   284→            $this->subLedger->postCustomerLedgerEntry([
   285→   285→                'customer_id' => $return->customer_id,
   286→   286→                'branch_id' => $return->branch_id,
   287→   287→                'transaction_date' => now()->format('Y-m-d'),
   288→   288→                'transaction_type' => 'sales_return_reversal',
   289→   289→                'reference_type' => 'sales_return',
   290→   290→                'reference_id' => $return->id,
   291→   291→                'debit' => (float) $return->total_amount,
   292→   292→                'credit' => 0,
   293→   293→                'description' => 'Return reversal ' . $return->return_code . ": {$reason}",
   294→   294→                'journal_entry_id' => $return->journal_entry_id,
   295→   295→                'created_by' => $reversedBy,
   296→   296→            ]);
   297→   297→
   298→   298→            // P1-5: Reverse linked damage write-offs FIRST (before return stock reversal).
   299→   299→            // Each linked damage_invoice has its own stock OUT + GL that must be reversed.
   300→   300→            $this->reverseLinkedDamageForReturn($returnId, $reversedBy, $reason);
   301→   301→
   302→   302→            // Reverse each stock movement (stock OUT at original cost — append-only reversal).
   303→   303→            $stockTxs = DB::table('stock_transactions')
   304→   304→                ->where('reference_type', 'sales_return')
   305→   305→                ->where('reference_id', $returnId)
   306→   306→                ->where('is_reversed', false)
   307→   307→                ->get();
   308→   308→
   309→   309→            foreach ($stockTxs as $tx) {
   310→   310→                $this->stockService->reverseTransaction(
   311→   311→                    $tx->id, $reversedBy,
   312→   312→                    "Return reversed: {$reason}"
   313→   313→                );
   314→   314→            }
   315→   315→
   316→   316→            // Mark return as reversed.
   317→   317→            DB::table('sales_returns')
   318→   318→                ->where('id', $returnId)
   319→   319→                ->update([
   320→   320→                    'status' => 'reversed',
   321→   321→                    'is_reversed' => true,
   322→   322→                    'reversed_at' => now(),
   323→   323→                    'reversed_by' => $reversedBy,
   324→   324→                    'reverse_reason' => $reason,
   325→   325→                    'updated_at' => now(),
   326→   326→                ]);
   327→   327→
   328→   328→            // P1-3: Audit log — return_reversed.
   329→   329→            $this->auditLogger->returnReversed(
   330→   330→                $reversedBy,
   331→   331→                $returnId, $return->return_code, (int) $return->branch_id,
   332→   332→                (float) $return->total_amount, $reason
   333→   333→            );
   334→   334→
   335→   335→            // P1-7: Notify accountants + admins that a return was reversed.
   336→   336→            $this->notifications->dispatch(
   337→   337→                'return_reversed',
   338→   338→                "Return {$return->return_code} reversed — stock + GL + ledger rolled back. "
   339→   339→                . "Reason: {$reason}",
   340→   340→                'sales_return',
   341→   341→                $returnId
   342→   342→            );
   343→   343→
   344→   344→            return SalesReturn::find($returnId);
   345→   345→        });
   346→   346→    }

   348→   348→    /**
   349→   349→     * Post GL Revenue Reversal: Dr Sales Return / Cr AR.
   350→   350→     *
   351→   351→     * @return int journal_entry_id
   352→   352→     */
   353→   353→    private function postRevenueReversalGL(SalesReturn $return, int $createdBy): int
   354→   354→    {
   355→   355→        $amount = (float) $return->total_amount;
   356→   356→        if ($amount < 0.01) return 0;
   357→   357→
   358→   358→        $returnLedgerId = $this->journalPosting->lookupLedgerByNature('sales_return');
   359→   359→        $arLedgerId = $this->journalPosting->lookupLedgerByNature('ar');
   360→   360→
   361→   361→        if (!$returnLedgerId) {
   362→   362→            throw new \RuntimeException('Sales Return ledger not found (nature: sales_return).');
   363→   363→        }
   364→   364→        if (!$arLedgerId) {
   365→   365→            throw new \RuntimeException('Accounts Receivable ledger not found (nature: ar).');
   366→   366→        }
   367→   367→
   368→   368→        return $this->journalPosting->createJournalEntry([
   369→   369→            'entry_date' => $return->return_date->format('Y-m-d'),
   370→   370→            'reference_type' => 'sales_return',
   371→   371→            'reference_id' => $return->id,
   372→   372→            'branch_id' => $return->branch_id,
   373→   373→            'description' => 'Sales Return Revenue Reversal ' . $return->return_code,
   374→   374→            'source' => 'sales_return',
   375→   375→            'created_by' => $createdBy,
   376→   376→        ], [
   377→   377→            [
   378→   378→                'ledger_id' => $returnLedgerId,
   379→   379→                'debit' => $amount, 'credit' => 0,
   380→   380→                'entity_type' => 'sales_return', 'entity_id' => $return->id,
   381→   381→                'memo' => 'Return ' . $return->return_code . ' — revenue reversal',
   382→   382→            ],
   383→   383→            [
   384→   384→                'ledger_id' => $arLedgerId,
   385→   385→                'debit' => 0, 'credit' => $amount,
   386→   386→                'entity_type' => 'customer', 'entity_id' => $return->customer_id,
   387→   387→                'memo' => 'Return ' . $return->return_code . ' — AR credit',
   388→   388→            ],
   389→   389→        ]);
   390→   390→    }
   391→   391→
   392→   392→    /**
   393→   393→     * Post GL COGS Reversal: Dr Inventory / Cr COGS (at ORIGINAL avg_cost).
   394→   394→     *
   395→   395→     * @return int journal_entry_id

   397→   397→    private function postCogsReversalGL(SalesReturn $return, int $createdBy): int

   401→   401→
   402→   402→        $inventoryLedgerId = $this->journalPosting->lookupLedgerByNature('inventory');
   403→   403→        $cogsLedgerId = $this->journalPosting->lookupLedgerByNature('cogs');
   404→   404→

   408→   408→        if (!$cogsLedgerId) {
   409→   409→            throw new \RuntimeException('COGS ledger not found (nature: cogs).');
   410→   410→        }
   411→   411→
   412→   412→        return $this->journalPosting->createJournalEntry([
   413→   413→            'entry_date' => $return->return_date->format('Y-m-d'),
   414→   414→            'reference_type' => 'sales_return',
   415→   415→            'reference_id' => $return->id,
   416→   416→            'branch_id' => $return->branch_id,
   417→   417→            'description' => 'Sales Return COGS Reversal ' . $return->return_code,
   418→   418→            'source' => 'sales_return',
   419→   419→            'created_by' => $createdBy,
   420→   420→        ], [
   421→   421→            [
   422→   422→                'ledger_id' => $inventoryLedgerId,
   423→   423→                'debit' => $cogsAmount, 'credit' => 0,
   424→   424→                'entity_type' => 'sales_return', 'entity_id' => $return->id,
   425→   425→                'memo' => 'Return ' . $return->return_code . ' — inventory restored',
   426→   426→            ],
   427→   427→            [
   428→   428→                'ledger_id' => $cogsLedgerId,
   429→   429→                'debit' => 0, 'credit' => $cogsAmount,
   430→   430→                'entity_type' => 'sales_return', 'entity_id' => $return->id,
   431→   431→                'memo' => 'Return ' . $return->return_code . ' — COGS reversal',
   432→   432→            ],
   433→   433→        ]);
   434→   434→    }

   436→   436→    /**
   437→   437→     * Generate atomic return code: SR-YYYYMMDD-NNNN.
   438→   438→     */
   439→   439→    private function generateReturnCode(): string
   440→   440→    {
   441→   441→        $datePart = now()->format('Ymd');
   442→   442→        $periodKey = now()->format('Y-m');
   443→   443→        $docType = 'sales_return';
   444→   444→
   445→   445→        return DB::transaction(function () use ($docType, $periodKey, $datePart) {
   446→   446→            $seqRow = DB::table('document_sequences')
   447→   447→                ->where('doc_type', $docType)
   448→   448→                ->where('branch_id', 0)
   449→   449→                ->where('period_key', $periodKey)
   450→   450→                ->lockForUpdate()
   451→   451→                ->first();
   452→   452→
   453→   453→            $nextNumber = $seqRow ? ((int) $seqRow->last_number + 1) : 1;
   454→   454→
   455→   455→            if ($seqRow) {
   456→   456→                DB::table('document_sequences')->where('id', $seqRow->id)
   457→   457→                    ->update(['last_number' => $nextNumber, 'updated_at' => now()]);
   458→   458→            } else {
   459→   459→                DB::table('document_sequences')->insert([
   460→   460→                    'doc_type' => $docType, 'branch_id' => 0,
   461→   461→                    'period_key' => $periodKey, 'last_number' => $nextNumber,
   462→   462→                    'updated_at' => now(),
   463→   463→                ]);
   464→   464→            }
   465→   465→
   466→   466→            return "SR-{$datePart}-" . str_pad((string) $nextNumber, 4, '0', STR_PAD_LEFT);
   467→   467→        });
   468→   468→    }

   470→   470→    /**
   471→   471→     * Validate items — look up ORIGINAL avg_cost from the challan's stock_transaction.
   472→   472→     *
   473→   473→     * @param array $items
   474→   474→     * @param int $invoiceId
   475→   475→     * @param int $challanId
   476→   476→     * @return array
   477→   477→     */
   478→   478→    private function validateItems(array $items, int $invoiceId, int $challanId): array
   479→   479→    {
   480→   480→        $validated = [];
   481→   481→
   482→   482→        foreach ($items as $item) {
   483→   483→            $productId = (int) ($item['product_id'] ?? 0);
   484→   484→            $qty = (float) ($item['qty'] ?? 0);
   485→   485→            $warehouseId = (int) ($item['warehouse_id'] ?? 0);
   486→   486→            $invoiceItemId = (int) ($item['sales_invoice_item_id'] ?? 0);
   487→   487→
   488→   488→            if ($productId <= 0 || $qty <= 0 || $warehouseId <= 0) continue;
   489→   489→
   490→   490→            // Check returnable_qty (invoice item qty - already returned).
   491→   491→            if ($invoiceItemId > 0) {
   492→   492→                $invoiceItem = DB::table('sales_invoice_items')->where('id', $invoiceItemId)->first();
   493→   493→                if ($invoiceItem) {
   494→   494→                    $alreadyReturned = (float) DB::table('sales_return_items as sri')
   495→   495→                        ->join('sales_returns as sr', 'sr.id', '=', 'sri.sales_return_id')
   496→   496→                        ->where('sri.sales_invoice_item_id', $invoiceItemId)
   497→   497→                        ->whereIn('sr.status', ['created', 'confirmed'])
   498→   498→                        ->where('sr.is_reversed', false)
   499→   499→                        ->sum('sri.qty');
   500→   500→
   501→   501→                    $returnable = (float) $invoiceItem->qty - $alreadyReturned;
   502→   502→                    if ($qty > $returnable + 0.0001) {
   503→   503→                        throw new \RuntimeException(
   504→   504→                            "Return qty {$qty} exceeds returnable qty {$returnable} for product {$productId}."
   505→   505→                        );
   506→   506→                    }
   507→   507→                }
   508→   508→            }
   509→   509→
   510→   510→            // CRITICAL: Look up ORIGINAL avg_cost from the challan's stock_transaction.
   511→   511→            $challanStockTx = DB::table('stock_transactions')
   512→   512→                ->where('reference_type', 'sales_challan')
   513→   513→                ->where('reference_id', $challanId)
   514→   514→                ->where('product_id', $productId)
   515→   515→                ->where('warehouse_id', $warehouseId)
   516→   516→                ->where('is_reversed', false)
   517→   517→                ->first();
   518→   518→
   519→   519→            $originalCost = $challanStockTx ? (float) $challanStockTx->rate : 0;
   520→   520→            if ($originalCost <= 0) {
   521→   521→                throw new \RuntimeException(
   522→   522→                    "Cannot determine original avg_cost for product {$productId} from challan. "
   523→   523→                    . "The challan stock transaction may be missing or reversed."
   524→   524→                );
   525→   525→            }
   526→   526→
   527→   527→            // Rate = sales rate from the invoice item (for revenue reversal).
   528→   528→            $rate = (float) ($item['rate'] ?? 0);
   529→   529→            if ($rate <= 0 && $invoiceItemId > 0) {
   530→   530→                $invoiceItem = DB::table('sales_invoice_items')->where('id', $invoiceItemId)->first();
   531→   531→                $rate = $invoiceItem ? (float) $invoiceItem->rate : 0;
   532→   532→            }
   533→   533→
   534→   534→            $validated[] = [
   535→   535→                'product_id' => $productId,
   536→   536→                'warehouse_id' => $warehouseId,
   537→   537→                'qty' => $qty,
   538→   538→                'rate' => $rate, // sales rate (for revenue reversal)
   539→   539→                'original_cost' => $originalCost, // ORIGINAL avg_cost (for COGS + stock IN)
   540→   540→                'sales_invoice_item_id' => $invoiceItemId > 0 ? $invoiceItemId : null,
   541→   541→            ];
   542→   542→        }
   543→   543→
   544→   544→        if (empty($validated)) {
   545→   545→            throw new \InvalidArgumentException('At least one valid item is required.');
   546→   546→        }
   547→   547→
   548→   548→        return $validated;
   549→   549→    }

   551→   551→    /**
   552→   552→     * P1-5: Create linked damage write-offs for 'Damage' condition return items.
   553→   553→     *
   554→   554→     * Mirrors legacy SalesReturnModel::createLinkedDamageWriteOff:1599.
   555→   555→     * For each Damage-condition item:
   556→   556→     *   1. Create a damage_invoices row (linked via sales_return_id)
   557→   557→     *   2. Create damage_invoice_items
   558→   558→     *   3. Stock OUT via DamageService (immediately back OUT the damaged goods)
   559→   559→     *   4. GL: Dr damage_loss / Cr inventory (via DamageService::confirmDamage)
   560→   560→     *   5. Link sales_return_items.damage_invoice_id → damage_invoices.id
   561→   561→     *
   562→   562→     * Groups damage items by warehouse (one damage_invoice per warehouse).
   563→   563→     *
   564→   564→     * @param SalesReturn $return
   565→   565→     * @param int $confirmedBy
   566→   566→     * @return array List of created damage_invoice IDs
   567→   567→     */
   568→   568→    private function createLinkedDamageWriteOffs(SalesReturn $return, int $confirmedBy): array
   569→   569→    {
   570→   570→        // Group Damage-condition items by warehouse.
   571→   571→        $damageByWarehouse = [];
   572→   572→        foreach ($return->items as $item) {
   573→   573→            if (($item->condition_state ?? 'Good') !== 'Damage') {
   574→   574→                continue;
   575→   575→            }
   576→   576→            $wid = (int) $item->warehouse_id;
   577→   577→            if ($wid <= 0) {
   578→   578→                continue;
   579→   579→            }
   580→   580→            $damageByWarehouse[$wid][] = $item;
   581→   581→        }
   582→   582→
   583→   583→        if (empty($damageByWarehouse)) {
   584→   584→            return []; // No damage items — nothing to write off.
   585→   585→        }
   586→   586→
   587→   587→        $linkedDamageIds = [];
   588→   588→        $returnDate = $return->return_date->format('Y-m-d');
   589→   589→
   590→   590→        foreach ($damageByWarehouse as $warehouseId => $damageItems) {
   591→   591→            // Build items array for DamageService::createDamage.
   592→   592→            $items = [];
   593→   593→            foreach ($damageItems as $item) {
   594→   594→                $items[] = [
   595→   595→                    'product_id' => (int) $item->product_id,
   596→   596→                    'qty' => (float) $item->qty,
   597→   597→                    'rate' => (float) $item->original_cost, // use original_cost (same as return stock IN)
   598→   598→                ];
   599→   599→            }
   600→   600→
   601→   601→            // Create the damage invoice (draft) via DamageService.
   602→   602→            $damage = $this->damageService->createDamage([
   603→   603→                'warehouse_id' => $warehouseId,
   604→   604→                'damage_date' => $returnDate,
   605→   605→                'reason' => 'Auto write-off for damaged sales return #' . $return->return_code,
   606→   606→                'created_by' => $confirmedBy,
   607→   607→                'items' => $items,
   608→   608→            ]);
   609→   609→
   610→   610→            // Link damage_invoices.sales_return_id → this return.
   611→   611→            DB::table('damage_invoices')
   612→   612→                ->where('id', $damage->id)
   613→   613→                ->update(['sales_return_id' => $return->id]);
   614→   614→
   615→   615→            // Confirm the damage (stock OUT + GL Dr damage_loss / Cr inventory).
   616→   616→            $damage = $this->damageService->confirmDamage($damage->id, $confirmedBy);
   617→   617→
   618→   618→            // Link each return item to the damage invoice.
   619→   619→            foreach ($damageItems as $item) {
   620→   620→                DB::table('sales_return_items')
   621→   621→                    ->where('id', $item->id)
   622→   622→                    ->update(['damage_invoice_id' => $damage->id]);
   623→   623→            }
   624→   624→
   625→   625→            $linkedDamageIds[] = $damage->id;
   626→   626→        }
   627→   627→
   628→   628→        return $linkedDamageIds;
   629→   629→    }

   631→   631→    /**
   632→   632→     * P1-5: Reverse linked damage write-offs when a return is reversed.
   633→   633→     *
   634→   634→     * Mirrors legacy SalesReturnModel::reverseLinkedDamageForReturn.
   635→   635→     * Finds all damage_invoices linked to this return (via sales_return_id)
   636→   636→     * and cancels each one via DamageService::cancelDamage (reverses stock + GL).
   637→   637→     *
   638→   638→     * Must be called BEFORE reversing the return's own stock movements
   639→   639→     * (so the damage stock OUT is reversed first, then the return stock IN).
   640→   640→     *
   641→   641→     * @param int $returnId
   642→   642→     * @param int $reversedBy
   643→   643→     * @param string $reason
   644→   644→     */
   645→   645→    private function reverseLinkedDamageForReturn(int $returnId, int $reversedBy, string $reason): void
   646→   646→    {
   647→   647→        $linkedDamages = DB::table('damage_invoices')
   648→   648→            ->where('sales_return_id', $returnId)
   649→   649→            ->where('is_reversed', false)
   650→   650→            ->get();
   651→   651→
   652→   652→        foreach ($linkedDamages as $damage) {
   653→   653→            // Cancel the damage (reverses stock OUT + GL Dr damage_loss / Cr inventory).
   654→   654→            $this->damageService->cancelDamage($damage->id, $reversedBy, "Return reversed: {$reason}");
   655→   655→
   656→   656→            // Clear the link on sales_return_items (set damage_invoice_id = null).
   657→   657→            DB::table('sales_return_items')
   658→   658→                ->where('damage_invoice_id', $damage->id)
   659→   659→                ->update(['damage_invoice_id' => null]);
   660→   660→        }
   661→   661→    }
