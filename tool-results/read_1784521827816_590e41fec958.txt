    44→    44→    44→    private function filterSalesAuditLogs(int $limit = 300): array
    45→    45→    45→    {
    46→    46→    46→        $all = $this->userAudit->getRecentLogs(500);
    47→    47→    47→        $prefixes = ['sale_', 'godown_', 'challan_', 'payment_'];
    48→    48→    48→        $filtered = [];
    49→    49→    49→        foreach ($all as $entry) {
    50→    50→    50→            $action = $entry['action'] ?? '';
    51→    51→    51→            foreach ($prefixes as $p) {
    52→    52→    52→                if (stripos($action, $p) === 0) {
    53→    53→    53→                    $filtered[] = $entry;
    54→    54→    54→                    break;
    55→    55→    55→                }
    56→    56→    56→            }
    57→    57→    57→            if (count($filtered) >= $limit) {
    58→    58→    58→                break;
    59→    59→    59→            }
    60→    60→    60→        }
    61→    61→    61→        return $filtered;
    62→    62→    62→    }
